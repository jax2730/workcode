#include <Common_Uniform_VS.inl>

float4	U_WorldViewProjectMatrix[4];
float4  U_WorldMatrix[3];
float4 	U_WorldScale;

struct VS_INPUT
{
	float3 i_Position	: POSITION0;
	float2 i_UV		: TEXCOORD0;
};

struct VS_OUTPUT
{
	float4 o_PosInClip	: SV_POSITION;
	float2 o_UV		: TEXCOORD0;
};

VS_OUTPUT main(VS_INPUT IN)
{
	VS_OUTPUT vsOut = (VS_OUTPUT)0;
    vsOut.o_UV = IN.i_UV;

    float4 position = float4(IN.i_Position.xyz, 1.0f);

	float4x4 matInvView = transpose((float4x4)U_ViewMatrix);
	position.xyz = mul((float3x3)matInvView, position.xyz);

	float4x4 translateMat = float4x4(
		float4(1.0f, 0.0f, 0.0f, U_WorldMatrix[0].w),
		float4(0.0f, 1.0f, 0.0f, U_WorldMatrix[1].w),
		float4(0.0f, 0.0f, 1.0f, U_WorldMatrix[2].w),
		float4(0, 0, 0, 1)
	);

	float4x4 scaleMat = float4x4(
		float4(U_WorldScale.x, 0.0f, 0.0f, 0),
		float4(0.0f, U_WorldScale.y, 0.0f, 0),
		float4(0.0f, 0.0f, U_WorldScale.z, 0),
		float4(0, 0, 0, 1)
	);

    float4x4 matWorld = mul(translateMat, scaleMat);

	matWorld[0][3] -= U_VS_CameraPosition[0];
	matWorld[1][3] -= U_VS_CameraPosition[1];
	matWorld[2][3] -= U_VS_CameraPosition[2];

	float4x4 matWVP = mul((float4x4)U_ZeroViewProjectMatrix, matWorld);
    
	vsOut.o_PosInClip = mul(matWVP, position);
#ifdef EDITOR_MODE
	#ifdef HierarchicalLOD
		vsOut.o_PosInClip = float4(2.0f * IN.i_UV.x - 1.0f, 1.0f - 2.0f * IN.i_UV.y, 1.0f, 1.0f);
	#endif
#endif
	return vsOut;
}
