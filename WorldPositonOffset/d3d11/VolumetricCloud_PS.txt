struct PS_INPUT
{
	float4 o_Position	: SV_POSITION;
#ifndef SHADOWPASS
	float2 o_UV			: TEXCOORD0;
	float3 o_WorldPos	: TEXCOORD1;
	float3 o_WorldNormal: NORMAL0;
	int	InstanceID 		: TEXCOORD2;
	float3 o_3DUV			: TEXCOORD3;
	#ifdef USESHADOWMAP
	#endif
	#ifdef VERTEX_COLOR
		float4 o_Diffuse	: COLOR0;
	#endif
	float4 o_Fog		: TEXCOORD9;
	float3 o_InstWorldPos : TEXCOORD10;
#endif
#ifdef PARABOLOID_MAPPING
	float	i_ClipDepth			: TEXCOORD12;
#endif
};

struct PS_OUTPUT
{
#ifndef SHADOWPASS
	float4 FragColor : SV_TARGET;
#endif
};


							 
float4 U_PSGeneralRegister1; // U_PSGeneralRegister1.x=_DrawLayer(int)
							 // U_PSGeneralRegister1.y=_MinAlpha
							 // U_PSGeneralRegister1.z=_MaxAlpha
							 
float4 U_PSGeneralRegister2; // U_PSGeneralRegister2.x= frontSSSIntensity
							 // U_PSGeneralRegister2.y= backSSSIntensity
							 // U_PSGeneralRegister2.z=_Power

float4 U_PSGeneralRegister3; // U_PSGeneralRegister3.x=_NoiseScale
							 // U_PSGeneralRegister3.y=_Speed
							 // U_PSGeneralRegister3.z=DiffuseIntensity  //diffuseColor

float4 U_PSGeneralRegister10; // U_PSGeneralRegister10.x=PreColorIntensity;  //PreColorIntensity*(ambient+diffus+SSS)=>ToneMapping
							 // U_PSGeneralRegister10.y=_AmbientIntensity;
							 // U_PSGeneralRegister10.z=*color  LastColorIntensity //lastColor

float4 U_PSGeneralRegister11; // U_PSGeneralRegister11.x=stainingColor.r; 
							 // U_PSGeneralRegister11.y=stainingColor.g;
							 // U_PSGeneralRegister11.z=stainingColor.b 

float4 U_PSGeneralRegister13; // noise scale
float4 U_PSGeneralRegister14; // noise distance

float4 U_PSGeneralRegister4;
float4 U_PSGeneralRegister5;
float4 U_PSGeneralRegister6;

float4 U_PSGeneralRegister7;
float4 U_PSGeneralRegister8;
float4 U_PSGeneralRegister9;

Texture3D S_Noise3DTex;
SamplerState S_Noise3DTexSamplerState;

Texture2D S_Custom;
SamplerState S_CustomSamplerState;
PS_OUTPUT main(PS_INPUT IN)											
{	
	PS_OUTPUT psOut = (PS_OUTPUT)0.f;

#ifdef PARABOLOID_MAPPING
	clip(IN.i_ClipDepth);
#endif

	#ifdef SHADOWPASS
		return psOut;//优化后的处理
	#else

	int i_InstanceID = IN.InstanceID;

	float clipThre = U_PSGeneralRegister1.y + (U_PSGeneralRegister1.z - U_PSGeneralRegister1.y) / (int(U_PSGeneralRegister1.x) - 1) * i_InstanceID;

	float3 vPosToColl = U_PS_CloudCollision.xyz - IN.o_InstWorldPos.xyz;
	float fCollision = max(length(vPosToColl), 0.0001f);
	vPosToColl /= fCollision;

	float3 vCollisionUV = vPosToColl * U_PSGeneralRegister8.z + U_PS_Time.x * U_PSGeneralRegister8.x;
	float fCollisionNoise = S_Noise3DTex.Sample(S_Noise3DTexSamplerState, vCollisionUV).x;
	// 0 ~ 1 from outer to inner
	fCollision /= max(U_PS_CloudCollision.w + fCollisionNoise * U_PSGeneralRegister8.y, 0.0001f);
	fCollision = 1.f - saturate(fCollision);
	fCollision *= U_PSGeneralRegister9.y;

	#ifdef ALPHA_TEST
		// const float MaxScale = 100.f;
		// float2 b20 = float2(0.f, 0.f);
		// float2 b21 = b20 + float2(100.f, 0.f);
		// float2 b22 = b21 + float2(max(U_PSGeneralRegister7.y, 0.0001f) * MaxScale, MaxScale);
		float3 v = IN.o_WorldPos - U_PS_CameraPosition.xyz;
		float l = max(length(v), 0.0001f);
		v /= l;
		l = saturate(dot(v, U_PS_ViewDirection.xyz)) * l;
		// float t = max(l - U_PSGeneralRegister7.z, 0.f) / b22.x;
		// float2 b10 = lerp(b20, b21, t);
		// float2 b11 = lerp(b21, b22, t);
		// float2 b00 = lerp(b10, b11, t);
		// float3 flowUV = IN.o_3DUV.xyz / (b00.y + U_PSGeneralRegister7.x) + U_PS_Time.x * U_PSGeneralRegister3.y;

		// float fNoiseScale = max(l - U_PSGeneralRegister7.z, 0.f) / max(U_PSGeneralRegister7.y, 0.0001f);
		// fNoiseScale += U_PSGeneralRegister7.x;

		float4 vNoiseScaleDistance = U_PSGeneralRegister14;
		vNoiseScaleDistance.y = max(vNoiseScaleDistance.x, vNoiseScaleDistance.y);
		vNoiseScaleDistance.z = max(vNoiseScaleDistance.y, vNoiseScaleDistance.z);
		vNoiseScaleDistance.w = max(vNoiseScaleDistance.z, vNoiseScaleDistance.w);
		float fDist = clamp(l, vNoiseScaleDistance.x, vNoiseScaleDistance.y);
		float fNoiseFactor = (fDist - vNoiseScaleDistance.x) / max(vNoiseScaleDistance.y - vNoiseScaleDistance.x, 0.0001f);
		float fNoiseScale = lerp(U_PSGeneralRegister13.x, U_PSGeneralRegister13.y, fNoiseFactor);
		fDist = clamp(l, vNoiseScaleDistance.y, vNoiseScaleDistance.z);
		fNoiseFactor = (fDist - vNoiseScaleDistance.y) / max(vNoiseScaleDistance.z - vNoiseScaleDistance.y, 0.0001f);
		fNoiseScale = lerp(fNoiseScale, U_PSGeneralRegister13.z, fNoiseFactor);
		fDist = clamp(l, vNoiseScaleDistance.z, vNoiseScaleDistance.w);
		fNoiseFactor = (fDist - vNoiseScaleDistance.z) / max(vNoiseScaleDistance.w - vNoiseScaleDistance.z, 0.0001f);
		fNoiseScale = lerp(fNoiseScale, U_PSGeneralRegister13.w, fNoiseFactor);

		//fNoiseScale = floor(fNoiseScale);
		float3 flowUV = IN.o_3DUV.xyz / fNoiseScale + U_PS_Time.x * U_PSGeneralRegister3.y;
		float noise = S_Noise3DTex.Sample(S_Noise3DTexSamplerState, flowUV.xyz).x;
		
		// 用坐标做个噪点做半透明抖动
		//float dither = frac((sin(IN.o_WorldPos.x + IN.o_WorldPos.y) * 99.0 + 11.0) * 99.0);
		//clip((noise - clipThre) - dither / 5.0);
		
		float dither = S_Custom.SampleLevel(S_CustomSamplerState, IN.o_UV * U_PSGeneralRegister10.x, 0.0).r;
		
		float fLayerFactor = float(i_InstanceID + 1.f) / max(U_PSGeneralRegister1.x, 1.f);
		dither = i_InstanceID == 0 ? 0.f : dither;
		
		float fAlphaClip = noise - clipThre - dither - min(fCollision, fLayerFactor);
		fAlphaClip -= 1.f - U_MaterialDiffuse.a;
		if (fAlphaClip < 0.f)
			discard;
		return psOut;//优化后的处理
	#endif


	float3 lightDir = normalize(-U_MainLightDirection.xyz);
	float3 viewLightDir = normalize(-U_PS_ViewDirection.xyz);
	float3 viewDir = normalize(U_PS_CameraPosition.xyz - IN.o_WorldPos.xyz);
	// float3 worldNormal = normalize(IN.o_WorldNormal);
	float fNormalFactor = smoothstep(0.f, 1.f, clamp(fCollision / U_PSGeneralRegister8.w, 0.f, 1.f));
	vPosToColl = normalize(vPosToColl + float3(0.f, U_PSGeneralRegister9.x, 0.f));
	float3 worldNormal = normalize(lerp(vPosToColl, IN.o_WorldNormal.xyz, 1.f - fNormalFactor));
	float NdotL = max(dot(worldNormal, lightDir), 0.0);
	float smoothNdotL = saturate(pow(NdotL,2-clipThre));

	float NdotviewL = max(dot(worldNormal, viewLightDir), 0.0);
	float smoothNdotviewL = saturate(pow(NdotviewL,2-clipThre));

	// 背面次表面散射
	float3 backLitDir = worldNormal*(1.0f-U_PSGeneralRegister2.y) + lightDir;
	float backSSS = saturate(dot(viewDir,-backLitDir));
	backSSS = saturate(pow(backSSS,2+clipThre*2)*1.5f);

	float finalLit = saturate(smoothNdotviewL * U_PSGeneralRegister11.w+
			saturate(smoothNdotL + backSSS) * (1 - /*NdotviewL */ U_PSGeneralRegister11.w));

	half4 diffuseColor = S_Diffuse.Sample(S_DiffuseSamplerState, IN.o_UV);
	float3 baseMap = pow(U_PSGeneralRegister11.rgb, 2.2)*
	U_PSGeneralRegister3.z * pow(diffuseColor.rgb,2.2);
	#ifdef TEXTURE_MASK
		half4 mask = S_Custom2.Sample(S_Custom2SamplerState, IN.o_UV);
		half3 colorR = pow(U_PSGeneralRegister4.rgb, 2.2) * mask.r;
		half3 colorG = pow(U_PSGeneralRegister5.rgb, 2.2) * mask.g;
		half3 colorB = pow(U_PSGeneralRegister6.rgb, 2.2) * mask.b;
		float f = min(mask.r + mask.g + mask.b, 1.0);
		half3 colorF = colorR + colorG + colorB;
		baseMap = lerp(baseMap, baseMap * colorF, f);
	#endif

	#ifdef VERTEX_COLOR
		baseMap *= IN.o_Diffuse.rgb;
	#endif
	float4 final_color = float4(1.0f, 1.0f, 1.0f, 1.0f);
	
	final_color.rgb = (baseMap + U_PBR_MainLightColor.rgb * finalLit *
		U_PSGeneralRegister10.z);// * (1.f - fCollision);
	#ifdef EMISSIVE
		float illumFactor = U_PSGeneralRegister1.x - U_PSGeneralRegister1.w*i_InstanceID + 1;
		half3 misColor = U_PSGeneralRegister0.xyz;//pow(U_PSGeneralRegister0.xyz, 2.2);
		half3 missiveColor = baseMap.rgb * misColor * illumFactor;
		half3 minMissiveColor = missiveColor * U_PSGeneralRegister0.w;
		half todFactor = lerp(1.0, U_LightPower.x, U_PSGeneralRegister12.z);
		half3 maxMissiveColor = missiveColor * U_PSGeneralRegister12.y;
		half factor = U_MaterialSpecular.w;	
		final_color.rgb += lerp(minMissiveColor, lerp(minMissiveColor, maxMissiveColor,
					half3(factor, factor, factor)), U_PSGeneralRegister12.w) * todFactor * diffuseColor.a;
	#endif

#ifdef CLUSTER_POINT_LIGHT
		final_color.xyz += U_PSGeneralRegister10.w * baseMap * 
		calculateShadowPointLightForVolumetricCloud(float4(IN.o_InstWorldPos, 1.0), worldNormal);
	
        final_color.xyz += 
            (U_PSGeneralRegister10.w * baseMap * calculateClusterPointLightForVolumetricClound(IN.o_Position.xy,
                                                                                               IN.o_InstWorldPos.xyz));
#endif
        
	half normalY = worldNormal.y * 0.5 + 0.5;
	final_color.rgb += baseMap * lerp(U_PBR_GroundLightColor.rgb,
			U_PBR_SkyLightColor.rgb, normalY) * U_PSGeneralRegister10.y;

	psOut.FragColor += final_color;
	psOut.FragColor.rgb = psOut.FragColor.rgb * IN.o_Fog.a + IN.o_Fog.rgb;
	float4 sphFog = ComputeSphFogBlocked(
		IN.o_WorldPos.xyz,
		U_PS_CameraPosition.xyz,
		-viewDir,
		IN.o_Position.xy
	);
	psOut.FragColor.rgb = lerp(psOut.FragColor.rgb, sphFog.rgb, sphFog.a);
	#ifdef EDITOR_MODE
	if(U_MaterialCustomParam.w > 0)
		psOut.FragColor.rgb = U_MaterialCustomParam.xyz;			
	#endif
#endif

	return psOut;
}
