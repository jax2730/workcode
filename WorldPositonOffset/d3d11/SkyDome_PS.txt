// intersect ray with shadow cone 
void GetSphereShadowInOut(
    in float3 viewDirection, 
    in float3 sunDirection, 
    out float shadowIn, 
    out float shadowOut)
{
    float3 P0 = CameraPos.xyz - EarthCenter.xyz;
    float P0DotS = dot(P0, sunDirection);
    float viewDotSun = dot(viewDirection, sunDirection);

    // Ray formula: P = P0 + tV
    // Shifted Cone formula: [P+(R/k)S]^2-(1+k^2){[P+(R/k)S]*S}^2 = 0
    // solve for intersection t0 and t1
    float k = SunSize.x;
    float l = 1.f + k * k;
    float a = 1.f - viewDotSun * viewDotSun;
    float b = dot(P0, viewDirection) - l * P0DotS * viewDotSun - k * EarthRadius * viewDotSun;
    float c = dot(P0, P0) - l * P0DotS * P0DotS - 2.f * k * EarthRadius * P0DotS - k * k * EarthRadius * EarthRadius;
    float discriminant = b * b - a * c;
    if(discriminant > 0.f)
    {
        shadowIn = max(0.f, (-b - sqrt(discriminant)) / a);
        shadowOut = (-b + sqrt(discriminant)) / a;

        // clamp shadow in umbra apex and sphere center
        float dBase = -P0DotS / viewDotSun;
        float dApex = -(P0DotS + EarthRadius / k) / viewDotSun;

        if(viewDotSun > 0.f)
        {
            shadowIn = max(shadowIn, dApex);
            shadowOut= a > 0.f ? min(shadowOut, dBase) : dBase;
        }
        else{
            shadowIn = a > 0.0f ? max(shadowIn, dBase) : dBase;
            shadowOut = min(shadowOut, dApex);
        }
    }
    else{
        shadowIn = 0.f;
        shadowOut = 0.f;
    }
}
