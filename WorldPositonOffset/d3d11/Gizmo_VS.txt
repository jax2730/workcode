//Gizmo_VS
//World View projection
#include <Common_Uniform_VS.inl>

float4 U_WorldMatrix[3];
float4 U_InvProjectMatrix[4];

#ifdef USEINSTANCE
	uniform float4 U_VSCustom0[InstanceCount];
#else
	uniform float4 U_WorldViewProjectMatrix[4];
#endif

struct VS_INPUT
{
    float3 i_Pos    : POSITION0;
    float2 i_UV		: TEXCOORD0;
};

struct VS_OUTPUT
{
    float4 o_PosInClip  : SV_POSITION;
    float2 o_UV		    : TEXCOORD0;
};

VS_OUTPUT main(VS_INPUT IN, uint i_InstanceID : SV_InstanceID)
{
    VS_OUTPUT vsOut = (VS_OUTPUT)0;
    vsOut.o_UV = IN.i_UV;

    float4 pos = float4(IN.i_Pos, 1.0f);
	float4x4 matInvView = transpose((float4x4)U_ViewMatrix);
	pos.xyz = mul((float3x3)matInvView, pos.xyz);

#ifdef USEINSTANCE
	float4 posScale = U_VSCustom0[i_InstanceID];
	float4x4 matInst = float4x4(
		posScale.w, 0, 0, posScale.x,
		0, posScale.w, 0, posScale.y,
	 	0, 0, posScale.w, posScale.z,
	 	0, 0, 0, 1);
		
	// float4x4 matWorld = float4x4(
	// 		U_WorldMatrix[0],
	// 		U_WorldMatrix[1],
	// 		U_WorldMatrix[2],
	// 		float4(0, 0, 0, 1));

	float4x4 matWorld = float4x4(
			float4(1, 0, 0, 0),
			float4(0, 1, 0, 0),
			float4(0, 0, 1, 0),
			float4(0, 0, 0, 1));

	matWorld[0][3] -= U_VS_CameraPosition[0];
	matWorld[1][3] -= U_VS_CameraPosition[1];
	matWorld[2][3] -= U_VS_CameraPosition[2];

	matWorld = mul(matWorld, matInst);

	float4x4 matWVP = mul((float4x4)U_ZeroViewProjectMatrix, matWorld);
	vsOut.o_PosInClip = mul(matWVP, float4(pos.xyz, 1.f));

#else

	float4x4 matWorld = float4x4(
			U_WorldMatrix[0],
			U_WorldMatrix[1],
			U_WorldMatrix[2],
			float4(0, 0, 0, 1));

	matWorld[0][3] -= U_VS_CameraPosition[0];
	matWorld[1][3] -= U_VS_CameraPosition[1];
	matWorld[2][3] -= U_VS_CameraPosition[2];

	float4x4 matWVP = mul((float4x4)U_ZeroViewProjectMatrix, matWorld);
	vsOut.o_PosInClip = mul(matWVP, pos);
#endif

#ifdef EDITOR_MODE
	#ifdef HierarchicalLOD
		vsOut.o_PosInClip = float4(2.0f * IN.i_UV.x - 1.0f, 1.0f - 2.0f * IN.i_UV.y, 1.0f, 1.0f);
	#endif
#endif
    return vsOut;
}
