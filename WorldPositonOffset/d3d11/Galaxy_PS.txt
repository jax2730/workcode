// License: Unknown, author: nmz (twitter: @stormoid), found: https://www.shadertoy.com/view/NdfyRM
float sRGB(float t)
{ 
    return lerp(1.055 * pow(t, 1.0 / 2.4) - 0.055, 12.92 * t, step(t, 0.0031308)); 
}
// License: Unknown, author: nmz (twitter: @stormoid), found: https://www.shadertoy.com/view/NdfyRM
float3 sRGB(in float3 c) 
{ 
    return float3 (sRGB(c.x), sRGB(c.y), sRGB(c.z)); 
}

// License: Unknown, author: Matt Taylor (https://github.com/64), found: https://64.github.io/tonemapping/
float3 aces_approx(float3 v) 
{
    v = max(v, 0.0);
    v *= 0.6f;
    float a = 2.51f;
    float b = 0.03f;
    float c = 2.43f;
    float d = 0.59f;
    float e = 0.14f;
    return clamp((v * (a * v + b)) / (v * (c * v + d) + e), 0.0f, 1.0f);
}

// License: Unknown, author: Unknown, found: don't remember
float tanh_approx(float x) {
  //  Found this somewhere on the interwebs
  //  return tanh(x);
    float x2 = x*x;
    return clamp(x*(27.0 + x2)/(27.0+9.0*x2), -1.0, 1.0);
}


// License: WTFPL, author: sam hocevar, found: https://stackoverflow.com/a/17897228/418488
static float4 hsv2rgb_K = float4(1.0, 2.0 / 3.0, 1.0 / 3.0, 3.0);

float3 hsv2rgb(float3 c) 
{
    float3 p = abs(frac(c.xxx + hsv2rgb_K.xyz) * 6.0 - hsv2rgb_K.www);
    return c.z * lerp(hsv2rgb_K.xxx, clamp(p - hsv2rgb_K.xxx, 0.0, 1.0), c.y);
}

// License: MIT OR CC-BY-NC-4.0, author: mercury, found: https://mercury.sexy/hg_sdf/
float2 mod2(inout float2 p, float2 size) 
{
    float2 c = floor((p + size*0.5) / size);
    p = ((p + size * 0.5) % size) - size * 0.5;
    return c;
}

// License: Unknown, author: Unknown, found: don't remember
float2 hash2(float2 p) 
{
    p = float2(dot (p, float2 (127.1, 311.7)), dot (p, float2 (269.5, 183.3)));
    return frac(sin(p)*43758.5453123);
}

float3 hash3(float3 p3)
{
	p3 = frac(p3 * float3(.1031, .1030, .0973));
    p3 += dot(p3, p3.yxz + 33.33);
    return frac((p3.xxy + p3.yxx) * p3.zyx);
}

float2 shash2(float2 p) 
{
    return -1.0 + 2.0 * hash2(p);
}

float3 shash3(float3 p)
{
    return -1.0 + 2.0 * hash3(p);
}

float3 toSpherical(float3 p) 
{
    float r   = length(p);
    float t   = acos(p.z/r);
    float ph  = atan2(p.y, p.x);
    return float3(r, t, ph);
}


// License: CC BY-NC-SA 3.0, author: Stephane Cuillerdier - Aiekick/2015 (twitter:@aiekick), found: https://www.shadertoy.com/view/Mt3GW2
float3 blackbody(float Temp) 
{
    float3 col = 255;
    col.x = 56100000. * pow(Temp,(-3. / 2.)) + 148.;
    col.y = 100.04 * log(Temp) - 623.6;
    if (Temp > 6500.) col.y = 35200000. * pow(Temp,(-3. / 2.)) + 184.;
    col.z = 194.18 * log(Temp) - 1448.6;
    col = clamp(col, 0., 255.)/255.;
    if (Temp < 1000.) col *= Temp/1000.;
    return col;
}


// License: MIT, author: Inigo Quilez, found: https://www.shadertoy.com/view/XslGRr
float noise(float3 p) {
  // Found at https://www.shadertoy.com/view/sdlXWX
  // Which then redirected to IQ shader
  float3 i = floor(p);
  float3 f = frac(p);
  float3 u = f * f * (3 - 2 * f);
  
  float n = lerp(
                lerp( lerp( dot(shash3(i + float3(0, 0, 0) ), f - float3(0, 0, 0)), 
                            dot(shash3(i + float3(1, 0, 0) ), f - float3(1, 0, 0)), u.x),
                    lerp(   dot(shash3(i + float3(0, 1, 0) ), f - float3(0, 1, 0)), 
                            dot(shash3(i + float3(1, 1, 0) ), f - float3(1, 1, 0)), u.x), u.y),
                lerp( lerp( dot(shash3(i + float3(0, 0, 1) ), f - float3(0, 0, 1)), 
                            dot(shash3(i + float3(1, 0, 1) ), f - float3(1, 0, 1)), u.x),
                    lerp(   dot(shash3(i + float3(0, 1, 1) ), f - float3(0, 1, 1)), 
                            dot(shash3(i + float3(1, 1, 1) ), f - float3(1, 1, 1)), u.x), u.y), u.z);
  return 2.0 * n;
}

float fbm(float3 p, float o, float s, int iters) 
{
    p *= s;
    p += o;

    const float aa = 0.5;

    float h = 0.0;
    float a = 1.0;
    float d = 0.0;
    for (int i = 0; i < iters; ++i) {
        h += a * noise(p);
        p += float3(10.7, 8.3, 7.1);
        a *= aa;
    }
    h /= iters;

    return h;
}

float height(float3 p) 
{
    float h = fbm(p, 0.0, 2.0, 5);
    h *= 0.3;
    h += 0.0;
    return (h);
}

float3 mod3(inout float3 p, float3 size) 
{
  float3 c = floor((p + size*0.5)/size);
  p = ((p + size*0.5) % size) - size*0.5;
  return c;
}

float3 stars(float3 sp) 
{
    float3 col = 0.0;

    const float m = LAYERS;

    for (float i = 0.0; i < m; ++i) 
    {
        float3 pp = sp + 0.5 * i;
        float s = i/(m-1.0);
        float3 dim  = lerp(0.05, 0.003, s) * PI;
        float3 np = mod3(pp, dim);
        float3 h = hash3(np + 127.0 + i);
        float3 o = -1.0+2.0*h;
        float y = sin(sp.x);
        pp += o*dim * 0.5;
        float l = length(pp);

        float3 scol = lerp(8.0*h.y, 0.25*h.y*h.y, s) * blackbody(lerp(3000.0, 22000.0, h.x*h.x));

        col += exp(-(6000.0 / lerp(2.0, 0.25, s)) * max(l - 0.001, 0.0)) * scol;
    }

    return col;
}

// License: MIT, author: Inigo Quilez, found: https://iquilezles.org/articles/spherefunctions
float2 raySphere(float3 ro, float3 rd, float4 sph) {
    float3 oc = ro - sph.xyz;
    float b = dot( oc, rd );
    float c = dot( oc, oc ) - sph.w*sph.w;
    float h = b*b - c;
    if( h<0.0 ) return -1.0;
    h = sqrt( h );
    return float2(-b - h, -b + h);
}


float4 moon(float3 ro, float3 rd, float3 lp, float4 md) 
{
    float2 mi = raySphere(ro, rd, md);
    if (mi.x <= 0) return 0;
    float3 p    = ro + mi.x*rd;
    float3 n    = normalize(p-md.xyz);
    float3 r    = reflect(rd, n);
    float3 ld   = normalize(lp - p);
    float fre = max(dot(n, rd), 0)+1.0;
    fre = pow(fre, 15.0);
    float dif = max(dot(ld, n), 0.0);
    float spe = pow(max(dot(ld, r), 0.0), 8.0);
    float i = 0.5 * tanh_approx(20.0*fre*spe+0.05*dif);
    float3 col = blackbody(1500.0)*i+hsv2rgb(float3(0.6, lerp(0.6, 0.0, i), i));

    float t = tanh_approx(0.25*(mi.y-mi.x));

    return float4(col, t);
}

float3 sky(float3 ro, float3 rd, float2 sp, float3 lp) 
{
    float ld = max(dot(normalize(lp), rd),0.0);
    float y = -0.5 + sp.x / PI;
    y = max(abs(y)-0.02, 0.0) + 0.1 * smoothstep(0.5, PI, abs(sp.y));
    return hsv2rgb(float3(0.6, 0.75, 0.35*exp(-15.0*y)));
}

float3 galaxy(float3 rayDir, float2 sp) 
{
    float2 gp = sp;

    float gcc = dot(gp, gp);
    float2 gc = exp(-abs(float2(1, 4)*gp));
    float gh = gc.y*gc.x;
    float3 col = 0.0;
    col += blackbody(lerp(300.0, 1500.0, gh))*gh;
    col += hsv2rgb(float3(0.6, 0.5, 0.00125 / gcc));
    col *= lerp(0, 1.0, gh);
    return col;
}

float3 color(float3 ro, float3 rd, float3 lp, float4 md, float2 positionHCS) 
{

    float3 col = 0.0;

    float4 gc = float4(lp, 0);  // dot receive translations
    float4 gcHCS;
    gcHCS.x = dot(U_WorldViewProjectMatrix[0], gc);
    gcHCS.y = dot(U_WorldViewProjectMatrix[1], gc);
    gcHCS.w = dot(U_WorldViewProjectMatrix[3], gc);
    gcHCS.xy /= gcHCS.w;

    float4 mcol = moon(ro, rd, lp, md);

    float2 offset = positionHCS - gcHCS.xy;

    if (gcHCS.w > 0)
    {
        col += galaxy(rd, offset);
    }
    col = lerp(col, mcol.xyz, mcol.w);
    float2 sp = toSpherical(rd.xyz).yz;
    col += sky(ro, rd, sp, lp);

    return col;
}

struct PS_INPUT
{
    float4 i_PosInClip		: SV_POSITION;
    float4 i_UV				: TEXCOORD0;
    float4 i_ViewDir        : TEXCOORD1;
    float4 i_PositionHCS    : TEXCOORD2;
};

float4 main(PS_INPUT IN): SV_TARGET
{
    float3 lp = float3(3, 1.5, 4);
    float4 md = 50.0 * float4(float3(0.4, 0.2, -1.0), 0.5);
    
    float3 col= color(0, normalize(IN.i_ViewDir.xyz), 500.0 * lp, md, IN.i_PositionHCS.xy);

    col = aces_approx(col);
    col = sRGB(col);

    // return float4(col, 1.0);
    return float4(col, 1.0);
}
