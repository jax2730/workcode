float3 calculateSSS(float3 lightDir, float3 viewDir, float3 normal, float3 albedo,float SSSWidth,float3 SSSColor) 
{
	float NdotL = max(dot(normal, lightDir), 0.0); 
	float NdotV = max(dot(normal, viewDir), 0.0);

	// 使用Wrap Lighting模型来增强SSS效果
	// 它可以使光线在表面边缘处产生更柔和的散射效果
	float wrap = 1.0 + SSSWidth * NdotL;
	float sssStrength = pow(NdotL / wrap, 2.0); 

	// 结合SSS颜色和物体的基本颜色
	float3 scatteredLight = SSSColor.xyz * sssStrength;//内部折射颜色变化

	// 将SSS效果与物体表面的基本颜色结合
	return lerp(albedo, scatteredLight, sssStrength);
}

struct PS_INPUT
{
    float4 i_PosInClip			: SV_POSITION;

	#ifdef SHADOWPASS
		#ifdef ALPHA_TEST
			float4 	i_UV		: TEXCOORD0;
		#endif // ALPHA_TEST
	#else // SHADOWPASS
	    float4 i_UV				: TEXCOORD0;
	    float4 i_Normal			: TEXCOORD1;
	    float4 i_WPos			: TEXCOORD2;
	    float4 i_Fog			: TEXCOORD3;	
	    float4 i_Tangent		: TEXCOORD4;
	    float4 i_Binormal		: TEXCOORD5;
	    float4 i_EyeDir			: TEXCOORD6;

	    #ifdef VERTEX_COLOR
	        float4 o_Diffuse    : TEXCOORD7;
		#endif // VERTEX_COLOR

		#ifdef USESHADOWMAP
			float	i_ViewSpaceZ					: TEXCOORD8;
		#endif // USESHADOWMAP
	#endif // SHADOWPASS

	#ifdef PARABOLOID_MAPPING
		float	i_ClipDepth			: TEXCOORD13;
	#endif // PARABOLOID_MAPPING
};

#if defined(HierarchicalLOD) && defined(EDITOR_MODE)
	struct PS_OUTPUT{
		float4 albedo			: SV_TARGET0;
		float4 roughnessMetal	: SV_TARGET1;
		float4 normal			: SV_TARGET2;
	};
#else // HierarchicalLOD
	struct PS_OUTPUT
	{
	#ifndef SHADOWPASS
	    float4 o_Color : SV_TARGET;
	#endif

PS_OUTPUT main(PS_INPUT IN) : SV_TARGET
{
    PS_OUTPUT psOut = (PS_OUTPUT) 1.f;

#ifdef PARABOLOID_MAPPING
	clip(IN.i_ClipDepth);
#endif

#ifdef SHADOWPASS
	#ifdef ALPHA_TEST
		float4 color = texDiffuse(IN.i_UV.xy);
		clip(color.a - U_MaterialDiffuse.a);
	#endif
#else
	#ifdef NoColor
		psOut.o_Color = float4(0.0, 0.0, 0.0, 0.0);
		return psOut;
	#endif
	float2 uv = IN.i_UV.xy;

	float2 underUV = uv;
	float roleUV = U_PSGeneralRegister1.z;
	underUV = float2(underUV.x * cos(roleUV) - underUV.y * sin(roleUV),
					underUV.x * sin(roleUV) + underUV.y * cos(roleUV));
	float3 viewDir = normalize(IN.i_EyeDir.xyz);
	float3 normal = normalize(IN.i_Normal.xyz);
	
	#ifdef USESHADOWMAP
		float4 shadowSpacePos[SHADOWSPLITE];
		ProjectToLightSpace(shadowSpacePos, IN.i_WPos.xyz, normal, IN.i_ViewSpaceZ);
	#endif

	uv *= U_PSGeneralRegister3.x;
	
	float4 iceColor = S_Diffuse.Sample(S_DiffuseSamplerState, uv);
	
	half4 bumpColor = 0.0;
	
	half3 nn = normal;
	half3 tn = normalize(IN.i_Tangent.xyz);
	half3 bn = normalize(IN.i_Binormal.xyz);

	tn = normalize(tn - nn * dot(tn, nn));
	bn = normalize(bn - nn * dot(bn, nn));

	half3x3 tbn =
	{
		tn.x, bn.x, nn.x,
		tn.y, bn.y, nn.y,
		tn.z, bn.z, nn.z 
	};
		
	bumpColor = S_Bump.Sample(S_BumpSamplerState, uv);
		
	half3 bump = bumpColor.rgb * 2.0 - 1.0;
	bump *= half3(U_PSGeneralRegister0.w, U_PSGeneralRegister0.w, 1.0);	
	bump.z = 1.h;
	bump = normalize(bump);
	normal = mul(tbn, bump);

	//计算一下法线影响下层冰面UV
	float2 normalInf = (normal.xy + 0.5) * 0.5;
	
	underUV *= U_PSGeneralRegister3.y;//下冰面UV缩放
	float3  parallax = mul(transpose(tbn), viewDir.xyz);
	underUV = underUV - parallax.xy * U_PSGeneralRegister0.x + normalInf * U_PSGeneralRegister4.x;
	underUV = float2(underUV.x + U_PSGeneralRegister4.y, underUV.y + U_PSGeneralRegister4.z);
	
	float4 mindIceColor = S_Custom.Sample(S_CustomSamplerState, underUV);
	
	float4 diffuseColor = lerp(iceColor, mindIceColor, U_PSGeneralRegister0.z);

	#ifdef ALPHA_TEST
		clip(diffuseColor.a - U_MaterialDiffuse.a);
	#endif

	half3 baseMap = pow(diffuseColor.rgb, 2.2);
	
	half3 colorMask = pow(U_PSGeneralRegister5.rgb, 2.2);

	#ifdef TEXTURE_MASK
		half4 mask = S_Custom2.Sample(S_Custom2SamplerState, uv);
		half3 colorR = colorMask * mask.r;
		half3 colorG = pow(U_PSGeneralRegister6.rgb, 2.2) * mask.g;
		half3 colorB = pow(U_PSGeneralRegister7.rgb, 2.2) * mask.b;
		float f = (min(mask.r + mask.g + mask.b, 1.0)) * U_PSGeneralRegister3.w;
		half3 colorF = colorR + colorG +  colorB;
		baseMap = lerp(baseMap, baseMap * colorF, f);
	#else
		colorMask = lerp(half3(1.0, 1.0, 1.0), colorMask, U_PSGeneralRegister3.w);
		baseMap = baseMap * colorMask;
	#endif
	half metallic = 1.0;
	half roughness = 0.0001;
	half ao = 1.0;
	
	#ifdef SIMPLE
		metallic = U_PSGeneralRegister2.z;
		roughness = U_PSGeneralRegister2.y;
	#else
		half3 mixMap = S_Specular.Sample(S_SpecularSamplerState, uv).rgb;
		metallic = mixMap.r;
		roughness = clamp(mixMap.g + U_PSGeneralRegister0.y, 0.0, 1.0);
		ao = mixMap.b;
	#endif
		
	float wetFinal = U_PSGeneralRegister1.x;
	roughness -= wetFinal;
	roughness = max(roughness, 1e-6);
	metallic = lerp(metallic, 0.5, wetFinal);
	
	#if defined(HierarchicalLOD) && defined(EDITOR_MODE)
		psOut.albedo = float4(pow(baseMap, 0.4545), 1.f);
		psOut.roughnessMetal = float4(metallic, roughness, 1.f, 1.f);
		psOut.normal = float4(normalize(bump) * 0.5h + 0.5h, 1.f);
	#else // HierarchicalLOD

	half3 color = 0.0;

	half3 Laa = half3(-U_MainLightDirection.x, 0.0, -U_MainLightDirection.z);
	half3 L = normalize(Laa);
	
	half4 mainLightColor = 0.0;
	half4 skyLightColor = 0.0;
	half4 groundLightColor = 0.0;
	half  blendThreshold = U_PS_AmbientBlendThreshold.y;

	mainLightColor = U_PBR_MainLightColor;
	skyLightColor = U_PBR_SkyLightColor;
	groundLightColor = U_PBR_GroundLightColor;
	
	#ifdef COVER_WEATHER
		float coverLerp = dot(normal.xyz, U_PS_GravityDirection.xyz);
		coverLerp += U_PSGeneralRegister10.x;
		coverLerp = clamp(coverLerp, 0.0, 1.0);
		coverLerp = coverLerp * U_PSGeneralRegister10.y;
		half3 coverMap = U_WeatherParam0.xyz;
		coverLerp *= U_WeatherParam0.w * (1.0 - U_PSGeneralRegister30.x);
		float maskRes = 1.0;
	// #define CUBE_RANGE
	#ifndef CUBE_RANGE
		float4 tsPos = U_CoverColorParams;
		// float3 tsPos = float3(-30.04, 11.98, 123.71);
		float dist = length(IN.i_WPos.xz - tsPos.xz);
		maskRes = smoothstep(tsPos.w - 1.0f, tsPos.w, dist);
		// maskRes = 1.0 - maskRes;
	#else
		float rectSize = 20.0f;
		float2 halfSize = rectSize * 0.5;
		float4 tsPos = U_CoverColorParams;
		// Calculate the distance to the nearest edge of the rectangle
		float2 distToEdge = abs(IN.i_WPos.xz - tsPos.xz) - halfSize;

		// Use length function to determine if the point is within the rectangle boundaries
		maskRes = length(max(distToEdge, 0.0));
		maskRes = clamp(maskRes, 0.0, 1.0);
		// maskRes = 1.0 - maskRes;
	#endif
		coverLerp *= maskRes;
	#endif

	half3 albedo = baseMap * (1.0 - metallic);
	half3 specularColor = lerp(half3(0.04h, 0.04h, 0.04h), baseMap, metallic);
	#ifdef COVER_WEATHER
		albedo = lerp(albedo, U_WeatherParam0.xyz, coverLerp);
	#endif
	color += calculateLightPower(normal, viewDir, L, mainLightColor.xyz * (1.0 - U_PS_ViewLightPercent.x), baseMap, roughness, metallic);
	color = color * U_MaterialDiffuse.rgb;

	#ifdef USESHADOWMAP
		float shadowFactor = calcShadowFactor(
			shadowSpacePos, 
			IN.i_ViewSpaceZ);
		shadowFactor = min(shadowFactor, 1.0);
		color *= shadowFactor;
	#endif

	color += calculateLightPower(normal, viewDir, -U_PS_ViewDirection.xyz, mainLightColor.xyz * U_PS_ViewLightPercent.x, baseMap, roughness, metallic);
	
	half3 ambientColor = ComputeAmbientLight(skyLightColor.xyz, groundLightColor.xyz, normal, blendThreshold, U_PS_GravityDirection.xyz);
	half3 envColor = 0.0;

	#ifdef ENVIRONMENT_MAP
		// set ao to be 1.0
		envColor += CalculateGlossyEnvColor(normal, viewDir, baseMap, metallic, roughness, 1.0,
            0.0, U_PSGeneralRegister2.x, U_GlobalEnvironmentCubeMipMapLevels.x, 
            S_GlobalEnvironmentCubeMap, S_GlobalEnvironmentCubeMapSamplerState);
        envColor += CalculateDiffuseEnvColor(normal, baseMap, metallic, 1.0, U_PBR_RMatrix, U_PBR_GMatrix,
            U_PBR_BMatrix, U_PSGeneralRegister2.x);
        envColor *= U_PBR_Environment_Explosure.x;
	#else
		envColor += CalculateDefaultEnvColor(normal, viewDir, baseMap, metallic, roughness, 1.0, 0.0);
	#endif
	color += envColor * ambientColor;
	color *= ao;
	
	float3 sss = calculateSSS(L, viewDir, normal, albedo, U_PSGeneralRegister8.y, U_PSGeneralRegister9.rgb);
	color += sss * U_PSGeneralRegister8.x;

	color *= U_MaterialCustomParam1.x;
	color = color * IN.i_Fog.a + IN.i_Fog.rgb;
	psOut.o_Color = float4(pow(color, 1.0 / 2.2), 1.0f);
		
	#ifdef VERTEX_COLOR
		psOut.o_Color.a *= (IN.i_Diffuse.g * IN.i_Diffuse.g);
	#endif
	#endif // HierarchicalLOD
#endif 
    return psOut;
}
