
#define KERNELSIZE 9



Texture2D	S_2DExt0;
SamplerState S_2DExt0SamplerState;

float4 U_PSCustom0;		//ViewPortRange
float4 U_PSCustom1;		//ViewPortWideHigh

float4  U_PSCustom2[KERNELSIZE];//GaussianBlurWeight

struct PS_INPUT
{
    float4 i_PosInClip			: SV_POSITION;
	float4 i_UV			: TEXCOORD0;
};

struct PS_OUTPUT
{
	float4 o_Color	: SV_TARGET;
};


PS_OUTPUT main(PS_INPUT IN)
{
	PS_OUTPUT psOut = (PS_OUTPUT)1.f;
	
	float2 uv = IN.i_UV.xy * U_PSCustom0.xy;
	
	//clip(U_PSCustom0.xy - uv);
	
	float weightall = 0.0;
	
	for (int i = 0; i < KERNELSIZE; i++)
    {    
        weightall += U_PSCustom2[i].x;
    }
	
	float4 color = 0.0;
	
	float4 pixelSize = U_PSCustom0 / U_PSCustom1;
    for (int j = 0; j < KERNELSIZE; j++)
    {    
        //color += S_2DExt0.Sample( S_2DExt0SamplerState, uv + float2(0.0f, U_PSCustom2[j].y) / KERNELSIZE * pixelSize.xy ) * U_PSCustom2[j].x;
		color += S_2DExt0.Sample( S_2DExt0SamplerState, uv + float2(0.0f, U_PSCustom2[j].y) * pixelSize.xy ) * U_PSCustom2[j].x;
    }
	
	color.rgba = color.rgba/ weightall;
	
	psOut.o_Color.rgb = color.rgb;
	
	psOut.o_Color.a = 1.0;

	return psOut;
}
