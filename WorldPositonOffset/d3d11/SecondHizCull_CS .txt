bool HiZ(InstanceData instData, ClusterData clusterData, float3 minBox, float3 maxBox)
{
	float4 BBox[8];
	float4x4 vpMatrix;
	vpMatrix[0] = U_CSCustom3[0];
	vpMatrix[1] = U_CSCustom3[1];
	vpMatrix[2] = U_CSCustom3[2];
	vpMatrix[3] = U_CSCustom3[3];
	BBox[0] = mul(vpMatrix, float4(minBox.x, minBox.y, minBox.z, 1.0f));
	BBox[1] = mul(vpMatrix, float4(maxBox.x, minBox.y, minBox.z, 1.0f));
	BBox[2] = mul(vpMatrix, float4(minBox.x, minBox.y, maxBox.z, 1.0f));
	BBox[3] = mul(vpMatrix, float4(maxBox.x, minBox.y, maxBox.z, 1.0f));
	
	BBox[4] = mul(vpMatrix, float4(minBox.x, maxBox.y, minBox.z, 1.0f));
	BBox[5] = mul(vpMatrix, float4(maxBox.x, maxBox.y, minBox.z, 1.0f));
	BBox[6] = mul(vpMatrix, float4(minBox.x, maxBox.y, maxBox.z, 1.0f));
	BBox[7] = mul(vpMatrix, float4(maxBox.x, maxBox.y, maxBox.z, 1.0f));
	
	
	float3 minPos = float3(100000, 100000, 100000);
    float3 maxPos = float3(-100000, -100000, -100000);
	
	for (int i=0; i<8; i++)
	{
		BBox[i].xyz /= BBox[i].w;
		minPos = min(minPos, BBox[i].xyz);
        maxPos = max(maxPos, BBox[i].xyz);
	}
		
	float minU = minPos.x * 0.5 + 0.5;
	float maxU = maxPos.x * 0.5 + 0.5;

	float v1 = 0.5 - minPos.y * 0.5;
	float v2 = 0.5 - maxPos.y * 0.5;

	float minV = min(v1, v2);
	float maxV = max(v1, v2);
		
	float2 minUV = float2(minU, minV);
    float2 maxUV = float2(maxU, maxV);
				  
	float InstanceDepth = CloserFunc(minPos.z, maxPos.z);
	
	float2 size = (maxUV - minUV) * float2(U_CSCustom5.x, U_CSCustom5.y);
	float maxDim = max(size.x, size.y);
	float mips = floor(log2(maxDim)); 
	mips = clamp(mips, 0.0, float(U_CSCustom4.z));
	//------------
	float4 Samples;
    Samples.x = S_2DExt0.SampleLevel(S_2DExt0SamplerState, float2(minUV.x, minUV.y), mips).r;
    Samples.y = S_2DExt0.SampleLevel(S_2DExt0SamplerState, float2(maxUV.x, minUV.y), mips).r;
    Samples.z = S_2DExt0.SampleLevel(S_2DExt0SamplerState, float2(minUV.x, maxUV.y), mips).r;
    Samples.w = S_2DExt0.SampleLevel(S_2DExt0SamplerState, float2(maxUV.x, maxUV.y), mips).r;
	//------------

	float maxDepth = CompFunc( CompFunc( Samples.x, Samples.y ), CompFunc( Samples.z, Samples.w ) );
	
	if(FilterFunc(maxDepth, InstanceDepth)) 
	{
		return false;
	}
	else
	{
		return true;
	}
}

void invalidCluster(uint clusterIdx)
{
    //B_CSCustom1[clusterIdx].ClusterID = 0xF0000000u;
	InstClusterIDStore_ClusterID(B_CSCustom1, clusterIdx, 0xF0000000u);
}

bool isValidCluster(uint clusterID)
{
	return (clusterID & 0xF0000000u) == 0u;
}


uint CullTriangle(InstanceData instData, ClusterData clusterData, float4 sphereData, out uint mask1, out uint mask2)
{
	float3 camDir =  U_CSCustom1 - sphereData.xyz;
	float3x3 invWorldMat;
	invWorldMat[0] = float3(instData.InvTransposeTransform[0],instData.InvTransposeTransform[1],instData.InvTransposeTransform[2]);
	invWorldMat[1] = float3(instData.InvTransposeTransform[4],instData.InvTransposeTransform[5],instData.InvTransposeTransform[6]);
	invWorldMat[2] = float3(instData.InvTransposeTransform[8],instData.InvTransposeTransform[9],instData.InvTransposeTransform[10]);
	//worldMat = transpose(worldMat);
	//mat3 invWorldMat = inverse(worldMat);
	float3 camInModel = mul(camDir, invWorldMat);
	float absX = abs(camInModel.x);
	float absY = abs(camInModel.y);
	float absZ = abs(camInModel.z);
	float maxV = max(max(absX, absY), absZ);
	int idx = 0;
	if(abs(maxV - absX) <= 1e-5)
	{
		if(camInModel.x > 0.0f)
		{
			idx = 2;
		}
		else
		{
			idx = 0;
		}
		//idx += int(sign(camInModel.x)) * -2 * int(instData.Flip & 1u);
	}
	else if(abs(maxV - absY) <= 1e-5)
	{
		if(camInModel.y > 0.0f)
		{
			idx = 6;
		}
		else
		{
			idx = 4;
		}
		//idx += int(sign(camInModel.y)) * -2 * int(instData.Flip>>1 & 1u);
	}
	else
	{
		if(camInModel.z > 0.0f)
		{
			idx = 10;
		}
		else
		{
			idx = 8;
		}
		//idx += int(sign(camInModel.z)) * -2 * int(instData.Flip>>2 & 1u);
	}
	uint triCount = 0u;
	for(int i = 0; i < 2; ++i)
	{
		uint mask = clusterData.CubeMap[idx+i];
		uint n = (mask & 0x55555555u) + ((mask & 0xAAAAAAAAu) >> 1);
		n = (n & 0x33333333u) + ((n & 0xCCCCCCCCu) >> 2);
		n = (n & 0x0F0F0F0Fu) + ((n & 0xF0F0F0F0u) >> 4);
		n = (n & 0x00FF00FFu) + ((n & 0xFF00FF00u) >> 8);
		n = (n & 0x0000FFFFu) + ((n & 0xFFFF0000u) >> 16);
		triCount += n;
	}
	mask1 = clusterData.CubeMap[idx];
	mask2 = clusterData.CubeMap[idx+1];
	return triCount;
}



[numthreads(thread_group_size_x, thread_group_size_y, 1)]
void main(uint3 threadIDInGroup : SV_GroupThreadID,
			uint3 groupID : SV_GroupID,
			uint groupIndex : SV_GroupIndex,
			uint3 dispatchThreadID : SV_DispatchThreadID)
{
	uint clusterIdx = dispatchThreadID.x;
	if (clusterIdx >= RawBufferLoadUint(B_CSCustom0, 0))
		return;
	
	InstClusterID inst2Cluster = InstClusterIDLoad(B_CSCustom1, clusterIdx); //B_CSCustom1[clusterIdx];
    
	InstanceData instData = B_CSCustom3[inst2Cluster.InstanceID];
	ClusterData clusterData = B_CSCustom2[inst2Cluster.ClusterID];
    
	float4 sphereData;
	float3 minBox;
	float3 maxBox;
	if(isValidCluster(inst2Cluster.ClusterID))
	{
		
		//Caculate SphereData
		float2 param_0 = unpackHalf2x16(clusterData.BoundBox[0]);
		float2 param_1 = unpackHalf2x16(clusterData.BoundBox[1]);
		float2 param_2 = unpackHalf2x16(clusterData.BoundBox[2]);
		
		float4 _tmpMin = float4(param_0.x, param_0.y, param_1.x, 1.0f);
		float4 _tmpMax = float4(param_1.y, param_2.x, param_2.y, 1.0f);
		
		minBox.x = dot(float4(instData.Transform[0], instData.Transform[1], instData.Transform[2], instData.Transform[3]), _tmpMin);
		minBox.y = dot(float4(instData.Transform[4], instData.Transform[5], instData.Transform[6], instData.Transform[7]), _tmpMin);
		minBox.z = dot(float4(instData.Transform[8], instData.Transform[9], instData.Transform[10], instData.Transform[11]), _tmpMin);
		
		maxBox.x = dot(float4(instData.Transform[0], instData.Transform[1], instData.Transform[2], instData.Transform[3]), _tmpMax);
		maxBox.y = dot(float4(instData.Transform[4], instData.Transform[5], instData.Transform[6], instData.Transform[7]), _tmpMax);
		maxBox.z = dot(float4(instData.Transform[8], instData.Transform[9], instData.Transform[10], instData.Transform[11]), _tmpMax);
		
		sphereData.xyz = minBox + maxBox;
		sphereData.xyz *= float3(0.5f, 0.5f, 0.5f);
		sphereData.w = length(maxBox - minBox);
		
		minBox -= U_CSCustom1;
		maxBox -= U_CSCustom1;
		
		if(HiZ(instData, clusterData, minBox, maxBox))
		{
			uint mask1, mask2;
			uint trangleCount;
			uint idxSum;
			uint IndexCount = clusterData.ConeData[3] >> 16u;
			//Two sides
			if((instData.Flip & 0x80u) == 0x80u)
			{
				mask1 = 0xFFFFFFFFu;
				mask2 = 0xFFFFFFFFu;
				trangleCount = 1u;
				RawInterlockedAdd(B_CSCustom5, DISPATCH_OFFSET * instData.Pipeline, IndexCount, idxSum); //InterlockedAdd(B_CSCustom5[DISPATCH_OFFSET * instData.Pipeline], IndexCount, idxSum);
			}
			else
			{
				trangleCount = CullTriangle(instData, clusterData, sphereData, mask1, mask2);
				RawInterlockedAdd(B_CSCustom5, DISPATCH_OFFSET * instData.Pipeline, trangleCount * 3u, idxSum); //InterlockedAdd(B_CSCustom5[DISPATCH_OFFSET * instData.Pipeline], trangleCount * 3u, idxSum);
			}
			if(trangleCount != 0u)
			{
				uint idxOffset = clusterData.VertexOffset * VERTEX_COUNT_PER_CLUSTER;
				uint offset; 
				RawInterlockedAdd(B_CSCustom4, 0, 1u, offset); //InterlockedAdd(B_CSCustom4[0], 1u, offset);

				ClusterCulledDataStore_ReadFrom(B_CSCustom6, offset, 0, inst2Cluster.ClusterID * INDEX_COUNT_PER_CLUSTER);
				ClusterCulledDataStore_WriteTo(B_CSCustom6, offset, 0, idxSum);
				ClusterCulledDataStore_IndexCount(B_CSCustom6, offset, 0, IndexCount);
				ClusterCulledDataStore_IndexOffset(B_CSCustom6, offset, 0, idxOffset);
				ClusterCulledDataStore_InstanceID(B_CSCustom6, offset, 0, inst2Cluster.InstanceID);
				ClusterCulledDataStore_Mask1(B_CSCustom6, offset, 0, mask1);
				ClusterCulledDataStore_Mask2(B_CSCustom6, offset, 0, mask2);
				ClusterCulledDataStore_Pipeline(B_CSCustom6, offset, 0, instData.Pipeline);

				uint flipCount = (instData.Flip & 0x5u) + ((instData.Flip & 0xAu) >> 1);
				flipCount = (flipCount & 0x3u) + ((flipCount & 0xCu) >> 2);
				ClusterCulledDataStore_IsFlip(B_CSCustom6, offset, 0, flipCount & 0x1u);

				
				//instance count
				//if(gl_GlobalInvocationID.x == 0u)
				RawBufferStoreUint(B_CSCustom5, DISPATCH_OFFSET * instData.Pipeline + 1, 1u); //B_CSCustom5[DISPATCH_OFFSET * instData.Pipeline + 1] = 1u;
			}
			
		}
	}
}
