
#include <Common_Uniform_VS.inl>
#include <Common_Function.inl>

uniform float4	U_VSCustom0; //x:SkySphereSize y:SkyFogHeight
uniform float4	U_VSCustom1;
uniform float4	U_VSCustom2;
uniform float4	U_VSCustom3;

struct VS_INPUT
{
	float4 i_Position	: POSITION0;
	float4 i_UV	        : TEXCOORD0;
};

struct VS_OUTPUT
{
	float4 o_Position	: SV_POSITION;
	float4 o_UV			: TEXCOORD0;
	float4 o_CustomParam: TEXCOORD1;
	float4 o_FogColor	: TEXCOORD2;
	float4 o_ViewDir	: TEXCOORD3;
};

float4 ComputeSkyFogColor(in float3 cameraToWorldPos,in float2 heightScale,in float2 volFogHeightDensity,
	in float4 fogRamp,in float4 vFogColorNear, in float4 vFogColorFar, in float value)
{
	float2 fFactor = fogExp3(cameraToWorldPos, heightScale, volFogHeightDensity, fogRamp);
	fFactor *= value;
	float2 fFactorDiff = 1.0f - fFactor;
	float4 fogColor = fFactor.x * fFactorDiff.y * vFogColorNear + fFactor.y * vFogColorFar;
	fogColor.a = fFactorDiff.x * fFactorDiff.y;
	return fogColor;
}



VS_OUTPUT main(VS_INPUT IN)
{
	VS_OUTPUT vsOut = (VS_OUTPUT)0;
	
	float4 vPos = IN.i_Position;
	


	float4x4 rotatePosMatrix;
	rotatePosMatrix[0] = float4(U_VSCustom1.xyz,0.f);
	rotatePosMatrix[1] = float4(U_VSCustom2.xyz,0.f);
	rotatePosMatrix[2] = float4(U_VSCustom3.xyz,0.f);
	rotatePosMatrix[3] = float4(0.f,0.f,0.f,1.f);
	vPos = mul(rotatePosMatrix,vPos);
	// note o_ViewDir pointing from camera to shading point 
	vsOut.o_ViewDir.xyz = normalize(vPos.xyz);
	
	vPos.w = 0.0f;
	float4 rotatedPos = vPos;
		
	float4 vVMatrixC0 = U_ViewMatrix[0];
	float4 vVMatrixC1 = U_ViewMatrix[1];
	float4 vVMatrixC2 = U_ViewMatrix[2];
	
	vsOut.o_Position.x = dot(vVMatrixC0, vPos);
	vsOut.o_Position.y = dot(vVMatrixC1, vPos);
	vsOut.o_Position.z = dot(vVMatrixC2, vPos);
	
	vPos = vsOut.o_Position;
	
	vPos.w = 1.0f;
	
	float4 vPMatrixC0 = U_ProjectMatrix[0];
	float4 vPMatrixC1 = U_ProjectMatrix[1];
	float4 vPMatrixC2 = U_ProjectMatrix[2];
	float4 vPMatrixC3 = U_ProjectMatrix[3];
	
	vsOut.o_Position.x = dot(vPMatrixC0, vPos);
	vsOut.o_Position.y = dot(vPMatrixC1, vPos);
	vsOut.o_Position.z = dot(vPMatrixC2, vPos);
	vsOut.o_Position.w = dot(vPMatrixC3, vPos);
	
#ifdef REVERSE_DEPTH
	vsOut.o_Position.z = 0.0f;
#else
	vsOut.o_Position.z = vsOut.o_Position.w;
#endif

	vsOut.o_UV = IN.i_UV;
	
	vsOut.o_CustomParam.xyz = rotatedPos.xyz;
	
	float3 skyDir = normalize(rotatedPos.xyz);		
	float factor = saturate(dot(skyDir,U_VS_GravityDirection.xyz));
	factor = factor > U_VSCustom0.y ? 0.0f : 1.0f;
		
	float3 vWPos = rotatedPos.xyz * U_VSCustom0.x + U_VS_CameraPosition.xyz;
	float3 camToWorld =  U_VS_CameraPosition.xyz - vWPos.xyz;

	float4x4 gravityRotatePosMatrix;
	gravityRotatePosMatrix[0] = U_VS_GravityRotationMatrix[0];
	gravityRotatePosMatrix[1] = U_VS_GravityRotationMatrix[1];
	gravityRotatePosMatrix[2] = U_VS_GravityRotationMatrix[2];
	gravityRotatePosMatrix[3] = U_VS_GravityRotationMatrix[3];
	camToWorld = mul(gravityRotatePosMatrix,float4(camToWorld.xyz,0.0f)).xyz;

	vsOut.o_FogColor = ComputeSkyFogColor(camToWorld, U_FogParam.xy, U_FogParam.zw, U_FogRampParam, U_FogColorNear, U_FogColorFar,factor);
#ifdef EDITOR_MODE
	#ifdef HierarchicalLOD
		vsOut.o_Position = float4(2.0f * IN.i_UV.x - 1.0f, 1.0f - 2.0f * IN.i_UV.y, 1.0f, 1.0f);
	#endif
#endif	
	return vsOut;
}
