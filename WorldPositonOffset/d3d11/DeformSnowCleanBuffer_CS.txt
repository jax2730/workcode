#define thread_group_size_x 4
#define thread_group_size_y 4

//x: clear data,y total clear float num,z per thread clean num(no use current),w dispatch dim x/y
float4 U_CSCustom0;
//To clear buffer
RWByteAddressBuffer B_CSCustom0;
#define ToClear B_CSCustom0
#define perThreadClearNum 32

[numthreads(thread_group_size_x, thread_group_size_y, 1)]
void main(uint3 gtId : SV_GroupThreadID,
			uint3 groupID : SV_GroupID,
			uint groupIndex : SV_GroupIndex,
			uint3 dtId : SV_DispatchThreadID
){
    uint2 dispatchDim = uint3((uint)U_CSCustom0.w,(uint)U_CSCustom0.w,1);
	uint gtId1d = gtId.x +gtId.y * thread_group_size_x;
	uint globalOffset = (groupID.y * dispatchDim .x  + groupID.x) * (thread_group_size_x * thread_group_size_y);
	uint curIdGlobal = globalOffset + gtId1d;

	if(curIdGlobal * (uint)U_CSCustom0.z > perThreadClearNum){
		return;
	}
	[unroll]
	for(uint i = 0;i < perThreadClearNum;++i){
		if(i > (uint)U_CSCustom0.y){
			return;
		}
		uint offset = 4 * (curIdGlobal +  i);
		ToClear.Store(offset,U_CSCustom0.x);
	}

}
