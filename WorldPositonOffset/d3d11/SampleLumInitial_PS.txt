Texture2D	S_2DExt0;
SamplerState S_2DExt0SamplerState;


float4 U_PSCustom0;
float4 U_PSCustom1;

struct PS_INPUT
{
    float4	i_PosInClip			: SV_POSITION;
	float4 i_UV			: TEXCOORD0;
};

struct PS_OUTPUT
{
	float4 o_Color	: SV_TARGET;
};

PS_OUTPUT main(PS_INPUT IN)
{
	PS_OUTPUT psOut = (PS_OUTPUT)1.f;
	
	float2 uv = IN.i_UV.xy;
	
	float lum = 0.0f;
	
	lum += dot(pow(S_2DExt0.Sample(S_2DExt0SamplerState, uv + U_PSCustom0.xy).rgb, 2.2f), float3(0.2126f,0.7152f,0.0722f));
	lum += dot(pow(S_2DExt0.Sample(S_2DExt0SamplerState, uv + U_PSCustom0.zw).rgb, 2.2f), float3(0.2126f,0.7152f,0.0722f));
	lum += dot(pow(S_2DExt0.Sample(S_2DExt0SamplerState, uv + U_PSCustom1.xy).rgb, 2.2f), float3(0.2126f,0.7152f,0.0722f));
	lum += dot(pow(S_2DExt0.Sample(S_2DExt0SamplerState, uv + U_PSCustom1.zw).rgb, 2.2f), float3(0.2126f,0.7152f,0.0722f));
	
	lum = min(lum * 0.25f, 64.0f);
  
	psOut.o_Color = float4(lum, lum, lum, lum);

	return psOut;
}
