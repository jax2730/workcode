struct ITSC
{
	float3 p;
	float dist;
	float3 n;
    float2 uv;
};

ITSC raycylh(float3 ro, float3 rd, float3 c, float r)
{
	ITSC i;
	i.dist = 1e38;
	float3 e = ro - c;
	float a = dot(rd.xy, rd.xy);
	float b = 2.0 * dot(e.xy, rd.xy);
	float cc = dot(e.xy, e.xy) - r;
	float f = b * b - 4.0 * a * cc;
	if(f > 0.0)
	{
		f = sqrt(f);
		float t = (-b + f) / (2.0 * a);
		
		if(t > 0.001)
		{
			i.dist = t;
			i.p = e + rd * t;
			i.n = -float3(normalize(i.p.xy), 0.0);
		}
	}
	return i;
}

void tPlane(inout ITSC hit, float3 ro, float3 rd, float3 o, float3 n, float3 tg, float2 si)
{
    float2 uv;
    ro -= o;
    float t = -dot(ro, n) / dot(rd, n);
    if(t < 0.0)
        return;
    float3 its = ro + rd * t;
    uv.x = dot(its, tg);
    uv.y = dot(its, cross(tg, n));
    if(abs(uv.x) > si.x || abs(uv.y) > si.y)
        return;
    
    //if(t < hit.dist)
    {
        hit.dist = t;
        hit.uv = uv;
    }
    return;
}

float hsh(float x)
{
    return frac(sin(x * 297.9712) * 90872.2961);
}

float2 rotate(float2 p, float a)
{
	return float2(p.x * cos(a) - p.y * sin(a), p.x * sin(a) + p.y * cos(a));
}

float nse3d(float3 x)
{
    float3 p = floor(x);
    float3 f = frac(x);
	f = f * f * (3.0 - 2.0 * f);
	float2 uv = (p.xy + float2(37.0, 17.0) * p.z) + f.xy;
	float4 uvlod = float4((uv+.5)/256.,0.0,0.0);
	//float2 rg = tex2D(S_2DExt0, uv).yx;
	float2 rg = S_2DExt0.SampleLevel(S_2DExt0SamplerState, uvlod.xy, 0.).yx;
	
	return lerp(rg.x, rg.y, f.z);
}

float fbm(float3 p)
{
    //float3 q = p;
    //q.xy = rotate(p.xy, U_PSCustom0.x);
    
    p += (nse3d(p * 3.0) - 0.5) * 0.3;
    
    //float v = nse3d(p) * 0.5 + nse3d(p * 2.0) * 0.25 + nse3d(p * 4.0) * 0.125 + nse3d(p * 8.0) * 0.0625;
    
    //p.y += U_PSCustom0.x * 0.2;
    
    float mtn = U_PSCustom0.x * _cloudSpeed;
    
    float v = 0.0;
    float fq = 1.0, am = 0.5;
	[unroll(10)]
    for(int i = 0; i < STP; i++)
    {
        v += nse3d(p * fq + mtn * fq) * am;
        fq *= 2.0;
        am *= 0.5;
    }
    return v;
}

float density(float3 p)
{
    float2 pol = float2(atan2(p.x, p.y), length(p.yx));
    
    float v = fbm(p);
    
    float fo = (pol.y - 1.5);//(densA + densB) * 0.5);
    //fo *= (densB - densA);
    v *= exp(fo * fo * -5.0);
    
    float edg = 0.3;
    return smoothstep(edg, edg + 0.1, v);
}

float hash1(float p)
{
	return frac(sin(p * 172.435) * 29572.683) - 0.5;
}

float ns(float p)
{
	float fr = frac(p);
	float fl = floor(p);
	return lerp(hash1(fl), hash1(fl + 1.0), fr);
}

float fbm(float p)
{
	return (ns(p) * 0.4 + ns(p * 2.0 - 10.0) * 0.125 + ns(p * 8.0 + 10.0) * 0.025);
}

float fbmd(float p)
{
	float h = 0.01;
	return atan2(h, fbm(p + h) - fbm(p - h));
}

float3 HSVConvertToRGB(float3 hsv)
{
	float R,G,B;
	//float3 rgb;
	if( hsv.y == 0 )
	{
		R=G=B=hsv.z;
	}
	else
	{
		hsv.x = hsv.x/60.0; 
		int i = (int)hsv.x;
		float f = hsv.x - (float)i;
		float a = hsv.z * ( 1 - hsv.y );
		float b = hsv.z * ( 1 - hsv.y * f );
		float c = hsv.z * ( 1 - hsv.y * (1 - f ) );
		switch(i)
		{
			case 0: R = hsv.z; G = c; B = a;
				break;
			case 1: R = b; G = hsv.z; B = a; 
				break;
			case 2: R = a; G = hsv.z; B = c; 
				break;
			case 3: R = a; G = b; B = hsv.z; 
				break;
			case 4: R = c; G = a; B = hsv.z; 
				break;
			default: R = hsv.z; G = a; B = b; 
				break;
		}
	}

	return float3(R,G,B);
}
