		float4 ComputeRipple(float2 uv, float t)
		{
			//波纹贴图采样，并把采样的高度值扩展到-1到1。
			float4 ripple =  S_Ripple.Sample(S_RippleSamplerState, uv);

			//获取波纹的时间,从A通道获取不同的波纹时间,
			float dropFrac = frac(ripple.a + t);
			//把时间限制在R通道内

			//做淡出处理
			float dropFactor = 1.0f-saturate( dropFrac);
			//计算最终的高度，用一个sin计算出随时间的振幅	 
			float final = dropFactor* sin(clamp(timeFrac * 9.0, 0.0, 4.0) * 3.1415926);
			return float4(ripple.yz * final, 1.0, (final ) * 0.2);
		}

float3 mod289(float3 x)
{
    return x - floor(x / 289.0) * 289.0;
}

float2 mod289(float2 x)
{
    return x - floor(x / 289.0) * 289.0;
}

float3 permute(float3 x)
{
    return mod289((x * 34.0 + 1.0) * x);
}

float3 taylorInvSqrt(float3 r)
{
    return 1.79284291400159 - 0.85373472095314 * r;
}

float snoise(float2 v)
{
    const float4 C = float4( 0.211324865405187,  // (3.0-sqrt(3.0))/6.0
                             0.366025403784439,  // 0.5*(sqrt(3.0)-1.0)
                            -0.577350269189626,  // -1.0 + 2.0 * C.x
                             0.024390243902439); // 1.0 / 41.0
    // First corner
    float2 i  = floor(v + dot(v, C.yy));
    float2 x0 = v -   i + dot(i, C.xx);

    // Other corners
    float2 i1;
    i1.x = step(x0.y, x0.x);
    i1.y = 1.0 - i1.x;

    // x1 = x0 - i1  + 1.0 * C.xx;
    // x2 = x0 - 1.0 + 2.0 * C.xx;
    float2 x1 = x0 + C.xx - i1;
    float2 x2 = x0 + C.zz;

    // Permutations
    i = mod289(i); // Avoid truncation effects in permutation
    float3 p =
      permute(permute(i.y + float3(0.0, i1.y, 1.0))
                    + i.x + float3(0.0, i1.x, 1.0));

    float3 m = max(0.5 - float3(dot(x0, x0), dot(x1, x1), dot(x2, x2)), 0.0);
    m = m * m;
    m = m * m;

    // Gradients: 41 points uniformly over a line, mapped onto a diamond.
    // The ring size 17*17 = 289 is close to a multiple of 41 (41*7 = 287)
    float3 x = 2.0 * frac(p * C.www) - 1.0;
    float3 h = abs(x) - 0.5;
    float3 ox = floor(x + 0.5);
    float3 a0 = x - ox;

    // Normalise gradients implicitly by scaling m
    m *= taylorInvSqrt(a0 * a0 + h * h);

    // Compute final noise value at P
    float3 g;
    g.x = a0.x * x0.x + h.x * x0.y;
    g.y = a0.y * x1.x + h.y * x1.y;
    g.z = a0.z * x2.x + h.z * x2.y;
    return 130.0 * dot(m, g);
}

float3 snoise_grad(float2 v)
{
    const float4 C = float4( 0.211324865405187,  // (3.0-sqrt(3.0))/6.0
                             0.366025403784439,  // 0.5*(sqrt(3.0)-1.0)
                            -0.577350269189626,  // -1.0 + 2.0 * C.x
                             0.024390243902439); // 1.0 / 41.0
    // First corner
    float2 i  = floor(v + dot(v, C.yy));
    float2 x0 = v -   i + dot(i, C.xx);

    // Other corners
    float2 i1;
    i1.x = step(x0.y, x0.x);
    i1.y = 1.0 - i1.x;

    // x1 = x0 - i1  + 1.0 * C.xx;
    // x2 = x0 - 1.0 + 2.0 * C.xx;
    float2 x1 = x0 + C.xx - i1;
    float2 x2 = x0 + C.zz;

    // Permutations
    i = mod289(i); // Avoid truncation effects in permutation
    float3 p =
      permute(permute(i.y + float3(0.0, i1.y, 1.0))
                    + i.x + float3(0.0, i1.x, 1.0));

    float3 m = max(0.5 - float3(dot(x0, x0), dot(x1, x1), dot(x2, x2)), 0.0);
    float3 m2 = m * m;
    float3 m3 = m2 * m;
    float3 m4 = m2 * m2;

    // Gradients: 41 points uniformly over a line, mapped onto a diamond.
    // The ring size 17*17 = 289 is close to a multiple of 41 (41*7 = 287)
    float3 x = 2.0 * frac(p * C.www) - 1.0;
    float3 h = abs(x) - 0.5;
    float3 ox = floor(x + 0.5);
    float3 a0 = x - ox;

    // Normalise gradients
    float3 norm = taylorInvSqrt(a0 * a0 + h * h);
    float2 g0 = float2(a0.x, h.x) * norm.x;
    float2 g1 = float2(a0.y, h.y) * norm.y;
    float2 g2 = float2(a0.z, h.z) * norm.z;

    // Compute noise and gradient at P
    float2 grad =
      -6.0 * m3.x * x0 * dot(x0, g0) + m4.x * g0 +
      -6.0 * m3.y * x1 * dot(x1, g1) + m4.y * g1 +
      -6.0 * m3.z * x2 * dot(x2, g2) + m4.z * g2;
    float3 px = float3(dot(x0, g0), dot(x1, g1), dot(x2, g2));
    return 130.0 * float3(grad, dot(m4, px));
}

float3 fresnelSchlick(float cosTheta, float3 F0)
{
	return F0 + (1.0-F0) * pow(1.0-cosTheta, 5.0);
}

float distributionGGX(float3 N, float3 H, float roughness)
{
	float a = roughness * roughness;
	float a2 = a * a;
	float NdotH = max(dot(N, H), 0.0);
	float NdotH2 = NdotH * NdotH;
	float nom = a2;
	float denom = (NdotH2 * (a2 - 1.0) + 1.0);
	denom = 3.141592654 * denom * denom;
	denom = max(denom, 0.00001);
	return nom / denom;
}

float geometrySchlickGGX(float NdotV, float roughness)
{
	float r = roughness + 1.0;
	float k = (r * r) / 8.0;
	
	float nom = NdotV;
	float denom = NdotV * (1.0 - k) + k;
	
	return nom / denom;
}

float geometrySmith(float3 N, float3 V, float3 L, float roughness)
{
	float NdotV = max(dot(N, V), 0.0);
	float NdotL = max(dot(N, L), 0.0);
	float ggx2 = geometrySchlickGGX(NdotV, roughness);
	float ggx1 = geometrySchlickGGX(NdotL, roughness);
	
	return ggx2 * ggx1;
}

float3 envBRDFApprox(float3 specularColor, float roughness, float NoV)
{
	float4 c0 = float4(-1.0,	-0.0275,	-0.572,		0.022);
	float4 c1 = float4(1.0,		0.0425,		1.04,		-0.04);
	
	float4 r = roughness * c0 + c1;
	float a004 = min(r.x * r.x, exp2(-9.28 * NoV)) * r.x + r.y;
	float2 AB = float2(-1.04, 1.04) * a004 + r.zw;
	return specularColor * AB.x + AB.y;
}

float3 thinTranslucency(float3 N, float3 L, float3 transmittanceColor)
{
	float transmittance = saturate(dot(-N, L) + 0.2);
	float fresnel = (1.0 - transmittance) * (1.0 - transmittance);
	fresnel *= fresnel;
	
	return transmittance * (1.0 - fresnel) * transmittanceColor;
}

float m_random(float2 m_xy)
{
    return frac(sin(dot(m_xy, float2(12.98980045318603515625, 78.233001708984375))) * 43758.546875);
}

float2 m_rotate(float2 m_v, float m_cosa, float m_sina)
{
    return float2((m_cosa * m_v.x) - (m_sina * m_v.y), (m_sina * m_v.x) + (m_cosa * m_v.y));
}

float m_blend_weights(float x, float y) {
    x = sqrt(x * 0.5);
    float m_result = max(0.100000001490116119384765625 * x, ((10.0 * (x + y)) + 1.0) - (x + 10.0));
    return m_result;
}

float4 getMat(float2 uv, float2 uv_center, Texture2D tex, SamplerState state, float2 tillingScale) 
{
	float m_r = m_random(uv_center) * 3.1415927410125732421875;
	float rotateOffset = tillingScale.y;
	float m_rand = m_r * rotateOffset;
	float2 m_rot = float2(cos(m_rand), sin(m_rand));
	float2 matUV = m_rotate(uv, m_rot.x, m_rot.y) * tillingScale.x;
	half4 decalColor2 = tex.Sample(state, matUV);
	return decalColor2;
}

float4 mixTillingResult(float2 uv, Texture2D tex, Texture2D noiseTex, SamplerState state, float2 tillingValue) 
{
	float2 m_texel_pos_floor = floor(uv * 1000.0);
	float2 m_texel_pos = uv * 1000.0;
	float4 m_mirror = float4(frac(m_texel_pos_floor * 0.5) * 2.0, 1.0, 1.0);
	m_mirror.zw = 1.0 - m_mirror.xy;
	float2 m_weights1 = clamp(m_texel_pos - m_texel_pos_floor, float2(0.0, 0.0), float2(1.0, 1.0));


	m_weights1 = lerp(m_weights1, float2(1.0, 1.0) - m_weights1, m_mirror.xy);
	float2 m_weights0 = float2(1.0, 1.0) - m_weights1;

	half4 decalColorR = getMat(uv, m_texel_pos_floor + m_mirror.xy, tex, state, tillingValue);
	half4 decalColorG = getMat(uv, m_texel_pos_floor + m_mirror.xw, tex, state, tillingValue);
	half4 decalColorB = getMat(uv, m_texel_pos_floor + m_mirror.zy, tex, state, tillingValue);
	half4 decalColorA = getMat(uv, m_texel_pos_floor + m_mirror.zw, tex, state, tillingValue);

	half4 m_noise3 = noiseTex.Sample(state, uv * tillingValue.x);
	
	float uvR = m_weights0.x * m_weights0.y;
	float noiseR = m_noise3;
	float blendR = m_blend_weights(uvR, noiseR);

	float uvG = m_weights0.x * m_weights1.y;
	float noiseG = m_noise3;
	float blendG = m_blend_weights(uvG, noiseG);

	float uvB = m_weights1.x * m_weights0.y;
	float noiseB = m_noise3;
	float blendB = m_blend_weights(uvB, noiseB);

	float uvA = m_weights1.x * m_weights1.y;
	float noiseA = m_noise3;
	float blendA = m_blend_weights(uvA, noiseA);

	float4 blendRes = float4(blendR, blendG, blendB, blendA);
	float blendRes_sum = ((blendRes.x + blendRes.y) + blendRes.z) + blendRes.w;
	float blendRes_inv = 1.0 / blendRes_sum;
	float4 m_albedo_height = ((((decalColorR * blendRes.x) + (decalColorG * blendRes.y)) + (decalColorB * blendRes.z)) + (decalColorA * blendRes.w)) * blendRes_inv;
	return m_albedo_height;
}

void applyGlobalWetEffect(inout float3 baseColor, inout float in_roughness, in float wetStrength)
{
	float3 hightLightColor = 1.0;
	float3 darkLightColor = 0.5;
	wetStrength = clamp(wetStrength, 0.001, 10.0);
	wetStrength /= 10.0;
	baseColor *= lerp(hightLightColor, darkLightColor, wetStrength);
	in_roughness = lerp(in_roughness, 0.001, wetStrength);
}

		half4 mask = S_Custom2.Sample(S_Custom2SamplerState, uv);
		half3 colorR = pow(U_PSGeneralRegister4.rgb, 2.2) * mask.r;
		half3 colorG = pow(U_PSGeneralRegister5.rgb, 2.2) * mask.g;
		half3 colorB = pow(U_PSGeneralRegister6.rgb, 2.2) * mask.b;
		float f = min(mask.r + mask.g + mask.b, 1.0);
		half3 colorF = colorR + colorG + colorB;
		baseMap = lerp(baseMap, baseMap * colorF, f);
	#endif
	
	half3 N = normalize(IN.i_Normal.xyz);
	
	#ifdef TWO_SIDES
		float frontValue = isFrontFacing ? 1.0f : -1.0f;
		N *= frontValue;
	#endif
	
	#ifdef USESHADOWMAP
		float4 shadowSpacePos[SHADOWSPLITE];
		ProjectToLightSpace(shadowSpacePos, IN.i_WPos.xyz, N, IN.i_ViewSpaceZ);
	#endif
	
	half3 V = normalize(U_PS_CameraPosition.xyz - IN.i_WPos.xyz);
	
    float maskRes = 1.0;
	float enableCube = 1.0;

	float4 tsPos = U_CoverColorParams;
	if(tsPos.w > 0.0)
	{
		float4 rotVec = float4(IN.i_WPos.xyz, 1.0);
		float4 qu = U_CoverColorRotation;
		rotVec.xyz = RotateWithQuaternion(rotVec.xyz, qu);
		float dist = length(rotVec.xyz - tsPos.xyz);
		float sphere_maskRes = smoothstep(tsPos.w - 1.0f, tsPos.w, dist);

		float rectSize = tsPos.w;
		float3 halfSize = rectSize * 0.5;
		float3 distToEdge = abs(rotVec.xyz - tsPos.xyz) - halfSize;

		float cube_maskRes = length(max(distToEdge, 0.0));
		cube_maskRes = clamp(cube_maskRes, 0.0, 1.0);

		maskRes = lerp(sphere_maskRes, cube_maskRes, enableCube);
	}	
	half4 bumpColor = 0.0;	
    half3 bump = half3(0.h, 0.h, 1.h);
    half3x3 tbn = (half3x3)0;

	#ifdef BUMP_MAP
		half3 nn = normalize(IN.i_Normal.xyz);
		half3 tn = normalize(IN.i_Tangent.xyz);
		half3 bn = normalize(IN.i_Binormal.xyz);

		tn = normalize(tn - nn * dot(tn, nn));
		bn = normalize(bn - nn * dot(bn, nn));

		tbn = half3x3(tn.x, bn.x, nn.x, tn.y, bn.y, nn.y, tn.z, bn.z, nn.z);
		
		bumpColor = S_Bump.Sample(S_BumpSamplerState, uv); 

		bump = bumpColor.rgb * 2.0 - 1.0;	
		bump *= half3(U_PSGeneralRegister2.w, U_PSGeneralRegister2.w, 1.0);
		bump.z = 1.h;
		
		#if defined(DETAIL_BUMP) || defined(DECAL1) || defined(DECAL2)
			half4 maskColor = S_Custom9.Sample(S_Custom9SamplerState, uv);
		#endif
		
		#ifdef DETAIL_BUMP
			float2 detailUV1 = float2(U_PSGeneralRegister13.x, U_PSGeneralRegister13.x);
			
			#ifdef UV_TILLING
				detailUV1 /= U_PSGeneralRegister2.x;
			#endif
			
			half4 detailBumpColor1 = S_Custom3.Sample(S_Custom3SamplerState, uv * detailUV1);
			bump.xy += (detailBumpColor1.xy - half2(0.5, 0.5)) * U_PSGeneralRegister2.z * maskColor.r;
		//bump.xy * (1.0 - maskColor.r) + 
		#endif

		#ifdef DECAL1	
		#ifdef MIXBLEND
			half4 decalBumpColor1 = mixTillingResult(uv, S_Custom5, S_Custom7, S_Custom5SamplerState, float2(U_PSGeneralRegister11.x, U_PSGeneralRegister18.y));
		#else
			half4 decalBumpColor1 = S_Custom5.Sample(S_Custom5SamplerState, uv * U_PSGeneralRegister11.x + U_PSGeneralRegister18.xy);
		#endif
			half3 decalBump1 = decalBumpColor1.rgb * 2.0 - 1.0;
			decalBump1 *= half3(U_PSGeneralRegister13.y, U_PSGeneralRegister13.y, 1.0);
			decalBump1.z = sqrt(1.0 - (decalBump1.x * decalBump1.x + decalBump1.y * decalBump1.y));
			
			float decalBumpMaskG = clamp(maskColor.g - (1.0 - U_PSGeneralRegister19.x), 0.0, 1.0);
			
			bump.xy += decalBump1.xy * decalBumpMaskG;
		// bump * (1.0 - decalBumpMaskG) + 
		#endif
		
		#ifdef DECAL2
		#ifdef MIXBLEND
			half4 decalBumpColor2 = mixTillingResult(uv, S_Custom8, S_Custom7, S_Custom8SamplerState, float2(U_PSGeneralRegister12.y, U_PSGeneralRegister18.w));
		#else
			half4 decalBumpColor2 = S_Custom8.Sample(S_Custom8SamplerState, uv * U_PSGeneralRegister12.y + U_PSGeneralRegister18.zw);
		#endif
			half3 decalBump2 = decalBumpColor2.rgb * 2.0 - 1.0;
			decalBump2 *= half3(U_PSGeneralRegister13.w, U_PSGeneralRegister13.w, 1.0);
			decalBump2.z = sqrt(1.0 - (decalBump2.x * decalBump2.x + decalBump2.y * decalBump2.y));
			
			float decalBumpMaskB = clamp(maskColor.b - (1.0 - U_PSGeneralRegister13.z), 0.0, 1.0);
			
			bump.xy += decalBump2.xy * decalBumpMaskB;
		//bump * (1.0 - decalBumpMaskB)
		#endif
		
		#ifdef MUSCLE_BUMP
			half4 secondBumpColor = S_Custom6.Sample(S_Custom6SamplerState, uv);
			float secondBumpAlpha = secondBumpColor.z;
			secondBumpColor.xy = secondBumpColor.xy * 2.0 - 1.0;
			//secondBumpColor.z = sqrt(1.0f - dot(secondBumpColor.xy, secondBumpColor.xy));
			//half3 secondbump = normalize(secondBumpColor.xyz);
		
			bump.xy += secondBumpColor.xy * U_PSGeneralRegister17.r * secondBumpAlpha;
			bump.xy *= 0.5;
		#endif
		
		#ifdef AGE
			half4 thridBumpColor = S_Custom8.Sample(S_Custom8SamplerState, uv);
			float thridBumpAlpha = thridBumpColor.z;
			thridBumpColor.xy = thridBumpColor.xy * 2.0 - 1.0;
			//thridBumpColor.z = sqrt(1.0f - dot(thridBumpColor.xy, thridBumpColor.xy));
			//half3 thridbump = normalize(thridBumpColor.xyz);
			bump.xy += thridBumpColor.xy * U_PSGeneralRegister17.g * thridBumpAlpha;
			bump.xy *= 0.5;
		#endif
		//当有涟漪的时候，减弱模型原本的法线贴图效果，或者直接砍掉法线贴图的计算
		#ifdef RIPPLE
			float s = clamp(dot(N, float3(0.0,1.0,0.0)), 0.0, 1.0) ;
			s = pow(s, 5.0);
			s *= (1.f - U_PSGeneralRegister30.x);
			bump.xy  *= (1.0 - s * 0.9f);
			double time = U_PS_Time.x;
			float time01 = U_PS_Time.y * 0.1591549; //U_PS_Time.y / PI / 2
			//涟漪整体的透明度使用sin函数进行0-1的变化，当透明度变为0的时候，坐标进行偏移，以减轻雨滴的重复感。单张贴图透明度为0的时候会没有涟漪，所以采样两次，交替显示
			float2 rUV = IN.i_WPos.xz * 0.25f;
			
			float rOffset = floor(U_PS_Time.x * 0.5f) * 0.1f;
			float4 ripple = ComputeRipple(rUV + rOffset, time01 * 3.0f);
			
			rUV = IN.i_WPos.xz * 0.25f + 0.25f;
			rOffset = floor(U_PS_Time.x * 0.5f + 0.25f) * 0.1f;
			float4 ripple2 = ComputeRipple(rUV + float2(rOffset, -rOffset), time01 * 3.0f);
			
			float rLerp1 =  sin(time01 * 3.1415);
			float rLerp2 =  1.0 - rLerp1;
			bump.xy += rLerp1 * ripple.xy * s;
			bump.xy += rLerp2 * ripple2.xy * s;
		#endif

	#endif
		
	float coverLerp = dot(N, U_PS_GravityDirection.xyz);	
	float agingLerp = coverLerp*U_PSGeneralRegister17.w;
	agingLerp = clamp(agingLerp, 0.0, 1.0);
	half3 agingColor = pow(U_GlobalAgingColour.rgb, 2.2);
	baseMap.rgb = lerp(baseMap.rgb, agingColor, agingLerp);	
	
	#ifdef COVER_WEATHER
		coverLerp += U_PSGeneralRegister16.z;
		coverLerp = clamp(coverLerp, 0.0, 1.0);
		
		half3 coverMap = U_WeatherParam0.xyz;
				
		#ifdef COVER_CUSTOM
			float2 coverUV = uv * U_PSGeneralRegister16.ww;
			half3 coverCustomMap = pow(S_Custom3.Sample(S_Custom3SamplerState, coverUV).rgb, 2.2);
			coverMap = lerp(coverCustomMap, coverMap, U_WeatherParam0.www * (1.0 - U_PSGeneralRegister30.x));
		#else
			coverLerp *= U_WeatherParam0.w * (1.0 - U_PSGeneralRegister30.x);
		#endif
	#endif
	
	half metallic = 1.0;
	half roughness = 0.0001;
	
	half ao = 1.0;
	#ifdef PBR_SIMPLE
		#ifdef	PBR_METAL	
			metallic = 1.0;
		#else
			metallic = 0.0;
		#endif
		
		#ifdef PBR_AO
			ao = bumpColor.b;
			roughness = U_PSGeneralRegister4.w;
		#endif
		
		#ifdef PBR_ROUGHNESS
			ao = 1.0;
			roughness = bumpColor.b;
		#endif
		
		#ifdef PBR_MIX
			metallic = bumpColor.b;
			roughness = U_PSGeneralRegister4.w;
			ao = 1.0;
		#endif
		
		roughness = clamp(roughness + U_PSGeneralRegister15.x, 0.0, 1.0);
	#else
		half3 mixMap = S_Specular.Sample(S_SpecularSamplerState, uv).rgb;
		metallic = mixMap.r;
		roughness = clamp(mixMap.g + U_PSGeneralRegister15.x, 0.0, 1.0);
		ao = mixMap.b;
	#endif
	
	#if defined(DECAL1) || defined(DECAL2)
		#ifndef BUMP_MAP
			half4 maskColor = S_Custom9.Sample(S_Custom9SamplerState, uv);
		#endif
	#endif
	float3 decalColorMix = baseMap;
	#ifdef DECAL1	
	#ifdef MINXBLEND
		half4 decalColor1 = mixTillingResult(uv, S_Custom4, S_Custom7, S_Custom4SamplerState, float2(U_PSGeneralRegister11.x, U_PSGeneralRegister18.x));
	#else
		half4 decalColor1 = S_Custom4.Sample(S_Custom4SamplerState, uv * U_PSGeneralRegister11.x + U_PSGeneralRegister18.xy);
	#endif
		#ifdef ALPHA_TEST
			clip(decalColor1.a - U_MaterialDiffuse.a);
		#endif	

		half3 decal1BaseMap = pow(decalColor1.rgb, 2.2);
		
		#ifdef TEXTURE_MASK
			half3 textureMask1 = pow(U_PSGeneralRegister14.rgb, 2.2) * maskColor.g;
			float f1 = min(maskColor.g, 1.0);
			decal1BaseMap = lerp(decal1BaseMap, decal1BaseMap * textureMask1, f1);
		#endif
		
		float decalColorMaskG = maskColor.g * U_PSGeneralRegister12.x * decalColor1.a;
		
		#ifdef DECALCOLORMIX1
			baseMap = baseMap * (1.0 - decalColorMaskG) + decal1BaseMap * decalColorMaskG;
		#else
			baseMap = baseMap * (1.0 - decalColorMaskG) + decal1BaseMap * decalColorMaskG * baseMap;
		#endif
		#ifdef	PBR_METAL	
			decalColor1.r = U_PSGeneralRegister14.w;
		#else
			decalColor1.r = 0.0;
		#endif
		metallic = metallic * (1.0 - decalColorMaskG) + decalColor1.r * decalColorMaskG;
		decalColor1.g = 0.0001;
		#ifdef PBR_ROUGHNESS
			decalColor1.b = 1.0;
			#ifndef BUMP_MAP
				decalColor1.g = 0.0;
			#else
				decalColor1.g = decalBumpColor1.b;
			#endif
		#endif
		roughness = roughness * (1.0 - decalColorMaskG) + clamp(decalColor1.g + U_PSGeneralRegister11.w, 0.0, 1.0) * decalColorMaskG;
		
		ao = ao * (1.0 - decalColorMaskG) + decalColor1.b * decalColorMaskG;
	#endif
	
	#ifdef DECAL2	
	#ifdef MIXBLEND
		half4 decalColor2 = mixTillingResult(uv, S_Custom6, S_Custom7, S_Custom6SamplerState, float2(U_PSGeneralRegister12.y, U_PSGeneralRegister18.z));
	#else
		half4 decalColor2 = S_Custom6.Sample(S_Custom6SamplerState, uv * U_PSGeneralRegister12.y + U_PSGeneralRegister18.zw);
	#endif
		#ifdef ALPHA_TEST
			clip(decalColor2.a - U_MaterialDiffuse.a);
		#endif

		half3 decal2BaseMap = pow(decalColor2.rgb , 2.2);
		
		#ifdef TEXTURE_MASK
			half3 textureMask2 = pow(U_PSGeneralRegister19.yzw, 2.2) * maskColor.b;
			float f2 = min(maskColor.b, 1.0);
			decal2BaseMap = lerp(decal2BaseMap, decal2BaseMap * textureMask2, f2);
		#endif
		
		float decalColorMaskB = maskColor.b * U_PSGeneralRegister12.w * decalColor2.a;
		
		#ifdef DECALCOLORMIX2
			baseMap =  baseMap * (1.0 - decalColorMaskB) + decal2BaseMap * decalColorMaskB;
		#else
			baseMap =  baseMap * (1.0 - decalColorMaskB) + decal2BaseMap * decalColorMaskB * baseMap;
		#endif
		
		#ifdef	PBR_METAL	
			decalColor2.r = U_PSGeneralRegister15.w;
		#else
			decalColor2.r = 0.0;
		#endif
		
		metallic = metallic * (1.0 - decalColorMaskB) + decalColor2.r * decalColorMaskB;
		decalColor2.g = 0.0001;
		#ifdef PBR_ROUGHNESS
			decalColor2.b = 1.0;
			#ifndef BUMP_MAP
				decalColor2.g = 0.0;
			#else
				decalColor2.g = decalBumpColor2.b;
			#endif
		#endif
		roughness = roughness * (1.0 - decalColorMaskB) + clamp(decalColor2.g + U_PSGeneralRegister12.z, 0.0, 1.0) * decalColorMaskB;
		
		ao = ao * (1.0 - decalColorMaskB) + decalColor2.b * decalColorMaskB;
	#endif

	#ifdef BUMP_MAP
		N = normalize(mul(tbn, bump));
	#endif

	#ifdef COVER_WEATHER
		float r = U_WeatherParam1.x;
		#ifdef COVER_CUSTOM
			r = U_PSGeneralRegister16.x;
		#endif
		roughness = lerp(roughness, r, coverLerp);
	#endif
	
	float wetFinal = U_PSGeneralRegister8.w + U_PBR_MainLightColor.w;
	half3 H = normalize(normalize(-U_MainLightDirection.xyz) + V);
	half NdotH = max(dot(N, H), 1e-4h);
	
	//roughness -= wetFinal;
	//有涟漪的时候，给材质的高光整体加0.4		
	#ifdef RIPPLE
		roughness = clamp(roughness - 0.4f * (1.f - U_PSGeneralRegister30.x), 0.0, 1.0);
	#endif
	roughness = max(roughness , 1e-4);
	
	#ifdef COVER_WEATHER
		float m = U_WeatherParam1.y;
		#ifdef COVER_CUSTOM
			m = U_PSGeneralRegister16.y;
		#endif
		// metallic = lerp(metallic, m, coverLerp);
	#endif
	applyGlobalWetEffect(baseMap.xyz, roughness, wetFinal);
	// metallic = lerp(metallic, 0.5, wetFinal);
	
	#ifdef COVER_WEATHER
		coverLerp *= maskRes;
		baseMap = lerp(baseMap, coverMap, coverLerp);
	#endif
	
	half3 grayColor = baseMap.rgb * half3(0.333, 0.333, 0.333);
	float gray = grayColor.r + grayColor.g + grayColor.b;
	
	grayColor.rgb =  half3(gray,gray,gray);
	
	baseMap.rgb = lerp(baseMap.rgb,grayColor,U_MaterialCustomParam.x);
    
    // 计算自发光
    half3 emission = 0.0h;
    #ifdef TEXTURE_FLOW
        uv *= U_PSGeneralRegister3.z;
        uv += frac(U_PS_Time.z * float2(U_PSGeneralRegister3.x, U_PSGeneralRegister3.y));
        half3 flowTextureColor = pow(S_Custom.Sample(S_CustomSamplerState, uv).rgb, 2.2);
        
        #ifdef TOD_TEXTURE_FLOW
            #ifdef TEXTURE_FLOW_WITHOUT_CHANNEL
                emission += flowTextureColor * U_PSGeneralRegister3.w * U_LightPower.x;
            #else
                emission += flowTextureColor * U_PSGeneralRegister3.w * U_LightPower.x * diffuseColor.a;
            #endif
        #else
            #ifdef TEXTURE_FLOW_WITHOUT_CHANNEL
                emission += lerp(flowTextureColor, flowTextureColor * U_PSGeneralRegister3.w, U_LightPower.x);
            #else
                emission += lerp(flowTextureColor, flowTextureColor * U_PSGeneralRegister3.w, U_LightPower.x) * diffuseColor.a;
            #endif
        #endif
    #endif
    
    #ifdef SEQUENCE_FRAME
        #ifndef SEQUENCE_FRAME_BASEMAP
            int index = floor(U_PS_Time.x * U_PSGeneralRegister10.z);
            int indexY = index / U_PSGeneralRegister10.w;
            int indexX = index - indexY * U_PSGeneralRegister10.w;
            float2 sfUV = float2(uv.x / U_PSGeneralRegister10.w, uv.y / U_PSGeneralRegister10.w);
            sfUV.x += indexX / U_PSGeneralRegister10.w;
            sfUV.y += indexY / U_PSGeneralRegister10.w;
            half3 sfTextureColor = pow(S_Custom.Sample(S_CustomSamplerState, sfUV).rgb, 2.2);
            emission += sfTextureColor * diffuseColor.a;
        #endif
    #endif
    
    #ifdef EMISSIVE
        half3 missiveColor = diffuseColor.rgb * U_PSGeneralRegister0.xyz * diffuseColor.a;
        half3 minMissiveColor = missiveColor * U_PSGeneralRegister0.w;
        #ifdef TOD_EMISSIVE
            half todFactor = U_LightPower.x;		
            #ifdef DYNAMIC_EMISSIVE
                half3 maxMissiveColor = missiveColor * U_PSGeneralRegister1.x;
                half factor = U_MaterialSpecular.w;	
                emission += lerp(minMissiveColor, maxMissiveColor, half3(factor, factor, factor)) * todFactor;
            #else
                emission += minMissiveColor * todFactor;
            #endif
        #else
            #ifdef DYNAMIC_EMISSIVE
                half3 maxMissiveColor = missiveColor * U_PSGeneralRegister1.x;
                half factor = U_MaterialSpecular.w;	
                emission += lerp(minMissiveColor, maxMissiveColor, half3(factor, factor, factor));
            #else
                emission += minMissiveColor;
            #endif
        #endif
    #endif

    float alphaaa = 1.0f;
    #ifdef SOFTBOARD
        float pixelDepth = 1.0f;
        float sceneDepth = 1.0f;
        float4 tempPos = 1.0f;
        tempPos = float4(IN.i_ClipPos.xy/IN.i_ClipPos.w, 0.0, 1.0);
        tempPos.x = tempPos.x * 0.5 + 0.5;
        tempPos.y = (1.0 - tempPos.y) * 0.5;
        sceneDepth = calcSceneDepth(IN.i_ClipPos);
        float4 vp;
        vp.x = dot(U_InvProjectMatrix[0], IN.i_ClipPos);
        vp.y = dot(U_InvProjectMatrix[1], IN.i_ClipPos);
        vp.z = dot(U_InvProjectMatrix[2], IN.i_ClipPos);
        vp.w = dot(U_InvProjectMatrix[3], IN.i_ClipPos);
        pixelDepth = vp.z / vp.w;
        alphaaa = saturate(abs(pixelDepth - sceneDepth) / IN.i_UV.z);
    #endif


    // #if defined(GBUFFER_PASS)
        float shadowFactor = 1.0;
        #ifdef USESHADOWMAP
            shadowFactor = calcShadowFactor(shadowSpacePos, IN.i_ViewSpaceZ);
            shadowFactor = min(shadowFactor, 1.0);
        #endif

        half3 encodedNormal = EncodeNormal(N);

        uint useRoleMainLight = 0;
        #ifdef ALONE_ROLE_LIGHT
            useRoleMainLight = 1;
        #endif

        uint receiveShadow = 0;
        #ifdef USESHADOWMAP
            receiveShadow = 1;
        #endif

        uint useEnvironmentMap = 0;
        #ifdef ENVIRONMENT_MAP
            useEnvironmentMap = 1;
        #endif

		uint usePointLight = 0;
		#ifdef CLUSTER_POINT_LIGHT
			usePointLight = 1;
		#endif

        half shadingModel = EncodeShadingModel(
            useRoleMainLight,
            receiveShadow,
            useEnvironmentMap,
			usePointLight,
            2);

        float alphaOut = alphaValue * alphaaa;

        psOut.g0.xyz = pow(baseMap.rgb, 1.0 / 2.2); // Albedo
        psOut.g0.w = alphaOut;

        psOut.g1.xyz = encodedNormal;   // Normal
        psOut.g1.w = shadowFactor;      // Shadow Factor

        psOut.g2.x = metallic;          // Metallic
        psOut.g2.y = roughness;         // Roughness
        psOut.g2.z = ao;                // AO
        psOut.g2.w = shadingModel;      // Shading Model

        psOut.sc.xyz = emission;        // Emission, 描边效果在延迟渲染中无法完美还原，此处为近似
        psOut.sc.w = 0.0h;

        if(U_MaterialCustomParam.w > 0)
        {
            psOut.sc.xyz = (1 - U_MaterialCustomParam.w) * psOut.g0.xyz + U_MaterialCustomParam.w * U_MaterialCustomParam.xyz;
            psOut.sc.a = 1.0;
            psOut.g2.w = 0; // unlit
        }

		#ifdef EditorHelp
	        psOut.sc.rgb = psOut.g0.rgb * 0.00001 + float3(1.0, 0.0, 0.0);
	        psOut.sc.a = 1.0;
			psOut.g2.w = 0; // unlit
		#endif // EditorHelp
	// #endif // GBUFFER_PASS
	return psOut;
}
