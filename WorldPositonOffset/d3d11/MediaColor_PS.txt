
struct PS_INPUT
{
	float4 o_Position	: SV_POSITION;
	float4 i_UV			: TEXCOORD0;
};

struct PS_OUTPUT
{
	float4 o_Color	: SV_TARGET;
};

Texture2D	S_2DExt0;
Texture2D	S_2DExt1;

SamplerState S_2DExt0SamplerState;
SamplerState S_2DExt1SamplerState;


float4 U_PSCustom0; //U_PSCustom0.x = OutputWidth, U_PSCustom0.y = scaledU, U_PSCustom0.z=scaledV

float3 sRGBToLinear( float3 Color ) 
{
	Color = max(6.10352e-5, Color); // minimum positive non-denormal (fixes black problem on DX11 AMD and NV)
	return Color > 0.04045 ? pow( Color * (1.0 / 1.055) + 0.0521327, 2.4 ) : Color * (1.0 / 12.92);
}

float3 YuvToRgb(float3 YUV)
{
	float4 ColorTransform[4];
	ColorTransform[0] = float4(1.164383f, 0.000000f, 1.596027f, 0.06274509803921568627f);
	ColorTransform[1] = float4(1.164383f, -0.391762f, -0.812968f, 0.5019607843137254902f);
	ColorTransform[2] = float4(1.164383f, 2.017232f, 0.000000f, 0.5019607843137254902f);
	ColorTransform[3] = float4(0.000000f, 0.000000f, 0.000000f, 1.000000f);
			
	float3x3 Coeff = float3x3(
		ColorTransform[0].xyz,
		ColorTransform[1].xyz,
		ColorTransform[2].xyz
	);

	// Offset in last column of matrix
	YUV -= float3(ColorTransform[0].w, ColorTransform[1].w, ColorTransform[2].w);
	float3 sRGB = mul(Coeff, YUV);

	return sRGB;//sRGBToLinear(sRGB);
}

// shader to convert a YUY2 texture to RGBA
PS_OUTPUT main(PS_INPUT IN)
{
	PS_OUTPUT psOut = (PS_OUTPUT)1.f;
	
	float2 uv = IN.i_UV.xy;
	
	float2 scaledUV = uv * float2(U_PSCustom0.y, U_PSCustom0.z);

	float3 YUV;
	YUV.yz = S_2DExt0.Sample(S_2DExt0SamplerState, scaledUV).yw; //B

	float4 YUY2P = S_2DExt1.Sample(S_2DExt1SamplerState, scaledUV); //P
	float Pos = scaledUV.x * U_PSCustom0.x;
	YUV.x = (floor(Pos % 2.0f) == 0) ? YUY2P.z : YUY2P.x;

	psOut.o_Color.xyz = YuvToRgb(YUV);
	psOut.o_Color.w = 1.0f;
	
	return psOut;
}
