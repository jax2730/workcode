	float SimpleRand(float2 uv) 
	{
    	return frac(sin(dot(uv, float2(12.9898, 78.233))) * 43758.5453);
	}
	

	float SphereThicknessMask(float3 camPos, float3 pixelPos, float3 center, float radius)
	{
		float3 dir = pixelPos - camPos;
		float dist = length(dir);
		if (dist <= 1e-4) return 0.0;
		dir /= dist;

		float3 oc = camPos - center;
		float b = dot(dir, oc);
		float c = dot(oc, oc) - radius * radius;
		float h = b * b - c;

		if (h < 0.0) return 0.0; // 未击中

		float sqrtH = sqrt(h);
		float tEnter = max(-b - sqrtH, 0.0);
		float tExit  = -b + sqrtH;

		if (tExit <= 0.0 || dist <= tEnter) return 0.0;

		float tEnd = min(tExit, dist);
		float lenInside = max(0.0, tEnd - tEnter);

		return saturate(lenInside * (0.5 / radius)); 
	}

	float3 DoExplosionBlur(float2 uv, float2 center, float strength, float Aberration, float RayLength, float JitterIntensity)
	{
		float dist = length(uv - center);
		float range = 0.001f;
		float centerMask = smoothstep(0.0, range, dist);
		strength *= centerMask;

		if (strength <= 1e-3) 
		{
			return S_BackbufferTex.Sample(S_BackbufferTexSamplerState, uv).rgb;
		}		

		float2 dir = uv - center;
		float len = length(dir);

		if (len < 1e-4) 
			dir = float2(0,0);
		else 
			dir /= len;

		float  noise = SimpleRand(uv); 
		float3 sum = 0;
		float  totalW = 0;
		float2 maxOffset = dir * RayLength * strength; 

		[unroll]
		for(int i=0; i < LOCAL_RADIALBLUR_SAMPLE_COUNT; i++)
		{
			float  t = float(i) / float(LOCAL_RADIALBLUR_SAMPLE_COUNT - 1);
			float  jitterT = t + (noise - 0.5) * JitterIntensity * t;
			float2 offsetBase = maxOffset * jitterT;

			float3 sColor;
			#ifdef LOCAL_RADIALBLUR_ABERRATION
				sColor.r = S_BackbufferTex.Sample(S_BackbufferTexSamplerState, uv - offsetBase).r;
				sColor.g = S_BackbufferTex.Sample(S_BackbufferTexSamplerState, uv - offsetBase * (1.0 + Aberration * 0.2)).g;
				sColor.b = S_BackbufferTex.Sample(S_BackbufferTexSamplerState, uv - offsetBase * (1.0 + Aberration)).b;
			#else
				sColor = S_BackbufferTex.Sample(S_BackbufferTexSamplerState, uv - offsetBase).rgb;
			#endif
			float lum = dot(sColor, float3(0.3, 0.59, 0.11));
			float w = 1.0 + lum * 2.0; 

			sum += sColor * w;
			totalW += w;
		}
		return sum / max(totalW, 1e-4);
	}


	float4 GetExplosisonBlurResult(float2 screen_uv, float3 InCenterPos, float2 InCenterUV, float sphereRadius, float3 worldPos)
	{
		float4 resultColor;

		//这里应该是粒子的世界位置
		float3 SphereCenter = InCenterPos;
		float2 CenterUV = InCenterUV;

		//这里应该是面片的半径
		float  SphereRadius = sphereRadius;

		//毛刺幅度
		float  JitterIntensity = U_PSGeneralRegister27.x ;

		// 控制中心亮斑衰减
		float CenterHotspot = U_PSGeneralRegister27.y;

		//中心清晰区域的大小, 这个是屏幕级别的
		//float  HollowRange = U_PSGeneralRegister27.z;

		//色散强度
		float  Aberration = U_PSGeneralRegister27.z;

		//射线长度
		float  RayLength = U_PSGeneralRegister27.w;
		float  BlurStrength = U_PSGeneralRegister28.x;	

		float4 baseColor = S_BackbufferTex.Sample(S_BackbufferTexSamplerState, screen_uv);
		//float  mask = SphereThicknessMask(U_PS_CameraPosition.xyz, worldPos, SphereCenter, SphereRadius);
		//衰减
		//mask = 1.0f;
		float  distance = length(worldPos - SphereCenter);
		float  mask = distance/SphereRadius;
		if (mask <= 1e-4 || mask >= 1.0f) 
		{
			resultColor = baseColor;
			return resultColor;
    	}
		
		
		// 3. 计算最终强度
		float curvePower = 1.0 / max(CenterHotspot, 0.01); 
		float finalStrength = BlurStrength * pow(mask, curvePower);

		// 4. 执行爆炸模糊
		float3 blurColor = DoExplosionBlur(screen_uv, CenterUV, finalStrength, Aberration, RayLength, JitterIntensity);

		// 5. 混合 (修复亮度断层)
		float3 blendedColor = lerp(baseColor.rgb, blurColor, saturate(finalStrength));
		float3 finalRGB = lerp(baseColor.rgb, blurColor, saturate(finalStrength));

		resultColor = float4(finalRGB, baseColor.a);

		return resultColor;
	}

float3 mod289(float3 x)
{
    return x - floor(x / 289.0) * 289.0;
}

float2 mod289(float2 x)
{
    return x - floor(x / 289.0) * 289.0;
}

float3 permute(float3 x)
{
    return mod289((x * 34.0 + 1.0) * x);
}

float3 taylorInvSqrt(float3 r)
{
    return 1.79284291400159 - 0.85373472095314 * r;
}

float snoise(float2 v)
{
    const float4 C = float4( 0.211324865405187,  // (3.0-sqrt(3.0))/6.0
                             0.366025403784439,  // 0.5*(sqrt(3.0)-1.0)
                            -0.577350269189626,  // -1.0 + 2.0 * C.x
                             0.024390243902439); // 1.0 / 41.0
    // First corner
    float2 i  = floor(v + dot(v, C.yy));
    float2 x0 = v -   i + dot(i, C.xx);

    // Other corners
    float2 i1;
    i1.x = step(x0.y, x0.x);
    i1.y = 1.0 - i1.x;

    // x1 = x0 - i1  + 1.0 * C.xx;
    // x2 = x0 - 1.0 + 2.0 * C.xx;
    float2 x1 = x0 + C.xx - i1;
    float2 x2 = x0 + C.zz;

    // Permutations
    i = mod289(i); // Avoid truncation effects in permutation
    float3 p =
      permute(permute(i.y + float3(0.0, i1.y, 1.0))
                    + i.x + float3(0.0, i1.x, 1.0));

    float3 m = max(0.5 - float3(dot(x0, x0), dot(x1, x1), dot(x2, x2)), 0.0);
    m = m * m;
    m = m * m;

    // Gradients: 41 points uniformly over a line, mapped onto a diamond.
    // The ring size 17*17 = 289 is close to a multiple of 41 (41*7 = 287)
    float3 x = 2.0 * frac(p * C.www) - 1.0;
    float3 h = abs(x) - 0.5;
    float3 ox = floor(x + 0.5);
    float3 a0 = x - ox;

    // Normalise gradients implicitly by scaling m
    m *= taylorInvSqrt(a0 * a0 + h * h);

    // Compute final noise value at P
    float3 g;
    g.x = a0.x * x0.x + h.x * x0.y;
    g.y = a0.y * x1.x + h.y * x1.y;
    g.z = a0.z * x2.x + h.z * x2.y;
    return 130.0 * dot(m, g);
}

float3 snoise_grad(float2 v)
{
    const float4 C = float4( 0.211324865405187,  // (3.0-sqrt(3.0))/6.0
                             0.366025403784439,  // 0.5*(sqrt(3.0)-1.0)
                            -0.577350269189626,  // -1.0 + 2.0 * C.x
                             0.024390243902439); // 1.0 / 41.0
    // First corner
    float2 i  = floor(v + dot(v, C.yy));
    float2 x0 = v -   i + dot(i, C.xx);

    // Other corners
    float2 i1;
    i1.x = step(x0.y, x0.x);
    i1.y = 1.0 - i1.x;

    // x1 = x0 - i1  + 1.0 * C.xx;
    // x2 = x0 - 1.0 + 2.0 * C.xx;
    float2 x1 = x0 + C.xx - i1;
    float2 x2 = x0 + C.zz;

    // Permutations
    i = mod289(i); // Avoid truncation effects in permutation
    float3 p =
      permute(permute(i.y + float3(0.0, i1.y, 1.0))
                    + i.x + float3(0.0, i1.x, 1.0));

    float3 m = max(0.5 - float3(dot(x0, x0), dot(x1, x1), dot(x2, x2)), 0.0);
    float3 m2 = m * m;
    float3 m3 = m2 * m;
    float3 m4 = m2 * m2;

    // Gradients: 41 points uniformly over a line, mapped onto a diamond.
    // The ring size 17*17 = 289 is close to a multiple of 41 (41*7 = 287)
    float3 x = 2.0 * frac(p * C.www) - 1.0;
    float3 h = abs(x) - 0.5;
    float3 ox = floor(x + 0.5);
    float3 a0 = x - ox;

    // Normalise gradients
    float3 norm = taylorInvSqrt(a0 * a0 + h * h);
    float2 g0 = float2(a0.x, h.x) * norm.x;
    float2 g1 = float2(a0.y, h.y) * norm.y;
    float2 g2 = float2(a0.z, h.z) * norm.z;

    // Compute noise and gradient at P
    float2 grad =
      -6.0 * m3.x * x0 * dot(x0, g0) + m4.x * g0 +
      -6.0 * m3.y * x1 * dot(x1, g1) + m4.y * g1 +
      -6.0 * m3.z * x2 * dot(x2, g2) + m4.z * g2;
    float3 px = float3(dot(x0, g0), dot(x1, g1), dot(x2, g2));
    return 130.0 * float3(grad, dot(m4, px));
}

#ifdef CONE_UV	
	float2 CalConeUV(in float2 _uv)
	{
		float u = _uv.x;
		float v = _uv.y;
		float radians = u * 2.0 * 3.141592653;
		float cosU = cos(radians);
		float sinU = sin(radians);
	
		return float2(cosU * v * 0.5 + 0.5, sinU * v * 0.5 + 0.5);
	}
#endif


float4 cal_color(float4 color1, float4 color1_coefficient, float4 color2, float4 color2_coefficient)
{
	return color1 * color1_coefficient * color2 * color2_coefficient;
}


float3 rotate (float4 qvec, float3 v) 
{
	float3 uv, uuv;
	 
	uv = cross(qvec.xyz, v);
	uuv = cross(qvec.xyz, uv);
	uv *= (2.0 * qvec.w);
	uuv *= 2.0;
		
	return v + uv + uuv;
}

	half3 nn = normalize(IN.i_Normal.xyz);
	half3 tn = normalize(IN.i_Tangent.xyz);
	half3 bn = normalize(IN.i_Binormal.xyz);
	tn = normalize(tn - nn * dot(tn, nn));
	bn = normalize(bn - nn * dot(bn, nn));	
	half3x3 tbn = 
	{
		tn.x, bn.x, nn.x,
		tn.y, bn.y, nn.y,
		tn.z, bn.z, nn.z 
	};
	tbn = transpose(tbn);

	{
		float3 mainLightDir = U_MainLightDirection.xyz;
		mainLightDir = normalize(mul(tbn, mainLightDir));

		float hMap = (mainLightDir.x > 0.0f) ? (lightMap2.x) : (lightMap1.x);
		//判断并取Top/Bottom方向灯光
		float vMap = (mainLightDir.y > 0.0f) ? (lightMap2.y) : (lightMap1.y);

		//判断并取Front/Back方向灯光
		float dMap = (mainLightDir.z > 0.0f) ? (lightMap1.z) : (lightMap2.z);
		//float  weight = hMap * mainLightDir.x * mainLightDir.x + vMap * mainLightDir.y * mainLightDir.y + dMap * mainLightDir.z * mainLightDir.z;
		float  weight = hMap * abs(mainLightDir.x) + vMap * abs(mainLightDir.y) + dMap * abs(mainLightDir.z);
		weight = weight * U_PSGeneralRegister25.x;
		float3  mainLightColor = U_MainLightColor.rgb * weight;
		psOut.o_Color.rgb += mainLightColor;

{

		float2 dir = float2(0.5,0.5) - uv;

		float2 newLocalUV1 = uv - dir*0.01;
		float2 newLocalUV2 = uv + dir*0.01;

		newScreenUV1.y = U_PSCustom2.y + (U_PSCustom2.w - U_PSCustom2.y) * newLocalUV1.y;
		newScreenUV1.y = 1.0f - newScreenUV1.y;
		
		float2 newScreenUV2;
		newScreenUV2.x = lerp( U_PSCustom2.x , U_PSCustom2.z , newLocalUV2.x);
		newScreenUV2.y = U_PSCustom2.y + (U_PSCustom2.w - U_PSCustom2.y) * newLocalUV2.y;
		newScreenUV2.y = 1.0f - newScreenUV2.y;
			
		float3 result=S_BackbufferTex.Sample(S_BackbufferTexSamplerState, newScreenUV1 * viewOffset).rgb;
		result += S_BackbufferTex.Sample(S_BackbufferTexSamplerState, newScreenUV2 * viewOffset).rgb;	
		result /= 2;
		
		float originalAlpha = S_2DExt0.Sample(S_2DExt0SamplerState, in_UV).a;
		
		colb = float4(result,saturate(U_PSCustom3.x * dist));
		colb.a *= originalAlpha;
	#endif

	cola.rgb *= U_PSCustom1[0].x;
	cola.a *= U_PSCustom1[0].y;
	
	
	colb.rgb *= U_PSCustom1[0].z;
	colb.a *= U_PSCustom1[0].w;
	
	float alphaAMutilB = cola.a * colb.a;
	float alphaAAddB = cola.a + colb.a;
	
	float3 colorAMutilB = cola.rgb * colb.rgb;
	float3 colorAAddB = cola.rgb + colb.rgb;
	
	float3 colorFinal = lerp(colorAMutilB, colorAAddB, U_PSCustom1[1].z);
	float alphaFinal = lerp(alphaAMutilB, alphaAAddB, U_PSCustom1[1].w);
	
	colorFinal += U_PSCustom1[1].x;
	alphaFinal += U_PSCustom1[1].y;
	
	alphaFinal = clamp(alphaFinal, 0.0, 1.0f);
	
	#ifndef SCREENDISTORT
		#ifdef EFFECT_TRAIL
			float len = sqrt(dot(IN.i_EyeDir.xyz,IN.i_EyeDir.xyz));
			len = len - IN.i_AlphaDis.x;
			float alphaLen = IN.i_AlphaDis.y - IN.i_AlphaDis.x;
			alphaLen = max(alphaLen,0.00001f);

			len = clamp(len,0.0,1.0f);
			alphaFinal *= len;	
		#endif
	#endif	
	
	#ifdef DOUBLETEX
	{
		float4 colc;
		colc.rgb = colorFinal;
		colc.a = alphaFinal;
		
		#ifdef DOUBLETEX_ENABLE_POLAR
			float2 double_tex_uv = InPolarUV2;
		#else
			float2 double_tex_uv = in_UV2;
		#endif

		float offset_v = U_PSCustom10[data_index].y;

		double_tex_uv.y += offset_v;
		float4 cold = S_2DExt1.Sample(S_2DExt1SamplerState, double_tex_uv);	
		cold = pow(cold, U_PSCustom11.y);

		colc.rgb *= U_PSCustom1[2].x;
		colc.a *= U_PSCustom1[2].y;
		
		cold.rgb *= U_PSCustom1[2].z;
		cold.a *= U_PSCustom1[2].w;
		
		float alphaCMutilD = colc.a * cold.a;
		float alphaCAddD = colc.a + cold.a;
		
		float3 colorCMutilD = colc.rgb * cold.rgb;
		float3 colorCAddD = colc.rgb + cold.rgb;
		
		colorFinal = lerp(colorCMutilD, colorCAddD, U_PSCustom1[3].z);
		alphaFinal = lerp(alphaCMutilD, alphaCAddD, U_PSCustom1[3].w);
		
		colorFinal += U_PSCustom1[3].x;
		alphaFinal += U_PSCustom1[3].y;
	}
	#endif
	psOut.o_Color =  float4(colorFinal, alphaFinal);
}

#ifdef DEPTH_FADE
{
	float2 depth_uv;
	depth_uv.x = ((IN.i_PosInClip.x / IN.i_PosInClip.w) + 1) * 0.5f;
	depth_uv.y = 1 - ((IN.i_PosInClip.y / IN.i_PosInClip.w) + 1) * 0.5f;

	float depth_val = S_Depth.Sample(S_DepthSamplerState, depth_uv).r;

	float4 target_ndc_pos = float4(IN.i_PosInClip.x / IN.i_PosInClip.w, IN.i_PosInClip.y / IN.i_PosInClip.w, depth_val, 1.0f);

	float4 target_view_pos; 
	//target_view_pos.x = dot(U_InvProjectMatrix[0], target_ndc_pos);
	//target_view_pos.y = dot(U_InvProjectMatrix[1], target_ndc_pos);
	target_view_pos.z = dot(U_InvProjectMatrix[2], target_ndc_pos);
	target_view_pos.w = dot(U_InvProjectMatrix[3], target_ndc_pos);
	target_view_pos.z = target_view_pos.z / target_view_pos.w;

	float4 view_pos;
	view_pos.z = dot(U_InvProjectMatrix[2], IN.i_PosInClip);
	view_pos.w = dot(U_InvProjectMatrix[3], IN.i_PosInClip);
	view_pos.z = view_pos.z / view_pos.w;

	float distance = U_PSGeneralRegister20.x;
	

	float factor = smoothstep(0.0f, distance, abs(view_pos.z - target_view_pos.z));
	
	#ifdef DEPTH_FADE_REVERSE
		factor = 1.0f - factor;
	#endif
	
	#ifdef DEPTH_FADE_OPACITY
		float opacity = U_PSGeneralRegister20.y;
		psOut.o_Color.a = psOut.o_Color.a * factor + psOut.o_Color.a*opacity ;
	#else 
		psOut.o_Color.a = psOut.o_Color.a * factor;
	#endif
	#ifdef DEPTH_FADE_COLOR
		psOut.o_Color.rgb += U_PSGeneralRegister21.xyz * factor * U_PSGeneralRegister20.z;
		//psOut.o_Color.rgba *= U_PSGeneralRegister21.xyzw;
		//psOut.o_Color.rgba *= U_PSGeneralRegister20.z * factor;
	#endif
}
#endif
	psOut.o_Color.a = min(1.0, psOut.o_Color.a);
 	float overdrawlerp = ceil(U_PS_Time.w);	
	psOut.o_Color = psOut.o_Color * (1.0 - overdrawlerp) + float4(U_PS_Time.w * overdrawlerp, U_PS_Time.w * overdrawlerp, U_PS_Time.w * overdrawlerp, overdrawlerp);
	return psOut;
