
#include <Common_Uniform_PS.inl> 

float4 U_PSCustom0;

struct PS_INPUT
{
    float4 i_Position	: SV_POSITION;
	float4 i_UV			: TEXCOORD0;
#ifdef PARABOLOID_MAPPING
	float	i_ClipDepth			: TEXCOORD12;
#endif
};

struct PS_OUTPUT
{
	float4 o_Color	: SV_TARGET;
};

PS_OUTPUT main(PS_INPUT IN)
{
#ifdef PARABOLOID_MAPPING
	clip(IN.i_ClipDepth);
#endif
	PS_OUTPUT psOut = (PS_OUTPUT)1.f;
	float scale = saturate(1.0f - length(IN.i_UV.xy)); 
	psOut.o_Color.rgb = U_PSCustom0.rgb;
	psOut.o_Color.a = U_PSCustom0.a * scale;
	return psOut;
}
