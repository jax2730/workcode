//HolographicEffect_PS
#include <Common_Uniform_PS.inl>

Texture2D S_Diffuse;
SamplerState S_DiffuseSamplerState;
Texture2D S_Custom; //噪声图像
SamplerState S_CustomSamplerState;
Texture2D S_Custom2;
SamplerState S_Custom2SamplerState;

float4	U_MaterialCustomParam;
float4	U_MaterialCustomParam1;

float4 U_PSGeneralRegister0;    //baseColor 噪声图像的颜色
float4 U_PSGeneralRegister1;    //fresnelBaseColor 菲涅尔效果的颜色
float4 U_PSGeneralRegister2;    //xy:tilingVector 噪声图像位移方向
                                //z:tilingSpeed 位移速度
                                //w:fresnelPower 菲涅尔次方
float4 U_PSGeneralRegister3;    //x:alpha mesh透明度
                                //y:fresnelBrightness 菲涅尔亮度
                                //z:brightness 效果亮度
                                //w:fresnelStep 菲涅尔效果阶梯数
float4 U_PSGeneralRegister4;    //xy:tiling 噪声图像位移
                                //zw:noiseTextScaling 噪声图像缩放
float4 U_PSGeneralRegister5;    //x:flashingSpeed 闪烁速度
                                //y:fresnelEdgeWidth 描边宽度
                                //z:progress 遮罩百分比
                                //w:maskGradientCritical 遮罩渐变临界范围
#ifdef PROGRESS_MASK
float4 U_PSGeneralRegister6;    //maskColor 遮罩颜色
float4 U_PSGeneralRegister7;    //x:parent worldposition.y
                                //y:holoGradientCritical 全息渐变临界范围
                                //z:轴向 0 y轴 1 x轴 2 z轴
                                //w:方向 0正向 1反向

float4 U_PSGeneralRegister8;    //x:是否使用双端平滑过渡  0:单端 1:双端
                                //y:明显分界的截断值，0.0-1.0之间
                                //z:是否让maskColor受fresnelPower影响 0:否 1:是
                                //w:预留
float4 U_PSGeneralRegister9;    //xyzw:fresnelMaskColor
                               
#endif

float randomRange(float2 seed, float min, float max)
{
    float randomno = frac(sin(dot(seed, float2(12.9898, 78.233))) * 43758.5453);
    float Out = lerp(min, max, randomno);
    return Out;
}


struct PS_INPUT
{
    float4	i_Pos    			: SV_POSITION;
#ifndef SHADOWPASS
	float4 	i_UV				: TEXCOORD0;
	float4	i_Normal			: TEXCOORD1;
	float4	i_WPos				: TEXCOORD2;
	float4	i_PosInClip			: TEXCOORD3;
    float4  i_PosInModel        : TEXCOORD4;
    float4  i_BoundBoxRange     : TEXCOORD14;
#endif
#ifdef PARABOLOID_MAPPING
	float	i_ClipDepth			: TEXCOORD12;
#endif
};

struct PS_OUTPUT
{
#ifndef SHADOWPASS
	float4 o_Color	: SV_TARGET;
#endif
};

PS_OUTPUT main(PS_INPUT IN)
{
	PS_OUTPUT psOut = (PS_OUTPUT)1.f;
    
#ifdef PARABOLOID_MAPPING
	clip(IN.i_ClipDepth);
#endif

#ifndef SHADOWPASS
    float4 baseColor = U_PSGeneralRegister0;
    float4 fresnelBaseColor = U_PSGeneralRegister1;
    float2 tilingVector = U_PSGeneralRegister2.xy;
    float tilingSpeed = U_PSGeneralRegister2.z;
    float fresnelPower = U_PSGeneralRegister2.w;
    float fresnelBrightness = U_PSGeneralRegister3.y;
    float brightness = U_PSGeneralRegister3.z;
    float fresnelStep = U_PSGeneralRegister3.w;
    float fresnelEdgeWidth = U_PSGeneralRegister5.y;
    //float2 tiling = U_PSGeneralRegister4.xy;
    float2 noiseTextScaling = pow(U_PSGeneralRegister4.zw, 0.454545);
    float alpha = U_PSGeneralRegister3.x;
    float flashingSpeed = U_PSGeneralRegister5.x;
#ifdef PROGRESS_MASK
    float progress = clamp(U_PSGeneralRegister5.z,0.0f,1.0f);
    float maskGradientCritical = clamp(U_PSGeneralRegister5.w,0.0f,1.0f);
    float holoGradientCritical = min(clamp(U_PSGeneralRegister7.y, 0.0f, 1.0f), 0.2f);
    bool doubleSmooth = (U_PSGeneralRegister8.x > 0.5f);
    float cutOffValue = clamp(U_PSGeneralRegister8.y,0.0f,1.0f);
    float bEnablefresneControl = U_PSGeneralRegister8.z;
    float4 fresnelMaskColor = U_PSGeneralRegister9;
#endif

    #ifdef TEXTURE_MASK
        float4 color = S_Diffuse.Sample(S_DiffuseSamplerState, IN.i_UV.xy);
    #else
        float4 color = baseColor;
        alpha = 1.0f;
    #endif

    // 菲涅尔效果
    fresnelPower = pow(fresnelPower, 2.2);
    float3 w_viewDir = normalize(U_PS_CameraPosition.xyz - IN.i_WPos.xyz);
    float3 w_normal = normalize(IN.i_Normal.xyz);
    float fresnel = pow(1.0 - dot(w_normal, w_viewDir), fresnelPower);
    //平滑过渡带
    float edgeStart = 1.0 - fresnelEdgeWidth * 2.0;
    float edgeEnd   = 1.0 - fresnelEdgeWidth;
    float fresnelEdgeBlend = smoothstep(edgeStart,edgeEnd,fresnel);    
   
    float4 fresnelStepColor = (float4)0.0;
    float4 fresnelMaskStepColor = (float4)0.0;
#ifdef ENABLESTEP
    fresnelStepColor.rgb = fresnelBaseColor.rgb * (floor(fresnel * fresnelStep) / fresnelStep) * fresnelBrightness;
    fresnelStepColor.a = fresnelBaseColor.a * (floor(fresnel * fresnelStep) / fresnelStep);
    fresnelStepColor.rgb += fresnelBaseColor.rgb * fresnelEdgeBlend * fresnelBrightness;
    fresnelStepColor.a = lerp(fresnelStepColor.a,fresnelBaseColor.a,fresnelEdgeBlend);
    #ifdef PROGRESS_MASK
        fresnelMaskStepColor.rgb = fresnelMaskColor.rgb * (floor(fresnel * fresnelStep) / fresnelStep) * fresnelBrightness;
        fresnelMaskStepColor.a = fresnelMaskColor.a * (floor(fresnel * fresnelStep) / fresnelStep);
        fresnelMaskStepColor.rgb += fresnelMaskColor.rgb * fresnelEdgeBlend * fresnelBrightness;
        fresnelMaskStepColor.a    = lerp(fresnelMaskStepColor.a, fresnelMaskColor.a, fresnelEdgeBlend);     
    #endif
#else
    fresnelStepColor.rgb = fresnelBaseColor.rgb * fresnel * fresnelBrightness;
    fresnelStepColor.a = fresnelBaseColor.a * fresnel;
    fresnelStepColor.rgb += fresnelBaseColor.rgb * fresnelEdgeBlend * fresnelBrightness;
    fresnelStepColor.a    = lerp(fresnelStepColor.a, fresnelBaseColor.a, fresnelEdgeBlend);
    #ifdef PROGRESS_MASK
        fresnelMaskStepColor.rgb = fresnelMaskColor.rgb * fresnel * fresnelBrightness;
        fresnelMaskStepColor.a   = fresnelMaskColor.a * fresnel;
        fresnelMaskStepColor.rgb += fresnelMaskColor.rgb * fresnelEdgeBlend * fresnelBrightness;
        fresnelMaskStepColor.a    = lerp(fresnelMaskStepColor.a, fresnelMaskColor.a, fresnelEdgeBlend);
    #endif
#endif

    float time = U_PS_Time.x;

    float2 screenTC = IN.i_PosInClip.xy / IN.i_PosInClip.w;
	screenTC.xy = 1.0 - (1.0 + screenTC.xy) * 0.5;

	float2 noiseTexture_uv = screenTC.xy * noiseTextScaling.xy /*+ tiling*/ + time * tilingSpeed * tilingVector;
	float4 noiseTexture = S_Custom.Sample(S_CustomSamplerState, noiseTexture_uv);
    
    #ifdef TEXTURE_MASK
    color.a = noiseTexture.r * alpha;
    noiseTexture.rgb = (baseColor.rgb * brightness + fresnelStepColor.rgb);
    #else
    color.a = noiseTexture.r * alpha * max(baseColor.a, fresnelStepColor.a * fresnel);
    noiseTexture.rgb = (baseColor.rgb * brightness + fresnelStepColor.rgb);
    #endif

    float randomNum = randomRange(float2(time, time), 0, 1) > (1.0 - flashingSpeed) ? 1.3 : 1.0;
    
    noiseTexture.rgb = noiseTexture.rgb * randomNum;

	float3 missiveColor = noiseTexture.rgb;
	float3 minMissiveColor = missiveColor * noiseTexture.w;
    #ifdef TEXTURE_MASK
    color.rgb = color.rgb + minMissiveColor.rgb;
    #else
    color.rgb = minMissiveColor.rgb;
    #endif

    psOut.o_Color = color;

#ifdef PROGRESS_MASK
    float4 maskColor = U_PSGeneralRegister6;
    #ifdef PROGRESS_TEX_MASK
    maskColor *= S_Custom2.Sample(S_Custom2SamplerState, noiseTexture_uv);
    #endif

    if(bEnablefresneControl > 0.5f)
    {
        maskColor.rgb = (maskColor.rgb * brightness + fresnelMaskStepColor.rgb);
        maskColor.rgb *= randomNum;
    }

    //maskColor.a *= U_PSGeneralRegister6.w;

    int axis = U_PSGeneralRegister7.z; // use for uniform static branch
    int showDirection = U_PSGeneralRegister7.w;
    float currentAxisLengthInModel = IN.i_PosInModel.y;
    if(axis == 0){
        currentAxisLengthInModel = IN.i_PosInModel.y;
    }else if(axis == 2){
        currentAxisLengthInModel = IN.i_PosInModel.z;
    }else if(axis == 1){
        currentAxisLengthInModel = IN.i_PosInModel.x;
    }
    currentAxisLengthInModel -= IN.i_BoundBoxRange.x;
    //currentAxisLengthInModel = abs(currentAxisLengthInModel);
    #ifdef EDITOR_MODE
        float curEntityHeight = U_PSGeneralRegister7.x;
        if(IN.i_BoundBoxRange.y - IN.i_BoundBoxRange.x > 0.0f)
        {
            curEntityHeight = IN.i_BoundBoxRange.y - IN.i_BoundBoxRange.x; //cur model height
        }
    #else
        float curEntityHeight = IN.i_BoundBoxRange.y - IN.i_BoundBoxRange.x; //cur model height
    #endif

    float texProgress = (currentAxisLengthInModel) / curEntityHeight; //当前像素的遮罩百分比
    if(showDirection == 0) //uniform static branch
    {

        //遮罩(MaskDownlimit - 1.0)
        //全息到遮罩(progress - maskDownlimit)
        //全息(holoDownlimit - progress)
        //透明到全息(transparentUplimit - holoDownlimit)
        //透明(0.0 - transparentUplimit)

        float maskDownlimit = progress + progress * maskGradientCritical; //由全息到遮罩的渐变宽度上限位置
        maskDownlimit = clamp(maskDownlimit, progress, progress + 0.15f);
        
        float holoDownlimit = progress - progress * holoGradientCritical; //全息效果显示的范围下限
        float transparentUplimit = progress - progress * holoGradientCritical * 2.0f; //全息到全透明的渐变宽度下限位置

        //全息到遮罩的插值系数        
        float holoToMaskRadio = abs(texProgress - progress) / abs(maskDownlimit - progress);
        //使用截断来实现明显边界
        float holoToMaskDiffCoe = clamp(holoToMaskRadio,0.0,cutOffValue); 

        if(progress >= 1.0f)
        {
            psOut.o_Color.a = 0.0f;
        }
        else
        {
            bool isColorInterp = (texProgress >= progress);

            if(isColorInterp)  
            {
                if(doubleSmooth)
                {
                    float t = smoothstep(progress,maskDownlimit,texProgress);
                    psOut.o_Color = lerp(psOut.o_Color,maskColor,t);
                }
                else
                {
                    float inRange = step(progress, texProgress) * step(texProgress, maskDownlimit); 
                    psOut.o_Color = lerp(maskColor, lerp(psOut.o_Color, maskColor, holoToMaskDiffCoe), inRange);
                }
            }
            else
            {
                float t = smoothstep(transparentUplimit,holoDownlimit,texProgress);
                float fadedAlpha = lerp(0.0f, psOut.o_Color.a, t);
                psOut.o_Color.a = lerp(1.0f, fadedAlpha, step(0.0f, holoGradientCritical));
            }
        }   
    }
    else
    {
        //透明(transparentDownlimit - 1.0)
        //全息到透明(holoUplimit - transparentDownlimit)
        //全息(holoDownlimit- holoUplimit)
        //遮罩到全息(progress-holoDownlimit)
        //遮罩(0.0-progress)

        float holoDownlimit = progress + progress * maskGradientCritical;
        holoDownlimit = clamp(holoDownlimit, progress, progress + 0.15f);

        float holoUplimit = holoDownlimit + progress * holoGradientCritical;
        float transparentDownlimit = holoDownlimit + progress * holoGradientCritical * 2.0f;
        
         //遮罩到全息的插值系数        
        float maskToHoloRadio = abs(texProgress - progress) / abs(holoDownlimit - progress);
        float masktoHoloDiffCoe = clamp(maskToHoloRadio,0.0,cutOffValue); 

        if(progress >= 1.0f)
        {
            psOut.o_Color = maskColor;
        }
        else
        {
            bool isColorInterp = (texProgress <= holoDownlimit);

            if (isColorInterp)
            {
                if(doubleSmooth)
                {
                    float t = smoothstep(progress, holoDownlimit, texProgress);
                    psOut.o_Color = lerp(maskColor,psOut.o_Color,t);
                }
                else
                {
                    float inRange = step(progress, texProgress) * step(texProgress, holoDownlimit); 
                    psOut.o_Color = lerp(maskColor, lerp(maskColor,psOut.o_Color,masktoHoloDiffCoe), inRange);
                }
            }
            else 
            {
                float t = smoothstep(holoUplimit, transparentDownlimit, texProgress);
                float fadedAlpha = lerp(psOut.o_Color.a, 0.0f, t);
                psOut.o_Color.a = lerp(1.0f,fadedAlpha, step(0.0f, holoGradientCritical));
            }
        }
    }
#endif

    psOut.o_Color.rgb = psOut.o_Color.rgb * U_MaterialCustomParam1.x;

    #ifdef EDITOR_MODE
        if(U_MaterialCustomParam.w > 0)
        {
            psOut.o_Color.rgb = U_MaterialCustomParam.xyz;
            psOut.o_Color.a = 1.0f;
        }
    #endif

#endif
    return psOut;
}
