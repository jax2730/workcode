{
    float4 o_Position	    : SV_POSITION;
	float4 o_UV				: TEXCOORD0;
	float4 o_ClipPos    	: TEXCOORD1;
	float4 o_FogColor		: TEXCOORD2;
    float4 o_WorldPos       : TEXCOORD4;
	#ifdef USE_SUN_BACK_COLOR
		float4 o_CustomParam	: TEXCOORD3;

	#endif

    float4 o_Color	: SV_TARGET;

float getSceneDepth(in float4 clipPos, in float2 uv)
{
	float4 tempPos = float4(uv, 0.0, 1.0);
	tempPos.z = S_Depth.Sample(S_DepthSamplerState, uv).r;
	
	tempPos.x = clipPos.x;
	tempPos.y = clipPos.y;
	
	float4 viewSpacePos;
	viewSpacePos.x = dot(U_InvProjectMatrix[0], tempPos);
	viewSpacePos.y = dot(U_InvProjectMatrix[1], tempPos);
	viewSpacePos.z = dot(U_InvProjectMatrix[2], tempPos);
	viewSpacePos.w = dot(U_InvProjectMatrix[3], tempPos);
	float depth = viewSpacePos.z / viewSpacePos.w;
	
	return depth;
}

//Dx only need to sample one times.
float calcSceneDepth(in float4 clipPos)
{	
	float2 screenTC = clipPos.xy / clipPos.w;
	screenTC.x = (screenTC.x + 1.0) * 0.5; //+ U_ViewParam.z;
	screenTC.y = 1.0 - (screenTC.y + 1.0) * 0.5; //+ U_ViewParam.w;
	screenTC *= U_ViewParam.xy;
	float depth = getSceneDepth(clipPos / clipPos.w, screenTC);
	
	return depth;
}


#ifdef USE_DISTORT
 
float3 mod289(float3 x)
{
    return x - floor(x / 289.0) * 289.0;
}

float2 mod289(float2 x)
{
    return x - floor(x / 289.0) * 289.0;
}

float3 permute(float3 x)
{
    return mod289((x * 34.0 + 1.0) * x);
}

float3 taylorInvSqrt(float3 r)
{
    return 1.79284291400159 - 0.85373472095314 * r;
}

float snoise(float2 v)
{
    const float4 C = float4( 0.211324865405187,  // (3.0-sqrt(3.0))/6.0
                             0.366025403784439,  // 0.5*(sqrt(3.0)-1.0)
                            -0.577350269189626,  // -1.0 + 2.0 * C.x
                             0.024390243902439); // 1.0 / 41.0
    // First corner
    float2 i  = floor(v + dot(v, C.yy));
    float2 x0 = v -   i + dot(i, C.xx);

    // Other corners
    float2 i1;
    i1.x = step(x0.y, x0.x);
    i1.y = 1.0 - i1.x;

    // x1 = x0 - i1  + 1.0 * C.xx;
    // x2 = x0 - 1.0 + 2.0 * C.xx;
    float2 x1 = x0 + C.xx - i1;
    float2 x2 = x0 + C.zz;

    // Permutations
    i = mod289(i); // Avoid truncation effects in permutation
    float3 p =
      permute(permute(i.y + float3(0.0, i1.y, 1.0))
                    + i.x + float3(0.0, i1.x, 1.0));

    float3 m = max(0.5 - float3(dot(x0, x0), dot(x1, x1), dot(x2, x2)), 0.0);
    m = m * m;
    m = m * m;

    // Gradients: 41 points uniformly over a line, mapped onto a diamond.
    // The ring size 17*17 = 289 is close to a multiple of 41 (41*7 = 287)
    float3 x = 2.0 * frac(p * C.www) - 1.0;
    float3 h = abs(x) - 0.5;
    float3 ox = floor(x + 0.5);
    float3 a0 = x - ox;

    // Normalise gradients implicitly by scaling m
    m *= taylorInvSqrt(a0 * a0 + h * h);

    // Compute final noise value at P
    float3 g;
    g.x = a0.x * x0.x + h.x * x0.y;
    g.y = a0.y * x1.x + h.y * x1.y;
    g.z = a0.z * x2.x + h.z * x2.y;
    return 130.0 * dot(m, g);
}

float3 snoise_grad(float2 v)
{
    const float4 C = float4( 0.211324865405187,  // (3.0-sqrt(3.0))/6.0
                             0.366025403784439,  // 0.5*(sqrt(3.0)-1.0)
                            -0.577350269189626,  // -1.0 + 2.0 * C.x
                             0.024390243902439); // 1.0 / 41.0
    // First corner
    float2 i  = floor(v + dot(v, C.yy));
    float2 x0 = v -   i + dot(i, C.xx);

    // Other corners
    float2 i1;
    i1.x = step(x0.y, x0.x);
    i1.y = 1.0 - i1.x;

    // x1 = x0 - i1  + 1.0 * C.xx;
    // x2 = x0 - 1.0 + 2.0 * C.xx;
    float2 x1 = x0 + C.xx - i1;
    float2 x2 = x0 + C.zz;

    // Permutations
    i = mod289(i); // Avoid truncation effects in permutation
    float3 p =
      permute(permute(i.y + float3(0.0, i1.y, 1.0))
                    + i.x + float3(0.0, i1.x, 1.0));

    float3 m = max(0.5 - float3(dot(x0, x0), dot(x1, x1), dot(x2, x2)), 0.0);
    float3 m2 = m * m;
    float3 m3 = m2 * m;
    float3 m4 = m2 * m2;

    // Gradients: 41 points uniformly over a line, mapped onto a diamond.
    // The ring size 17*17 = 289 is close to a multiple of 41 (41*7 = 287)
    float3 x = 2.0 * frac(p * C.www) - 1.0;
    float3 h = abs(x) - 0.5;
    float3 ox = floor(x + 0.5);
    float3 a0 = x - ox;

    // Normalise gradients
    float3 norm = taylorInvSqrt(a0 * a0 + h * h);
    float2 g0 = float2(a0.x, h.x) * norm.x;
    float2 g1 = float2(a0.y, h.y) * norm.y;
    float2 g2 = float2(a0.z, h.z) * norm.z;

    // Compute noise and gradient at P
    float2 grad =
      -6.0 * m3.x * x0 * dot(x0, g0) + m4.x * g0 +
      -6.0 * m3.y * x1 * dot(x1, g1) + m4.y * g1 +
      -6.0 * m3.z * x2 * dot(x2, g2) + m4.z * g2;
    float3 px = float3(dot(x0, g0), dot(x1, g1), dot(x2, g2));
    return 130.0 * float3(grad, dot(m4, px));
}

PS_OUTPUT main(PS_INPUT IN)
{		
	PS_OUTPUT psOut = (PS_OUTPUT)1.f;
	
	float2 uv = IN.o_UV.xy;	
	
	uv *= U_PSCustom0[1].z;	
	
	#ifdef USE_DISTORT
		float _distortSpeed = U_PSCustom2.x;
		float _distortAmt = U_PSCustom2.y;
		float _distortScale = U_PSCustom2.z;
		 
		float noise = snoise(uv * _distortScale + float2(_distortSpeed, - _distortSpeed));
		
		uv += noise * _distortAmt;
	#endif
	
	uv += U_PSCustom0[1].xy;
	
	float4 diffuseColor = S_Diffuse.Sample(S_DiffuseSamplerState, uv);
	
	psOut.o_Color.rgb = pow(U_PSCustom0[0].rgb, 2.2) + pow(diffuseColor.rgb, 2.2);
		
	#ifdef USE_SUN_BACK_COLOR
		float3 cloudDir = normalize(IN.o_CustomParam.xyz); 
		
		float factor = max(dot(cloudDir,-U_MainLightDirection.xyz), 0.0f);
		factor = factor * factor * factor;
		
		float emendation = (1.0f - factor) * U_PSCustom0[2].z;
		
		psOut.o_Color.rgb = pow(U_PSCustom0[0].rgb, 2.2) * emendation + pow(U_PSCustom1.rgb, 2.2) * pow(diffuseColor.rgb, 2.2) * factor;	
	#endif

	float3 viewDir = normalize(IN.o_WorldPos.xyz - U_PS_CameraPosition.xyz);

	psOut.o_Color.rgb = psOut.o_Color.rgb * IN.o_FogColor.a + IN.o_FogColor.rgb;

    psOut.o_Color.rgb = pow(psOut.o_Color.rgb, 0.4545f);

	// Note: our mainRT is marked as linear color space, but we are writting sRGB color into it. 
	// and spherical fog post process uses alpha blend to blend fog color and scene color.
	// So the final result of post process is a linear interpolation of sRGB format scene color and sRGB format fog color. 
	// So when it comes to transparent objects, we also transform scene color and fog color to sRGB space and do linear interpolation. 
    float4 sphFog = ComputeSphFogBlocked(
		IN.o_WorldPos.xyz,
		U_PS_CameraPosition.xyz,
		viewDir,
        IN.o_Position.xy
	);
    psOut.o_Color.rgb = lerp(psOut.o_Color.rgb, sphFog.rgb, sphFog.a);

	float sceneDepth = calcSceneDepth(IN.o_ClipPos);
		
	float4 vp;
	vp.x = dot(U_InvProjectMatrix[0], IN.o_ClipPos);
	vp.y = dot(U_InvProjectMatrix[1], IN.o_ClipPos);
	vp.z = dot(U_InvProjectMatrix[2], IN.o_ClipPos);
	vp.w = dot(U_InvProjectMatrix[3], IN.o_ClipPos);
	
	float pixelDepth = vp.z / vp.w;
		
	float alpha = saturate((pixelDepth - sceneDepth) / U_PSCustom0[2].y);
	psOut.o_Color.a = alpha * diffuseColor.a * U_PSCustom0[2].x;
	
	#ifdef VERTEX_COLOR
		psOut.o_Color.a *= (IN.o_Diffuse.g * IN.o_Diffuse.g);
	#endif
	float overdrawlerp = ceil(U_PS_Time.w);
	psOut.o_Color = psOut.o_Color * (1.0 - overdrawlerp) + float4(U_PS_Time.w * overdrawlerp, U_PS_Time.w * overdrawlerp, U_PS_Time.w * overdrawlerp, overdrawlerp);
	return psOut;
}
