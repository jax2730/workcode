	float3 fresnelSchlick(float cosTheta, float3 F0)
	{
		return F0 + (1.0-F0) * pow(1.0-cosTheta, 5.0);
	}

	float distributionGGX(float3 N, float3 H, float roughness)
	{
		float a = roughness * roughness;
		float a2 = a * a;
		float NdotH = max(dot(N, H), 0.0);
		float NdotH2 = NdotH * NdotH;
		float nom = a2;
		float denom = (NdotH2 * (a2 - 1.0) + 1.0);
		denom = 3.141592654 * denom * denom;
		denom = max(denom, 0.00001);
		return nom / denom;
	}

	float geometrySchlickGGX(float NdotV, float roughness)
	{
		float r = roughness + 1.0;
		float k = (r * r) / 8.0;
		
		float nom = NdotV;
		float denom = NdotV * (1.0 - k) + k;
		
		return nom / denom;
	}

	float geometrySmith(float3 N, float3 V, float3 L, float roughness)
	{
		float NdotV = max(dot(N, V), 0.0);
		float NdotL = max(dot(N, L), 0.0);
		float ggx2 = geometrySchlickGGX(NdotV, roughness);
		float ggx1 = geometrySchlickGGX(NdotL, roughness);
		
		return ggx2 * ggx1;
	}

	float3 calculateLightPower(float3 N, float3 V, float3 L, float3 lightPower, float3 albedo, float roughness, float metallic)
	{
		float3 H    = normalize(L+V);
		float NdotH = max(dot(N, H), 0.0);
		float LdotH = max(dot(L,H),0.0);
		float NdotL = max(dot(N, L), 0.0);
		
		float a2    = roughness*roughness;
		float a4    = a2*a2;
		
		float NdotH2 = NdotH*NdotH;
		float LdotH2 = LdotH*LdotH;
		
		 float denom = 4.0f*3.1415926f*pow(NdotH2*(a4-1.f)+1.f,2.f)*(LdotH2*(1.0f-a2)+a2);
		denom = max(denom, 0.00001f);
		float brdf  = a4/denom;
		float3 F0 = float3(0.04, 0.04, 0.04);
		F0 = lerp(F0, albedo, metallic);
		
		float3 kd    = 1.0f - min(F0/(LdotH + 0.00001f), 1.0f);
		kd = kd * (1.0f-metallic);
		
		return (albedo*kd / 3.141592654 + brdf*F0) * NdotL * lightPower;	
	}
	
    float3 envBRDFApprox(float3 specularColor, float roughness, float NoV)
    {
       return pow(1.0f-max(roughness,NoV),3.f)+specularColor;
    }
    

	float GetLuminance(float3 color)
	{
		const float3 lumWeight = float3(0.2125f, 0.7154f, 0.0721f);
		return dot(color, lumWeight);
	}

#endif
};

PS_OUTPUT main(PS_INPUT IN,  bool isFrontFacing : SV_IsFrontFace)
{
	PS_OUTPUT psOut = (PS_OUTPUT)0;

#ifdef SHADOWPASS

#else
	psOut.color.a = 1.0;
	#ifdef NoColor
		psOut.color = float4(0.0, 0.0, 0.0, 0.0);
		return psOut;
	#endif
	
	float2 uv = IN.i_UV.xy;
	
	
	half3 normal = normalize(IN.i_Normal.xyz);
	
		
	#ifdef TWO_SIDES
		float frontValue = isFrontFacing ? 1.0f : -1.0f;
		normal *= frontValue;
	#endif
	
	float3 N = normal;

	#ifdef USESHADOWMAP
		float4 shadowSpacePos[SHADOWSPLITE];
		ProjectToLightSpace(shadowSpacePos, IN.i_WPos.xyz, N, IN.i_ViewSpaceZ);
	#endif

	float fRoughnessBase = 0.0f;
	
	float metallic = 1.0;
	float roughness = 0.0001;
	float ao = 1.0;
	
	#ifdef PBR_METAL
		metallic = 1.0f;
	#else
		metallic = 0.0f;
	#endif
	#ifdef PBR_ROUGHNESS
		ao = 1.0f;
		roughness = cTexBump.b;
	#endif
	
	roughness = clamp(roughness + fRoughnessBase, 0.0, 1.0);
	

	
	half4 diffuseColor = half4(1,1,1,1);
	
	#ifdef VERTEX_COLOR
		diffuseColor.rgb = IN.i_Diffuse.rgb;
	#else
		diffuseColor.rgb =  pow(U_MaterialDiffuse.rgb,2.2);	
	#endif		
	
	float4 cAgingDiffuse = S_AgingTex.Sample(S_AgingTexSamplerState, IN.i_UV.xy);
	diffuseColor.rgb = lerp(diffuseColor.rgb, pow(cAgingDiffuse.rgb, 2.2f), U_PSGeneralRegister17.w);
	

	diffuseColor.rgb = lerp(diffuseColor.rgb, U_PSGeneralRegister0.rgb, U_PSGeneralRegister0.w);
	
		
	float3 cBaseDiffuse = diffuseColor.rgb;
	

	float coverLerp = dot(N, U_PS_GravityDirection.xyz);
	coverLerp += U_PSGeneralRegister11.z;
	coverLerp = clamp(coverLerp, 0.0, 1.0);
		
	coverLerp = coverLerp * U_PSGeneralRegister5.w;
	
	coverLerp *= U_WeatherParam0.w;
	cBaseDiffuse = lerp(cBaseDiffuse, U_WeatherParam0.xyz, coverLerp);
	

	float3 albedo = cBaseDiffuse.rgb * (1.0 - metallic);
	float3 specularColor = lerp(float3(0.04f, 0.04f, 0.04f), cBaseDiffuse.rgb, metallic);	

	float3 L = -U_MainLightDirection.xyz;
	float3 V = normalize(IN.i_EyeDir.xyz);
	psOut.color.rgb += calculateLightPower(N, V, L, U_PBR_MainLightColor.xyz * 0.95, cBaseDiffuse.rgb, roughness, metallic);
	//psOut.color.rgb *= U_MaterialDiffuse.rgb;	
	
	float3 envBRDF = envBRDFApprox(specularColor, roughness, max(dot(N, V), 0.0));

	float3 ambientColor = ComputeAmbientLight(U_PBR_SkyLightColor.xyz, U_PBR_GroundLightColor.xyz, N, U_PS_AmbientBlendThreshold.y, U_PS_GravityDirection.xyz);

	envBRDF *= normalize(ambientColor);


	#ifdef USESHADOWMAP
		float shadowFactor = calcShadowFactor(
			shadowSpacePos, 
			IN.i_ViewSpaceZ);
			
		psOut.color.rgb *= shadowFactor;
	#endif

	psOut.color.rgb += (albedo + envBRDF) * ambientColor;
	psOut.color.rgb *= ao;
		

	psOut.color.rgb *= U_MaterialCustomParam1.x;	
	psOut.color.rgb = psOut.color.rgb * IN.i_Fog.a + IN.i_Fog.rgb;
	psOut.color.rgb = pow(psOut.color.rgb, 1.0 / 2.2);
	psOut.color.a = diffuseColor.a * U_MaterialFresnel.r * psOut.color.a;
		
	#ifdef EditorHelp
		psOut.color.rgb = psOut.color.rgb * 0.00001 + float3(1.0,0.0,0.0);
		psOut.color.a = 1.0;
	#endif
	#ifdef EDITOR_MODE
		if(U_MaterialCustomParam.w > 0)
			psOut.color.rgb = U_MaterialCustomParam.xyz;			
	#endif
	
	float overdrawlerp = ceil(U_PS_Time.w);
	psOut.color = psOut.color * (1.0 - overdrawlerp) + float4(U_PS_Time.w * overdrawlerp, U_PS_Time.w * overdrawlerp, U_PS_Time.w * overdrawlerp, overdrawlerp);
#endif

	return psOut;
}
