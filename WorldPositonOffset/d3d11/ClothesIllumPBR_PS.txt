float3 mod289(float3 x)
{
    return x - floor(x / 289.0) * 289.0;
}

float2 mod289(float2 x)
{
    return x - floor(x / 289.0) * 289.0;
}

float3 permute(float3 x)
{
    return mod289((x * 34.0 + 1.0) * x);
}

float3 taylorInvSqrt(float3 r)
{
    return 1.79284291400159 - 0.85373472095314 * r;
}

float snoise(float2 v)
{
    const float4 C = float4( 0.211324865405187,  // (3.0-sqrt(3.0))/6.0
                             0.366025403784439,  // 0.5*(sqrt(3.0)-1.0)
                            -0.577350269189626,  // -1.0 + 2.0 * C.x
                             0.024390243902439); // 1.0 / 41.0
    // First corner
    float2 i  = floor(v + dot(v, C.yy));
    float2 x0 = v -   i + dot(i, C.xx);

    // Other corners
    float2 i1;
    i1.x = step(x0.y, x0.x);
    i1.y = 1.0 - i1.x;

    // x1 = x0 - i1  + 1.0 * C.xx;
    // x2 = x0 - 1.0 + 2.0 * C.xx;
    float2 x1 = x0 + C.xx - i1;
    float2 x2 = x0 + C.zz;

    // Permutations
    i = mod289(i); // Avoid truncation effects in permutation
    float3 p =
      permute(permute(i.y + float3(0.0, i1.y, 1.0))
                    + i.x + float3(0.0, i1.x, 1.0));

    float3 m = max(0.5 - float3(dot(x0, x0), dot(x1, x1), dot(x2, x2)), 0.0);
    m = m * m;
    m = m * m;

    // Gradients: 41 points uniformly over a line, mapped onto a diamond.
    // The ring size 17*17 = 289 is close to a multiple of 41 (41*7 = 287)
    float3 x = 2.0 * frac(p * C.www) - 1.0;
    float3 h = abs(x) - 0.5;
    float3 ox = floor(x + 0.5);
    float3 a0 = x - ox;

    // Normalise gradients implicitly by scaling m
    m *= taylorInvSqrt(a0 * a0 + h * h);

    // Compute final noise value at P
    float3 g;
    g.x = a0.x * x0.x + h.x * x0.y;
    g.y = a0.y * x1.x + h.y * x1.y;
    g.z = a0.z * x2.x + h.z * x2.y;
    return 130.0 * dot(m, g);
}

float3 snoise_grad(float2 v)
{
    const float4 C = float4( 0.211324865405187,  // (3.0-sqrt(3.0))/6.0
                             0.366025403784439,  // 0.5*(sqrt(3.0)-1.0)
                            -0.577350269189626,  // -1.0 + 2.0 * C.x
                             0.024390243902439); // 1.0 / 41.0
    // First corner
    float2 i  = floor(v + dot(v, C.yy));
    float2 x0 = v -   i + dot(i, C.xx);

    // Other corners
    float2 i1;
    i1.x = step(x0.y, x0.x);
    i1.y = 1.0 - i1.x;

    // x1 = x0 - i1  + 1.0 * C.xx;
    // x2 = x0 - 1.0 + 2.0 * C.xx;
    float2 x1 = x0 + C.xx - i1;
    float2 x2 = x0 + C.zz;

    // Permutations
    i = mod289(i); // Avoid truncation effects in permutation
    float3 p =
      permute(permute(i.y + float3(0.0, i1.y, 1.0))
                    + i.x + float3(0.0, i1.x, 1.0));

    float3 m = max(0.5 - float3(dot(x0, x0), dot(x1, x1), dot(x2, x2)), 0.0);
    float3 m2 = m * m;
    float3 m3 = m2 * m;
    float3 m4 = m2 * m2;

    // Gradients: 41 points uniformly over a line, mapped onto a diamond.
    // The ring size 17*17 = 289 is close to a multiple of 41 (41*7 = 287)
    float3 x = 2.0 * frac(p * C.www) - 1.0;
    float3 h = abs(x) - 0.5;
    float3 ox = floor(x + 0.5);
    float3 a0 = x - ox;

    // Normalise gradients
    float3 norm = taylorInvSqrt(a0 * a0 + h * h);
    float2 g0 = float2(a0.x, h.x) * norm.x;
    float2 g1 = float2(a0.y, h.y) * norm.y;
    float2 g2 = float2(a0.z, h.z) * norm.z;

    // Compute noise and gradient at P
    float2 grad =
      -6.0 * m3.x * x0 * dot(x0, g0) + m4.x * g0 +
      -6.0 * m3.y * x1 * dot(x1, g1) + m4.y * g1 +
      -6.0 * m3.z * x2 * dot(x2, g2) + m4.z * g2;
    float3 px = float3(dot(x0, g0), dot(x1, g1), dot(x2, g2));
    return 130.0 * float3(grad, dot(m4, px));
}

float3 fresnelSchlick(float cosTheta, float3 F0)
{
	return F0 + (1.0-F0) * pow(1.0-cosTheta, 5.0);
}

float distributionGGX(float3 N, float3 H, float roughness)
{
	float a = roughness * roughness;
	float a2 = a * a;
	float NdotH = max(dot(N, H), 0.0);
	float NdotH2 = NdotH * NdotH;
	float nom = a2;
	float denom = (NdotH2 * (a2 - 1.0) + 1.0);
	denom = 3.141592654 * denom * denom;
	denom = max(denom, 0.00001);
	return nom / denom;
}

float geometrySchlickGGX(float NdotV, float roughness)
{
	float r = roughness + 1.0;
	float k = (r * r) / 8.0;
	
	float nom = NdotV;
	float denom = NdotV * (1.0 - k) + k;
	
	return nom / denom;
}

float geometrySmith(float3 N, float3 V, float3 L, float roughness)
{
	float NdotV = max(dot(N, V), 0.0);
	float NdotL = max(dot(N, L), 0.0);
	float ggx2 = geometrySchlickGGX(NdotV, roughness);
	float ggx1 = geometrySchlickGGX(NdotL, roughness);
	
	return ggx2 * ggx1;
}

float3 envBRDFApprox(float3 specularColor, float roughness, float NoV)
{
	float4 c0 = float4(-1.0,	-0.0275,	-0.572,		0.022);
	float4 c1 = float4(1.0,		0.0425,		1.04,		-0.04);
	
	float4 r = roughness * c0 + c1;
	float a004 = min(r.x * r.x, exp2(-9.28 * NoV)) * r.x + r.y;
	float2 AB = float2(-1.04, 1.04) * a004 + r.zw;
	return specularColor * AB.x + AB.y;
}

float3 thinTranslucency(float3 N, float3 L, float3 transmittanceColor)
{
	float transmittance = saturate(dot(-N, L) + 0.2);
	float fresnel = (1.0 - transmittance) * (1.0 - transmittance);
	fresnel *= fresnel;
	
	return transmittance * (1.0 - fresnel) * transmittanceColor;
}

PS_OUTPUT main(PS_INPUT IN)
{
	PS_OUTPUT psOut = (PS_OUTPUT)0;

#ifdef PARABOLOID_MAPPING
	clip(IN.i_ClipDepth);
#endif

#ifdef SHADOWPASS
	#ifdef ALPHA_TEST
		float4 color = texDiffuse(IN.i_UV.xy);
		clip(color.a - U_MaterialDiffuse.a);
	#endif
#else

	#ifdef MOTIONBLUR
		float2 vQX = IN.i_VelUV.xy;
	
		float fLenQX = length(vQX);
		vQX /= (fLenQX+0.01f);
		float fWeight = max(0.5f, min(fLenQX, 20.0f));
		vQX *= fWeight;
		psOut.color.xy = vQX;
		return psOut;
	#endif
		
	float2 uv = IN.i_UV.xy;
	#ifdef UV_ROLL
		uv += frac(U_PS_Time.z * float2(U_PSGeneralRegister1.z, U_PSGeneralRegister1.w));
	#endif
	
	#ifdef DISTORT
		float _distortAmt = U_PSGeneralRegister2.y;
		float _distortScale = U_PSGeneralRegister2.z;
		float _distortSpeed = U_PSGeneralRegister1.y;
		float Time = U_PS_Time.z;
 
		float noise = snoise(uv * _distortScale + float2(Time *_distortSpeed, - Time *_distortSpeed));
		
		uv += noise * _distortAmt;
	#endif

	#ifdef UV_TILLING
		uv *= U_PSGeneralRegister2.x;
	#endif
	#ifdef DISSOLVE
	
		if(U_MaterialCustomParam1.y > 0)
		{
			float4x4 thresholdMatrix =
			{  1.0 / 17.0,  9.0 / 17.0,  3.0 / 17.0, 11.0 / 17.0,
			  13.0 / 17.0,  5.0 / 17.0, 15.0 / 17.0,  7.0 / 17.0,
			   4.0 / 17.0, 12.0 / 17.0,  2.0 / 17.0, 10.0 / 17.0,
			  16.0 / 17.0,  8.0 / 17.0, 14.0 / 17.0,  6.0 / 17.0
			};
			float2 pos = IN.i_PosInClip.xy;
			
			
			pos.xy = pos.xy/U_MaterialCustomParam1.z;
			
			int xindex = fmod(pos.x, 4);
			int yindex =fmod(pos.y, 4);
			
			float alpha = thresholdMatrix[xindex][yindex];//* _RowAccess[yindex];
			
			if(alpha < U_MaterialCustomParam1.y)
				discard;
		}
		else
		{
			if(S_Custom7.Sample(S_Custom7SamplerState, uv).a < U_MaterialFresnel.y)
			discard;	
		}
		
		bool needDiscard = ComputeBitmapCullMask(U_PSGeneralRegister29, IN.i_BoundBoxRange, 
		U_MaterialCustomParam1.z, IN.i_PosInModel.xyz, IN.i_PosInClip.xyz);
		if(needDiscard)
			discard;

	#endif
	#ifdef SEQUENCE_FRAME
		#ifdef SEQUENCE_FRAME_BASEMAP
			int index = floor(U_PS_Time.x * U_PSGeneralRegister10.z);
			int indexY = index / U_PSGeneralRegister10.w;
			int indexX = index - indexY * U_PSGeneralRegister10.w;
			uv = float2(uv.x / U_PSGeneralRegister10.w, uv.y / U_PSGeneralRegister10.w);
			uv.x += indexX / U_PSGeneralRegister10.w;
			uv.y += indexY / U_PSGeneralRegister10.w;
		#endif
	#endif
	
	half4 diffuseColor = S_Diffuse.Sample(S_DiffuseSamplerState, uv);
	
		
	#ifdef ALPHA_TEST
		clip(diffuseColor.a - U_MaterialDiffuse.a);
	#endif
	
	half3 baseMap = pow(diffuseColor.rgb, 2.2);
	
	#ifdef TEXTURE_MASK
		half4 mask = S_Custom2.Sample(S_Custom2SamplerState, uv);
		half3 colorR = pow(U_PSGeneralRegister4.rgb, 2.2) * mask.r;
		half3 colorG = pow(U_PSGeneralRegister5.rgb, 2.2) * mask.g;
		half3 colorB = pow(U_PSGeneralRegister6.rgb, 2.2) * mask.b;
		float f = min(mask.r + mask.g + mask.b, 1.0);
		half3 colorF = colorR + colorG + colorB;
		//half3 lumWeight = half3(0.2125f, 0.7154f, 0.0721f);
		//float hslcolor = dot(baseMap.rgb, lumWeight);
		//hslcolor += mask.w;
		//hslcolor = min(hslcolor, 1.0);
		//half3 maskColor = lerp(baseMap, hslcolor * colorF, f);
		//baseMap = lerp(baseMap, maskColor, U_PSGeneralRegister7.w);
		baseMap = lerp(baseMap, baseMap * colorF, f);

	#endif
	
	half3 N = normalize(IN.i_Normal.xyz);
	half3 V = normalize(IN.i_EyeDir.xyz);
	
	#ifdef USESHADOWMAP
		float4 shadowSpacePos[SHADOWSPLITE];
		ProjectToLightSpace(shadowSpacePos, IN.i_WPos.xyz, N, IN.i_ViewSpaceZ);
	#endif

	half4 bumpColor = 0.0;	
	#ifdef BUMP_MAP
		half3 nn = normalize(IN.i_Normal.xyz);
		half3 tn = normalize(IN.i_Tangent.xyz);
		half3 bn = normalize(IN.i_Binormal.xyz);

		tn = normalize(tn - nn * dot(tn, nn));
		bn = normalize(bn - nn * dot(bn, nn));

		half3x3 tbn = 
		{
			tn.x, bn.x, nn.x,
			tn.y, bn.y, nn.y,
			tn.z, bn.z, nn.z 
		};
		
		bumpColor = S_Bump.Sample(S_BumpSamplerState, uv); 

		half3 bump = bumpColor.rgb * 2.0 - 1.0;	
		bump *= half3(U_PSGeneralRegister2.w, U_PSGeneralRegister2.w, 1.0);
		bump.z = sqrt(1.0 - (bump.x * bump.x + bump.y * bump.y));
		
		#if defined(DETAIL_BUMP) || defined(DECAL1) || defined(DECAL2)
			half4 maskColor = S_Custom9.Sample(S_Custom9SamplerState, uv);
		#endif
		
		#ifdef DETAIL_BUMP
			float2 detailUV1 = float2(U_PSGeneralRegister2.y, U_PSGeneralRegister2.y);
			
			#ifdef UV_TILLING
				detailUV1 /= U_PSGeneralRegister2.x;
			#endif
			
			half4 detailBumpColor1 = S_Custom3.Sample(S_Custom3SamplerState, uv * detailUV1);
			bump.xy += (detailBumpColor1.xy - half2(0.5, 0.5)) * U_PSGeneralRegister2.z * maskColor.r;
			//bump.xy * (1.0 - maskColor.r) + 
		#endif

		#ifdef DECAL1	
			half4 decalBumpColor1 = S_Custom5.Sample(S_Custom5SamplerState, uv * U_PSGeneralRegister11.x);
			
			half3 decalBump1 = decalBumpColor1.rgb * 2.0 - 1.0;	
			decalBump1 *= half3(U_PSGeneralRegister13.y, U_PSGeneralRegister13.y, 1.0);
			decalBump1.z = sqrt(1.0 - (decalBump1.x * decalBump1.x + decalBump1.y * decalBump1.y));
			
			float decalBumpMaskG = clamp(maskColor.g - (1.0 - U_PSGeneralRegister13.x), 0.0, 1.0);
			
			bump += decalBump1 * decalBumpMaskG;
			// bump * (1.0 - decalBumpMaskG) + 
		#endif
		
		#ifdef DECAL2
			half4 decalBumpColor2 = S_Custom8.Sample(S_Custom8SamplerState, uv * U_PSGeneralRegister12.y);	
			
			half3 decalBump2 = decalBumpColor2.rgb * 2.0 - 1.0;	
			decalBump2 *= half3(U_PSGeneralRegister13.w, U_PSGeneralRegister13.w, 1.0);
			decalBump2.z = sqrt(1.0 - (decalBump2.x * decalBump2.x + decalBump2.y * decalBump2.y));
			
			float decalBumpMaskB = clamp(maskColor.b - (1.0 - U_PSGeneralRegister13.z), 0.0, 1.0);
			
			bump += decalBump2 * decalBumpMaskB;
			//bump * (1.0 - decalBumpMaskB)
		#endif
		
		N = normalize(mul(tbn, bump));
	#endif
	
	#ifdef COVER_WEATHER
		float coverLerp = dot(N, U_PS_GravityDirection.xyz);
		coverLerp += U_PSGeneralRegister16.z;
		coverLerp = clamp(coverLerp, 0.0, 1.0);
		
		half3 coverMap = U_WeatherParam0.xyz;
				
		#ifdef COVER_CUSTOM
			float2 coverUV = uv * U_PSGeneralRegister16.ww;
			half3 coverCustomMap = pow(S_Custom3.Sample(S_Custom3SamplerState, coverUV).rgb, 2.2);
			coverMap = lerp(coverCustomMap, coverMap, U_WeatherParam0.www);
		#else
			coverLerp *= U_WeatherParam0.w;
		#endif
	#endif
	
	half metallic = 1.0;
	half roughness = 0.0001;
	
	half ao = 1.0;
	#ifdef PBR_SIMPLE
		#ifdef	PBR_METAL	
			metallic = 1.0;
		#else
			metallic = 0.0;
		#endif
		
		#ifdef PBR_AO
			ao = bumpColor.b;
			roughness = U_PSGeneralRegister4.w;
		#endif
		
		#ifdef PBR_ROUGHNESS
			ao = 1.0;
			roughness = bumpColor.b;
		#endif
		
		roughness = clamp(roughness + U_PSGeneralRegister6.w, 0.0, 1.0);
	#else
		half3 mixMap = S_Specular.Sample(S_SpecularSamplerState, uv).rgb;
		metallic = mixMap.r;
		roughness = clamp(mixMap.g + U_PSGeneralRegister6.w, 0.0, 1.0);
		ao = mixMap.b;
	#endif
	
	#if defined(DECAL1) || defined(DECAL2)
		#ifndef BUMP_MAP
			half4 maskColor = S_Custom9.Sample(S_Custom9SamplerState, uv);
		#endif
	#endif
	float3 decalColorMix = baseMap;
	#ifdef DECAL1	
		half4 decalColor1 = S_Custom4.Sample(S_Custom4SamplerState, uv * U_PSGeneralRegister11.x);
		
		#ifdef ALPHA_TEST
			clip(decalColor1.a - U_MaterialDiffuse.a);
		#endif	

		half3 decal1BaseMap = pow(decalColor1.rgb, 2.2);
		
		#ifdef TEXTURE_MASK
			half3 textureMask1 = pow(U_PSGeneralRegister14.rgb, 2.2) * maskColor.g;
			float f1 = min(maskColor.g, 1.0);
			decal1BaseMap = lerp(decal1BaseMap, decal1BaseMap * textureMask1, f1);
		#endif
		
		float decalColorMaskG = maskColor.g * U_PSGeneralRegister12.x * decalColor1.a;
		
		#ifdef DECALCOLORMIX1
			baseMap = baseMap * (1.0 - decalColorMaskG) + decal1BaseMap * decalColorMaskG;
		#else
			baseMap = baseMap * (1.0 - decalColorMaskG) + decal1BaseMap * decalColorMaskG * baseMap;
		#endif
		#ifdef	PBR_METAL	
			decalColor1.r = 1.0;
		#else
			decalColor1.r = 0.0;
		#endif
		metallic = metallic * (1.0 - decalColorMaskG) + decalColor1.r * decalColorMaskG;
		#ifdef PBR_ROUGHNESS
			decalColor1.b = 1.0;
			#ifndef BUMP_MAP
				decalColor1.g = 0.0;
			#else
				decalColor1.g = decalBumpColor1.b;
			#endif
		#endif
		roughness = roughness * (1.0 - decalColorMaskG) + clamp(decalColor1.g + U_PSGeneralRegister11.w, 0.0, 1.0) * decalColorMaskG;
		
		ao = ao * (1.0 - decalColorMaskG) + decalColor1.b * decalColorMaskG;
	#endif
	
	#ifdef DECAL2	
		half4 decalColor2 = S_Custom6.Sample(S_Custom6SamplerState, uv * U_PSGeneralRegister12.y);
		#ifdef ALPHA_TEST
			clip(decalColor2.a - U_MaterialDiffuse.a);
		#endif

		half3 decal2BaseMap = pow(decalColor2.rgb , 2.2);
		
		#ifdef TEXTURE_MASK
			half3 textureMask2 = pow(U_PSGeneralRegister15.rgb, 2.2) * maskColor.b;
			float f2 = min(maskColor.b, 1.0);
			decal2BaseMap = lerp(decal2BaseMap, decal2BaseMap * textureMask2, f2);
		#endif
		
		float decalColorMaskB = maskColor.b * U_PSGeneralRegister12.w * decalColor2.a;
		
		#ifdef DECALCOLORMIX2
			baseMap =  baseMap * (1.0 - decalColorMaskB) + decal2BaseMap * decalColorMaskB;
		#else
			baseMap =  baseMap * (1.0 - decalColorMaskB) + decal2BaseMap * decalColorMaskB * baseMap;
		#endif
		
		#ifdef	PBR_METAL	
			decalColor2.r = 1.0;
		#else
			decalColor2.r = 0.0;
		#endif
		
		metallic = metallic * (1.0 - decalColorMaskB) + decalColor2.r * decalColorMaskB;
		#ifdef PBR_ROUGHNESS
			decalColor2.b = 1.0;
			#ifndef BUMP_MAP
				decalColor2.g = 0.0;
			#else
				decalColor2.g = decalBumpColor2.b;
			#endif
		#endif
		roughness = roughness * (1.0 - decalColorMaskB) + clamp(decalColor2.g + U_PSGeneralRegister12.z, 0.0, 1.0) * decalColorMaskB;
		
		ao = ao * (1.0 - decalColorMaskB) + decalColor2.b * decalColorMaskB;
	#endif
	
	#ifdef COVER_WEATHER
		float r = U_WeatherParam1.x;
		#ifdef COVER_CUSTOM
			r = U_PSGeneralRegister16.x;
		#endif
		roughness = lerp(roughness, r, coverLerp);
	#endif
	
	float wetFinal = U_PSGeneralRegister8.w;
	roughness -= wetFinal;
	roughness = max(roughness , 1e-6);
	
	#ifdef COVER_WEATHER
		float m = U_WeatherParam1.y;
		#ifdef COVER_CUSTOM
			m = U_PSGeneralRegister16.y;
		#endif
		metallic = lerp(metallic, m, coverLerp);
	#endif
	
	metallic = lerp(metallic, 0.5, wetFinal);
	
	half3 color = 0.0;
	
	half3 L = normalize(-U_MainLightDirection.xyz);
	
	#ifdef	THIN_TRANSLUCENCY
		color += thinTranslucency(N, L, U_PSGeneralRegister7.xyz) * metallic;
		metallic = 0.0;
	#endif
	
	half4 mainLightColor   = 0.0;
	half4 skyLightColor    = 0.0;
	half4 groundLightColor = 0.0;
	
	#ifdef ALONE_ROLE_LIGHT
		mainLightColor   = U_RoleMainLightColor;
		skyLightColor    = U_RoleSkyLightColor;
		groundLightColor = U_RoleGroundLightColor;
	#else
		mainLightColor   = U_PBR_MainLightColor;
		skyLightColor    = U_PBR_SkyLightColor;
		groundLightColor = U_PBR_GroundLightColor;
	#endif
	
	#ifdef COVER_WEATHER
		baseMap = lerp(baseMap, coverMap, coverLerp);
	#endif
	
	
	half3 grayColor = baseMap.rgb * half3(0.333, 0.333, 0.333);
	float gray = grayColor.r + grayColor.g + grayColor.b;
	
	grayColor.rgb =  half3(gray,gray,gray);
	
	baseMap.rgb = lerp(baseMap.rgb,grayColor,U_MaterialCustomParam.x);	
	
	
	half3 albedo = baseMap * (1.0 - metallic);
	half3 specularColor = lerp(half3(0.04h, 0.04h, 0.04h), baseMap, metallic);	
	
	color += calculateLightPower(N, V, L, mainLightColor.xyz * 0.95, baseMap, roughness, metallic);
	half3 VV = L;
	VV = -VV;
	color += calculateLightPower(N, V, VV, mainLightColor.xyz * U_PSGeneralRegister8.x, baseMap, roughness, metallic);
	color = color *U_MaterialDiffuse.rgb;
		
	//half shadowOffset = 0.0;
	//half lightlum = GetLuminance(mainLightColor.xyz);
	
	#ifdef USESHADOWMAP
		float shadowFactor = calcShadowFactor(shadowSpacePos, IN.i_ViewSpaceZ);
		color *= shadowFactor;
	#endif

#ifdef CLUSTER_POINT_LIGHT
		color += calculateShadowPBRPointLight(IN.i_WPos, N, V, baseMap, roughness, metallic);
	
        float3 clusterLightColor = 
            calculateClusterPBRPointLight(IN.i_PosInClip.xy,
                                          IN.i_WPos.xyz,
                                          N,
                                          V,
                                          baseMap,
                                          roughness,
                                          metallic);
        color += clusterLightColor;
#endif

        
	half3 envBRDF = envBRDFApprox(specularColor, roughness, max(dot(N, V), 0.0));
	
	half normalY = N.y * 0.5 + 0.5;
	normalY = clamp(normalY, 0.0001f, 1.0f);
	half3 ambientColor = groundLightColor.rgb * (1.0-normalY) + skyLightColor.rgb * normalY;
	
	#ifdef ENVIRONMENT_MAP
		half4 refl;
		refl.xyz = reflect(-V, N);
		refl.w = U_EnvironmentCubeMapMipLevels.x - 1.0 - (1.0 - 1.2 * log2(roughness));
		refl.w = max(0.0, refl.w);
		half3 reflColor = S_EnvironmentCubeMap.SampleLevel(S_EnvironmentCubeMapSamplerState, refl.xyz,refl.w).rgb;
		envBRDF *= reflColor;
	#else
		envBRDF *= normalize(ambientColor);
	#endif
	
	color += (albedo + envBRDF) * ambientColor;
	color *= ao;
	
	#ifdef TEXTURE_FLOW
		uv *= U_PSGeneralRegister3.z;
		uv += frac(U_PS_Time.z * float2(U_PSGeneralRegister3.x, U_PSGeneralRegister3.y));
		half3 flowTextureColor = pow(S_Custom.Sample(S_CustomSamplerState, uv).rgb, 2.2);
		
		#ifdef TOD_TEXTURE_FLOW
			#ifdef TEXTURE_FLOW_WITHOUT_CHANNEL
				color += flowTextureColor * U_PSGeneralRegister3.w * U_LightPower.x;
			#else
				color += flowTextureColor * U_PSGeneralRegister3.w * U_LightPower.x * diffuseColor.a;
			#endif
		#else
			#ifdef TEXTURE_FLOW_WITHOUT_CHANNEL
				color += lerp(flowTextureColor, flowTextureColor * U_PSGeneralRegister3.w, U_LightPower.x);
			#else
				color += lerp(flowTextureColor, flowTextureColor * U_PSGeneralRegister3.w, U_LightPower.x) * diffuseColor.a;
			#endif
		#endif
	#endif
	
	#ifdef SEQUENCE_FRAME
		#ifndef SEQUENCE_FRAME_BASEMAP
			int index = floor(U_PS_Time.x * U_PSGeneralRegister10.z);
			int indexY = index / U_PSGeneralRegister10.w;
			int indexX = index - indexY * U_PSGeneralRegister10.w;
			float2 sfUV = float2(uv.x / U_PSGeneralRegister10.w, uv.y / U_PSGeneralRegister10.w);
			sfUV.x += indexX / U_PSGeneralRegister10.w;
			sfUV.y += indexY / U_PSGeneralRegister10.w;
		
			half3 sfTextureColor = pow(S_Custom.Sample(S_CustomSamplerState, sfUV).rgb, 2.2);
			color += sfTextureColor * diffuseColor.a;
		#endif
	#endif
	
	#ifdef EMISSIVE
		
		half3 missiveColor = diffuseColor.rgb * U_PSGeneralRegister0.xyz * diffuseColor.a;
		half3 minMissiveColor = missiveColor * U_PSGeneralRegister0.w;

		#ifdef TOD_EMISSIVE
			half todFactor = U_LightPower.x;		
			
			#ifdef DYNAMIC_EMISSIVE
				half3 maxMissiveColor = missiveColor * U_PSGeneralRegister1.x;
				half factor = U_MaterialSpecular.w;	
				color += lerp(minMissiveColor, maxMissiveColor, half3(factor, factor, factor)) * todFactor;
			#else
				color += minMissiveColor * todFactor;
			#endif
		#else
		
			#ifdef DYNAMIC_EMISSIVE
				half3 maxMissiveColor = missiveColor * U_PSGeneralRegister1.x;
				half factor = U_MaterialSpecular.w;	
				color += lerp(minMissiveColor, maxMissiveColor, half3(factor, factor, factor));
			#else
				color += minMissiveColor;
			#endif
			
		#endif
		
	#endif
	
	#ifdef STROKE
			half stroke = max(dot(V, N)*U_PSGeneralRegister9.a + U_PSGeneralRegister10.r,0.0f);	
			
			//psOut.color.rgb =float3(0,0,1-IN.i_UV.z);   //float3(0,1 - IN.i_UV.z,0);
			//psOut.color.a = 1;
		
		//return psOut;
		
		//half stroke = max(IN.i_UV.z*U_PSGeneralRegister9.a + U_PSGeneralRegister10.r,0.0f);
		
		stroke = min(stroke,1.0f);
		
		stroke = (1.0 - stroke)*U_MaterialFresnel.r;
		
		color.rgb = lerp(color.rgb, U_PSGeneralRegister9.rgb, stroke);
		
		color.rgb *= (U_PSGeneralRegister10.g*stroke*10.0+1.0);

	#endif
	
	color *= U_MaterialCustomParam1.x;
	color = color * IN.i_Fog.a + IN.i_Fog.rgb;
	psOut.color.rgb = pow(color, 1.0 / 2.2);
	psOut.color.a = diffuseColor.a;
	
	#ifdef EditorHelp
		psOut.color.rgb = psOut.color.rgb * 0.00001 + float3(1.0,0.0,0.0);
		psOut.color.a = 1.0;
	#endif
	#ifdef EDITOR_MODE
		if(U_MaterialCustomParam.w > 0)
			psOut.color.rgb = U_MaterialCustomParam.xyz;
		
	#endif
	float overdrawlerp = ceil(U_PS_Time.w);
	psOut.color = psOut.color * (1.0 - overdrawlerp) + float4(U_PS_Time.w * overdrawlerp, U_PS_Time.w * overdrawlerp, U_PS_Time.w * overdrawlerp, overdrawlerp);
#endif 

	return psOut;
}
