#include <Common_Uniform_VS.inl>
#include <Common_Function.inl>

float4 U_WorldViewProjectMatrix[4];

#if defined(EMISSION_COLOR) || defined(FLOW_COLOR)
	float4 U_LightPower;
#endif

uniform float4 U_VSCustom0;	// xy: patch uv start, z: uv factor, w:Patch size
uniform float4 U_VSCustom1;   //xy: patch world position at XZ axis, y is 0;
								//z: squared transition start dist;
								//w: 1 / (squared max detail distance - squared transition start dist);
#ifdef FLOW_COLOR
	
	uniform float4 U_VSCustom2;		// x: flow uv tilling
									// y: flow uv speed ratio
									// z: squared flow fade start distance
									// w: 1 / (squared flow fade end distance - squared flow fade start distance)
#endif

uniform float4 U_VSCustom3;		// x: rotation degree


struct VS_INPUT
{
	uint3	i_UV		: POSITION0;
	float	i_Height	: POSITION1;
#ifdef VERTEX_COLOR
	unorm float4 i_Diffuse	: COLOR0;
#endif
#ifdef EMISSION_COLOR
	unorm float4 i_Emission	: COLOR1;
#endif
#ifdef FLOW_COLOR
	unorm float4 i_Flow		: COLOR2;
#endif
};

struct VS_OUTPUT
{
	float4	o_PosInClip	: SV_POSITION;
	float4	o_UVBase	: TEXCOORD0;		// w: viewSpaceZ
	float3	o_PosUV		: TEXCOORD1;
	float4	o_WorldPos	: TEXCOORD2;
	float4	o_FogColor	: TEXCOORD3;
#ifdef VERTEX_COLOR
	float4	o_Diffuse	: COLOR0;
#endif

#ifdef EMISSION_COLOR
	float4	o_Emission	: COLOR1;
#endif

#ifdef FLOW_COLOR
	float4	o_Flow		: COLOR2;
	float4  o_FlowUV	: COLOR3;
#endif
};



VS_OUTPUT main(VS_INPUT IN)
{
	VS_OUTPUT vsOut = (VS_OUTPUT)0;

	float4 PatchInfo = U_VSCustom0;
	float4 PatchPosition = U_VSCustom1;

	float2 vUV = float2(IN.i_UV.xy) / (float)IN.i_UV.z;
	float4 vObjPos = float4(
		vUV.x * PatchInfo.w,
		IN.i_Height,
		vUV.y * PatchInfo.w,
		1.0);
		
	vsOut.o_WorldPos.xyz = vObjPos.xyz + float3(PatchPosition.x, 0.f, PatchPosition.y);
	vsOut.o_WorldPos.w = 1.0;

	
	float2 vCamToPosXZ = vsOut.o_WorldPos.xz - U_VS_CameraPosition.xz;
	float fSquareLenXZ = dot(vCamToPosXZ, vCamToPosXZ);		
	float fHeightLower = max(fSquareLenXZ - U_CameraParam.x * U_CameraParam.x + 250000.0, 0.0) * 0.0001;	
	vObjPos.y -= fHeightLower;
	vsOut.o_WorldPos.y -= fHeightLower;
	
	float4 	vWVPMatrixC0 = U_WorldViewProjectMatrix[0];
	float4	vWVPMatrixC1 = U_WorldViewProjectMatrix[1];
	float4	vWVPMatrixC2 = U_WorldViewProjectMatrix[2];
	float4	vWVPMatrixC3 = U_WorldViewProjectMatrix[3];

	vsOut.o_PosInClip.x = dot(vWVPMatrixC0, vObjPos);
	vsOut.o_PosInClip.y = dot(vWVPMatrixC1, vObjPos);
	vsOut.o_PosInClip.z = dot(vWVPMatrixC2, vObjPos);
	vsOut.o_PosInClip.w = dot(vWVPMatrixC3, vObjPos);

	float3 vViewDir = U_VS_CameraPosition.xyz - vsOut.o_WorldPos.xyz;

	float4x4 gravityRotatePosMatrix;	
	gravityRotatePosMatrix[0] = U_VS_GravityRotationMatrix[0];
	gravityRotatePosMatrix[1] = U_VS_GravityRotationMatrix[1];
	gravityRotatePosMatrix[2] = U_VS_GravityRotationMatrix[2];
	gravityRotatePosMatrix[3] = U_VS_GravityRotationMatrix[3];
	vViewDir = mul(gravityRotatePosMatrix,float4(vViewDir.xyz,0.0f)).xyz;

	vsOut.o_FogColor = ComputeFogColor(vViewDir, U_FogParam.xy, U_FogParam.zw, U_FogRampParam, U_FogColorNear, U_FogColorFar);	
	
	vsOut.o_UVBase.xy = vUV.xy * PatchInfo.z + PatchInfo.xy;
	// rotate base uv
	float3x2 matUVTrans[4] =
	{
		float3x2( 1,  0,  0,  1,  0,  0),
		float3x2( 0, -1,  1,  0,  0,  1),
		float3x2(-1,  0,  0, -1,  1,  1),
		float3x2( 0,  1, -1,  0,  1,  0),
	};
	vsOut.o_UVBase.xy = mul(float3(vsOut.o_UVBase.xy, 1), matUVTrans[(int)U_VSCustom3.x]);
	float fViewDirSquaredLen = dot(vViewDir, vViewDir);
	vsOut.o_UVBase.z = saturate((fViewDirSquaredLen - PatchPosition.z) * PatchPosition.w);
	vsOut.o_UVBase.z *= vsOut.o_UVBase.z;	// smooth it
	vsOut.o_PosUV = float3(vUV.x, IN.i_Height / PatchInfo.w, vUV.y);

#ifdef VERTEX_COLOR
	vsOut.o_Diffuse = pow(IN.i_Diffuse, 2.2f);
#endif

#if defined(EMISSION_COLOR) || defined(FLOW_COLOR)
	float fLightPowerClamped = saturate(U_LightPower.x);
#endif

#ifdef EMISSION_COLOR
	vsOut.o_Emission.rgb = lerp(pow(IN.i_Emission.rgb, 2.2f) * 10.f, float3(0.0f, 0.0f, 0.0f), vsOut.o_UVBase.z);
	vsOut.o_Emission.rgb *= lerp(fLightPowerClamped, 1.f, IN.i_Emission.a);
#endif

#ifdef FLOW_COLOR
	vsOut.o_Flow = IN.i_Flow;
	// Multiply a large ratio to increase flow strength, you can use
	// flow-brush to paint the strength in editor, then we don't need
	// a extra parameter to adjust it.
	vsOut.o_Flow.rgb *= 10.f;

	// day & night ratio
	float fDayNightRatio = IN.i_Flow.a * 255.f;
	float fDayRatio = floor(fDayNightRatio / 16.f);
	float fNightRatio = fDayNightRatio - fDayRatio * 16.f;
	vsOut.o_Flow.rgb *= lerp(fDayRatio, fNightRatio, fLightPowerClamped) / 15.f;

	// fade
	float fDistRatio = saturate((fViewDirSquaredLen - U_VSCustom2.z) * U_VSCustom2.w);
	vsOut.o_Flow.rgb *= 1.f - fDistRatio * fDistRatio;

	// uv
	float2x2 matWind = float2x2(U_VS_WindParams.x, U_VS_WindParams.z,
								-U_VS_WindParams.z, U_VS_WindParams.x);
	vsOut.o_FlowUV.xy = mul(matWind, vsOut.o_WorldPos.xz * U_VSCustom2.xx);
	vsOut.o_FlowUV.x -= frac(U_VS_WindParams.w * U_VS_Time.x * U_VSCustom2.y);
#endif

#ifdef USESHADOWMAP
	vsOut.o_UVBase.w = -dot(U_MainCamViewMatrix, vsOut.o_WorldPos);
#endif

	return vsOut;
}
