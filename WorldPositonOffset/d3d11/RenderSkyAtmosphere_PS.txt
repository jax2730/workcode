float4 PrepareOutput(float3 preExposedLuminance, float3 transmittance = float3(1.f, 1.f, 1.f))
{
    const float Max10BitsFloat = 64512.f;

    // sky materials can result in high luminance values, e.g. the sun disk.
    // this is so we makesure to at least stay winth in boundaries of fp10 and not cause NaN on some platforms
    // we also half that range to also make sure we have room for other additive elements such as bloom, cloud or particle value
    const float3 safePreExposedLuminance = min(preExposedLuminance, Max10BitsFloat.xxx * 0.5f);

    const float greyScaleTransmittance  = dot(transmittance, float3(1.f / 3.0f, 1.f / 3.f, 1.f / 3.f));
    float4 luminanceAlpha = float4(safePreExposedLuminance, greyScaleTransmittance);

    return luminanceAlpha;
}

float3 GetLightDiskLuminance(float3 planetCenterToCamera, float3 worldDir)
{
    float t = RaySphereIntersectNearest(planetCenterToCamera, worldDir, float3(0.f, 0.f, 0.f), Atmosphere.bottomRadius);
    
    // ray is not blocked by planet
    if(t < 0.f)
    {
        float3 lightDiskLuminance = GetLightDiskLuminance(
            planetCenterToCamera,
            worldDir,
            Atmosphere.bottomRadius,
            Atmosphere.topRadius,
            TransmittanceTex,
            BilinearSampler,
            -U_MainLightDirection.xyz,
            Atmosphere.sunAngularRadius,
            U_PBR_MainLightColor.rgb
        );

        const float3 maxLightLuminance = 64000.f;
        float3 exposedLightLuminance = min(lightDiskLuminance, maxLightLuminance);
        return exposedLightLuminance;
    }
    return 0.f;
}

float4 linearTosRGB(float4 color)
{
    return color;
    float3 rgb = pow(color.rgb, 1.0f / 2.2f);
    return float4(rgb, color.a);
}

float4 main(PS_INPUT IN):SV_TARGET
{
    float4 outLuminance = 0.f;

    float2 pixelPos = IN.svPos.xy;
    float2 uvBuffer = IN.uv.xy * Viewport.uvSizeAndInvSize.xy + Viewport.uvMinAndMax.xy;

    float3 worldPos = GetTranslatedCameraPlanetPos();
    float3 worldDir = GetScreenWorldDirection(IN.svPos);

    float3 preExposedL = 0.f;
    float3 luminanceScale = 1.f;
    #ifdef SAMPLE_ATMOSPHERE_ON_CLOUDS
        // todo: Support clouds
    #else
        float deviceZ = S_Depth.SampleLevel(BilinearSampler, uvBuffer, 0).r;
        if(deviceZ == FAR_DEPTH_VALUE)
        {
            luminanceScale = SkyAtmosphere.skyLuminanceFactor_distanceScale.xyz;
            if(SourceDiskEnabled > 0)
            {
                preExposedL += GetLightDiskLuminance(worldPos, worldDir);
            }
        }

        #ifdef AERIAL_PERSPECTIVE_DISABLED
            else{
                outLuminance = PrepareOutput(float3(0.f, 0.f, 0.f), float3(1.f, 1.f, 1.f));
                // clip(-1);
                return linearTosRGB(outLuminance);
            }
        #endif
    #endif // SAMPLE_ATMOSPHERE_ON_CLOUDS

    float viewHeight = length(worldPos);

    #ifdef FAST_LUTS_ENABLED
        // render sky
        if(viewHeight < (Atmosphere.topRadius * PLANET_RADIUS_RATIO_SAFE_EDGE) && deviceZ == FAR_DEPTH_VALUE)
        {   
            // using sky view lut enabled and current pixel at far 
            float2 uv = uvBuffer;

            // the referencial used to build the sky view lut

            float3x3 localReferential = GetSkyViewReferential(View.skyViewLutReferential);

            // input vectors expressed in this referencial: Up is always Z.
            // also note that viewHeight is unchanged in this referential.
            float3 worldPosLocal = float3(0.f, 0.f, viewHeight);
            float3 upVectorLocal = float3(0.f, 0.f, 1.f);
            float3 worldDirLocal = mul(localReferential, worldDir);
            float viewZenithCosAngle = dot(worldDirLocal, upVectorLocal);

            // now evaluate inputs in the referential
            bool intersectGround = RaySphereIntersectNearest(
                worldPosLocal, 
                worldDirLocal, 
                float3(0.f, 0.f, 0.f), 
                Atmosphere.bottomRadius) > 0.f;

            // SkyViewLutParamsToUv(
            //     intersectGround, 
            //     viewZenithCosAngle, 
            //     worldDirLocal, 
            //     viewHeight, 
            //     Atmosphere.bottomRadius,
            //     SkyAtmosphere.skyViewLutSizeAndInvSize,
            //     uv);

            float4 skyLuminanceTransmittance = SkyViewLutTex.SampleLevel(BilinearSampler, uv, 0);
            float3 skyLuminance = skyLuminanceTransmittance.rgb;

            float3 skyGreyTransmittance = 1.f;
            // #if POST_PROCESS_ALPHA
            //     if(View.RenderingReflectionCaptureMask == 0.f)
            //     {
                    skyGreyTransmittance = skyLuminanceTransmittance.aaa;
            //     }
            // #endif

            preExposedL += skyLuminance * luminanceScale;//  * (ViewOneOverPreExposure * outputPreExposure);

            outLuminance = PrepareOutput(preExposedL, skyGreyTransmittance);
            return linearTosRGB(outLuminance);
        }
        
        // compute aerial perspective
        {
            // return float4(0.f, 0.f, 0.f, 1.f);
            float3 depthBufferTranslatedWorldPos = GetScreenTranslatedWorldPos(IN.svPos.xy, deviceZ).xyz;
            float3 depthBufferTranslatedWorldPosKm = depthBufferTranslatedWorldPos * 0.001f; // world position from depth
            depthBufferTranslatedWorldPosKm *= SkyAtmosphere.skyLuminanceFactor_distanceScale.w;
            float4 NDCPos = GetNdcPos(IN.svPos.xy, deviceZ);

            const float nearFadeOutRangeInvDepth = 1.f / 0.00001f; // 1 centimeter fade region
            float4 AP = GetAerialPerspectiveLuminanceTransmittance(
                false,
                SkyAtmosphere.aerialPerspectiveVolumeScreenSizeAndInvSize,
                NDCPos,
                (depthBufferTranslatedWorldPosKm - View.cameraWorldPosKm_frameIdMod8.xyz),
                CameraAerialPerspectiveVolumeTex,
                BilinearSampler,
                SkyAtmosphere.aerialPerspectiveVolumeDepthResolutionAndInvResolution.y,
                SkyAtmosphere.aerialPerspectiveVolumeDepthResolutionAndInvResolution.x,
                AerialPerspectiveStartDepthKm,
                SkyAtmosphere.aerialPerspectiveVolumeDepthSliceLengthAndInvLength.x,
                SkyAtmosphere.aerialPerspectiveVolumeDepthSliceLengthAndInvLength.y,
                1.f,    // todo :: add preExposure
                nearFadeOutRangeInvDepth
            );

            preExposedL += AP.rgb * luminanceScale;
            float transmittance = AP.a;

            outLuminance = PrepareOutput(preExposedL, float3(transmittance, transmittance, transmittance));
            return linearTosRGB(outLuminance);
        }
    #else // !FAST_LUTS_ENABLED
        // move camera to intersect point of view and atmosphere top
        // use this point as the starting point for ray marching
        // this is critical to be after the above to not disrupt above atmosphere tests and voxel selection
        if(!MoveToTopAtmosphere(worldPos, worldDir, Atmosphere.topRadius))
        {
            // ray is not intersecting the atmosphere
            outLuminance = PrepareOutput(preExposedL);
            return linearTosRGB(outLuminance);
        }

        // apply the start depth offset after moving to the to atmosphere for consistancy
        worldPos += worldDir * AerialPerspectiveStartDepthKm;

        SamplingSetup sampling = (SamplingSetup)0;
        {
            sampling.variableSampleCount = true;
            sampling.minSampleCount = SkyAtmosphere.rayMarchingSampleCountMin;
            sampling.maxSampleCount = SkyAtmosphere.rayMarchingSampleCountMax;
            sampling.distanceToSampleCountMaxInv = SkyAtmosphere.rayMarchingDistanceToSampleCountMaxInv;
        }

        const bool ground = false;
        const bool mieRayPhase = true;
        const float aerialPerspectiveViewDistanceScale = deviceZ == FAR_DEPTH_VALUE ?
            1.f : SkyAtmosphere.aerialPerspectiveDistanceScale;
        SingleScatteringResult ss = IntegrateSingleScatteredLuminance(
            IN.svPos,
            worldPos, 
            worldDir,
            ground,
            sampling,
            deviceZ,
            mieRayPhase,
            -U_MainLightDirection.xyz,
            U_PBR_MainLightColor.rgb,
            aerialPerspectiveViewDistanceScale);
        
        preExposedL += ss.L * luminanceScale;

        #if SAMPLE_ATMOSPHERE_ON_CLOUDS
            // todo: add supports to volume clouds
        #endif

        outLuminance = PrepareOutput(preExposedL, ss.transmittance);
        return linearTosRGB(outLuminance);
    #endif // !FAST_LUTS_ENABLED
}
