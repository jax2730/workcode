
#include <Common_Uniform_PS.inl> 

uniform float4  U_PSCustom0;

struct PS_INPUT
{
    float4 o_Position	: SV_POSITION;
	float4 o_UV			: TEXCOORD0;
	float4 o_Color		: TEXCOORD1;
	float4 o_CustomParam: TEXCOORD2;
};

struct PS_OUTPUT
{
    float4 o_Color	: SV_TARGET;
};

PS_OUTPUT main(PS_INPUT IN)
{
	PS_OUTPUT psOut = (PS_OUTPUT)1.f;
	
	float2 pos = IN.o_UV.xy;
	
	float2 distCenter = 3.5 * pos.xy - 3.5 * float2(0.5, 0.5);
	float scale = exp(-dot(distCenter, distCenter));

	half3 colCool = IN.o_Color.xyz;
	half3 colHot = IN.o_Color.w;
	half3 colFinal = U_PSCustom0.x * (colCool * scale + 5 * colHot * pow(scale, 10));
	
	float3 starDir = normalize(IN.o_CustomParam.xyz); 	
	float factor = max(dot(starDir,-U_MainLightDirection.xyz),0.9f);
	factor = (factor - 0.9f)/0.1f;
	factor = factor * factor;
	
	psOut.o_Color = float4(colFinal, scale*(1.0f - factor ));
	psOut.o_Color.rgb *= U_PSCustom0.y;
	
	
	//psOut.o_Color.rgb = lerp(psOut.o_Color.rgb, U_FogColor.rgb, IN.o_CustomParam.w); //差值雾效
	float overdrawlerp = ceil(U_PS_Time.w);
	psOut.o_Color = psOut.o_Color * (1.0 - overdrawlerp) + float4(U_PS_Time.w * overdrawlerp, U_PS_Time.w * overdrawlerp, U_PS_Time.w * overdrawlerp, overdrawlerp);	
	return psOut;
}
