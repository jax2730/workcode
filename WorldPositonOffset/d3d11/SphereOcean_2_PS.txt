float getSceneDepth(in float4 clipPos, in float2 uv, out float dd)
{
	float4 tempPos = float4(uv, 0.0, 1.0);
	tempPos.z = S_Depth.Sample(S_DepthSamplerState, uv).r;
	
	tempPos.x = clipPos.x;
	tempPos.y = clipPos.y;
	
	float4 viewSpacePos;
	viewSpacePos.x = dot(U_InvProjectMatrix[0], tempPos);
	viewSpacePos.y = dot(U_InvProjectMatrix[1], tempPos);
	viewSpacePos.z = dot(U_InvProjectMatrix[2], tempPos);
	viewSpacePos.w = dot(U_InvProjectMatrix[3], tempPos);
	float depth = viewSpacePos.z / viewSpacePos.w;
	float3 vvv = viewSpacePos.xyz / viewSpacePos.w;
	dd = length(vvv);
	return depth;
}

//Dx only need to sample one times.
float calcSceneDepth(in float4 clipPos, out float dd)
{	
	float2 screenTC = clipPos.xy / clipPos.w;
	screenTC.x = (screenTC.x + 1.0) * 0.5;//+ U_ViewParam.z;
	screenTC.y = 1.0 - (screenTC.y + 1.0) * 0.5;//+ U_ViewParam.w;
	screenTC *= U_ViewParam.xy;	
	float depth = getSceneDepth(clipPos / clipPos.w, screenTC, dd);
	
	return depth;
}

float3 mod289(float3 x)
{
    return x - floor(x / 289.0) * 289.0;
}

float2 mod289(float2 x)
{
    return x - floor(x / 289.0) * 289.0;
}

float3 permute(float3 x)
{
    return mod289((x * 34.0 + 1.0) * x);
}

float3 taylorInvSqrt(float3 r)
{
    return 1.79284291400159 - 0.85373472095314 * r;
}

float snoise(float2 v)
{
    const float4 C = float4( 0.211324865405187,  // (3.0-sqrt(3.0))/6.0
                             0.366025403784439,  // 0.5*(sqrt(3.0)-1.0)
                            -0.577350269189626,  // -1.0 + 2.0 * C.x
                             0.024390243902439); // 1.0 / 41.0
    // First corner
    float2 i  = floor(v + dot(v, C.yy));
    float2 x0 = v -   i + dot(i, C.xx);

    // Other corners
    float2 i1;
    i1.x = step(x0.y, x0.x);
    i1.y = 1.0 - i1.x;

    // x1 = x0 - i1  + 1.0 * C.xx;
    // x2 = x0 - 1.0 + 2.0 * C.xx;
    float2 x1 = x0 + C.xx - i1;
    float2 x2 = x0 + C.zz;

    // Permutations
    i = mod289(i); // Avoid truncation effects in permutation
    float3 p =
      permute(permute(i.y + float3(0.0, i1.y, 1.0))
                    + i.x + float3(0.0, i1.x, 1.0));

    float3 m = max(0.5 - float3(dot(x0, x0), dot(x1, x1), dot(x2, x2)), 0.0);
    m = m * m;
    m = m * m;

    // Gradients: 41 points uniformly over a line, mapped onto a diamond.
    // The ring size 17*17 = 289 is close to a multiple of 41 (41*7 = 287)
    float3 x = 2.0 * frac(p * C.www) - 1.0;
    float3 h = abs(x) - 0.5;
    float3 ox = floor(x + 0.5);
    float3 a0 = x - ox;

    // Normalise gradients implicitly by scaling m
    m *= taylorInvSqrt(a0 * a0 + h * h);

    // Compute final noise value at P
    float3 g;
    g.x = a0.x * x0.x + h.x * x0.y;
    g.y = a0.y * x1.x + h.y * x1.y;
    g.z = a0.z * x2.x + h.z * x2.y;
    return 130.0 * dot(m, g);
}

float4 LowCostTripSample(Texture2D tex,SamplerState s,float3 worldPos,float3 normal, float texScale, float speed)
{
    float3 absN = abs(normal);
    float3 pn = 0;
    if (absN.x > absN.y && absN.x > absN.z) pn.x = 1.0;
    else if (absN.y > absN.z) pn.y = 1.0;
    else pn.z = 1.0;

    float scale = max(texScale * 0.1f, 0.0001);
    float scroll = speed * 0.01;

    float2 uv;
    if(pn.x > 0.5) 
        uv = worldPos.yz * scale + float2(scroll, 0.0); // Side (YZ) - Scroll along Y
    else if(pn.y > 0.5)
        uv = worldPos.xz * scale + float2(0.0, scroll); // Top (XZ) - Scroll along Z
    else
        uv = worldPos.xy * scale + float2(scroll, 0.0); // Front (XY) - Scroll along X

    return tex.Sample(s, uv);
}

float4 HighCostTripSample(Texture2D tex,SamplerState s,float3 worldPos,float3 normal,float texScale, float speed)
{
	float2 directionX = float2(1.0f, 0.0f) * speed * 0.01;
    float2 directionY = float2(0.0f, 1.0f) * speed * 0.01;
    float2 directionZ = float2(1.0f, 0.0f) * speed * 0.01;

	float scale = max(texScale * 0.1f,0.0001);

	float2 uv1 = worldPos.xy * scale + directionZ;
	float2 uv2 = worldPos.xz * scale + directionY;
	float2 uv3 = worldPos.yz * scale + directionX;
	float4 r1 = tex.Sample(s,uv1);
	float4 r2 = tex.Sample(s,uv2);
	float4 r3 = tex.Sample(s,uv3);

	float nl1 =  max((abs(dot(normal,float3(0,0,1))) - 0.56) * 20.0f,0);
	float nl2 =  max((abs(dot(normal,float3(0,1,0))) - 0.56) * 20.0f,0);
	float nl3 =  max((abs(dot(normal,float3(1,0,0))) - 0.56) * 20.0f,0);

	float l1 = saturate(nl1 - nl2 - nl3);
	float l2 = saturate(nl3 - nl2 - nl1 + 1.679f);

	float4 result1 = lerp(r2,r1,l1);
	float4 result2 = lerp(result1,r3,l2);

	return result2;
}

//The UV coord in the center gets less stretch than the one in the edge
float2 centrateUV(float2 uv){
	return saturate((float2)0.25f - 9.f * pow((uv - (float2)0.5),6));
}

float2 screenEdgeUVSmooth(float2 uvIn,float refractionBorder){
	float uvOutBound = 1.f - 2.f*refractionBorder;
	float2 proccessUV = centrateUV(uvIn);
	return lerp((uvIn * uvOutBound) + float2(refractionBorder,refractionBorder),uvIn,max(proccessUV.x,proccessUV.y));
}

float SmoothTransition(float x, float threshold, float steepness)
{
    return 1.0 / (1.0 + exp(steepness * (x - threshold)));
}

float4 SampleFlowingNormal(Texture2D tex, Texture2D noiseTex, SamplerState ss, float2 baseUV, float2 noiseUV, float time, float period, float speed)
{
    // 1. 使用噪声生成随机的 Flow 方向
    // 采样两次噪声以获得二维向量
    // float n1 = saturate(noiseTex.Sample(ss,noiseUV * 0.1).a);
    // float n2 = saturate(noiseTex.Sample(ss,noiseUV * 0.2).a); 
    float2 flowDir = noiseUV; 
    
    // 2. 计算两个相位的时间变量
    float halfPeriod = period * 0.5;
	
    float t1 = fmod(time * speed, period);
    float t2 = fmod(time * speed + halfPeriod, period);
    
    // 3. 计算 UV 偏移 (Bias * Speed)
    float2 offset1 = flowDir * t1;
    float2 offset2 = flowDir * t2;
    
    // 4. 分别采样
    float4 c1 = tex.Sample(ss, (baseUV - offset1 * 0.2));
    float4 c2 = tex.Sample(ss, (baseUV - offset2 * 0.2));
    
    // 5. 混合 (三角波权重)
    float lerpFactor = abs(halfPeriod - t1) / halfPeriod;
    return lerp(c1, c2, lerpFactor);
}

// Using for create caustic effect
float3 ReconstructWorldPos(float2 screenPos, float depth, float4 U_InvViewProjectMatrix[4])
{
	 #ifndef REVERSE_DEPTH
        depth = 1.0 - depth;
    #endif

    // Using depth and screen position to create clip space position
    float4 clipPos = float4(screenPos, depth, 1.0);
    
    // Using inverse view-projection matrix to get world position
    float4 worldPos;
    worldPos.x = dot(U_InvViewProjectMatrix[0], clipPos);
    worldPos.y = dot(U_InvViewProjectMatrix[1], clipPos);
    worldPos.z = dot(U_InvViewProjectMatrix[2], clipPos);
    worldPos.w = dot(U_InvViewProjectMatrix[3], clipPos);
    
    return worldPos.xyz / worldPos.w;
}

// Based on Shadertoy "Water Caustics" by David Hoskins
// https://www.shadertoy.com/view/MdKXDm
float3 ProceduralCaustic(float2 uv, float time)
{
    float3 k = float3(uv * 20.0, time * 2.0); 
    float3x3 m = float3x3(-2,-1,2, 3,-2,1, 1,2,2);

    k = mul(k, m);
    float v1 = length(0.5 - frac(k * 0.1));

    k = mul(k, m);
    float v2 = length(0.5 - frac(k * 0.025));

    k = mul(k, m);
    float v3 = length(0.5 - frac(k * 0.01));
    
    float val = pow(min(min(v1, v2), v3), 7.0) * 5.0;
    val = clamp(val, 0.0, 5.0);
    
    return float3(val, val, val);
}
void AdvectUV3_DetailWave(float2 uv, float time, float speed, float2 uvScale, float offsetStrength, inout float3 weight, inout float2 uv1, inout float2 uv2, inout float2 uv3)
{
    float period = 100.0;
    float fixedSpeed = floor(speed * period + 0.5) / period;

    float3 fracTime = frac(time * fixedSpeed + float3(0.0, 1.0, 2.0) / 3.0);

    // 2. Weight logic: -cos ensures 0 weight at phase reset (0 and 1)
    weight = (-cos(2*PI*fracTime) * 0.5 + 0.5) * 2/3;

    // 3. Directional Flow: Offset based on phase (-0.5 to 0.5)
    float2 flowDir = float2(0.0, 1.0);
    float3 flowOffset = (fracTime - 0.5) * offsetStrength;

    uv1 = uv * uvScale + flowDir * flowOffset.x;
    uv2 = uv * uvScale + float2(0.3,0.7) + flowDir * flowOffset.y;
    uv3 = uv * uvScale + float2(0.7,0.3) + flowDir * flowOffset.z; 
}


float2 CalculateDerivate(float3 color, float normalScale, float derivateScale)
{

    float2 normalXY = (color.xy - 0.5) * 2.0 * normalScale;

    float2 derivate = normalXY / max(0.0001, derivateScale);
    return derivate;
}


float2 SampleAdvectNormals(Texture2D tex, SamplerState ss, float2 planePos, float time, float speed, float2 uvScale, float offsetStrength, float normalScale, float derivateScale)
{
    float3 weight;
    float2 uv1, uv2, uv3;
    
    AdvectUV3_DetailWave(planePos, time, speed, uvScale, offsetStrength, weight, uv1, uv2, uv3);
    
    float3 c1 = tex.Sample(ss, uv1).rgb;
    float3 c2 = tex.Sample(ss, uv2).rgb;
    float3 c3 = tex.Sample(ss, uv3).rgb;

    float2 d1 = CalculateDerivate(c1, normalScale, derivateScale);
    float2 d2 = CalculateDerivate(c2, normalScale, derivateScale);
    float2 d3 = CalculateDerivate(c3, normalScale, derivateScale);
    
    return d1 * weight.x + d2 * weight.y + d3 * weight.z;
}
