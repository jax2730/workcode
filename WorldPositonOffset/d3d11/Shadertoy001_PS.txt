#include<Common_Uniform_PS.inl>

struct PS_INPUT
{
    float4 i_PosInClip	   : SV_POSITION;
    float4 i_UV            : TEXCOORD0;
    float4 i_ViewDir       : TEXCOORD1;
    float4 i_PositionHCS   : TEXCOORD2;
};

float4 linear2sRGB(float4 linearColor)
{
    return float4(pow(linearColor.rgb, 1.f / 2.2), linearColor.a);
}

float4 main(PS_INPUT IN): SV_TARGET
{
    float2 uv = (IN.i_UV.xy  - 0.5) * 2.0;
    float time = U_PS_Time.x;
    float t = time * .1 + (.25 + .05 * sin(time * .1))/(length(uv + .07)) * 2.2;
	float si = sin(t);
	float co = cos(t);
    float2x2 ma = {co, -si, si, co};

    float v1 = 0;
    float v2 = 0; 
    float v3 = 0;
    float s = 0.0;
    for (int i = 0; i < 90; i++)
    {
        float3 p = s * float3(uv, 0.0);
        p.xy = mul(p.xy, ma);
        p += float3(0.22, 0.3, s-1.5-sin(time * 0.13) * 0.1);
        for (int i = 0; i < 8; i++)
        {
            p = abs(p) / dot(p, p) - 0.659;
        }
        v1 += dot(p,p) * .0015 * (1.8 + sin(length(uv * 13.0) + .5  - time * .2));
        v2 += dot(p,p) * .0013 * (1.5 + sin(length(uv * 14.5) + 1.2 - time * .3));
        v3 += length(p.xy*10.) * .0003;
        s  += .035;
    }
    float len = length(uv);
    v1 *= smoothstep(.7, .0, len);
    v2 *= smoothstep(.5, .0, len);
    v3 *= smoothstep(.9, .0, len);
    float3 col = float3( v3 * (1.5 + sin(time * .2) * .4),
            (v1 + v3) * .3,
            v2) + smoothstep(0.2, .0, len) * .85 + smoothstep(.0, .6, v3) * .3;
    return float4(min(pow(abs(col), float3(1.2, 1.2, 1.2)), 1.0), 1.0);
    //return float4(1.0, 0.0, 0.0, 1.0);
}
