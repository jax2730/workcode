#define AnimeSpeed U_PSCustom1
#define Color1 U_PSCustom2
#define Color2 U_PSCustom3

static int MAX_ITER = 10;

float field(float3 p, float s)
{
	float accum = s * 0.25;
	float prev = 0.0;
	float tw = 0.0;
	float f = exp(-0.2);
	float w = 1.0;
	for (int i = 0; i < 10; ++i) 
  	{
		float mag = dot(p, p);
		p = abs(p) / mag + float3(-0.5, -0.4, -1.487);
		w *= f;
		float dif = abs(mag - prev);
		accum += w * exp(-9.025 * dif * dif);
		tw += w;
		prev = mag;
	}
	return max(0.0, 5.2 * accum / tw - 0.65);
}

struct PS_INPUT
{
    float4 i_PosInClip		: SV_POSITION;
    float4 i_UV				: TEXCOORD0;
    float4 i_ViewDir        : TEXCOORD1;
    float4 i_PositionHCS    : TEXCOORD2;
};

float4 main(PS_INPUT IN): SV_TARGET
{
    float time = U_PS_Time.x * AnimeSpeed;
	float4 freqs = float4(0.45, 0.4, 0.15, 0.9);

    // first layer of the kaliset fractal
	float3 p = normalize(IN.i_ViewDir.xyz) + float3(0.8, -1.3, 0.0) + U_PS_CameraPosition.xyz * 0.0001f;
	float3 TSIN = sin(time / float3(32, 24, 64));
	float2 v2 = (1.0 - exp((abs(IN.i_ViewDir.xy) - 1.0) * 6.0));
	float v = v2.x * v2.y;

	p += 0.45 * TSIN;

	float t = field(p, freqs.z);
	
    // second layer of the kaliset fractal
	float3 p2 = normalize(IN.i_ViewDir.xyz) + float3(2.0, -1.3, -1.0) + U_PS_CameraPosition.xyz * 0.0001f;
	p2 += 0.16 * TSIN;
	
	float t2 = field(p2, freqs.w);
	float3 c2 = Color2 * lerp(0.8, 0.2, v) * t2 * t2 * t2;
	
	// add stars (source: https://glslsandbox.com/e#6904.0)
	// float4 starColour = 0.0;
	// starColour += starLayer(p.xy, time); // add first layer of stars
	// starColour += starLayer(p2.xy, time); // add second layer of stars

	float3 colour = Color1 * (lerp(freqs.w - 0.3, 1.0, v) * t * t) + c2;// + starColour;
	return float4(colour.xyz, 1.0);
}
