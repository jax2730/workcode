#include <Common_Uniform_VS.inl>
#include <Common_Function.inl>

float4	U_WorldViewProjectMatrix[4];
  
float4	U_VSCustom0[5];

struct VS_INPUT
{
	float4 i_Offset		: POSITION0;
};

struct VS_OUTPUT
{
	float4 o_Position	: SV_POSITION;
	float4 o_UV			: TEXCOORD0;
#ifdef PARABOLOID_MAPPING
	float	o_ClipDepth			: TEXCOORD12;
#endif
};

VS_OUTPUT main(VS_INPUT IN)
{
	VS_OUTPUT vsOut = (VS_OUTPUT)0;
   	
	float3 volumeLightPos = U_VSCustom0[2].xyz;
	volumeLightPos += U_VSCustom0[1].xyz * U_VSCustom0[0].x * U_VSCustom0[1].w; 
	float2 offset = float2(U_VSCustom0[0].y * IN.i_Offset.x , U_VSCustom0[0].y * IN.i_Offset.y);
	volumeLightPos = volumeLightPos + offset.x * U_VSCustom0[3].xyz + offset.y * U_VSCustom0[4].xyz;
	
	float4 vPos;	
	vPos.xyz = volumeLightPos;
	vPos.w = 1.0f;	
		
	vsOut.o_Position.x = dot(U_WorldViewProjectMatrix[0], vPos);
	vsOut.o_Position.y = dot(U_WorldViewProjectMatrix[1], vPos);
	vsOut.o_Position.z = dot(U_WorldViewProjectMatrix[2], vPos);
	vsOut.o_Position.w = dot(U_WorldViewProjectMatrix[3], vPos);

#ifdef PARABOLOID_MAPPING
	vsOut.o_ClipDepth = vsOut.o_PosInClip.z;
	ParaboloidMapping(vsOut.o_PosInClip, U_ShadowLightDepthMulAdd.x, U_ShadowLightDepthMulAdd.z, U_ShadowLightDepthMulAdd.w);
#endif
	
	vsOut.o_UV.xy = IN.i_Offset.xy;	
	
	return vsOut;
}
