#include "SSSRCommon.h"

// Inputs
Texture2D<float> S_Depth;
#define g_depth_buffer S_Depth
Texture2D<float> S_2DExt0;
#define g_roughness S_2DExt0
Texture2D<float4> S_GBuffer1;
#define g_normal S_GBuffer1
Texture2D<float3> S_2DExt1;
#define g_average_radiance S_2DExt1
Texture2D<float4> S_2DExt2;
#define g_in_radiance S_2DExt2
Texture2D<float> S_2DExt3;
#define g_in_variance S_2DExt3
Texture2D<float> S_2DExt4;
#define g_in_sample_count S_2DExt4

// Samplers
SamplerState S_2DExt0SamplerState;
#define g_linear_sampler S_2DExt0SamplerState

// Outputs
RWTexture2D<float4> S_2DExt5;
#define g_out_radiance S_2DExt5
RWTexture2D<float> S_2DExt6;
#define g_out_variance S_2DExt6
RWTexture2D<float> S_2DExt7;
#define g_out_sample_count S_2DExt7

StructuredBuffer<uint> B_CSCustom0;
#define g_denoiser_tile_list B_CSCustom0

float3 FFX_DNSR_Reflections_SampleAverageRadiance(float2 uv) {
    return g_average_radiance.SampleLevel(g_linear_sampler, uv, 0.0f).xyz;
}

float FFX_DNSR_Reflections_LoadRoughness(int2 pixel_coordinate) {
    float roughness = g_roughness.Load(int3(pixel_coordinate, 0));
	return roughness * roughness;
}

void FFX_DNSR_Reflections_LoadNeighborhood(
    int2 pixel_coordinate,
    out float3 radiance,
    out float variance,
    out float3 normal,
    out float depth,
    int2 screen_size) {
    
    radiance = g_in_radiance.Load(int3(pixel_coordinate, 0)).xyz;
    variance = g_in_variance.Load(int3(pixel_coordinate, 0)).x;

	normal = normalize(2.0 * (float3) g_normal.Load(int3(pixel_coordinate, 0)) - 1.0);

    float2 uv = (pixel_coordinate.xy + (0.5f).xx) / float2(screen_size.xy);
    depth = FFX_DNSR_Reflections_GetLinearDepth(uv, g_depth_buffer.Load(int3(pixel_coordinate, 0)));
}

void FFX_DNSR_Reflections_StorePrefilteredReflections(int2 pixel_coordinate, float3 radiance, float variance) {
    g_out_radiance[pixel_coordinate] = radiance.xyzz;
    g_out_variance[pixel_coordinate] = variance.x;
}

#include "ffx_denoiser_reflections_config.h"
#include "ffx_denoiser_reflections_common.h"
#include "ffx_denoiser_reflections_prefilter.h"

[numthreads(8, 8, 1)]
void main(int2 group_thread_id      : SV_GroupThreadID,
                uint group_index    : SV_GroupIndex,
                uint    group_id    : SV_GroupID) {
    uint  packed_coords               = g_denoiser_tile_list[group_id];
    int2  dispatch_thread_id          = int2(packed_coords & 0xffffu, (packed_coords >> 16) & 0xffffu) + group_thread_id;
    int2  dispatch_group_id           = dispatch_thread_id / 8;
    uint2 remapped_group_thread_id    = FFX_DNSR_Reflections_RemapLane8x8(group_index);
    uint2 remapped_dispatch_thread_id = dispatch_group_id * 8 + remapped_group_thread_id;

	FFX_DNSR_Reflections_Prefilter(remapped_dispatch_thread_id, remapped_group_thread_id, g_sssr_constants.g_buffer_dimensions);
}
