Texture2D S_Diffuse;
SamplerState S_DiffuseSamplerState;
Texture2D S_Bump;
SamplerState S_BumpSamplerState;
Texture2D S_Specular;
SamplerState S_SpecularSamplerState;


float4 	U_PSCustom0;
float4 	U_PSCustom1;

#ifdef RIPPLE
	Texture2D	S_Ripple;
	SamplerState S_RippleSamplerState;
	
	float4 ComputeRipple(float2 uv, float t)
	{
		//波纹贴图采样，并把采样的高度值扩展到-1到1。
		float4 ripple = S_Ripple.Sample(S_RippleSamplerState, uv);
		float gloss = ripple.y;
		ripple.yz = ripple.yz * 2.0 - 1.0;
		//获取波纹的时间,从A通道获取不同的波纹时间,
		float dropFrac = frac(ripple.a + t);
		//把时间限制在R通道内	
		float timeFrac = dropFrac - 1.0 + ripple.x;
		//做淡出处理
		float dropFactor = 1.0f-saturate( dropFrac);
		//计算最终的高度，用一个sin计算出随时间的振幅	 
		float final = dropFactor* sin(clamp(timeFrac * 9.0, 0.0, 4.0) * 3.1415926);
		return float4(ripple.yz * final, 1.0, (final ) * 0.2);
	}
#endif

#ifdef CLUSTER_POINT_LIGHT

#endif

half GetLuminance(half3 color)
{
    const half3 lumWeight = half3(0.2125f, 0.7154f, 0.0721f);
    return dot(color, lumWeight);
}

struct PS_INPUT
{ 
    float4	i_PosInClip			: SV_POSITION;
	float4	i_Color				: TEXCOORD0;
	float4	i_UV0				: TEXCOORD1;
	float3	i_Normal			: TEXCOORD2;
	float3	i_Tangent			: TEXCOORD3;
	float3	i_Binormal			: TEXCOORD4;
	float4	i_EyeDir			: TEXCOORD5;
	float4  i_FogParams 		: TEXCOORD6;
	float4  i_WPos 				: TEXCOORD7;
#ifdef PARABOLOID_MAPPING
	float	i_ClipDepth			: TEXCOORD12;
#endif
};

float4 main(PS_INPUT IN) : SV_TARGET
{
	
#ifdef PARABOLOID_MAPPING
	clip(IN.i_ClipDepth);
#endif

	float2 uvTemp = IN.i_UV0.xy;
	uvTemp.xy *= U_PSCustom0.yz;
	
	half3 nn = normalize(IN.i_Normal);

	#ifdef USESHADOWMAP
		float4 shadowSpacePos[SHADOWSPLITE];
		ProjectToLightSpace(shadowSpacePos, IN.i_WPos.xyz, nn, IN.i_UV0.z);
	#endif

	half3 tn = normalize(IN.i_Tangent);
	half3 bn = normalize(IN.i_Binormal);

	tn = normalize(tn - nn * dot(tn, nn));
	bn = normalize(bn - nn * dot(bn, nn));

	half3x3 tbn = 
	{
		tn.x, bn.x, nn.x,
		tn.y, bn.y, nn.y,
		tn.z, bn.z, nn.z 
	};
		
	half4 bumpColor = S_Bump.Sample(S_BumpSamplerState, uvTemp.xy);
	half3 bump = bumpColor.rgb * 2.0 - 1.0;
	//当有涟漪的时候，减弱模型原本的法线贴图效果，或者直接砍掉法线贴图的计算
	#ifdef RIPPLE
		float s = clamp(dot(nn, float3(0.0,1.0,0.0)), 0.0, 1.0) ;
		s = pow(s, 5.0);	
		bump.xy  *= (1.0 - s * 0.9f);
		double time = U_PS_Time.x;
		float time01 = U_PS_Time.y * 0.1591549; //U_PS_Time.y / PI / 2
		//涟漪整体的透明度使用sin函数进行0-1的变化，当透明度变为0的时候，坐标进行偏移，以减轻雨滴的重复感。单张贴图透明度为0的时候会没有涟漪，所以采样两次，交替显示
		float2 rUV = IN.i_WPos.xz * 0.25f;
		
		float rOffset = floor(U_PS_Time.x * 0.5f) * 0.1f;
		float4 ripple = ComputeRipple(rUV + rOffset, time01 * 3.0f);
		
		rUV = IN.i_WPos.xz * 0.25f + 0.25f;
		rOffset = floor(U_PS_Time.x * 0.5f + 0.25f) * 0.1f;
		float4 ripple2 = ComputeRipple(rUV + float2(rOffset, -rOffset), time01 * 3.0f);
		
		float rLerp1 =  sin(time01 * 3.1415);
		float rLerp2 =  1.0 - rLerp1;
		bump.xy += rLerp1 * ripple.xy * s;
		bump.xy += rLerp2 * ripple2.xy * s;
	#endif
	bump.z = sqrt(1.0 - (bump.x * bump.x + bump.y * bump.y));
	half3 normal = normalize(mul(tbn, bump));
	
	half3 vn = normalize(IN.i_EyeDir.xyz);
	
	half3 cDiffuseAcc = 0;
	half3 cSpecularAcc = 0;

	half Glossiness = U_PSCustom0.x;
	//有涟漪的时候，给材质的高光整体加0.4		
	#ifdef RIPPLE
		Glossiness *= 16.0;
	#endif	
	Glossiness += 1e-6;

	half4 diffuse_color = S_Diffuse.Sample(S_DiffuseSamplerState, uvTemp.xy);
	diffuse_color.rgb = pow(diffuse_color.rgb, 2.2f);
	diffuse_color.a *= IN.i_Color.a;
	diffuse_color.rgb *= IN.i_Color.rgb;
		
	half fCoverLerp = dot(normal,U_PS_GravityDirection.xyz);

	fCoverLerp = clamp(fCoverLerp, 0.0h, 1.0h);
	fCoverLerp *= U_WeatherParam0.w;
	diffuse_color.rgb = lerp(diffuse_color.rgb, U_WeatherParam0.xyz, fCoverLerp);

	half4 specular_color = S_Specular.Sample(S_SpecularSamplerState, uvTemp.xy);
	specular_color.rgb = pow(specular_color.rgb, 2.2f);
	
	half3 ln = normalize(-U_MainLightDirection.xyz);	
	half fNdotL = dot(ln, normal);
	half fNdotE = dot(vn,normal);
	
	half4 litV = lit(fNdotL, dot(normalize(vn + ln), normal), Glossiness);

	half3 sunColor = U_MainLightColor.xyz;
	cDiffuseAcc += sunColor * litV.y;
	
	cSpecularAcc += sunColor * litV.z;
	half3 ambient = lerp( U_GroundLightColor.xyz, U_SkyLightColor.xyz, 0.5h * normal.y + 0.5h);

	half4 color = 0;
	half3 ambientAcc = diffuse_color.xyz * ambient;
	half3 cDiffuse = diffuse_color.xyz * cDiffuseAcc;

	half sidealpha = 1-abs(IN.i_UV0.y-0.5)*2;
	sidealpha = U_PSCustom1.z * sidealpha + U_PSCustom1.w - U_PSCustom1.w * sidealpha;
	sidealpha = saturate(sidealpha);
	diffuse_color.a *= sidealpha;

	half fresnel = U_PSCustom1.x + (1-U_PSCustom1.x) * pow(1.001h - saturate(fNdotE),U_PSCustom1.y);
	cSpecularAcc *= fresnel;
	
	//specular_color.rgb *= cSpecularAcc;
	
	color.xyz += cDiffuse;
	color.a = diffuse_color.a;
	color.xyz += specular_color.rgb * cSpecularAcc * U_PSCustom0.w * U_MainLightColor.a;
	
	#ifdef USESHADOWMAP
		float shadowFactor = calcShadowFactor(
			shadowSpacePos, 
			IN.i_UV0.z);

	shadowFactor = min(shadowFactor, 1.0);
	color.rgb *= shadowFactor;
#endif

	color.xyz += ambientAcc;

#ifdef CLUSTER_POINT_LIGHT
	PointLightAndSpecularOutput shadowLightResult = calculateShadowPointLightAndSpecular(
		IN.i_WPos, normal, vn, Glossiness);
	color.rgb += shadowLightResult.color;
	color.rgb += shadowLightResult.specular * U_PSCustom0.w * U_MainLightColor.a;
	
    PointLightAndSpecularOutput lightResult = 
        calculateClusterPointLightAndSpecular(IN.i_PosInClip.xy,
                                              IN.i_WPos.xyz,
                                              normal,
                                              Glossiness,
                                              vn);
    color.rgb += lightResult.color;
    color.rgb += lightResult.specular * U_PSCustom0.w * U_MainLightColor.a;
#endif

	
	color.rgb = color.rgb * IN.i_FogParams.a + IN.i_FogParams.rgb;
	color.rgb = pow(color.rgb, 1.0 / 2.2);
	
	#ifdef EditorHelp
		color.rgb = color.rgb*0.001 + float3(1.0,0.0,0.0);
		color.a = 1.0;
	#endif
	float overdrawlerp = ceil(U_PS_Time.w);
	color = color * (1.0 - overdrawlerp) + float4(U_PS_Time.w * overdrawlerp, U_PS_Time.w * overdrawlerp, U_PS_Time.w * overdrawlerp, overdrawlerp);
	return color;
}
