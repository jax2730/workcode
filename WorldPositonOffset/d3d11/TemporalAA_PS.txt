#define HistoryReprojection 1
#define FilterHistory 1
#define FilterInput 1
#define UseYCoCg 0


#define ColorBuffer S_2DExt0
#define ColorSampler S_2DExt0SamplerState

#define DepthBuffer S_2DExt1
#define DepthSampler S_2DExt1SamplerState

#define HistoryBuffer S_2DExt2
#define HistorySampler S_2DExt2SamplerState

#define ViewPortRange	U_PSCustom0.xy
#define Reprojection	U_PSCustom1
#define Feedback		U_PSCustom2.x
#define EnableHDR		U_PSCustom2.y
#define FilterWeights	U_PSCustom3

Texture2D       ColorBuffer;
SamplerState    ColorSampler;

Texture2D       DepthBuffer;
SamplerState    DepthSampler;

Texture2D       HistoryBuffer;
SamplerState    HistorySampler;

float4 U_PSCustom0;		// viewPortRange
float4 U_PSCustom1[4];
float4 U_PSCustom2;
float4 U_PSCustom3[9];

float max3(float3 v)
{
	return max(v.x, max(v.y, v.z));
}

float rcp(float x)
{
	if (abs(x) < 1e-4f)
		return 10000.0f;
	return 1.0f / x;
}

float3 tonemap(float3 c)
{
    return c * rcp(1.0f + max3(c));
}

float4 tonemap(float4 c)
{
    return float4(c.rgb * rcp(1.0f + max3(c.rgb)), c.a);
}

float3 tonemap(float w, float3 c)
{
    return c * (w * rcp(1.0f + max3(c)));
}

float4 tonemap(float w, float4 c)
{
    return float4(c.rgb * (w * rcp(1.0f + max3(c.rgb))), c.a);
}

float3 untonemap(float3 c)
{
    const float epsilon = 1.0f / 65504.0f;
    return c * rcp(max(epsilon, 1.0f - max3(c)));
}

float3 RGB_YCoCg(float3 c)
{
    float Y  = dot(c.rgb, float3( 1, 2,  1) * 0.25f);
    float Co = dot(c.rgb, float3( 2, 0, -2) * 0.25f);
    float Cg = dot(c.rgb, float3(-1, 2, -1) * 0.25f);
    return float3(Y, Co, Cg);
}

float3 YCoCg_RGB(float3 c)
{
    float Y  = c.x;
    float Co = c.y;
    float Cg = c.z;
    float r = Y + Co - Cg;
    float g = Y + Cg;
    float b = Y - Co - Cg;
    return float3(r, g, b);
}

float4 clipToBox(float3 boxmin, float3 boxmax, float4 c, float4 h)
{
	const float epsilon = 0.0001f;

	float4 r = c - h;
	float3 ir = 1.0f / (epsilon + r.rgb);
	float3 rmax = (boxmax - h.rgb) * ir;
	float3 rmin = (boxmin - h.rgb) * ir;
	float3 imin = min(rmax, rmin);
	return h + r * saturate(max3(imin));
}

float4 SampleTextureCatmullRom(Texture2D tex, SamplerState samp, float2 uv, float2 texSize)
{
    float2 samplePos = uv * texSize;
    float2 texPos1 = floor(samplePos - 0.5f) + 0.5f;

    float2 f  = samplePos - texPos1;
    float2 f2 = f * f;
    float2 f3 = f2 * f;

    float2 w0 = f2 - 0.5f * (f3 + f);
    float2 w1 = 1.5f * f3 - 2.5f * f2 + 1.0f;
    float2 w3 = 0.5f * (f3 - f2);
    float2 w2 = 1.0f - w0 - w1 - w3;

    float2 w12 = w1 + w2;

    float2 texPos0  = texPos1 - float2(1.0f, 1.0f);
    float2 texPos3  = texPos1 + float2(2.0f, 2.0f);
    float2 texPos12 = texPos1 + w2 / w12;

    float2 invTexSize = 1.0f / texSize;
    texPos0  *= invTexSize;
    texPos3  *= invTexSize;
    texPos12 *= invTexSize;

    float k0 = w12.x * w0.y;
    float k1 = w0.x  * w12.y;
    float k2 = w12.x * w12.y;
    float k3 = w3.x  * w12.y;
    float k4 = w12.x * w3.y;

    float4 s[5];
    s[0] = tex.SampleLevel(samp, float2(texPos12.x, texPos0.y), 0.0f);
    s[1] = tex.SampleLevel(samp, float2(texPos0.x,  texPos12.y), 0.0f);
    s[2] = tex.SampleLevel(samp, float2(texPos12.x, texPos12.y), 0.0f);
    s[3] = tex.SampleLevel(samp, float2(texPos3.x,  texPos12.y), 0.0f);
    s[4] = tex.SampleLevel(samp, float2(texPos12.x, texPos3.y), 0.0f);

    float4 result = k0 * s[0] + k1 * s[1] + k2 * s[2] + k3 * s[3] + k4 * s[4];
    result *= rcp(k0 + k1 + k2 + k3 + k4);

    result = max(result, float4(0, 0, 0, 0));
    return result;
}

struct PS_INPUT
{
    float4 i_Position	: SV_POSITION; 
	float4 i_UV	        : TEXCOORD0;
	float4 i_ScreenPos	: TEXCOORD1;
};

struct PS_OUTPUT
{
	float4 o_Color	: SV_TARGET;
};

PS_OUTPUT main(PS_INPUT IN)
{
    PS_OUTPUT psOut = (PS_OUTPUT)0;

	float4 pos = IN.i_Position;
	float2 i_uv = IN.i_UV.xy * ViewPortRange.xy;
	float4 uv = i_uv.xyxy;

#if HistoryReprojection
	float4x4 reprojectionMatrix;
	reprojectionMatrix[0] = Reprojection[0];
	reprojectionMatrix[1] = Reprojection[1];
	reprojectionMatrix[2] = Reprojection[2];
	reprojectionMatrix[3] = Reprojection[3];

	float depth = DepthBuffer.Sample(DepthSampler, uv.xy).r;

	uv.zw = float2(uv.z * 2.0 - 1.0, 1.0 - uv.w * 2.0);
	float4 q = mul(reprojectionMatrix, float4(uv.zw, depth, 1.0));
	if (q.w != 0.0) {
		q.xyz /= q.w;
		uv.zw = float2(0.5 + q.x * 0.5, 0.5 - q.y * 0.5);

		if (uv.z < 0.f || uv.z > 1.f || uv.w < 0.f || uv.w > 1.f) {
			psOut.o_Color = ColorBuffer.Sample(ColorSampler, uv.xy);
			return psOut;
		}
	}
#endif

	// read center color and history samples
	float4 history;
#if FilterHistory
	float2 texSize = float2(0.0f, 0.0f);
	HistoryBuffer.GetDimensions(texSize.x, texSize.y);
	history = SampleTextureCatmullRom(HistoryBuffer, HistorySampler, uv.zw, texSize);
#else
	history = HistoryBuffer.Sample(HistorySampler, uv.zw);
#endif

	float2 size;
	ColorBuffer.GetDimensions(size.x, size.y);
    float2 p = (floor(uv.xy * size) + 0.5) / size;
    float4 filtered = ColorBuffer.Sample(ColorSampler, p);

	float3 s[9];
	{
		s[0] = ColorBuffer.SampleLevel(ColorSampler, p, 0.0, int2(-1, -1)).rgb;
		s[1] = ColorBuffer.SampleLevel(ColorSampler, p, 0.0, int2(0, -1)).rgb;
		s[2] = ColorBuffer.SampleLevel(ColorSampler, p, 0.0, int2(1, -1)).rgb;
		s[3] = ColorBuffer.SampleLevel(ColorSampler, p, 0.0, int2(-1, 0)).rgb;
		s[4] = filtered.rgb;
		s[5] = ColorBuffer.SampleLevel(ColorSampler, p, 0.0, int2(1, 0)).rgb;
		s[6] = ColorBuffer.SampleLevel(ColorSampler, p, 0.0, int2(-1, 1)).rgb;
		s[7] = ColorBuffer.SampleLevel(ColorSampler, p, 0.0, int2(0, 1)).rgb;
		s[8] = ColorBuffer.SampleLevel(ColorSampler, p, 0.0, int2(1, 1)).rgb;
	}

	const int j = 0;
	const float confidence = 1.0;
#if FilterInput
	// unjitter/filter input
	filtered = float4(0, 0, 0, filtered.a);
	for (int i = 0; i < 9; ++i)
	{
		float w = FilterWeights[i][j];
		filtered.rgb += s[i] * w;
	}
	filtered.rgb = max(filtered.rgb, float3(0, 0, 0));
#endif

	// build the history clamping box
	{
#if UseYCoCg
		history.rgb = RGB_YCoCg(history.rgb);
		filtered.rgb = RGB_YCoCg(filtered.rgb);
		for (int i = 0; i < 9; i++)
		{
			s[i] = RGB_YCoCg(s[i]);
		}
#endif
		float3 boxmin;
		float3 boxmax;

		boxmin = min(s[4], min(min(s[1], s[3]), min(s[5], s[7])));
		boxmax = max(s[4], max(max(s[1], s[3]), max(s[5], s[7])));
		float3 box9min = min(boxmin, min(min(s[0], s[2]), min(s[6], s[8])));
		float3 box9max = max(boxmax, max(max(s[0], s[2]), max(s[6], s[8])));
		boxmin = (boxmin + box9min) * 0.5f;
		boxmax = (boxmax + box9max) * 0.5f;

		// history clamp
		history = clipToBox(boxmin, boxmax, filtered, history);
	}

	float alpha = Feedback * confidence;

#if UseYCoCg
	filtered.rgb = YCoCg_RGB(filtered.rgb);
	history.rgb = YCoCg_RGB(history.rgb);
#endif

	if (EnableHDR > 0.5f)
	{
		// Tonemap before mixing
		filtered.rgb = tonemap(filtered.rgb);
		history.rgb = tonemap(history.rgb);
	}

	// Combine history and current frame
	float4 result = lerp(history, filtered, alpha);

	if (EnableHDR > 0.5f)
	{
		// Untonemap result
		result.rgb = untonemap(result.rgb);
	}

	result.a = 1.0f;

	psOut.o_Color = result;
	return psOut;
}
