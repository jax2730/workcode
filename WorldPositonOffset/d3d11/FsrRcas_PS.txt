AF4 FsrRcasLoadF(ASU2 p) 
{
    return ColorBuffer.Load(int3(p, 0));
}

void FsrRcasInputF(inout AF1 r, inout AF1 g, inout AF1 b)
{}

struct PS_INPUT
{
    float4 i_Position	: SV_POSITION; 
	float4 i_UV	        : TEXCOORD0;
	float4 i_ScreenPos	: TEXCOORD1;
};

struct PS_OUTPUT
{
	float4 o_Color	: SV_TARGET;
};

PS_OUTPUT main(PS_INPUT IN)
{
    PS_OUTPUT psOut = (PS_OUTPUT)0;

    float2 i_uv = IN.i_UV.xy;
	uint2 i_pos = uint2(ViewResolution.xy * i_uv);

    FsrRcasF(psOut.o_Color.r, psOut.o_Color.g, psOut.o_Color.b,
            i_pos, RcasCon);
    psOut.o_Color.a = 1.0;
    return psOut;
}
