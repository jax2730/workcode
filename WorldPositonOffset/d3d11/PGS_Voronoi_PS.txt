float3 hash3(float3 uvw)
{
    return frac(sin(float3(dot(uvw, float3(127.1, 311.7, 74.7)), dot(uvw, float3(269.5, 183.3, 246.1)), dot(uvw, float3(113.5, 271.9, 124.6)))) * 43758.5453);
}
