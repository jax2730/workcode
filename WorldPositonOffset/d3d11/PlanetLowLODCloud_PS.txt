#include <Common_Uniform_PS.inl>

#include <Light_Function_PS.inl>
#include <LandformPBRLighting.inl>
#include <EnvironmentLight_Function.inl>
#include <SphericalFog_Function.inl>
struct PS_INPUT
{
	float4 o_Position	: SV_POSITION;

	float2 o_UV			: TEXCOORD0;
	float3 o_WorldPos	: TEXCOORD1;
	float3 o_WorldNormal: NORMAL0;
	float3 o_3DUV		: TEXCOORD2;
};

struct PS_OUTPUT
{
	float4 FragColor : SV_TARGET;
};

float4 U_PSGeneralRegister0;
float4 U_PSGeneralRegister1;
float4 U_PSGeneralRegister2;
float4 U_PSGeneralRegister3;
float4 U_MaterialCustomParam1;

Texture2D	S_Diffuse;
SamplerState S_DiffuseSamplerState;

Texture3D S_Noise3DTex128; // /asset/Editor/CloudNoise3DTexR8_128
SamplerState S_Noise3DTex128SamplerState;

Texture2D S_Custom;
SamplerState S_CustomSamplerState;

#define UVScale U_PSGeneralRegister0.x
#define FlowSpeed U_PSGeneralRegister0.y
#define HDRFactor U_PSGeneralRegister0.z
#define metallic U_PSGeneralRegister0.w

#define roughness U_PSGeneralRegister1.x
#define ambientFactor U_PSGeneralRegister1.y
#define noiseOffsetScale U_PSGeneralRegister1.z
#define cloudDir U_PSGeneralRegister2.xyz

#define altitusPercent U_PSGeneralRegister3.x
#define DissolveStrength U_PSGeneralRegister3.y

PS_OUTPUT main(PS_INPUT IN)
{	
	PS_OUTPUT psOut = (PS_OUTPUT)0.f;

	float3 flowUV = IN.o_3DUV.xyz;

    float4x4 thresholdMatrix =
    {  1.0 / 17.0,  9.0 / 17.0,  3.0 / 17.0, 11.0 / 17.0,
        13.0 / 17.0,  5.0 / 17.0, 15.0 / 17.0,  7.0 / 17.0,
        4.0 / 17.0, 12.0 / 17.0,  2.0 / 17.0, 10.0 / 17.0,
        16.0 / 17.0,  8.0 / 17.0, 14.0 / 17.0,  6.0 / 17.0
    };
    float2 posInClip = IN.o_Position.xy;

    posInClip.xy = posInClip.xy/U_MaterialCustomParam1.z;
    
    int xindex = fmod(posInClip.x, 4);
    int yindex = fmod(posInClip.y, 4);
    
    float alpha_dissolve = thresholdMatrix[xindex][yindex];//* _RowAccess[yindex];
        
    if(alpha_dissolve < DissolveStrength)
        discard;

	float noise = S_Noise3DTex128.Sample(S_Noise3DTex128SamplerState,
                                         flowUV.xyz).x;
 
	flowUV += U_PS_Time.x * FlowSpeed * 0.01f * cloudDir;
	flowUV += noise * noiseOffsetScale;
	noise = S_Noise3DTex128.Sample(S_Noise3DTex128SamplerState,
                                   flowUV.xyz * UVScale).x;
	float3 baseMap = S_Diffuse.Sample(S_DiffuseSamplerState, float2(noise, 0.5f));
	baseMap = pow(baseMap.rgb, 2.2f);

	//Light
    float3 w_viewDir = normalize(U_PS_CameraPosition.xyz - IN.o_WorldPos.xyz);
    float3 w_normal = normalize(IN.o_WorldNormal.xyz);

    float3 directionLightLo = pbr_lighting_fresnel_weighted(w_normal,
                                            normalize(-U_MainLightDirection.xyz),
                                            w_viewDir,
                                            baseMap,
                                            metallic,
                                            roughness,
                                            U_PBR_MainLightColor.rgb);
    

    float3 pointLightLo = float3(0.f, 0.f,0.f);
#ifdef CLUSTER_POINT_LIGHT
    pointLightLo = calculateClusterPBRPointLight(IN.o_Position.xy,
                                                 IN.o_WorldPos.xyz,
                                                 w_normal,
                                                 w_viewDir,
                                                 baseMap,
                                                 roughness,
                                                 metallic);
#endif
    float3 environmentLo = baseMap * ambientFactor;
    
	psOut.FragColor.rgb = (directionLightLo + pointLightLo + environmentLo) * HDRFactor;

    //Scale noise value to [0,1]
    float normalNoise = saturate((noise - 0.3294117647058824f) * 1.655844155844156f);

	float edgeBlendAlpha = dot(w_normal, w_viewDir);

    const float blendAngle = 0.2f;
    const float scaleBlendAnble = 1.0f / blendAngle;
    edgeBlendAlpha = clamp(edgeBlendAlpha, 0.0f, blendAngle);
    edgeBlendAlpha *= scaleBlendAnble;
    
    psOut.FragColor.a = normalNoise * edgeBlendAlpha;

    
    float blendFactor = smoothstep(0.f, 1.f, altitusPercent);
    psOut.FragColor.a *= blendFactor;

    psOut.FragColor.rgb = pow(psOut.FragColor.rgb, 0.4545);
    float4 sphFog = ComputeSphFogBlocked(IN.o_WorldPos,U_PS_CameraPosition.xyz, -w_viewDir,IN.o_Position.xy);
    psOut.FragColor.rgb = lerp(psOut.FragColor.rgb, sphFog.rgb, sphFog.a);
    
	return psOut;
}
