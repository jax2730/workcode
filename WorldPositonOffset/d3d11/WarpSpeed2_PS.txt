#include<Common_Uniform_PS.inl>
#define SCREEN_EFFECT 0
#define NUM_LAYERS 4.0

struct PS_INPUT
{
    float4 i_PosInClip	   : SV_POSITION;
    float4 i_UV            : TEXCOORD0;
	float4 i_ViewDir       : TEXCOORD1;
	float4 i_PositionHCS   : TEXCOORD2;
};

float4 linear2sRGB(float4 linearColor)
{
    return float4(pow(linearColor.rgb, 1.0f / 2.2), linearColor.a);
}


float4 main(PS_INPUT IN): SV_TARGET{
    
    float time = (U_PS_Time.x-2.0) * 58.0;
    float2 uv = float2(IN.i_UV.xy) * 2.0f - 1.0f;
    
    float s = 0.0, v = 0.0;
	float3 col = float3(0.0,0.0,0.0);
    float3 init = float3(sin(time *.0032)*0.3, 0.35 - cos(time *0.005)*0.3, time * 0.002);
	for (int r = 0; r < 100; r++) 
	{
		float3 p = init + s * float3(uv, 0.05);
		p.z = frac(p.z);
		for (int i=0; i < 10; i++)	p = abs(p * 2.04) / dot(p, p) - 0.9;
		v += pow(dot(p, p), 0.7) * 0.06;
		col +=  float3(v * 0.2+0.4, 12.0-s*2.0, 0.1 + v * 1.0) * v * 0.00003;

		s += 0.025;
	}
	
    return linear2sRGB(float4(clamp(col, 0.0, 1.0), 1.0));
}
