			
#include <Common_Uniform_VS.inl>
#include <Common_Function.inl> 
																																																					

uniform	float4	U_FogColorNear;
uniform	float4	U_FogColorFar;
uniform	float4  U_FogParam;
uniform	float4  U_FogRampParam;	

uniform float4 	U_VSCustom0[44]; //VSCBPreMaterial																
uniform float4  U_VSCustom1[16 * 8]; //VSCBPreDrawCall
																								
struct VS_INPUT
{
	float3	Pos : POSITION0;
};
																										
struct VS_OUTPUT																						
{																										
	float4	Col : TEXCOORD0;																			
	float4	Tc  : TEXCOORD1;																								
	float4	Pos : SV_POSITION;	

#ifdef 	DOUBLETEX
	float2	Tc1  : TEXCOORD2;
#endif
	float4	o_Fog : TEXCOORD3;
};																										
																										
																										
																										
static float4	vecTileParam = U_VSCustom0[33 + 0];													
static float4	vecTileRot = U_VSCustom0[33 + 1];													
static float4	vecTCMoveMax = U_VSCustom0[33 + 2];													
static float4	vecTCMoveMin = U_VSCustom0[33 + 3];													
static float4	vecTCScaleMax = U_VSCustom0[33 + 4];												
static float4	vecTCScaleMin = U_VSCustom0[33 + 5];												
static float4	vecTCRotateMax = U_VSCustom0[33 + 6];												
static float4	vecTCRotateMin = U_VSCustom0[33 + 7];												
static float4	vecTCSelect = U_VSCustom0[33 + 8];													
static float4	vecPlayRate = U_VSCustom0[33 + 9];													
static float4	vecPlayRand = U_VSCustom0[33 + 10];													
																										
void CalParam(in float  _Time																			
	, in float2 _Tc																						
	, in float4 _Rand0																					
	, in float4 _Rand1																					
	, out float4 Color_																					
	, out float4 Scale_																					
	, out float4 Pose_																					
	, out float4 Tc_)																					
{																										
																										
	float	Time = _Time;																				
	float4	Rand0 = _Rand0;																				
	float4	Rand1 = _Rand1;																				
																										
	float4	KFTime = frac(Time * vecPlayRate + Rand0 * vecPlayRand) * 9.99;								
	int4	KFIndex = int4(KFTime);																		
	float4 	KFLerp = frac(KFTime);																		
																										
	float4	Color0 = U_VSCustom0[0 + KFIndex.x];													
	float4	Color1 = U_VSCustom0[0 + KFIndex.x + 1];												
	Color_ = lerp(Color0, Color1, KFLerp.x);															
																										
	float 	Alpha0 = U_VSCustom0[0 + KFIndex.y].a;													
	float 	Alpha1 = U_VSCustom0[0 + KFIndex.y + 1].a;												
	Color_.a = lerp(Alpha0, Alpha1, KFLerp.y);															
																										
	float4 	Scale0 = U_VSCustom0[11 + KFIndex.z];													
	float4 	Scale1 = U_VSCustom0[11 + KFIndex.z + 1];												
	Scale_ = lerp(Scale0, Scale1, KFLerp.z);															
																										
	float4 	Pose0 = U_VSCustom0[22 + KFIndex.w];													
	float4 	Pose1 = U_VSCustom0[22 + KFIndex.w + 1];												
	Pose_ = lerp(Pose0, Pose1, KFLerp.w);																
																										
	float2	tc = _Tc;																					
																										
																										
	float 	Tile = frac(Time * vecTileParam.x + Rand1.w * vecTileParam.y);								
	float 	Row = vecTileParam.w;																		
	float 	Col = vecTileParam.z;																		
	float	nIdx = floor(Tile*Row*Col);																	
	float 	nRow = floor(nIdx / Col);																	
	float 	nCol = nIdx - nRow*Col;																		
	float 	fRow = nRow / Row;																			
	float 	fCol = nCol / Col;																			
	float2 	Tc0 = float2(tc.x, tc.y) * vecTileRot.x +													
		float2(1.0 - tc.y, tc.x) * vecTileRot.y +														
		float2(1.0 - tc.x, 1.0 - tc.y) * vecTileRot.z +													
		float2(tc.y, 1.0 - tc.x) * vecTileRot.w;														
	float2	TcTile = float2(fCol, fRow) + Tc0*float2(1.0 / Col, 1.0 / Row);								
																										
	float4 	TcMove = lerp(vecTCMoveMax, vecTCMoveMin, Rand1.x);											
	float4 	TcScale = lerp(vecTCScaleMax, vecTCScaleMin, Rand1.y);										
	float4 	TcRotate = lerp(vecTCRotateMax, vecTCRotateMin, Rand1.z);									
																										
	float2 	TcTran = tc - float2(0.5, 0.5);																
	TcTran *= TcScale.zw + TcScale.xy * Time;															
	TcTran += TcMove.zw + TcMove.xy * Time;																
	float RotDeg = TcRotate.x * Time + TcRotate.y;														
	float2 RotMatR0 = float2(+cos(RotDeg), -sin(RotDeg));												
	float2 RotMatR1 = float2(+sin(RotDeg), +cos(RotDeg));												
	TcTran = float2(dot(TcTran, RotMatR0), dot(TcTran, RotMatR1));										
	TcTran += float2(0.5, 0.5);																			
																										
	Tc_.xy = lerp(TcTile, TcTran, vecTCSelect.x);	

#ifdef DOUBLETEX
	Tc_.zw = lerp(TcTran, TcTile, vecTCSelect.x);
#else
	Tc_.zw = 0.0;
#endif		
}																										
																										
float3 EulaRot(float3 Pose, float3 iVtx)																
{																										
	float3 oVtx;																						
																										
	float 	sy = sin(Pose.x);	float 	cy = cos(Pose.x);												
	float 	sp = sin(Pose.y);	float 	cp = cos(Pose.y);												
	float 	sr = sin(Pose.z);	float 	cr = cos(Pose.z);												
																										
																										
	float3 eula_x = float3(sr*sp*sy + cr*cy, cr*sp*sy - sr*cy, cp*sy);									
	float3 eula_y = float3(sr*cp, cr*cp, -sp);															
	float3 eula_z = float3(sr*sp*cy - cr*sy, cr*sp*cy + sr*sy, cp*cy);									
																										
	oVtx = float3(dot(eula_x, iVtx), dot(eula_y, iVtx), dot(eula_z, iVtx));								
																										
	return oVtx;																						
}


VS_OUTPUT main(VS_INPUT iVtx)
{
	VS_OUTPUT oVtx;

	float2	InExt = iVtx.Pos.xy;
	float 	InID = iVtx.Pos.z;

	int		Id = int(InID);

	float4	mWVPC0 = U_VSCustom1[Id * 8 + 0];
	float4	mWVPC1 = U_VSCustom1[Id * 8 + 1];
	float4	mWVPC2 = U_VSCustom1[Id * 8 + 2];
	float4	mWVPC3 = U_VSCustom1[Id * 8 + 3];
	float4	color  = U_VSCustom1[Id * 8 + 4];
	float4	rand0  = U_VSCustom1[Id * 8 + 5];
	float4	rand1  = U_VSCustom1[Id * 8 + 6];
	float4	time_scale = U_VSCustom1[Id * 8 + 7];

	float	time  = time_scale.x;
	float	scale = time_scale.y;

	float   rad_sin = sin(InExt.x);
	float   rad_cos = cos(InExt.x);
	float2	tce		= float2(rad_sin + 1.0, rad_cos + 1.0)*0.5;
	float2	tcc		= float2(0.5, 0.5);
	float2	tc		= lerp(tce, tcc, InExt.y);

	float4 Pose;
	float4 Scale;

	float4 outTC;

	CalParam(time, tc, rand0, rand1, oVtx.Col, Scale, Pose, outTC);

	oVtx.Tc.xy = outTC.xy;
	oVtx.Tc.zw = tc;
	
	#ifdef DOUBLETEX
		oVtx.Tc1 = outTC.zw;
	#endif
	
	oVtx.Col *= color;
	Scale *= scale;
	float 	radii  = lerp(Scale.x, Scale.z, InExt.y);
	float	height = lerp(Scale.y, Scale.w, InExt.y);
	float3 	pos	   = float3(rad_sin*radii, height, rad_cos*radii);

	pos = EulaRot(Pose.xyz, pos);

	float4 wpos = float4(pos, 1.0);
	
	float4	vCamPos	=	U_VS_CameraPosition;
	float3 camToWorld =  vCamPos.xyz - wpos.xyz;

	float4x4 gravityRotatePosMatrix;
	gravityRotatePosMatrix[0] = U_VS_GravityRotationMatrix[0];
	gravityRotatePosMatrix[1] = U_VS_GravityRotationMatrix[1];
	gravityRotatePosMatrix[2] = U_VS_GravityRotationMatrix[2];
	gravityRotatePosMatrix[3] = U_VS_GravityRotationMatrix[3];
	camToWorld = mul(gravityRotatePosMatrix,float4(camToWorld.xyz,0.0f)).xyz;

	oVtx.o_Fog = ComputeFogColor(camToWorld, U_FogParam.xy, U_FogParam.zw, U_FogRampParam, U_FogColorNear, U_FogColorFar);

	oVtx.Pos.x = dot(mWVPC0, wpos);
	oVtx.Pos.y = dot(mWVPC1, wpos);
	oVtx.Pos.z = dot(mWVPC2, wpos);
	oVtx.Pos.w = dot(mWVPC3, wpos);

	//oVtx.Col.a *= CalFade( wpos.xyz );

	return oVtx;
}
