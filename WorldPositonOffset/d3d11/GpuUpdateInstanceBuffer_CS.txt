#include <Common_GpuCull_CS.inl>

#define thread_group_size_x 32u
#define thread_group_size_y 1u

cbuffer UniformBlock_Private_CS
{
    uint U_CSCustom0; //Count
	uint U_CSCustom1; //operate type
};

StructuredBuffer<InstanceBlockUpdate> B_CSCustom0;				
RWStructuredBuffer<InstanceBlock> B_CSCustom1;

[numthreads(thread_group_size_x, thread_group_size_y, 1)]
void main(uint3 threadIDInGroup : SV_GroupThreadID,
			uint3 groupID : SV_GroupID,
			uint groupIndex : SV_GroupIndex,
			uint3 dispatchThreadID : SV_DispatchThreadID)
{

	uint thread_index = dispatchThreadID.x;
	if (thread_index >= U_CSCustom0)
        return;
	
	InstanceBlockUpdate blockUpdate = B_CSCustom0[thread_index];
	
	if (U_CSCustom1 == 0u)
	{
		B_CSCustom1[blockUpdate.memIdx] = blockUpdate.data;
	}
	else
	{
		for (uint i = 0u; i < 12u; i ++)
			B_CSCustom1[blockUpdate.memIdx].Transform[i] = blockUpdate.data.Transform[i];

		for (uint i = 0u; i < 12u; i ++)
			B_CSCustom1[blockUpdate.memIdx].InvTransposeTransform[i] = blockUpdate.data.InvTransposeTransform[i];

		B_CSCustom1[blockUpdate.memIdx].Flip = blockUpdate.data.Flip;
		B_CSCustom1[blockUpdate.memIdx].Pipeline = blockUpdate.data.Pipeline;	
		B_CSCustom1[blockUpdate.memIdx].Begin = blockUpdate.data.Begin;
		B_CSCustom1[blockUpdate.memIdx].Lens = blockUpdate.data.Lens;		
	}

}
