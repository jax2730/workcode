void IncrementRayCounter(uint value, out uint original_value) {
    InterlockedAdd(g_ray_counter[0], value, original_value);
}

void IncrementDenoiserTileCounter(out uint original_value) {
    InterlockedAdd(g_ray_counter[2], 1, original_value);
}

void StoreRay(int index, uint2 ray_coord, bool copy_horizontal, bool copy_vertical, bool copy_diagonal) {
    g_ray_list[index] = PackRayCoords(ray_coord, copy_horizontal, copy_vertical, copy_diagonal); // Store out pixel to trace
}

void StoreDenoiserTile(int index, uint2 tile_coord) {
    g_denoiser_tile_list[index] = ((tile_coord.y& 0xffffu) << 16) | ((tile_coord.x& 0xffffu) << 0); // Store out pixel to trace
}

bool IsReflectiveSurface(int2 pixel_coordinate, float roughness)
{
#ifdef REVERSE_DEPTH
	const float far_plane = 0.0f;
	return g_depth_buffer[pixel_coordinate] > far_plane;
#else
    const float far_plane = 1.0f;
    return g_depth_buffer[pixel_coordinate] < far_plane;
#endif
}

bool IsBaseRay(uint2 dispatch_thread_id, uint samples_per_quad) {
    switch (samples_per_quad) {
    case 1:
        return ((dispatch_thread_id.x & 1) | (dispatch_thread_id.y & 1)) == 0; // Deactivates 3 out of 4 rays
    case 2:
        return (dispatch_thread_id.x & 1) == (dispatch_thread_id.y & 1); // Deactivates 2 out of 4 rays. Keeps diagonal.
    default: // case 4:
        return true;
    }
}

groupshared uint g_TileCount;

float3 SampleEnvironmentMap(uint2 dispatch_thread_id, float roughness, float3 world_space_normal, ShadingModelInfo shadingModel) {
	float2 uv = (dispatch_thread_id + 0.5) * g_sssr_constants.g_inv_buffer_dimensions;
    float  z = g_depth_buffer.Load(int3(dispatch_thread_id, 0));
    float3 screen_uv_space_ray_origin = float3(uv, z);
    float3 view_space_ray = FFX_DNSR_Reflections_ScreenSpaceToViewSpace(screen_uv_space_ray_origin);
    float3 view_space_ray_direction = normalize(view_space_ray);
	float3 view_space_surface_normal = mul(float4(world_space_normal, 0), g_sssr_constants.g_view).xyz;
    float3 view_space_reflected_direction = reflect(view_space_ray_direction, view_space_surface_normal);
	float3 world_space_reflected_direction = mul(float4(view_space_reflected_direction, 0), g_sssr_constants.g_inv_view).xyz;

	float environmentExplosure = U_PBR_Environment_Explosure.x;

	float3 skyColor = shadingModel.useRoleMainLight ? U_RoleSkyLightColor.rgb : U_PBR_SkyLightColor.rgb;
	float3 groundColor = shadingModel.useRoleMainLight ? U_RoleGroundLightColor.rgb : U_PBR_GroundLightColor.rgb;
	float blendThreshold = shadingModel.useRoleMainLight ? U_PS_AmbientBlendThreshold.z : U_PS_AmbientBlendThreshold.y;

	float3 ambientColor = ComputeAmbientColorInternal(world_space_normal, groundColor, skyColor, blendThreshold, U_PS_GravityDirection.xyz);

    const float mip_count = 10;
	int cubemapMipLevel, cubemapW, cubemapH;
	g_environment_map.GetDimensions(0, cubemapW, cubemapH, cubemapMipLevel);
	float lod = clamp(cubemapMipLevel * roughness, 0.0f, cubemapMipLevel);
	float4 rgbmColor = g_environment_map.SampleLevel(g_environment_map_sampler, world_space_reflected_direction, lod);
	float3 ibl_GlossyIrradiance = rgbmColor.xyz * rgbmColor.w * 10;
	return ambientColor * environmentExplosure * ibl_GlossyIrradiance;
}

// Shared memory for counting rays and storing intermediate results
groupshared uint g_need_ray_mask_lo; // Lower 32 bits of the need ray mask
groupshared uint g_need_ray_mask_hi; // Upper 32 bits of the need ray mask
groupshared uint g_require_copy_mask_lo; // Lower 32 bits of the copy mask
groupshared uint g_require_copy_mask_hi; // Upper 32 bits of the copy mask
groupshared uint g_base_ray_index;

void ClassifyTiles(uint2 dispatch_thread_id, uint2 group_thread_id, uint dispatch_group_thread_id, float roughness, 
	float3 world_normal, ShadingModelInfo shadingModel, float percentageRoughness)
{
	g_TileCount = 0;
	g_need_ray_mask_lo = 0;
	g_need_ray_mask_hi = 0;
	g_require_copy_mask_lo = 0;
	g_require_copy_mask_hi = 0;
	g_base_ray_index = 0;

	bool is_first_thread_of_group = dispatch_group_thread_id == 0;
	uint thread_index = dispatch_group_thread_id;

    // First we figure out on a per thread basis if we need to shoot a reflection ray.
    // Disable offscreen pixels
	bool needs_ray = !(dispatch_thread_id.x >= g_sssr_constants.g_buffer_dimensions.x || dispatch_thread_id.y >= g_sssr_constants.g_buffer_dimensions.y);

    // Dont shoot a ray on very rough surfaces.
	bool is_reflective_model = !shadingModel.emptyGBuffer && shadingModel.useEnvironmentMap && shadingModel.lightModel == LightModel_PBR_HDR_Environment;
	needs_ray = needs_ray && is_reflective_model;
	bool is_reflective_surface = IsReflectiveSurface(dispatch_thread_id, roughness);
	bool is_glossy_reflection = FFX_DNSR_Reflections_IsGlossyReflection(roughness);
	needs_ray = needs_ray && is_glossy_reflection && is_reflective_surface;

    // Also we dont need to run the denoiser on mirror reflections.
	bool needs_denoiser = needs_ray && !FFX_DNSR_Reflections_IsMirrorReflection(roughness);

    // Decide which ray to keep
	bool is_base_ray = IsBaseRay(dispatch_thread_id, g_sssr_constants.g_samples_per_quad);
	needs_ray = needs_ray && (!needs_denoiser || is_base_ray); // Make sure to not deactivate mirror reflection rays.

	if (g_sssr_constants.g_temporal_variance_guided_tracing_enabled && needs_denoiser && !needs_ray)
	{
		bool has_temporal_variance = g_variance_history.Load(int3(dispatch_thread_id, 0)) > g_sssr_constants.g_temporal_variance_threshold;
		needs_ray = needs_ray || has_temporal_variance;
	}

	GroupMemoryBarrierWithGroupSync(); // Wait until g_TileCount is cleared - allow some computations before and after

    // Now we know for each thread if it needs to shoot a ray and wether or not a denoiser pass has to run on this pixel.

	if (is_glossy_reflection && is_reflective_surface)
		InterlockedAdd(g_TileCount, 1);

    // Next we have to figure out for which pixels that ray is creating the values for. Thus, if we have to copy its value horizontal, vertical or across.
	bool require_copy = !needs_ray && needs_denoiser; // Our pixel only requires a copy if we want to run a denoiser on it but don't want to shoot a ray for it.
	if (thread_index < 32)
	{
		if (require_copy)
			InterlockedOr(g_require_copy_mask_lo, 1 << thread_index);
		if (needs_ray)
			InterlockedOr(g_need_ray_mask_lo, 1 << thread_index);
	}
	else
	{
		if (require_copy)
			InterlockedOr(g_require_copy_mask_hi, 1 << (thread_index - 32));
		if (needs_ray)
			InterlockedOr(g_need_ray_mask_hi, 1 << (thread_index - 32));
	}
	bool copy_horizontal = false;
	bool copy_vertical = false;
	bool copy_diagonal = false;

	GroupMemoryBarrierWithGroupSync();

	if (is_base_ray)
	{
		if (g_sssr_constants.g_samples_per_quad != 4)
		{
		    // use group shared mask
			uint horizontal_index = thread_index ^ 1;
			uint h_mask = horizontal_index < 32 ? g_require_copy_mask_lo : g_require_copy_mask_hi;
			horizontal_index = horizontal_index < 32 ? horizontal_index : horizontal_index - 32;
			copy_horizontal = (h_mask & (1 << horizontal_index)) != 0;
		}
		if (g_sssr_constants.g_samples_per_quad == 1)
		{
		    // use group shared mask
			uint vertical_index = thread_index ^ 2;
			uint v_mask = vertical_index < 32 ? g_require_copy_mask_lo : g_require_copy_mask_hi;
			vertical_index = vertical_index < 32 ? vertical_index : vertical_index - 32;
			copy_vertical = (v_mask & (1 << vertical_index)) != 0;
		}
		if (g_sssr_constants.g_samples_per_quad == 1)
		{
			uint diagonal_index = thread_index ^ 3;
			uint d_mask = diagonal_index < 32 ? g_require_copy_mask_lo : g_require_copy_mask_hi;
			diagonal_index = diagonal_index < 32 ? diagonal_index : diagonal_index - 32;
			copy_diagonal = (d_mask & (1 << diagonal_index)) != 0;
		}
	}

    // Thus, we need to compact the rays and append them all at once to the ray list.
	uint local_ray_index_in_group = //WavePrefixCountBits(needs_ray);
	    thread_index < 32
		    ? countbits(g_need_ray_mask_lo & ((1 << thread_index) - 1)) // Count bits before this thread in the lower mask
		    : countbits(g_need_ray_mask_lo) + countbits(g_need_ray_mask_hi & ((1 << (thread_index - 32)) - 1)); // Count bits in both masks

	if (is_first_thread_of_group)
	{
		uint group_ray_count = countbits(g_need_ray_mask_lo) + countbits(g_need_ray_mask_hi); //WaveActiveCountBits(needs_ray);
		IncrementRayCounter(group_ray_count, g_base_ray_index);
	}

	GroupMemoryBarrierWithGroupSync();

	uint base_ray_index = g_base_ray_index;
	if (needs_ray)
	{
		int ray_index = base_ray_index + local_ray_index_in_group;
		StoreRay(ray_index, dispatch_thread_id, copy_horizontal, copy_vertical, copy_diagonal);
	}

	float4 intersection_output = 0;
	if (is_reflective_surface && !is_glossy_reflection)
	{
        // Fall back to environment map without preparing a ray
		intersection_output.xyz = SampleEnvironmentMap(dispatch_thread_id, percentageRoughness, world_normal, shadingModel);		
	}
	
    g_intersection_output[dispatch_thread_id] = intersection_output;

	GroupMemoryBarrierWithGroupSync(); // Wait until g_TileCount

	if (all(group_thread_id == 0) && g_TileCount > 0)
	{
		uint tile_offset;
		IncrementDenoiserTileCounter(tile_offset);
		StoreDenoiserTile(tile_offset, dispatch_thread_id.xy);
	}
}

[numthreads(8, 8, 1)]
void main(uint2 group_id : SV_GroupID, uint group_index : SV_GroupIndex) {
    uint2 group_thread_id = FFX_DNSR_Reflections_RemapLane8x8(group_index); // Remap lanes to ensure four neighboring lanes are arranged in a quad pattern
    uint2 dispatch_thread_id = group_id * 8 + group_thread_id;

	float4 gBuffer2Data = g_roughness.Load(int3(dispatch_thread_id, 0));
	// Roughness is authored as perceptual roughness; as is convention,
    // convert to material roughness by squaring the perceptual roughness [2].
	float percetageRoughness = gBuffer2Data.y;
	float roughness = percetageRoughness * percetageRoughness;
	ShadingModelInfo shadingModel = DecodeShadingModel(gBuffer2Data.w);
	float3 world_normal = DecodeNormal(g_normal.Load(int3(dispatch_thread_id, 0)).xyz);

	ClassifyTiles(dispatch_thread_id, group_thread_id, group_index, roughness, world_normal, shadingModel, percetageRoughness);

    // Extract only the channel containing the roughness to avoid loading all 4 channels in the follow up passes.
    g_extracted_roughness[dispatch_thread_id] = gBuffer2Data.y;
	g_extracted_normal[dispatch_thread_id] = world_normal * 0.5 + 0.5; // Store the normal for later use
}
