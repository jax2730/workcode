		float4 ComputeRipple(float2 uv, float t)
		{
			//波纹贴图采样，并把采样的高度值扩展到-1到1。
			float4 ripple = S_Ripple.Sample(S_RippleSamplerState, uv);
			float gloss = ripple.y;
			ripple.yz = ripple.yz * 2.0 - 1.0;
			//获取波纹的时间,从A通道获取不同的波纹时间,
			float dropFrac = frac(ripple.a + t);
			//把时间限制在R通道内	
			float timeFrac = dropFrac - 1.0 + ripple.x;
			//做淡出处理
			float dropFactor = 1.0f-saturate( dropFrac);
			//计算最终的高度，用一个sin计算出随时间的振幅	 
			float final = dropFactor* sin(clamp(timeFrac * 9.0, 0.0, 4.0) * 3.1415926);
			return float4(ripple.yz * final, 1.0, (final ) * 0.2);
		}

	float3 mod289(float3 x)
	{
		return x - floor(x / 289.0) * 289.0;
	}
	
	float2 mod289(float2 x)
	{
		return x - floor(x / 289.0) * 289.0;
	}
	
	float3 permute(float3 x)
	{
		return mod289((x * 34.0 + 1.0) * x);
	}
	
	float3 taylorInvSqrt(float3 r)
	{
		return 1.79284291400159 - 0.85373472095314 * r;
	}
	
	float snoise(float2 v)
	{
		const float4 C = float4( 0.211324865405187,  // (3.0-sqrt(3.0))/6.0
								0.366025403784439,  // 0.5*(sqrt(3.0)-1.0)
								-0.577350269189626,  // -1.0 + 2.0 * C.x
								0.024390243902439); // 1.0 / 41.0
		// First corner
		float2 i  = floor(v + dot(v, C.yy));
		float2 x0 = v -   i + dot(i, C.xx);
	
		// Other corners
		float2 i1;
		i1.x = step(x0.y, x0.x);
		i1.y = 1.0 - i1.x;
	
		// x1 = x0 - i1  + 1.0 * C.xx;
		// x2 = x0 - 1.0 + 2.0 * C.xx;
		float2 x1 = x0 + C.xx - i1;
		float2 x2 = x0 + C.zz;
	
		// Permutations
		i = mod289(i); // Avoid truncation effects in permutation
		float3 p =
		permute(permute(i.y + float3(0.0, i1.y, 1.0))
						+ i.x + float3(0.0, i1.x, 1.0));
	
		float3 m = max(0.5 - float3(dot(x0, x0), dot(x1, x1), dot(x2, x2)), 0.0);
		m = m * m;
		m = m * m;
	
		// Gradients: 41 points uniformly over a line, mapped onto a diamond.
		// The ring size 17*17 = 289 is close to a multiple of 41 (41*7 = 287)
		float3 x = 2.0 * frac(p * C.www) - 1.0;
		float3 h = abs(x) - 0.5;
		float3 ox = floor(x + 0.5);
		float3 a0 = x - ox;
	
		// Normalise gradients implicitly by scaling m
		m *= taylorInvSqrt(a0 * a0 + h * h);
	
		// Compute final noise value at P
		float3 g;
		g.x = a0.x * x0.x + h.x * x0.y;
		g.y = a0.y * x1.x + h.y * x1.y;
		g.z = a0.z * x2.x + h.z * x2.y;
		return 130.0 * dot(m, g);
	}
	
	float3 snoise_grad(float2 v)
	{
		const float4 C = float4( 0.211324865405187,  // (3.0-sqrt(3.0))/6.0
								0.366025403784439,  // 0.5*(sqrt(3.0)-1.0)
								-0.577350269189626,  // -1.0 + 2.0 * C.x
								0.024390243902439); // 1.0 / 41.0
		// First corner
		float2 i  = floor(v + dot(v, C.yy));
		float2 x0 = v -   i + dot(i, C.xx);
	
		// Other corners
		float2 i1;
		i1.x = step(x0.y, x0.x);
		i1.y = 1.0 - i1.x;
	
		// x1 = x0 - i1  + 1.0 * C.xx;
		// x2 = x0 - 1.0 + 2.0 * C.xx;
		float2 x1 = x0 + C.xx - i1;
		float2 x2 = x0 + C.zz;
	
		// Permutations
		i = mod289(i); // Avoid truncation effects in permutation
		float3 p =
		permute(permute(i.y + float3(0.0, i1.y, 1.0))
						+ i.x + float3(0.0, i1.x, 1.0));
	
		float3 m = max(0.5 - float3(dot(x0, x0), dot(x1, x1), dot(x2, x2)), 0.0);
		float3 m2 = m * m;
		float3 m3 = m2 * m;
		float3 m4 = m2 * m2;
	
		// Gradients: 41 points uniformly over a line, mapped onto a diamond.
		// The ring size 17*17 = 289 is close to a multiple of 41 (41*7 = 287)
		float3 x = 2.0 * frac(p * C.www) - 1.0;
		float3 h = abs(x) - 0.5;
		float3 ox = floor(x + 0.5);
		float3 a0 = x - ox;
	
		// Normalise gradients
		float3 norm = taylorInvSqrt(a0 * a0 + h * h);
		float2 g0 = float2(a0.x, h.x) * norm.x;
		float2 g1 = float2(a0.y, h.y) * norm.y;
		float2 g2 = float2(a0.z, h.z) * norm.z;
	
		// Compute noise and gradient at P
		float2 grad =
		-6.0 * m3.x * x0 * dot(x0, g0) + m4.x * g0 +
		-6.0 * m3.y * x1 * dot(x1, g1) + m4.y * g1 +
		-6.0 * m3.z * x2 * dot(x2, g2) + m4.z * g2;
		float3 px = float3(dot(x0, g0), dot(x1, g1), dot(x2, g2));
		return 130.0 * float3(grad, dot(m4, px));
	}

float3 fresnelSchlick(float cosTheta, float3 F0)
{
	return F0 + (1.0-F0) * pow(1.0-cosTheta, 5.0);
}

float distributionGGX(float3 N, float3 H, float roughness)
{
	float a = roughness * roughness;
	float a2 = a * a;
	float NdotH = max(dot(N, H), 0.0);
	float NdotH2 = NdotH * NdotH;
	float nom = a2;
	float denom = (NdotH2 * (a2 - 1.0) + 1.0);
	denom = 3.141592654 * denom * denom;
	denom = max(denom, 0.00001);
	return nom / denom;
}

float geometrySchlickGGX(float NdotV, float roughness)
{
	float r = roughness + 1.0;
	float k = (r * r) / 8.0;
	
	float nom = NdotV;
	float denom = NdotV * (1.0 - k) + k;
	
	return nom / denom;
}

float geometrySmith(float3 N, float3 V, float3 L, float roughness)
{
	float NdotV = max(dot(N, V), 0.0);
	float NdotL = max(dot(N, L), 0.0);
	float ggx2 = geometrySchlickGGX(NdotV, roughness);
	float ggx1 = geometrySchlickGGX(NdotL, roughness);
	
	return ggx2 * ggx1;
}

float3 envBRDFApprox(float3 specularColor, float roughness, float NoV)
{
	float4 c0 = float4(-1.0,	-0.0275,	-0.572,		0.022);
	float4 c1 = float4(1.0,		0.0425,		1.04,		-0.04);
	
	float4 r = roughness * c0 + c1;
	float a004 = min(r.x * r.x, exp2(-9.28 * NoV)) * r.x + r.y;
	float2 AB = float2(-1.04, 1.04) * a004 + r.zw;
	return max(specularColor * AB.x + AB.y, 1e-6f);
}

float3 thinTranslucency(float3 N, float3 L, float3 transmittanceColor)
{
	float transmittance = saturate(dot(-N, L) + 0.2);
	float fresnel = (1.0 - transmittance) * (1.0 - transmittance);
	fresnel *= fresnel;
	
	return transmittance * (1.0 - fresnel) * transmittanceColor;
}

PS_OUTPUT main(PS_INPUT IN,bool isFrontFacing : SV_IsFrontFace)
{
	PS_OUTPUT psOut = (PS_OUTPUT)0;

#ifdef PARABOLOID_MAPPING
	clip(IN.i_ClipDepth);
#endif

#ifdef SHADOWPASS
	#ifdef ALPHA_TEST
		float4 color = texDiffuse(IN.i_UV.xy);
		clip(color.a - U_MaterialDiffuse.a);
	#endif
	
	#ifdef VSM
		//VSM
		float depth_ShadowPass = IN.i_depth.z / IN.i_depth.w;
		depth_ShadowPass = depth_ShadowPass * 0.5 + 0.5;
		float moment1 = depth_ShadowPass;
		float moment2 = depth_ShadowPass * depth_ShadowPass;
		
		float dx = ddx(depth_ShadowPass);
		float dy = ddy(depth_ShadowPass);
		moment2 += 0.25*(dx*dx+dy*dy);
		psOut.color.r = moment1;
		psOut.color.g = moment2;
	#endif
		
	#ifdef ESM
		//ESM
		float linearDepth = distance(IN.i_depth.xyz, U_GlobalAgingColour.xyz) * U_GlobalAgingColour.w;
		psOut.color.r = exp(c * linearDepth);
	#endif
#else
	#ifdef MOTIONBLUR
		float2 vQX = IN.i_VelUV.xy;
	
		float fLenQX = length(vQX);
		vQX /= (fLenQX+0.01f);
		float fWeight = max(0.5f, min(fLenQX, 20.0f));
		vQX *= fWeight;
		psOut.color.xy = vQX;
		return psOut;
	#endif
	#ifdef NoColor
		psOut.color = float4(0.0, 0.0, 0.0, 0.0);
		return psOut;
	#endif
	float2 uv = IN.i_UV.xy;
	
	uv *= U_PSGeneralRegister2.x;

	#ifdef DISTORT
		uv += (U_PS_Time.z * float2(U_PSGeneralRegister1.z, U_PSGeneralRegister1.w));
	#else
		uv += frac(U_PS_Time.z * float2(U_PSGeneralRegister1.z, U_PSGeneralRegister1.w));
	#endif
	
	#ifdef DISTORT
		float _distortAmt = U_PSGeneralRegister2.y;
		float _distortScale = U_PSGeneralRegister2.z;
		float _distortSpeed = U_PSGeneralRegister1.y;
		float Time = U_PS_Time.z;
 
		float noise = snoise(uv * _distortScale + float2(Time *_distortSpeed, - Time *_distortSpeed));
		
		uv += noise * _distortAmt;
	#endif
	
	{
		//dither discard
		if(ComputeDitherCullMask(IN.i_PosInClip.xy,U_MaterialCustomParam1.y,U_MaterialCustomParam1.z))
			discard;

		//Lattice Cull Mask ///all shader use 29
		bool needDiscard = ComputeBitmapCullMask(U_PSGeneralRegister29, IN.i_BoundBoxRange, 
		U_MaterialCustomParam1.z, IN.i_PosInModel.xyz, IN.i_PosInClip.xyz);
		if(needDiscard)
			discard;
	}

	#ifdef DISSOLVE
		if(S_Custom7.Sample(S_Custom7SamplerState, uv).a < U_MaterialFresnel.y)
		discard;	
	#endif

	#ifdef SEQUENCE_FRAME
		#ifdef SEQUENCE_FRAME_BASEMAP
			float fmodtime = fmod(U_PS_Time.x, 60.0f);
			int index = floor(fmodtime * U_PSGeneralRegister10.z);
			int indexY = index / U_PSGeneralRegister10.w;
			int indexX = index - indexY * U_PSGeneralRegister10.w;
			uv = float2(uv.x / U_PSGeneralRegister10.w, uv.y / U_PSGeneralRegister10.w);
			uv.x += indexX / U_PSGeneralRegister10.w;
			uv.y += indexY / U_PSGeneralRegister10.w;
		#endif
	#endif
	
	half4 diffuseColor = S_Diffuse.Sample(S_DiffuseSamplerState, uv);

	#ifdef ALPHA_TEST
		clip(diffuseColor.a - U_MaterialDiffuse.a);
	#endif
	
	half3 baseMap = pow(diffuseColor.rgb, 2.2);
	
	half3 colorMask = pow(U_PSGeneralRegister4.rgb, 2.2);
	
	#ifdef TEXTURE_MASK
		half4 mask = S_Custom2.Sample(S_Custom2SamplerState, uv);
		half3 colorR = colorMask * mask.r;
		half3 colorG = pow(U_PSGeneralRegister5.rgb, 2.2) * mask.g;
		half3 colorB = pow(U_PSGeneralRegister6.rgb, 2.2) * mask.b;
		float f = min(mask.r + mask.g + mask.b, 1.0);
		half3 colorF = colorR + colorG + colorB;
		baseMap = lerp(baseMap, baseMap * colorF, f);
	#endif
	
	colorMask = lerp(half3(1.0, 1.0, 1.0), colorMask, U_PSGeneralRegister13.z);
	baseMap = baseMap * colorMask;
	
	half3 N = normalize(IN.i_Normal.xyz);
	
	#ifdef TWO_SIDES
		float frontValue = isFrontFacing ? 1.0f : -1.0f;
		N *= frontValue;
	#endif

	#ifdef USESHADOWMAP
		float4 shadowSpacePos[SHADOWSPLITE];
		ProjectToLightSpace(shadowSpacePos, IN.i_WPos.xyz, N, IN.i_ViewSpaceZ);
	#endif

	half3 V = normalize(U_PS_CameraPosition.xyz - IN.i_WPos.xyz);
	half3 L = normalize(-U_MainLightDirection.xyz);
	
	half3 H = normalize(L+V);
	half NdotH = max(dot(N, H), 0.0h);
	half wetFinal = U_PSGeneralRegister8.w * NdotH;
	
	half4 bumpColor = 0.0;	
	#ifdef BUMP_MAP
		half3 nn = N;//normalize(IN.i_Normal.xyz);
		half3 tn = normalize(IN.i_Tangent.xyz);
		half3 bn = normalize(IN.i_Binormal.xyz);

		tn = normalize(tn - nn * dot(tn, nn));
		bn = normalize(bn - nn * dot(bn, nn));

		half3x3 tbn = 
		{
			tn.x, bn.x, nn.x,
			tn.y, bn.y, nn.y,
			tn.z, bn.z, nn.z 
		};
		
		bumpColor = S_Bump.Sample(S_BumpSamplerState, uv); 

		half3 bump = bumpColor.rgb * 2.f - 1.f;	
		bump.xy *= U_PSGeneralRegister2.w;
		bump.z = 1.f;

		#ifdef DETAIL_BUMP
			float2 detailUV = U_PSGeneralRegister13.x;
			
			detailUV /= U_PSGeneralRegister2.x;
			
			half4 detailBumpColor = S_Custom.Sample(S_CustomSamplerState, uv * detailUV);
			bump.xy += (detailBumpColor.xy - half2(0.5, 0.5)) * U_PSGeneralRegister13.y;
		#endif
				
		#ifdef MUSCLE_BUMP
			half4 secondBumpColor = S_Custom4.Sample(S_Custom4SamplerState, uv);
			float secondBumpAlpha = secondBumpColor.z;
			secondBumpColor.xy = secondBumpColor.xy * 2.0 - 1.0;
			float2 newBump = bump.xy;
			newBump.xy += secondBumpColor.xy * U_PSGeneralRegister17.r;
			newBump.xy *= 0.5;
			
			bump.xy = lerp(bump.xy, newBump.xy, secondBumpAlpha);
		#endif
		
		#ifdef AGE
			half4 thridBumpColor = S_Custom5.Sample(S_Custom5SamplerState, uv);
			float thridBumpAlpha = thridBumpColor.z;
			thridBumpColor.xy = thridBumpColor.xy * 2.0 - 1.0;
			float2 newBump2 = bump.xy;
			newBump2.xy += thridBumpColor.xy * U_PSGeneralRegister17.g;
			newBump2.xy *= 0.5;
			
			bump.xy = lerp(bump.xy, newBump2.xy, thridBumpAlpha);
		#endif

		#ifdef RIPPLE
			//float3 modelN = normalize(IN.i_Normal.xyz);
			float s = clamp(dot(N, float3(0.0,1.0,0.0)), 0.0, 1.0) ;
			s = pow(s, 5.0);
			s *= (1.f - U_PSGeneralRegister30.x);
			bump.xy  *= (1.0 - s * 0.9f);
			
			double time = U_PS_Time.x;
			float time01 = U_PS_Time.y * 0.1591549; //U_PS_Time.y / PI / 2
			//涟漪整体的透明度使用sin函数进行0-1的变化，当透明度变为0的时候，坐标进行偏移，以减轻雨滴的重复感。单张贴图透明度为0的时候会没有涟漪，所以采样两次，交替显示
			float2 rUV = IN.i_WPos.xz * 0.25f;
			
			float rOffset = floor(U_PS_Time.x * 0.5f) * 0.1f;
			float4 ripple = ComputeRipple(rUV + rOffset, time01 * 3.0f);
			
			rUV = IN.i_WPos.xz * 0.25f + 0.25f;
			rOffset = floor(U_PS_Time.x * 0.5f + 0.25f) * 0.1f;
			float4 ripple2 = ComputeRipple(rUV + float2(rOffset, -rOffset), time01 * 3.0f);
			
			float rLerp1 =  sin(time01 * 3.1415);
			float rLerp2 =  1.0 - rLerp1;
			bump.xy += rLerp1 * ripple.xy * s;
			bump.xy += rLerp2 * ripple2.xy * s;
		#endif	
		
		bump = normalize(bump);
		N = mul(tbn, bump);
	#endif

	float coverLerp = dot(N, U_PS_GravityDirection.xyz);
	float agingLerp = coverLerp*U_PSGeneralRegister17.w;	
	agingLerp = clamp(agingLerp, 0.0, 1.0);
	half3 agingColor = pow(U_GlobalAgingColour.rgb, 2.2);
	baseMap.rgb = lerp(baseMap.rgb, agingColor, agingLerp);	

	
	coverLerp += U_PSGeneralRegister11.z;
	coverLerp = clamp(coverLerp, 0.0, 1.0);
	
	coverLerp = coverLerp * U_PSGeneralRegister5.w;
	
	half3 coverMap = U_WeatherParam0.xyz;
			
	#ifdef COVER_CUSTOM
		float2 coverUV = uv * U_PSGeneralRegister11.ww;
		half3 coverCustomMap = pow(S_Custom3.Sample(S_Custom3SamplerState, coverUV).rgb, 2.2);
		coverMap = lerp(coverCustomMap, coverMap, U_WeatherParam0.www * (1.0 - U_PSGeneralRegister30.x));
	#else
		coverLerp *= U_WeatherParam0.w * (1.0 - U_PSGeneralRegister30.x);
	#endif

	
	half metallic = 1.0;
	half roughness = 0.0001;
	half ao = 1.0;
	#ifdef PBR_SIMPLE
		metallic = U_PSGeneralRegister8.y;
		ao = lerp(1.0, bumpColor.b, U_PSGeneralRegister8.z);
		roughness = lerp(U_PSGeneralRegister4.w, bumpColor.b, U_PSGeneralRegister15.y);
		metallic = lerp(metallic, bumpColor.b, U_PSGeneralRegister17.z);
		roughness = clamp(roughness + U_PSGeneralRegister6.w, 0.0, 1.0);
	#else
		half3 mixMap = S_Specular.Sample(S_SpecularSamplerState, uv).rgb;
		metallic = mixMap.r;
		roughness = clamp(mixMap.g + U_PSGeneralRegister6.w, 0.0, 1.0);
		ao = mixMap.b;
	#endif
	

	float r = U_WeatherParam1.x;
	#ifdef COVER_CUSTOM
		r = U_PSGeneralRegister11.x;
	#endif
	roughness = lerp(roughness, r, coverLerp);
	roughness -= wetFinal;
	//有涟漪的时候，给材质的高光整体加0.4		
	#ifdef RIPPLE
		roughness = clamp(roughness - 0.4f * (1.f - U_PSGeneralRegister30.x), 0.0, 1.0);
	#endif
	roughness = max(roughness , 1e-6);
	
	float m = U_WeatherParam1.y;
	#ifdef COVER_CUSTOM
		m = U_PSGeneralRegister11.y;
	#endif
	metallic = lerp(metallic, m, coverLerp);
	
	metallic = lerp(metallic, 0.5, wetFinal);
	
	half3 color = 0.0;
	
	#ifdef	THIN_TRANSLUCENCY
		color += thinTranslucency(N, L, U_PSGeneralRegister7.xyz) * metallic;
		metallic = 0.0;
	#endif
	
	half4 mainLightColor = lerp(U_PBR_MainLightColor, U_RoleMainLightColor, U_PSGeneralRegister7.w);
	half4 skyLightColor = lerp(U_PBR_SkyLightColor, U_RoleSkyLightColor, U_PSGeneralRegister7.w);
	half4 groundLightColor = lerp(U_PBR_GroundLightColor, U_RoleGroundLightColor, U_PSGeneralRegister7.w);
	
	float maskRes = 1.0;

	// #define CUBE_RANGE
	#ifndef CUBE_RANGE
	    float4 tsPos = U_CoverColorParams;
	    float dist = length(IN.i_WPos.xz - tsPos.xz);
	    maskRes = smoothstep(tsPos.w - 1.0f, tsPos.w, dist);
	    // maskRes = 1.0 - maskRes;
	#else
	    float rectSize = 20.0f;
	    float2 halfSize = rectSize * 0.5;
	    float4 tsPos = U_CoverColorParams;
	    // Calculate the distance to the nearest edge of the rectangle
	    float2 distToEdge = abs(IN.i_WPos.xz - tsPos.xz) - halfSize;

	    // Use length function to determine if the point is within the rectangle boundaries
	    maskRes = length(max(distToEdge, 0.0));
	    maskRes = clamp(maskRes, 0.0, 1.0);
	    // maskRes = 1.0 - maskRes;
	#endif

	coverLerp *= maskRes;
	baseMap = lerp(baseMap, coverMap, coverLerp);
	
	#ifdef COVER_CUSTOM_BOTTOM
		float2 coverBottomUV = uv * U_PSGeneralRegister14.yy;
		half3 coverBottomCustomMap = pow(S_Custom4.Sample(S_Custom4SamplerState, coverBottomUV).rgb, 2.2);
		float s = dot(IN.i_Normal.xyz, float3(0.0f, -1.0f, 0.0f)) - U_PSGeneralRegister14.x;
		baseMap = lerp(baseMap, coverBottomCustomMap, clamp(s, 0.0, 1.0));
	#endif

	half3 grayColor = baseMap.rgb * half3(0.333, 0.333, 0.333);
	float gray = grayColor.r + grayColor.g + grayColor.b;
	
	grayColor.rgb =  half3(gray,gray,gray);
	
	baseMap.rgb = lerp(baseMap.rgb,grayColor,U_MaterialCustomParam.x);		
	
	half3 albedo = baseMap * (1.0 - metallic);
	half3 specularColor = lerp(half3(0.04h, 0.04h, 0.04h), baseMap, metallic);

	specularColor = lerp(specularColor, half3(0.5h, 0.5h, 0.5h), wetFinal);	
	
	color += calculateLightPower(N, V, L, mainLightColor.xyz * (1.0 - U_PSGeneralRegister8.x), baseMap, roughness, metallic);
	color = color * U_MaterialDiffuse.rgb;
		
	#ifdef USESHADOWMAP
		float shadowFactor = calcShadowFactor(
			shadowSpacePos, 
			IN.i_ViewSpaceZ);
	
		shadowFactor = min(shadowFactor, 1.0);
		color *= shadowFactor;
	#endif

	color += calculateLightPower(N, V, -U_PS_ViewDirection.xyz, mainLightColor.xyz * U_PSGeneralRegister8.x, baseMap, roughness, metallic);

	#ifdef CLUSTER_POINT_LIGHT
		color += calculateShadowPBRPointLight(IN.i_WPos, N, V, baseMap, roughness, metallic);

        float3 clusterLightColor = 
            calculateClusterPBRPointLight(IN.i_PosInClip.xy,
                                          IN.i_WPos.xyz,
                                          N,
                                          V,
                                          baseMap,
                                          roughness,
                                          metallic);
        color += clusterLightColor;
	#endif
	
	half3 envBRDF = envBRDFApprox(specularColor, roughness, max(dot(N, V), 0.0));
	
	half normalY = N.y * 0.5 + 0.5;
	normalY = clamp(normalY, 0.0001f, 1.0f);
	half3 ambientColor = groundLightColor.rgb * (1.0-normalY) + skyLightColor.rgb * normalY;
	
	#ifdef ENVIRONMENT_MAP
		half4 refl;
		refl.xyz = reflect(-V, N);
		refl.w = U_EnvironmentCubeMapMipLevels.x - 1.0 - (1.0 - 1.2 * log2(roughness));
		refl.w = max(0.0, refl.w);
		half3 reflColor = S_EnvironmentCubeMap.SampleLevel(S_EnvironmentCubeMapSamplerState, refl.xyz,refl.w).rgb;
		reflColor = pow(reflColor, 2.2);
		envBRDF *= reflColor;
	#else
		envBRDF *= normalize(ambientColor);
	#endif
	
	color += (albedo + envBRDF) * ambientColor;
	color *= ao;
	
	#ifdef TEXTURE_FLOW
		uv *= U_PSGeneralRegister3.z;
		uv += frac(U_PS_Time.z * float2(U_PSGeneralRegister3.x, U_PSGeneralRegister3.y));
		half3 flowTextureColor = pow(S_Custom.Sample(S_CustomSamplerState, uv).rgb, 2.2);
		
		half3 nottod_texture_flow = lerp((lerp(flowTextureColor, flowTextureColor * U_PSGeneralRegister3.w, U_LightPower.x) * diffuseColor.a), lerp(flowTextureColor, flowTextureColor * U_PSGeneralRegister3.w, U_LightPower.x), U_PSGeneralRegister12.x);
		half3 tod_texture_flow = lerp((flowTextureColor * U_PSGeneralRegister3.w * U_LightPower.x * diffuseColor.a), (flowTextureColor * U_PSGeneralRegister3.w * U_LightPower.x), U_PSGeneralRegister12.x);
        color += lerp(nottod_texture_flow, tod_texture_flow, U_PSGeneralRegister12.y);
	#endif
	
	#ifdef SEQUENCE_FRAME
		#ifndef SEQUENCE_FRAME_BASEMAP
			float fmodtime = fmod(U_PS_Time.x, 60.0f);
			int index = floor(fmodtime * U_PSGeneralRegister10.z);
			int indexY = index / U_PSGeneralRegister10.w;
			int indexX = index - indexY * U_PSGeneralRegister10.w;
			float2 sfUV = float2(uv.x / U_PSGeneralRegister10.w, uv.y / U_PSGeneralRegister10.w);
			sfUV.x += indexX / U_PSGeneralRegister10.w;
			sfUV.y += indexY / U_PSGeneralRegister10.w;
		
			half3 sfTextureColor = pow(S_Custom.Sample(S_CustomSamplerState, sfUV).rgb, 2.2);
			color += sfTextureColor * diffuseColor.a;
		#endif
	#endif
	
	#ifdef EMISSIVE
		half3 misColor = U_PSGeneralRegister0.xyz;//pow(U_PSGeneralRegister0.xyz, 2.2);
		half3 missiveColor = diffuseColor.rgb * misColor * diffuseColor.a;
		half3 minMissiveColor = missiveColor * U_PSGeneralRegister0.w;
		half todFactor = lerp(1.0, U_LightPower.x, U_PSGeneralRegister12.z);
		half3 maxMissiveColor = missiveColor * U_PSGeneralRegister1.x;
		half factor = U_MaterialSpecular.w;	
		color += lerp(minMissiveColor, lerp(minMissiveColor, maxMissiveColor, half3(factor, factor, factor)), U_PSGeneralRegister12.w) * todFactor;
	#endif
	
	#ifdef SPARKLE		
		float ss = 0.0f;		
		ss += S_Custom3.Sample(S_Custom3SamplerState, uv * U_PSGeneralRegister16.x + U_PS_Time.x * -U_PSGeneralRegister16.z).r;
		ss *= S_Custom3.Sample(S_Custom3SamplerState, uv * U_PSGeneralRegister16.y + U_PS_Time.x * +U_PSGeneralRegister16.z).g;
		ss = pow(ss, U_PSGeneralRegister16.w) - U_PSGeneralRegister18.x;
		color.rgb += max(ss, 0.0f) * U_PSGeneralRegister18.w;
	#endif
	
	#ifdef STROKE
		half3 strokeColor = U_PSGeneralRegister9.rgb;//pow(U_PSGeneralRegister9.rgb, 2.2);
		half stroke = max(dot(V, N)* U_PSGeneralRegister9.a + U_PSGeneralRegister10.r, 0.0f);
		stroke = min(stroke, 1.0f);
		stroke = (1.0 - stroke) * U_MaterialFresnel.r;
		color.rgb = lerp(color.rgb, strokeColor ,stroke);
		color.rgb *= (U_PSGeneralRegister10.g * stroke * 10.0 + 1.0);
	#endif
	float alphaaa = 1.0f;
	#ifdef SOFTBOARD
		float pixelDepth = 1.0f;
		float sceneDepth = 1.0f;
		float4 tempPos = 1.0f;
		tempPos = float4(IN.i_ClipPos.xy/IN.i_ClipPos.w, 0.0, 1.0);
		tempPos.x = tempPos.x * 0.5 + 0.5;
		tempPos.y = (1.0 - tempPos.y) * 0.5;
		sceneDepth = calcSceneDepth(IN.i_ClipPos);
		float4 vp;
		vp.x = dot(U_InvProjectMatrix[0], IN.i_ClipPos);
		vp.y = dot(U_InvProjectMatrix[1], IN.i_ClipPos);
		vp.z = dot(U_InvProjectMatrix[2], IN.i_ClipPos);
		vp.w = dot(U_InvProjectMatrix[3], IN.i_ClipPos);
		
		pixelDepth = vp.z / vp.w;
			
		alphaaa = saturate(abs(pixelDepth - sceneDepth) / IN.i_UV.z);
	#endif

	color *= U_MaterialCustomParam1.x;
	color = color * IN.i_Fog.a + IN.i_Fog.rgb;
	psOut.color.rgb = pow(color, 1.0 / 2.2);
	psOut.color.a = diffuseColor.a * U_MaterialFresnel.r * alphaaa;
	
	if(U_MaterialCustomParam.w > 0)
		psOut.color.rgb = (1 - U_MaterialCustomParam.w) * psOut.color.rgb + U_MaterialCustomParam.w * U_MaterialCustomParam.xyz;			
	
	#ifdef EditorHelp
		psOut.color.rgb = psOut.color.rgb * 0.00001 + float3(1.0,0.0,0.0);
		psOut.color.a = 1.0;
	#endif
	
	#ifdef VERTEX_COLOR
		psOut.color.a *= (IN.i_Diffuse.g * IN.i_Diffuse.g);
	#endif

	float overdrawlerp = ceil(U_PS_Time.w);
	psOut.color = psOut.color * (1.0 - overdrawlerp) + float4(U_PS_Time.w * overdrawlerp, U_PS_Time.w * overdrawlerp, U_PS_Time.w * overdrawlerp, overdrawlerp);

#endif

	return psOut;
}
