#include<Common_Uniform_PS.inl>
#define SCREEN_EFFECT 0
#define NUM_LAYERS 4.0
#define iterations 17
#define formuparam 0.53

#define volsteps 20
#define stepsize 0.1

#define zoom   0.800
#define tile   0.850
#define speed  0.010 

#define brightness 0.0015
#define darkmatter 0.300
#define distfading 0.730
#define saturation 0.850

struct PS_INPUT
{
    float4 i_PosInClip	   : SV_POSITION;
    float4 i_UV            : TEXCOORD0;
	float4 i_ViewDir	   : TEXCOORD1;
	float4 i_PositionHCS   : TEXCOORD2;
};

float4 linear2sRGB(float4 linearColor)
{
    return float4(pow(linearColor.rgb, 1.0f / 2.2), linearColor.a);
}




float4 main(PS_INPUT IN): SV_TARGET{
    
    float time = U_PS_Time.x*speed+0.25;
    float2 uv = float2(IN.i_UV.xy) * 2.0f - 1.0f;
    // return float4(uv, 0.f, 0.f);
	float3 dir=float3(uv*zoom,1.0);

	//mouse rotation
	float a1=0.5;//+iMouse.x/iResolution.x*2.;
	float a2=0.8;//+iMouse.y/iResolution.y*2.;
	float2x2 rot1=float2x2(cos(a1),sin(a1),-sin(a1),cos(a1));
	float2x2 rot2=float2x2(cos(a2),sin(a2),-sin(a2),cos(a2));
	dir.xz=mul(rot1, dir.xz);
	dir.xy=mul(rot2, dir.xy);
	float3 from=float3(1.0,0.5,0.5);
	from+=float3(time*2.0,time,-2.0);
	from.xz=mul(rot1, from.xz);
	from.xy=mul(rot2, from.xy);
	
	//volumetric rendering
	float s=0.01,fade=1.0;
	float3 v=float3(0.0, 0.0, 0.0);
	for (int r=0; r<volsteps; r++) {
		float3 p=from+s*dir*0.5;
		p = abs(float3(tile,tile,tile)- fmod(p,float3(tile,tile,tile)*2.0)); // tiling fold
		float pa = 0.0,a=0.0;
		for (int i=0; i<iterations; i++) { 
			p=abs(p)/dot(p,p)-formuparam; // the magic formula
			a+=abs(length(p)-pa); // absolute sum of average change
			pa=length(p);
		}
		float dm=max(0.,darkmatter-a*a*0.001); //dark matter
		a=a*a*a; // add contrast
		if (r>6) fade=fade*(1.0-dm); // dark matter, don't render near
		//v+=float3(dm,dm*.5,0.);
		v+=fade;
		v=v+float3(s,s*s,s*s*s*s)*a*brightness*fade; // coloring based on distance
		fade*=distfading; // distance fading
		s+=stepsize;
	}
	v=lerp(float3(length(v),length(v),length(v)),v,saturation); //color adjust
    
    return float4(v*0.01,1.0);
}
