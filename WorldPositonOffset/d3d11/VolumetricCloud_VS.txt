#include <Common_Uniform_VS.inl>
#include <Common_Function.inl>

struct VS_INPUT
{
	float3 i_Pos		: POSITION0;
#ifndef SHADOWPASS
	float3 i_Normal		: NORMAL0;
	float2 i_UV			: TEXCOORD0;
	#ifdef VERTEX_COLOR
		float4 i_Diffuse: COLOR0;
	#endif
#endif

#ifdef HWSKINNED
	float4	blend	: BLENDWEIGHT0;
	uint4	index	: BLENDINDICES0;
#endif

};

struct VS_OUTPUT
{
	float4 o_Position	: SV_POSITION;
#ifndef SHADOWPASS
	float2 o_UV			: TEXCOORD0;
	float3 o_WorldPos	: TEXCOORD1;
	float3 o_WorldNormal: NORMAL0;
	int	InstanceID 		: TEXCOORD2;
	float3 o_3DUV			: TEXCOORD3;
	#ifdef USESHADOWMAP
	#endif
	#ifdef VERTEX_COLOR
		float4 o_Diffuse: COLOR0;
	#endif
	float4 o_Fog		 : TEXCOORD9;
	float3 o_InstWorldPos : TEXCOORD10;
#endif
#ifdef PARABOLOID_MAPPING
	float	o_ClipDepth			: TEXCOORD12;
#endif
};

#ifdef HWSKINNED
	float4	U_BoneMatrix[3 * Bone_Matrix_Max_Cnt];
#endif

float4	U_WorldViewProjectMatrix[4];	
float4  U_WorldMatrix[3];
float4  U_InvWorldMatrix[3];
float4  U_VSGeneralRegister0;	// x: layer offset factor
								// y: layer number in runtime
								// z: layer number scale
								// w: layer number in material
float4  U_VSGeneralRegister1;	// layer offset
float4  U_VSGeneralRegister2;	// layer distance
float4  U_VSGeneralRegister3;	// x: uv tilling

VS_OUTPUT main(VS_INPUT IN, uint _InstanceID : SV_InstanceID)
{
	VS_OUTPUT vsOut = (VS_OUTPUT)0;
   	float4 pos = float4(IN.i_Pos.xyz, 1.0f);

	#ifndef SHADOWPASS
		vsOut.o_3DUV = pos.xyz * 0.5 + 0.5;	
		vsOut.o_UV = IN.i_UV * U_VSGeneralRegister3.x;
	#endif
	#ifdef HWSKINNED
		float4	Weight		= IN.blend;
		uint4  Index		= IN.index;

		float4 BoneMatC0	= float4(0.0,0.0,0.0,0.0);		
		float4 BoneMatC1	= float4(0.0,0.0,0.0,0.0);
		float4 BoneMatC2	= float4(0.0,0.0,0.0,0.0);

		for(int i=0;i<4;++i)
		{
			int idx = Index[i];
			BoneMatC0	+=	U_BoneMatrix[idx*3+0]*Weight[i];	
			BoneMatC1	+=	U_BoneMatrix[idx*3+1]*Weight[i];	
			BoneMatC2	+=	U_BoneMatrix[idx*3+2]*Weight[i];	
		}

		pos = float4(dot(BoneMatC0,pos),dot(BoneMatC1,pos),dot(BoneMatC2,pos),1.0f);		
	#endif	

	int i_InstanceID = (int(U_VSGeneralRegister0.y) - 1) - _InstanceID;

#ifndef SHADOWPASS	
	vsOut.InstanceID = i_InstanceID;

	vsOut.o_WorldPos = mul((float3x4)U_WorldMatrix, pos);
	float4  normal =  float4(IN.i_Normal.xyz, 0.0f);
	#ifdef  HWSKINNED
		normal = float4(dot(BoneMatC0,normal),dot(BoneMatC1,normal),dot(BoneMatC2,normal),0.0f);
	#endif

	float3 vCamPosInObj = (float3)1.f;
	vCamPosInObj.x = dot(U_InvWorldMatrix[0], float4(U_VS_CameraPosition.xyz, 1.f));
	vCamPosInObj.y = dot(U_InvWorldMatrix[1], float4(U_VS_CameraPosition.xyz, 1.f));
	vCamPosInObj.z = dot(U_InvWorldMatrix[2], float4(U_VS_CameraPosition.xyz, 1.f));
	float fDistCamToObj = distance(vCamPosInObj, pos.xyz);
	// float fLayerDist = saturate((fDistCamToObj - U_VSGeneralRegister0.z) /
	//         (max(U_VSGeneralRegister0.w - U_VSGeneralRegister0.z, 0.0001f)));
	// fLayerDist = lerp(U_VSGeneralRegister0.x, U_VSGeneralRegister1.w, fLayerDist);
	// pos.xyz += normal.xyz * (fLayerDist * i_InstanceID);

	float4 vLayerOffset = U_VSGeneralRegister1 * U_VSGeneralRegister0.xxxx;
	float4 vLayerDistance = U_VSGeneralRegister2;
	vLayerDistance.y = max(vLayerDistance.x, vLayerDistance.y);
	vLayerDistance.z = max(vLayerDistance.y, vLayerDistance.z);
	vLayerDistance.w = max(vLayerDistance.z, vLayerDistance.w);
	float fDist = clamp(fDistCamToObj, vLayerDistance.x, vLayerDistance.y);
	float fLayerFactor = (fDist - vLayerDistance.x) / max(vLayerDistance.y - vLayerDistance.x, 0.0001f);
	float fLayerOffset = lerp(vLayerOffset.x, vLayerOffset.y, fLayerFactor);
	fDist = clamp(fDistCamToObj, vLayerDistance.y, vLayerDistance.z);
	fLayerFactor = (fDist - vLayerDistance.y) / max(vLayerDistance.z - vLayerDistance.y, 0.0001f);
	fLayerOffset = lerp(fLayerOffset, vLayerOffset.z, fLayerFactor);
	fDist = clamp(fDistCamToObj, vLayerDistance.z, vLayerDistance.w);
	fLayerFactor = (fDist - vLayerDistance.z) / max(vLayerDistance.w - vLayerDistance.z, 0.0001f);
	fLayerOffset = lerp(fLayerOffset, vLayerOffset.w, fLayerFactor);
	pos.xyz += normal.xyz * (fLayerOffset * i_InstanceID);

	vsOut.o_WorldNormal = mul((float3x4)U_WorldMatrix, float4(normal.xyz, 0.0f));
	vsOut.o_InstWorldPos = mul((float3x4)U_WorldMatrix, pos);

	#ifdef VERTEX_COLOR
		vsOut.o_Diffuse = IN.i_Diffuse;
	#endif

	float3 camToWorld = U_VS_CameraPosition.xyz - vsOut.o_WorldPos.xyz;
	
	float4x4 gravityRotatePosMatrix;	
	gravityRotatePosMatrix[0] = U_VS_GravityRotationMatrix[0];
	gravityRotatePosMatrix[1] = U_VS_GravityRotationMatrix[1];
	gravityRotatePosMatrix[2] = U_VS_GravityRotationMatrix[2];
	gravityRotatePosMatrix[3] = U_VS_GravityRotationMatrix[3];
	camToWorld = mul(gravityRotatePosMatrix,float4(camToWorld.xyz,0.0f)).xyz;

	vsOut.o_Fog = ComputeFogColor(camToWorld,
			U_FogParam.xy, U_FogParam.zw, U_FogRampParam, U_FogColorNear, U_FogColorFar);
#endif

	vsOut.o_Position.x = dot(U_WorldViewProjectMatrix[0], pos);
	vsOut.o_Position.y = dot(U_WorldViewProjectMatrix[1], pos);
	vsOut.o_Position.z = dot(U_WorldViewProjectMatrix[2], pos);
	vsOut.o_Position.w = dot(U_WorldViewProjectMatrix[3], pos);

#ifdef PARABOLOID_MAPPING
	vsOut.o_ClipDepth = vsOut.o_Position.z;
	ParaboloidMapping(vsOut.o_Position, U_ShadowLightDepthMulAdd.x, U_ShadowLightDepthMulAdd.z, U_ShadowLightDepthMulAdd.w);
#endif
	
#ifdef SHADOWPASS
	#ifndef PARABOLOID_MAPPING
		//depthclamp
		vsOut.o_Position.z = ClampToNearPlane(vsOut.o_Position.z);
	#endif
#endif

#ifndef SHADOWPASS
	#ifdef EDITOR_MODE
		#ifdef HierarchicalLOD
			vsOut.o_Position = float4(2.0f * IN.i_UV.x - 1.0f, 1.0f - 2.0f * IN.i_UV.y, 1.0f, 1.0f);
		#endif
	#endif
#endif


	return vsOut;
}
