    	uint4   U_VSCustom1[(InstanceCount + 3) >> 2];//// animationData [cpu data type is uint32 , need recalculate index ]
	#else
    	float4 	U_VSCustom0[InstanceCount * UniformCount];
	#endif
#else
	float4	U_WorldViewProjectMatrix[4];	
	float4  U_WorldMatrix[3];
	float4  U_InvTransposeWorldMatrix[3];
	float4  U_ObjectMirrored;

float4 	U_VSGeneralRegister7; 	//x ---boundingbox min.y (current entity box) local box
								//y ---boundingbox max.y (current entity box)
								//z ---boundingbox min.y (parent entity box)

struct VS_INPUT
{
	float4 i_Pos		: POSITION0;
#ifdef SHADOWPASS
	#ifdef ALPHA_TEST
		float4 i_UV		: TEXCOORD0;	
	#endif
#else
	float4 i_Normal		: NORMAL0;
	float4 i_UV			: TEXCOORD0;	
	
	#ifdef SECOND_UV
		float4 i_Second_UV			: TEXCOORD1;
	#endif
	
	float4 i_Tangent	: TANGENT0;
	
	#ifdef VERTEX_COLOR
		float4	i_Diffuse	: COLOR0;
	#endif
#endif

#ifdef HWSKINNED
	#ifndef HWSKINNED_DUAL
		float4 i_BlendWeight	: BLENDWEIGHT0;
		uint4 i_BlendIndices	: BLENDINDICES0;
	#else
		float4 i_BlendWeight0	: BLENDWEIGHT0;
		float4 i_BlendWeight1	: BLENDWEIGHT1;
		uint4 i_BlendIndices0	: BLENDINDICES0;
		uint4 i_BlendIndices1	: BLENDINDICES1;
	#endif
#endif
};

struct VS_OUTPUT
{
	float4	o_PosInClip			: SV_POSITION;
#ifdef SHADOWPASS
	#ifdef ALPHA_TEST
		float4 o_UV		: TEXCOORD0;	
	#endif
#else
	float4 	o_UV				: TEXCOORD0;
	float4	o_Fog				: TEXCOORD1;
	float4	o_WNormal			: TEXCOORD2;
	float4	o_WPos				: TEXCOORD3;
	
	float4	o_Tangent			: TEXCOORD4;
	float4	o_Binormal			: TEXCOORD5;
	
	float4	o_EyeDir			: TEXCOORD6;
	#ifdef USESHADOWMAP
		float   o_ViewSpaceZ		: TEXCOORD7;
	#endif
	
	#if defined(SCREENDISTORT) || defined(DEPTH_FADE)
		float4 	o_PosInClipOut : TEXCOORD7;
	#endif
	
	#ifdef VERTEX_COLOR
		float4 o_Diffuse		: COLOR0;
	#endif
#endif
#ifdef PARABOLOID_MAPPING
	float	o_ClipDepth			: TEXCOORD12;
#endif
	float4	o_PosInModel		: TEXCOORD13;
	float4  o_BoundBoxRange		: TEXCOORD14;
};



float3 cal_maskValue(float3 input1, float3 input1_coefficient, float3 input2, float3 input2_coefficient)
{
	return input1 * input1_coefficient * input2 * input2_coefficient;
}

#ifdef SKINNEDINSTANCE
	float3x4 loadBoneMatrix (uint animationData , uint index)
	{
		float3x4 rvt ;

		uint hIndex = animationData>>16;
		uint wIndex = animationData&0xffff;

		rvt[0] = S_2DExt0.Load(uint3(index*3 + 0+wIndex, hIndex, 0));
		rvt[1] = S_2DExt0.Load(uint3(index*3 + 1+wIndex, hIndex, 0));
		rvt[2] = S_2DExt0.Load(uint3(index*3 + 2+wIndex, hIndex, 0));
		return rvt ;
	}
#endif

VS_OUTPUT main(VS_INPUT IN, uint i_InstanceID : SV_InstanceID)
{
	VS_OUTPUT vsOut = (VS_OUTPUT)0;
	float4  pos =  float4(IN.i_Pos.xyz, 1.0f);


#ifdef PARTICLE_VERTEX_OFFSET_TEXTURE
{
	float2 vertex_offset_texture_uv = IN.i_UV.xy;

	float vertex_offset_texture_tilling_u = U_VSGeneralRegister1.x;
	float vertex_offset_texture_tilling_v = U_VSGeneralRegister1.y;
	float vertex_offset_texture_warp_u = U_VSGeneralRegister1.z;
	float vertex_offset_texture_warp_v = U_VSGeneralRegister1.w;
	float vertex_offset_texture_offset_u = U_VSGeneralRegister2.x;
	float vertex_offset_texture_offset_v = U_VSGeneralRegister2.y;
	float vertex_offset_texture_normal_affect = U_VSGeneralRegister2.z;
	float vertex_offset_texture_use_r_channel = U_VSGeneralRegister2.w;
	float vertex_offset_texture_x = U_VSGeneralRegister3.x;
	float vertex_offset_texture_y = U_VSGeneralRegister3.y;
	float vertex_offset_texture_z = U_VSGeneralRegister3.z;
	float vertex_offset_texture_all = U_VSGeneralRegister3.w;

	vertex_offset_texture_uv.x *= vertex_offset_texture_tilling_u;
	vertex_offset_texture_uv.y *= vertex_offset_texture_tilling_v;

	vertex_offset_texture_uv.x += vertex_offset_texture_offset_u;
	vertex_offset_texture_uv.y += vertex_offset_texture_offset_v;

	vertex_offset_texture_uv.x = lerp(clamp(vertex_offset_texture_uv.x, 0.0f, 1.0f), frac(vertex_offset_texture_uv.x), vertex_offset_texture_warp_u);
	vertex_offset_texture_uv.y = lerp(clamp(vertex_offset_texture_uv.y, 0.0f, 1.0f), frac(vertex_offset_texture_uv.y), vertex_offset_texture_warp_v);

	float4 vertex_offset_texture_noise_rgba = S_Custom9.SampleLevel(S_Custom9SamplerState, vertex_offset_texture_uv, 0);
	float vertex_offset_texture_noise = lerp(vertex_offset_texture_noise_rgba.a, vertex_offset_texture_noise_rgba.r, vertex_offset_texture_use_r_channel);

	float3 normal_factor = lerp(float3(1.0f, 1.0f, 1.0f), IN.i_Normal.xyz, vertex_offset_texture_normal_affect);
	float3 factor = normal_factor * vertex_offset_texture_noise;

	factor.x *= vertex_offset_texture_x;
	factor.y *= vertex_offset_texture_y;
	factor.z *= vertex_offset_texture_z;

	factor *= vertex_offset_texture_all;

	#ifdef PARTICLE_VERTEX_OFFSET_MASK_TEXTURE
	{
		float2 mask_texture_uv = IN.i_UV.xy;
		float vertex_offset_mask_texture_tilling_u = U_VSGeneralRegister4.x;
		float vertex_offset_mask_texture_tilling_v = U_VSGeneralRegister4.y;
		float vertex_offset_mask_texture_warp_u =  U_VSGeneralRegister4.z;
		float vertex_offset_mask_texture_warp_v = U_VSGeneralRegister4.w;
		float vertex_offset_mask_texture_offset_u = U_VSGeneralRegister5.x;
		float vertex_offset_mask_texture_offset_v = U_VSGeneralRegister5.y;

		mask_texture_uv.x *= vertex_offset_mask_texture_tilling_u;
		mask_texture_uv.y *= vertex_offset_mask_texture_tilling_v;

		mask_texture_uv.x += vertex_offset_mask_texture_offset_u;
		mask_texture_uv.y += vertex_offset_mask_texture_offset_v;

		mask_texture_uv.x = lerp(clamp(mask_texture_uv.x, 0.0f, 1.0f), frac(mask_texture_uv.x), vertex_offset_mask_texture_warp_u);
		mask_texture_uv.y = lerp(clamp(mask_texture_uv.y, 0.0f, 1.0f), frac(mask_texture_uv.y), vertex_offset_mask_texture_warp_v);

		float3 vertex_coefficient = float3(0.0f, 0.0f, 0.0f);
		vertex_coefficient.x = U_VSGeneralRegister5.z;
		vertex_coefficient.y = U_VSGeneralRegister5.w;
		vertex_coefficient.z = U_VSGeneralRegister6.x;

		float3 mask_coefficient = float3(0.0f, 0.0f, 0.0f);
		mask_coefficient.x = U_VSGeneralRegister6.y;
		mask_coefficient.y = U_VSGeneralRegister6.z;
		mask_coefficient.z = U_VSGeneralRegister6.w;

		float4 mask_val = S_Custom10.SampleLevel(S_Custom10SamplerState, mask_texture_uv, 0);
		factor = cal_maskValue(factor, vertex_coefficient, mask_val.rgb, mask_coefficient);
	}
	#endif

	pos.x += factor.x;
	pos.y += factor.y;
	pos.z += factor.z;
}
#endif 


	vsOut.o_PosInModel = pos;
	float4 WorldMatrix[3];
	float4 InvTransposeWorldMatrix[3];
	float4 fHandedness = 1.f;

	#ifdef HWSKINNED
		#ifndef HWSKINNED_DUAL
			float4	Weight		= IN.i_BlendWeight;
			uint4   Index		= IN.i_BlendIndices;
			#define BlendCount 4
		#else
			float Weight[8] ;
			Weight[0] = IN.i_BlendWeight0[0]; Weight[1] = IN.i_BlendWeight0[1]; Weight[2] = IN.i_BlendWeight0[2]; Weight[3] = IN.i_BlendWeight0[3];
			Weight[4] = IN.i_BlendWeight1[0]; Weight[5] = IN.i_BlendWeight1[1]; Weight[6] = IN.i_BlendWeight1[2]; Weight[7] = IN.i_BlendWeight1[3];
			uint  Index[8];
			Index[0] = IN.i_BlendIndices0[0]; Index[1] = IN.i_BlendIndices0[1]; Index[2] = IN.i_BlendIndices0[2]; Index[3] = IN.i_BlendIndices0[3];
			Index[4] = IN.i_BlendIndices1[0]; Index[5] = IN.i_BlendIndices1[1]; Index[6] = IN.i_BlendIndices1[2]; Index[7] = IN.i_BlendIndices1[3];
			#define BlendCount 8
		#endif

		float4 BoneMatC0	= float4(0.0,0.0,0.0,0.0);		
		float4 BoneMatC1	= float4(0.0,0.0,0.0,0.0);
		float4 BoneMatC2	= float4(0.0,0.0,0.0,0.0);

        #ifdef SKINNEDINSTANCE
		   	// recalculate index
		   	uint 	animationData = U_VSCustom1[i_InstanceID >> 2][i_InstanceID & 3];
			for(int i=0;i<BlendCount;++i)
			{
		     	float3x4 res=loadBoneMatrix(animationData,Index[i]);
		     	BoneMatC0 += Weight[i] * res[0];
			 	BoneMatC1 += Weight[i] * res[1];
			 	BoneMatC2 += Weight[i] * res[2];
			}
	    #else
			for(int i=0;i<BlendCount;++i)
		    {
		    	int idx = Index[i];
		    	BoneMatC0	+=	U_BoneMatrix[idx*3+0]*Weight[i];	
		    	BoneMatC1	+=	U_BoneMatrix[idx*3+1]*Weight[i];	
		    	BoneMatC2	+=	U_BoneMatrix[idx*3+2]*Weight[i];	
		    }
		#endif
	#endif	
	
#ifdef USEINSTANCE	
	#ifdef  HWSKINNED
		pos = float4(dot(BoneMatC0,pos),dot(BoneMatC1,pos),dot(BoneMatC2,pos),1.0f);		
	#endif
	#ifndef SHADOWPASS
		float4  normal =  float4(IN.i_Normal.xyz, 0.0f);
		#ifdef  HWSKINNED
			normal = float4(dot(BoneMatC0,normal),dot(BoneMatC1,normal),dot(BoneMatC2,normal),0.0f);
		#endif
		#ifdef VERTEX_EXCITATION
			pos.xyz += normal.xyz * U_VSGeneralRegister0.x * sin(IN.i_UV.x * U_VSGeneralRegister0.y * U_VS_Time.x + IN.i_UV.y * U_VSGeneralRegister0.z * U_VS_Time.x);
		#endif
	#endif

	int idxInst = i_InstanceID * UniformCount;
	#ifdef HWSKINNED
		WorldMatrix[0] = U_VSCustom0[idxInst + 0];		// now is WroldViewMatrix
		WorldMatrix[1] = U_VSCustom0[idxInst + 1];
		WorldMatrix[2] = U_VSCustom0[idxInst + 2];
		InvTransposeWorldMatrix[0] = U_VSCustom0[idxInst + 3];
		InvTransposeWorldMatrix[1] = U_VSCustom0[idxInst + 4];
		InvTransposeWorldMatrix[2] = U_VSCustom0[idxInst + 5];
		
		float4 vObjPosInCam ;
		vObjPosInCam.xyz  = mul((float3x4)WorldMatrix, pos);
		vObjPosInCam.w    = 1.f;
		vsOut.o_PosInClip = mul((float4x4)U_ZeroViewProjectMatrix, vObjPosInCam);

		WorldMatrix[0][3] += U_VS_CameraPosition[0];
		WorldMatrix[1][3] += U_VS_CameraPosition[1];
		WorldMatrix[2][3] += U_VS_CameraPosition[2]; 	// now is WroldMatrix

		float4 Wpos;
		Wpos.xyz = mul((float3x4)WorldMatrix, pos);
		Wpos.w = 1.0;
	#else
		WorldMatrix[0] = U_VSCustom0[idxInst + 0];
		WorldMatrix[1] = U_VSCustom0[idxInst + 1];
		WorldMatrix[2] = U_VSCustom0[idxInst + 2];
		InvTransposeWorldMatrix[0] = U_VSCustom0[idxInst + 3];
		InvTransposeWorldMatrix[1] = U_VSCustom0[idxInst + 4];
		InvTransposeWorldMatrix[2] = U_VSCustom0[idxInst + 5];

		// WorldMatrix is now camera-relative (CPU did high-precision subtraction)
		// No need to subtract camera position again!
		float4 vObjPosInCam;
		vObjPosInCam.xyz = mul((float3x4)WorldMatrix, pos);
		vObjPosInCam.w = 1.f;
		vsOut.o_PosInClip = mul((float4x4)U_ZeroViewProjectMatrix, vObjPosInCam);
		
		// Reconstruct world position for lighting calculations
		float4 Wpos;
		Wpos.xyz = vObjPosInCam.xyz + U_VS_CameraPosition.xyz;
		Wpos.w = 1.0;
	#endif

	fHandedness = determinant(float3x3(WorldMatrix[0].xyz, WorldMatrix[1].xyz, WorldMatrix[2].xyz));
	fHandedness = fHandedness >= 0.f ? 1.f : -1.f;
#else

    #ifdef  HWSKINNED
		pos = float4(dot(BoneMatC0,pos),dot(BoneMatC1,pos),dot(BoneMatC2,pos),1.0f);		
	#endif
	#ifndef SHADOWPASS
		float4  normal =  float4(IN.i_Normal.xyz, 0.0f);
		#ifdef  HWSKINNED
			normal = float4(dot(BoneMatC0,normal),dot(BoneMatC1,normal),dot(BoneMatC2,normal),0.0f);
		#endif
		#ifdef VERTEX_EXCITATION
			pos.xyz += normal.xyz * U_VSGeneralRegister0.x * sin(IN.i_UV.x * U_VSGeneralRegister0.y * U_VS_Time.x + IN.i_UV.y * U_VSGeneralRegister0.z * U_VS_Time.x);
		#endif
	#endif
	
	vsOut.o_PosInClip.x = dot(U_WorldViewProjectMatrix[0], pos);
	vsOut.o_PosInClip.y = dot(U_WorldViewProjectMatrix[1], pos);
	vsOut.o_PosInClip.z = dot(U_WorldViewProjectMatrix[2], pos);
	vsOut.o_PosInClip.w = dot(U_WorldViewProjectMatrix[3], pos);	

	fHandedness = U_ObjectMirrored.w;
#endif

#ifdef PARABOLOID_MAPPING
	vsOut.o_ClipDepth = vsOut.o_PosInClip.z;
	ParaboloidMapping(vsOut.o_PosInClip, U_ShadowLightDepthMulAdd.x, U_ShadowLightDepthMulAdd.z, U_ShadowLightDepthMulAdd.w);
#endif
	
#ifdef SHADOWPASS
	#ifdef ALPHA_TEST
		vsOut.o_UV = IN.i_UV;
	#endif
	
	#ifndef PARABOLOID_MAPPING
		//depthclamp
		vsOut.o_PosInClip.z = ClampToNearPlane(vsOut.o_PosInClip.z);
	#endif
	
	
#else
	#ifdef USEINSTANCE
		vsOut.o_WPos = Wpos;
	#else
		WorldMatrix[0] = U_WorldMatrix[0];
		WorldMatrix[1] = U_WorldMatrix[1];
		WorldMatrix[2] = U_WorldMatrix[2];

		InvTransposeWorldMatrix[0] = U_InvTransposeWorldMatrix[0];
		InvTransposeWorldMatrix[1] = U_InvTransposeWorldMatrix[1];
		InvTransposeWorldMatrix[2] = U_InvTransposeWorldMatrix[2];

		vsOut.o_WPos.xyz = mul((float3x4)WorldMatrix, pos);
		vsOut.o_WPos.w = 1.0;
	#endif
	
	//float4  normal =  float4(IN.i_Normal.xyz, 0.0f);
	float4  tangent =  float4(IN.i_Tangent.xyz, 0.0f);
	
	#ifdef  HWSKINNED
		//normal = float4(dot(BoneMatC0,normal),dot(BoneMatC1,normal),dot(BoneMatC2,normal),0.0f);	
		tangent = float4(dot(BoneMatC0,tangent),dot(BoneMatC1,tangent),dot(BoneMatC2,tangent),0.0f);		
	#endif
	
	float4	vCamPos				=	U_VS_CameraPosition;
	
	vsOut.o_WNormal.xyz = mul((float3x4)InvTransposeWorldMatrix, float4(normal.xyz, 0.0f));
	vsOut.o_Tangent.xyz = mul((float3x4)WorldMatrix, float4(tangent.xyz, 0.0f));
	
	vsOut.o_Binormal.xyz = cross(vsOut.o_Tangent.xyz, vsOut.o_WNormal.xyz) * IN.i_Tangent.w;
	vsOut.o_Binormal.xyz *= fHandedness;
	
	vsOut.o_UV = IN.i_UV;
		
	float3 camToWorld =  vCamPos.xyz - vsOut.o_WPos.xyz;
	
	vsOut.o_EyeDir.xyz = camToWorld;
	
	#ifdef VERTEX_COLOR
		vsOut.o_Diffuse = IN.i_Diffuse;
	#endif
	
	float4x4 gravityRotatePosMatrix;	
	gravityRotatePosMatrix[0] = U_VS_GravityRotationMatrix[0];
	gravityRotatePosMatrix[1] = U_VS_GravityRotationMatrix[1];
	gravityRotatePosMatrix[2] = U_VS_GravityRotationMatrix[2];
	gravityRotatePosMatrix[3] = U_VS_GravityRotationMatrix[3];
	camToWorld = mul(gravityRotatePosMatrix,float4(camToWorld.xyz,0.0f)).xyz;

	vsOut.o_Fog = ComputeFogColor(camToWorld, U_FogParam.xy, U_FogParam.zw, U_FogRampParam, U_FogColorNear, U_FogColorFar);	
	
	#ifdef USESHADOWMAP
		float viewZ = -dot(U_MainCamViewMatrix, float4(vsOut.o_WPos.xyz, 1.0f));
		vsOut.o_ViewSpaceZ = viewZ;
	#endif

	#if defined(SCREENDISTORT) || defined(DEPTH_FADE)
		vsOut.o_PosInClipOut = vsOut.o_PosInClip;
	#endif
#endif



#ifndef SHADOWPASS
	#ifdef EDITOR_MODE
		#ifdef HierarchicalLOD
			vsOut.o_PosInClip = float4(2.0f * IN.i_UV.x - 1.0f, 1.0f - 2.0f * IN.i_UV.y, 1.0f, 1.0f);
		#endif
	#endif
#endif
#ifndef SHADOWPASS
	vsOut.o_BoundBoxRange.x = U_VSGeneralRegister7.x;//dot(WorldMatrix[1], float4(0.0f, U_VSGeneralRegister7.x, 0.0f, 1.0f));
	vsOut.o_BoundBoxRange.y = U_VSGeneralRegister7.y;//dot(WorldMatrix[1], float4(0.0f, U_VSGeneralRegister7.y, 0.0f, 1.0f));
	vsOut.o_BoundBoxRange.z = U_VSGeneralRegister7.z;
	vsOut.o_BoundBoxRange.w = U_VSGeneralRegister7.w;
#endif
	return vsOut;
}
