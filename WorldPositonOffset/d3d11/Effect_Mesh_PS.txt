		}

float3 mod289(float3 x)
{
    return x - floor(x / 289.0) * 289.0;
}

float2 mod289(float2 x)
{
    return x - floor(x / 289.0) * 289.0;
}

float3 permute(float3 x)
{
    return mod289((x * 34.0 + 1.0) * x);
}

float3 taylorInvSqrt(float3 r)
{
    return 1.79284291400159 - 0.85373472095314 * r;
}

float snoise(float2 v)
{
    const float4 C = float4( 0.211324865405187,  // (3.0-sqrt(3.0))/6.0
                             0.366025403784439,  // 0.5*(sqrt(3.0)-1.0)
                            -0.577350269189626,  // -1.0 + 2.0 * C.x
                             0.024390243902439); // 1.0 / 41.0
    // First corner
    float2 i  = floor(v + dot(v, C.yy));
    float2 x0 = v -   i + dot(i, C.xx);

    // Other corners
    float2 i1;
    i1.x = step(x0.y, x0.x);
    i1.y = 1.0 - i1.x;

    // x1 = x0 - i1  + 1.0 * C.xx;
    // x2 = x0 - 1.0 + 2.0 * C.xx;
    float2 x1 = x0 + C.xx - i1;
    float2 x2 = x0 + C.zz;

    // Permutations
    i = mod289(i); // Avoid truncation effects in permutation
    float3 p =
      permute(permute(i.y + float3(0.0, i1.y, 1.0))
                    + i.x + float3(0.0, i1.x, 1.0));

    float3 m = max(0.5 - float3(dot(x0, x0), dot(x1, x1), dot(x2, x2)), 0.0);
    m = m * m;
    m = m * m;

    // Gradients: 41 points uniformly over a line, mapped onto a diamond.
    // The ring size 17*17 = 289 is close to a multiple of 41 (41*7 = 287)
    float3 x = 2.0 * frac(p * C.www) - 1.0;
    float3 h = abs(x) - 0.5;
    float3 ox = floor(x + 0.5);
    float3 a0 = x - ox;

    // Normalise gradients implicitly by scaling m
    m *= taylorInvSqrt(a0 * a0 + h * h);

    // Compute final noise value at P
    float3 g;
    g.x = a0.x * x0.x + h.x * x0.y;
    g.y = a0.y * x1.x + h.y * x1.y;
    g.z = a0.z * x2.x + h.z * x2.y;
    return 130.0 * dot(m, g);
}

float3 snoise_grad(float2 v)
{
    const float4 C = float4( 0.211324865405187,  // (3.0-sqrt(3.0))/6.0
                             0.366025403784439,  // 0.5*(sqrt(3.0)-1.0)
                            -0.577350269189626,  // -1.0 + 2.0 * C.x
                             0.024390243902439); // 1.0 / 41.0
    // First corner
    float2 i  = floor(v + dot(v, C.yy));
    float2 x0 = v -   i + dot(i, C.xx);

    // Other corners
    float2 i1;
    i1.x = step(x0.y, x0.x);
    i1.y = 1.0 - i1.x;

    // x1 = x0 - i1  + 1.0 * C.xx;
    // x2 = x0 - 1.0 + 2.0 * C.xx;
    float2 x1 = x0 + C.xx - i1;
    float2 x2 = x0 + C.zz;

    // Permutations
    i = mod289(i); // Avoid truncation effects in permutation
    float3 p =
      permute(permute(i.y + float3(0.0, i1.y, 1.0))
                    + i.x + float3(0.0, i1.x, 1.0));

    float3 m = max(0.5 - float3(dot(x0, x0), dot(x1, x1), dot(x2, x2)), 0.0);
    float3 m2 = m * m;
    float3 m3 = m2 * m;
    float3 m4 = m2 * m2;

    // Gradients: 41 points uniformly over a line, mapped onto a diamond.
    // The ring size 17*17 = 289 is close to a multiple of 41 (41*7 = 287)
    float3 x = 2.0 * frac(p * C.www) - 1.0;
    float3 h = abs(x) - 0.5;
    float3 ox = floor(x + 0.5);
    float3 a0 = x - ox;

    // Normalise gradients
    float3 norm = taylorInvSqrt(a0 * a0 + h * h);
    float2 g0 = float2(a0.x, h.x) * norm.x;
    float2 g1 = float2(a0.y, h.y) * norm.y;
    float2 g2 = float2(a0.z, h.z) * norm.z;

    // Compute noise and gradient at P
    float2 grad =
      -6.0 * m3.x * x0 * dot(x0, g0) + m4.x * g0 +
      -6.0 * m3.y * x1 * dot(x1, g1) + m4.y * g1 +
      -6.0 * m3.z * x2 * dot(x2, g2) + m4.z * g2;
    float3 px = float3(dot(x0, g0), dot(x1, g1), dot(x2, g2));
    return 130.0 * float3(grad, dot(m4, px));
}

half GetLuminance(half3 color)
{
    const half3 lumWeight = half3(0.2125f, 0.7154f, 0.0721f);
    return dot(color, lumWeight);
}

{
	return color1 * color1_coefficient * color2 * color2_coefficient;
}


PS_OUTPUT main(PS_INPUT IN ,  bool isFrontFacing : SV_IsFrontFace)
{
	PS_OUTPUT psOut = (PS_OUTPUT)0.0;
	
#ifdef PARABOLOID_MAPPING
	clip(IN.i_ClipDepth);
#endif
#ifdef SHADOWPASS
	#ifdef ALPHA_TEST
		float4 color = texDiffuse(IN.i_UV.xy);
		clip(color.a - U_MaterialDiffuse.a);
	#endif
#else
	psOut.color.a = 1.0;
	#ifdef NoColor
		psOut.color = float4(0.0, 0.0, 0.0, 0.0);
		return psOut;
	#endif
	
	float2 uv = IN.i_UV.xy;
	float2 in_UV = IN.i_UV.xy;

	#ifdef POLAR_COORDINATES
		float2 InPolarUV = in_UV;
	#endif

	#ifdef POLAR_COORDINATES
	{
		float2 center = float2(U_PSGeneralRegister17.z, U_PSGeneralRegister17.w);
		float radialScale = U_PSGeneralRegister30.z;
		float lengthScale = U_PSGeneralRegister30.w;
		InPolarUV = PolarCoordinates_float(uv, center, radialScale, lengthScale);
		#ifdef DIFFUSE_MAP_ENABLE_POLAR
			uv = InPolarUV;
		#endif
	}
	#endif

	// 扰动贴图
	#ifdef DISTURBANCE_TEXTURE
	float2 disturbance_texture_to_diffuse_texture_strength_uv_offset = float2(0.0f, 0.0f); 
	{
		#ifdef DISTURBANCE_TEXTURE_ENABLE_POLAR
			float2 distance_texture_uv = InPolarUV;
		#else
			float2 distance_texture_uv = in_UV.xy;
		#endif

		// float disturbance_texture_to_mask_texture_strength = U_PSGeneralRegister18.z;
		float particle_age = U_PSGeneralRegister18.w;
		float distance_texture_uv_tilling_u = U_PSGeneralRegister19.x;
		float distance_texture_uv_tilling_v = U_PSGeneralRegister19.y;
		float distance_texture_uv_roll_speed_u = U_PSGeneralRegister19.z;
		float distance_texture_uv_roll_speed_v = U_PSGeneralRegister19.w;
		float distance_texture_uv_offset_u = U_PSGeneralRegister20.x;
		float distance_texture_uv_offset_v = U_PSGeneralRegister20.y;

		distance_texture_uv.x *= distance_texture_uv_tilling_u;
		distance_texture_uv.y *= distance_texture_uv_tilling_v;

		distance_texture_uv.x += particle_age * distance_texture_uv_roll_speed_u + distance_texture_uv_offset_u;
		distance_texture_uv.y += particle_age * distance_texture_uv_roll_speed_v + distance_texture_uv_offset_v;

		float disturbance_noise = texSample2D(S_Custom4, distance_texture_uv, S_Custom4SamplerState).r;

		// 对漫反射uv的影响
		float disturbance_texture_to_diffuse_texture_strength = U_PSGeneralRegister18.x;
		uv += disturbance_noise * disturbance_texture_to_diffuse_texture_strength;

		// 对溶解uv的影响
		float disturbance_texture_to_disolve_texture_strength = U_PSGeneralRegister18.y;
		float temp_val = float(disturbance_noise * disturbance_texture_to_disolve_texture_strength);
		disturbance_texture_to_diffuse_texture_strength_uv_offset = float2(temp_val, temp_val);
	}
	#endif 
	
	#ifdef UV_OFFSET
		uv += float2(U_PSGeneralRegister9.y, U_PSGeneralRegister9.z);
	#endif
	
	#ifdef UV_TILLING
		uv.x *= U_PSGeneralRegister17.x;
		uv.y *= U_PSGeneralRegister17.y;
	#endif
	
	// #ifdef UV_ROLL_TIME
	// 	uv += frac(U_PSGeneralRegister12.w * float2(U_PSGeneralRegister12.x, U_PSGeneralRegister12.y));
	// #endif
	
	#ifdef UV_ROLL
		uv += float2(U_PSGeneralRegister1.z, U_PSGeneralRegister1.w);
	#endif

	//溶解贴图
	#ifdef BROKEN	
		float dissolve_tilling_u = U_PSGeneralRegister24.z;
		float dissolve_tilling_v = U_PSGeneralRegister24.w;
		float dissolve_offset_u = U_PSGeneralRegister25.x;
		float dissolve_offset_v = U_PSGeneralRegister25.y;
		float dissolve_rotate_angle = U_PSGeneralRegister25.z;
		
		float2 dissolve_base_uv = uv;
		//原先溶解贴图的uv跟着diffuseMap的走的，这里拆出来，加个参数让它不跟着diffuseMap走，然后在这个基础上再去转极坐标
		#ifdef BROKEN_TEXTURE_SINGLE_UV
			#ifdef DISSOLVE_TEXTURE_ENABLE_POLAR
				dissolve_base_uv = InPolarUV;
			#else
				dissolve_base_uv = in_UV;
			#endif
		#endif
		float2 dissolve_uv = cal_uv(dissolve_base_uv, dissolve_tilling_u, dissolve_tilling_v, dissolve_offset_u, dissolve_offset_v, dissolve_rotate_angle);

		#ifdef DISTURBANCE_TEXTURE
		{
			dissolve_uv += disturbance_texture_to_diffuse_texture_strength_uv_offset;
		}
		#endif

		float dissolve_noise = texSample2D(S_Custom2, dissolve_uv, S_Custom2SamplerState).r;

		float4 dissolve_edge_color = U_PSGeneralRegister23;
		dissolve_edge_color *= U_PSGeneralRegister30.x;
		float dissolve_edge_width = U_PSGeneralRegister24.x;
		float dissolve_soft_value = U_PSGeneralRegister24.y;
		float dissolve_val = U_PSGeneralRegister9.x;

		#ifdef DISSOLVE_DIRECT
			#ifdef DISSOLVE_TEXTURE_ENABLE_POLAR
				float2 dissolve_direct_uv = InPolarUV;
			#else
				float2 dissolve_direct_uv = in_UV;
			#endif
			float dissolve_direct = S_Custom5.Sample(S_Custom5SamplerState, dissolve_direct_uv).r;
			dissolve_noise = (dissolve_noise + dissolve_direct) / 2.0f;
		#endif

		clip(dissolve_noise - dissolve_val);
	#endif
	
	//点阵剔除
	{
		//dither discard
		if(ComputeDitherCullMask(IN.i_PosInClip.xy,U_MaterialCustomParam1.y,U_MaterialCustomParam1.z))
			discard;
		bool needDiscard = ComputeBitmapCullMask(U_PSGeneralRegister29, IN.i_BoundBoxRange, 
		U_MaterialCustomParam1.z, IN.i_PosInModel.xyz, IN.i_PosInClip.xyz);
		if(needDiscard)
			discard;
	}

	//通用的溶解
	#ifdef DISSOLVE
		if(S_Custom7.Sample(S_Custom7SamplerState, uv).a < U_MaterialFresnel.y)
		discard;	
	#endif

	//序列帧
	#ifdef SEQUENCE_FRAME_PARTICLE
		float 	Tile = frac(U_PSGeneralRegister9.w * U_PSGeneralRegister10.x + U_PSGeneralRegister11.x / 255.0 * U_PSGeneralRegister10.y);							
		float 	Row = U_PSGeneralRegister10.w;				
		float 	Col = U_PSGeneralRegister10.z;							float	nIdx = floor(Tile*Row*Col);					
		float 	nRow = floor(nIdx / Col);							
		float 	nCol = nIdx - nRow*Col;									float 	fRow = nRow / Row;
		float 	fCol = nCol / Col;			
		uv = float2(fCol, fRow) + uv*float2(1.0 / Col, 1.0 / Row);	
	#endif
	
	// #ifdef SEQUENCE_FRAME
	// 	#ifdef SEQUENCE_FRAME_BASEMAP
	// 		float fmodtime = fmod(U_PS_Time.x, 60.0f);
	// 		int index = floor(fmodtime * U_PSGeneralRegister2.y);
	// 		int indexY = index / U_PSGeneralRegister2.z;
	// 		int indexX = index - indexY * U_PSGeneralRegister2.z;
	// 		uv = float2(uv.x / U_PSGeneralRegister2.z, uv.y / U_PSGeneralRegister2.z);
	// 		uv.x += indexX / U_PSGeneralRegister2.z;
	// 		uv.y += indexY / U_PSGeneralRegister2.z;
	// 	#endif
	// #endif

	//屏幕扰动
	#ifdef SCREENDISTORT
		float2 screen_distort_uv = (IN.i_PosInClipOut.xy / IN.i_PosInClipOut.w) * 0.5f + 0.5f;
		screen_distort_uv.y = 1.0f - screen_distort_uv.y;
	#endif

	
	#ifdef DISTORT
		float _distortAmt = U_PSGeneralRegister2.y;
		float _distortScale = U_PSGeneralRegister2.z;
		float _distortSpeed = U_PSGeneralRegister1.y;
		float Time = U_PSGeneralRegister12.w;
 
		#ifdef SCREENDISTORT
			float noise = snoise(screen_distort_uv * _distortScale + float2(Time * _distortSpeed, - Time * _distortSpeed));
			screen_distort_uv += noise * _distortAmt;
		#endif
	#endif
	//计算diffuseColor
	#ifdef SCREENDISTORT
		half4 diffuseColor = texSample2D(S_BackbufferTex, screen_distort_uv, S_BackbufferTexSamplerState);	
		
		#ifdef SCREENDISTORT_R
			diffuseColor.a = texSample2D(S_Diffuse, uv, S_DiffuseSamplerState).r;
		#else
			diffuseColor.a = texSample2D(S_Diffuse, uv, S_DiffuseSamplerState).a;
		#endif
		diffuseColor = pow(diffuseColor, U_PSGeneralRegister26.x);
	#else
		half4 diffuseColor = texSample2D(S_Diffuse, uv, S_DiffuseSamplerState);	
		diffuseColor = pow(diffuseColor, U_PSGeneralRegister26.x);
	#endif
		
	//basemap2
	#ifdef BASE_MAP2_DOUBLETEX
		float4 diffuse_color = diffuseColor;

		#ifdef BASE_MAP2_ENABLE_POLAR
			float2 base_map2_uv = InPolarUV.xy;

			float currentAlphaEnd = alphaDistanceFar + (alphaSmoothDistance + alphaBridgeDistance) * alphaNeedTwoWay;
			alphaFactor = smoothstep(currentAlphaStart, currentAlphaEnd, cam_to_obj_len);
			alphaFactor = alphaNeedTwoWay* (1.0f - alphaFactor) + (1.0f- alphaNeedTwoWay) * alphaFactor;
		#else
			alphaFactor = smoothstep(alphaDistanceStart, alphaDistanceFar, cam_to_obj_len);
		#endif

		#ifdef ALPHA_DISTANCE_REVERSE
			alphaFactor = 1.0f - alphaFactor;
		#endif
		psOut.color.a = psOut.color.a * lerp((1.0f - alphaDistanceStrength), 1.0f, alphaFactor);
	}
	#endif

	#ifdef DEPTH_FADE
	{
		float2 depth_uv;
		depth_uv.x = ((IN.i_PosInClipOut.x / IN.i_PosInClipOut.w) + 1) * 0.5f;
		depth_uv.y = 1 - ((IN.i_PosInClipOut.y / IN.i_PosInClipOut.w) + 1) * 0.5f;

		float depth_val = S_Depth.Sample(S_DepthSamplerState, depth_uv).r;

		float4 target_ndc_pos = float4(IN.i_PosInClipOut.x / IN.i_PosInClipOut.w, IN.i_PosInClipOut.y / IN.i_PosInClipOut.w, depth_val, 1.0f);

		float4 target_view_pos; 
		//target_view_pos.x = dot(U_InvProjectMatrix[0], target_ndc_pos);
		//target_view_pos.y = dot(U_InvProjectMatrix[1], target_ndc_pos);
		target_view_pos.z = dot(U_InvProjectMatrix[2], target_ndc_pos);
		target_view_pos.w = dot(U_InvProjectMatrix[3], target_ndc_pos);
		target_view_pos.z = target_view_pos.z / target_view_pos.w;

		float4 view_pos;
		view_pos.z = dot(U_InvProjectMatrix[2], IN.i_PosInClipOut);
		view_pos.w = dot(U_InvProjectMatrix[3], IN.i_PosInClipOut);
		view_pos.z = view_pos.z / view_pos.w;

		float distance = U_PSGeneralRegister27.x;
		
		float factor = smoothstep(0.0f, distance, abs(view_pos.z - target_view_pos.z));
		
		#ifdef DEPTH_FADE_REVERSE
			factor = 1.0f - factor;
		#endif
		
		#ifdef DEPTH_FADE_OPACITY
			float opacity = U_PSGeneralRegister27.y;
			psOut.color.a = psOut.color.a * factor + psOut.color.a*opacity ;
		#else 
			psOut.color.a = psOut.color.a * factor;
		#endif
		#ifdef DEPTH_FADE_COLOR
			//psOut.color.rgba *= U_PSGeneralRegister28.xyzw;
			//psOut.color.rgba *= U_PSGeneralRegister27.z* factor;
			psOut.color.rgb += U_PSGeneralRegister28.xyz * factor * U_PSGeneralRegister27.z;
		#endif
	}
	#endif

	float overdrawlerp = ceil(U_PS_Time.w);
	psOut.color = psOut.color * (1.0 - overdrawlerp) + float4(U_PS_Time.w * overdrawlerp, U_PS_Time.w * overdrawlerp, U_PS_Time.w * overdrawlerp, overdrawlerp);

	#ifdef SKYBOX
        float i_Depth = IN.i_PosInClip.z/IN.i_PosInClip.w;
		i_Depth = i_Depth*0.5f + 0.5f;
		i_Depth = 1.0f - i_Depth;
        psOut.o_Depth = i_Depth*0.0001f + 0.9999f;
    #endif
#endif

	return psOut;
}
