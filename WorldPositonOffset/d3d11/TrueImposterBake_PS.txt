#include <Common_Uniform_PS.inl>

struct PS_INPUT
{
	float4 position		: SV_POSITION;
	float3 normal		: TEXCOORD0;
	float2 uv			: TEXCOORD1;
};

float4 main(PS_INPUT IN) : SV_TARGET
{
    float4 psOut = 0.0;
    #ifdef BAKE_IMPOSTER_DEPTH_NORMAL_MAP
	    psOut = float4(normalize(IN.normal.xyz) * 0.5 + 0.5, IN.position.z);
    #endif
	return psOut;
}
