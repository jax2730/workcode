// U_PSCustom0.xy : Per Pass Full Resolution Coord Offset
// U_PSCustom0.zw : Half Viewport Pixel Size
uniform float4 U_PSCustom0;

// U_PSCustom1.xy : Viewport2xPixelSize
// U_PSCustom1.zw : Viewport2xPixelSize x 0.25 (for fusing add + mul into mad)
uniform float4 U_PSCustom1;

// U_PSCustom2.xy : NDCToView Mul
// U_PSCustom2.zw : NDCToView Add
uniform float4 U_PSCustom2;

// U_PSCustom3.x : EffectRadius
// U_PSCustom3.y : EffectSamplingRadiusNearLimitRec
//      if viewspace pixel closer than this,
//      don't enlarge shadow sampling radius anymore
//      (makes no sense to grow beyond some distance,
//      not enough samples to cover everything,
//      so just limit the shadow growth;
//      could be SSAOSettingsFadeOutFrom * 0.1 or less)
//
// U_PSCustom3.z : EffectShadowStrength   global strength of the effect (0 - 5)
// U_PSCustom3.w : EffectHorizonAngleThreshold :
//      limit errors on slopes and caused by
//      insufficient geometry tessellation (0.05 to 0.5)
uniform float4 U_PSCustom3;

// U_PSCustom4.x : EffectFadeOutMul       effect fade out from distance (ex. 25)
// U_PSCustom4.y : EffectFadeOutAdd       effect fade out to distance   (ex. 100)
// U_PSCustom4.z : EffectShadowPow
// U_PSCustom4.w : EffectShadowClamp
uniform float4 U_PSCustom4;
// U_PSCustom5 : PatternRotSclMatrices
uniform float4 U_PSCustom5[5];
// TODO : remove and use macro on specific platforms & quality settings
// U_PSCustom10.x : TapCount
uniform float4 U_PSCustom10;

uniform float4x4 U_PSCustom11; // view matrix

// packing/unpacking for edges; 2 bits per edge mean 4 gradient values (0, 0.33, 0.66, 1) for smoother transitions!
float PackEdges(float4 edgesLRTB)
{
//    int4 edgesLRTBi = int4( saturate( edgesLRTB ) * 3.0 + 0.5 );
//    return ( (edgesLRTBi.x << 6) + (edgesLRTBi.y << 4) + (edgesLRTBi.z << 2) + (edgesLRTBi.w << 0) ) / 255.0;

    // optimized, should be same as above
	edgesLRTB = round(saturate(edgesLRTB) * 3.05);
	return dot(edgesLRTB, float4(64.0 / 255.0, 16.0 / 255.0, 4.0 / 255.0, 1.0 / 255.0));
}

float3 NdcToViewSpace(const float2 pos, const float viewSpaceDepth)
{
	float3 ret;

	ret.xy = (U_PSCustom2.xy * pos + U_PSCustom2.zw) * viewSpaceDepth;
	ret.z = viewSpaceDepth;

	return ret;
}

// calculate effect radius and fit our screen sampling pattern inside it
void CalculateRadiusParameters(
    const float pixCenterLength,
    const float2 pixelDirRbViewspaceSizeAtCenterZ,
    out float pixLookupRadiusMod,
    out float effectRadius,
    out float falloffCalcMulSq)
{
	effectRadius = U_PSCustom3.x;

    // leaving this out for performance reasons: use something similar if radius needs to scale based on distance
    //effectRadius *= pow( pixCenterLength, g_ASSAOConsts.RadiusDistanceScalingFunctionPow);

    // when too close, on-screen sampling disk will grow beyond screen size; limit this to avoid closeup temporal artifacts
	const float tooCloseLimitMod = saturate(pixCenterLength * U_PSCustom3.y) * 0.8 + 0.2;

	effectRadius *= tooCloseLimitMod;

    // 0.85 is to reduce the radius to allow for more samples on a slope to still stay within influence
	pixLookupRadiusMod = (0.85 * effectRadius) / pixelDirRbViewspaceSizeAtCenterZ.x;

    // used to calculate falloff (both for AO samples and per-sample weights)
	falloffCalcMulSq = -1.0f / (effectRadius * effectRadius);
}

float4 CalculateEdges(const float centerZ, const float leftZ, const float rightZ, const float topZ, const float bottomZ)
{
    // slope-sensitive depth-based edge detection
	float4 edgesLRTB = float4(leftZ, rightZ, topZ, bottomZ) - centerZ;
	float4 edgesLRTBSlopeAdjusted = edgesLRTB + edgesLRTB.yxwz;
	edgesLRTB = min(abs(edgesLRTB), abs(edgesLRTBSlopeAdjusted));
	return saturate((1.3 - edgesLRTB / (centerZ * 0.040)));

    // cheaper version but has artifacts
    // edgesLRTB = abs( float4( leftZ, rightZ, topZ, bottomZ ) - centerZ; );
    // return saturate( ( 1.3 - edgesLRTB / (pixZ * 0.06 + 0.1) ) );
}

// all vectors in viewspace
float CalculatePixelObscurance(const float3 pixelNormal, const float3 hitDelta, const float falloffCalcMulSq)
{
	const float lengthSq = dot(hitDelta, hitDelta);
	const float nDotD = dot(pixelNormal, hitDelta) / sqrt(lengthSq);
	const float falloffMul = max(0.0, lengthSq * falloffCalcMulSq + 1.0);
	return max(0, nDotD - U_PSCustom3.w) * falloffMul;
}

void SSAOTapInner(
    inout float obscuranceSum,
    inout float weightSum,
    const float2 samplingUv,
    const float3 pixCenterPos,
    const float3 pixelNormal,
    const float falloffCalcMulSq,
    const float weightMod)
{
    // get depth at sample
	float viewspaceDepth = Depth.SampleLevel(PointClamp, samplingUv, 0).x;
	if (any(samplingUv < 0) || any(samplingUv > 1))
		viewspaceDepth = -65535.0; // far plane
    // convert to viewspace
	const float3 hitPos = NdcToViewSpace(samplingUv, viewspaceDepth);
	const float3 hitDelta = hitPos - pixCenterPos;

	const float obscurance = CalculatePixelObscurance(pixelNormal, hitDelta, falloffCalcMulSq);
	float weight = 1.0;

	weight *= weightMod;
	obscuranceSum += obscurance * weight;
	weightSum += weight;
}

// the actual number of texture samples is two times this value
// (each "tap" has two symmetrical depth texture samples)
void SSAOTap(
    inout float obscuranceSum,
    inout float weightSum,
    const int tapIndex,
    const float2x2 rotScale,
    const float3 pixCenterPos,
    const float3 pixelNormal,
    const float2 normalizedScreenPos,
    const float falloffCalcMulSq,
    float weightMod)
{
    // patterns
	float4 newSample = SamplePattern[tapIndex];
	float2 sampleOffset = mul(rotScale, newSample.xy);
	weightMod *= newSample.z;

    // snap to pixel center (more correct obscurance math, avoids artifacts)
	sampleOffset = round(sampleOffset);

	const float2 samplingUv = sampleOffset * U_PSCustom1.xy + normalizedScreenPos;

	SSAOTapInner(obscuranceSum, weightSum, samplingUv, pixCenterPos,
        pixelNormal, falloffCalcMulSq, weightMod);

    // for the second tap, just use the mirrored offset
	const float2 sampleOffsetMirroredUv = -sampleOffset;

    // snap to pixel center (more correct obscurance math, avoids artifacts)
	const float2 samplingMirroredUv = sampleOffsetMirroredUv * U_PSCustom1.xy + normalizedScreenPos;

	SSAOTapInner(obscuranceSum, weightSum, samplingMirroredUv, pixCenterPos,
        pixelNormal, falloffCalcMulSq, weightMod);
}

PS_OUTPUT main(PS_INPUT pIn)
{
	PS_OUTPUT pOut;

	const float2 posRounded = trunc(pIn.Position.xy);
	const int2 iPosRounded = int2(posRounded);

	const float2 centerUv = pIn.TexCoord.xy;

	float pixZ, pixLZ, pixTZ, pixRZ, pixBZ;

	float4 valuesUL = Depth.GatherRed(PointMirror, iPosRounded * U_PSCustom0.zw);
	float4 valuesBR = Depth.GatherRed(PointMirror, iPosRounded * U_PSCustom0.zw, int2(1, 1));
	
	// get this pixel's viewspace depth
	pixZ = valuesUL.y; //float pixZ = Depth.SampleLevel(PointClamp, centerUv, 0).x;

    // get left right top bottom neighbouring pixels for edge detection (gets compiled out on qualityLevel == 0)
	pixLZ = valuesUL.x;
	pixTZ = valuesUL.z;
	pixRZ = valuesBR.z;
	pixBZ = valuesBR.x;
    
	const float viewSpaceDepth = pixZ;
	float3 pixCenterPos = NdcToViewSpace(centerUv, viewSpaceDepth);
    // reconstruct this pixel's view-space normal
	int3 fullResCoord = int3(iPosRounded * 2 + int2(U_PSCustom0.xy), 0);
	float3 pixelWorldNormal = DecodeNormal(WorldNormal.Load(fullResCoord).rgb);
	float3 pixelNormal = normalize(mul(pixelWorldNormal, (float3x3) U_PSCustom11));
	
    //  optimized approximation of:
    //  float2 pixelDirRBViewspaceSizeAtCenterZ =
    //      NDCToViewspace( normalizedScreenPos.xy + g_ASSAOConsts.ViewportPixelSize.xy,
    //          pixCenterPos.z ).xy - pixCenterPos.xy;
	const float2 pixelDirRbViewspaceSizeAtCenterZ = -pixCenterPos.z * U_PSCustom2.xy * U_PSCustom0.zw;
	float pixLookupRadiusMod, falloffCalcMulSq, effectViewspaceRadius;
	CalculateRadiusParameters(length(pixCenterPos), pixelDirRbViewspaceSizeAtCenterZ,
        pixLookupRadiusMod, effectViewspaceRadius, falloffCalcMulSq);

    // calculate samples rotation/scaling
    // load & update pseudo-random rotation matrix
	const uint pseudoRandomIndex = uint(iPosRounded.y * 2 + iPosRounded.x) % 5;
	const float4 rs = U_PSCustom5[pseudoRandomIndex];
	const float2x2 rotScale = float2x2(rs.x * pixLookupRadiusMod, rs.y * pixLookupRadiusMod, rs.z * pixLookupRadiusMod,
        rs.w * pixLookupRadiusMod);

    // the main obscurance & sample weight storage
	float obscuranceSum = 0.0;
	float weightSum = 0.0;
	
	// edge mask for between this and left/right/top/bottom neighbour pixels - not used in quality level 0 so initialize to "no edge" (1 is no edge, 0 is edge)
	float4 edgesLRTB = float4(1.0, 1.0, 1.0, 1.0);
	
    // Move center pixel slightly towards camera to avoid imprecision artifacts
    // due to using of 16bit depth buffer; a lot smaller offsets needed when using 32bit floats
	pixCenterPos *= DEPTH_PRECISION_OFFSET_MOD;
	
	if (true)
	{
		edgesLRTB = CalculateEdges(pixZ, pixLZ, pixRZ, pixTZ, pixBZ);
	}
	
	// adds a more high definition sharp effect, which gets blurred out (reuses left/right/top/bottom samples that we used for edge detection)
	if (true)
	{
		//approximate neighbouring pixels positions (actually just deltas or "positions - pixCenterPos" )
		float3 viewspaceDirZNormalized = float3(pixCenterPos.xy / pixCenterPos.zz, 1.0);
		float3 pixLDelta = float3(-pixelDirRbViewspaceSizeAtCenterZ.x, 0.0, 0.0) + viewspaceDirZNormalized * (pixLZ - pixCenterPos.z); // very close approximation of: float3 pixLPos  = NDCToViewspace( normalizedScreenPos + float2( -g_ASSAOConsts.HalfViewportPixelSize.x, 0.0 ), pixLZ ).xyz - pixCenterPos.xyz;
		float3 pixRDelta = float3(+pixelDirRbViewspaceSizeAtCenterZ.x, 0.0, 0.0) + viewspaceDirZNormalized * (pixRZ - pixCenterPos.z); // very close approximation of: float3 pixRPos  = NDCToViewspace( normalizedScreenPos + float2( +g_ASSAOConsts.HalfViewportPixelSize.x, 0.0 ), pixRZ ).xyz - pixCenterPos.xyz;
		float3 pixTDelta = float3(0.0, -pixelDirRbViewspaceSizeAtCenterZ.y, 0.0) + viewspaceDirZNormalized * (pixTZ - pixCenterPos.z); // very close approximation of: float3 pixTPos  = NDCToViewspace( normalizedScreenPos + float2( 0.0, -g_ASSAOConsts.HalfViewportPixelSize.y ), pixTZ ).xyz - pixCenterPos.xyz;
		float3 pixBDelta = float3(0.0, +pixelDirRbViewspaceSizeAtCenterZ.y, 0.0) + viewspaceDirZNormalized * (pixBZ - pixCenterPos.z); // very close approximation of: float3 pixBPos  = NDCToViewspace( normalizedScreenPos + float2( 0.0, +g_ASSAOConsts.HalfViewportPixelSize.y ), pixBZ ).xyz - pixCenterPos.xyz;
		
		const float rangeReductionConst = 4.0f; // this is to avoid various artifacts
		const float modifiedFalloffCalcMulSq = rangeReductionConst * falloffCalcMulSq;
		
		float4 additionalObscurance;
		additionalObscurance.x = CalculatePixelObscurance(pixelNormal, pixLDelta, modifiedFalloffCalcMulSq);
		additionalObscurance.y = CalculatePixelObscurance(pixelNormal, pixRDelta, modifiedFalloffCalcMulSq);
		additionalObscurance.z = CalculatePixelObscurance(pixelNormal, pixTDelta, modifiedFalloffCalcMulSq);
		additionalObscurance.w = CalculatePixelObscurance(pixelNormal, pixBDelta, modifiedFalloffCalcMulSq);
		
		obscuranceSum += 0.5f * dot(additionalObscurance, edgesLRTB);
	}
	
	    // Sharp normals also create edges - but this adds to the cost as well
	if (true)
	{
		float3 neighbourNormalL = DecodeNormal(WorldNormal.Load(fullResCoord, int2(-2, 0)).rgb);
		float3 neighbourNormalR = DecodeNormal(WorldNormal.Load(fullResCoord, int2(2, 0)).rgb);
		float3 neighbourNormalT = DecodeNormal(WorldNormal.Load(fullResCoord, int2(0, -2)).rgb);
		float3 neighbourNormalB = DecodeNormal(WorldNormal.Load(fullResCoord, int2(0, 2)).rgb);

		const float dotThreshold = SSAO_NORMAL_BASED_EDGES_DOT_THRESHOLD;

		float4 normalEdgesLRTB;
		normalEdgesLRTB.x = saturate((dot(pixelWorldNormal, neighbourNormalL) + dotThreshold));
		normalEdgesLRTB.y = saturate((dot(pixelWorldNormal, neighbourNormalR) + dotThreshold));
		normalEdgesLRTB.z = saturate((dot(pixelWorldNormal, neighbourNormalT) + dotThreshold));
		normalEdgesLRTB.w = saturate((dot(pixelWorldNormal, neighbourNormalB) + dotThreshold));

		edgesLRTB *= normalEdgesLRTB;
	}

    [unroll]
	for (int i = 0; i < int(U_PSCustom10.x); i++)
	{
		SSAOTap(obscuranceSum, weightSum, i, rotScale, pixCenterPos, pixelNormal, centerUv,
            falloffCalcMulSq, 1.0);
	}

    // calculate weighted average
	float obscurance = obscuranceSum / weightSum;

    // calculate fadeout (1 close, gradient, 0 far)
	float fadeOut = 1; //saturate(pixCenterPos.z * U_PSCustom4.x + U_PSCustom4.y);
	
	// Reduce the SSAO shadowing if we're on the edge to remove artifacts on edges (we don't care for the lower quality one)
	if (true)
	{
        // float edgeCount = dot( 1.0-edgesLRTB, float4( 1.0, 1.0, 1.0, 1.0 ) );

        // when there's more than 2 opposite edges, start fading out the occlusion to reduce aliasing artifacts
		float edgeFadeoutFactor = saturate((1.0 - edgesLRTB.x - edgesLRTB.y) * 0.35) + saturate((1.0 - edgesLRTB.z - edgesLRTB.w) * 0.35);

        // (experimental) if you want to reduce the effect next to any edge
        // edgeFadeoutFactor += 0.1 * saturate( dot( 1 - edgesLRTB, float4( 1, 1, 1, 1 ) ) );

		fadeOut *= saturate(1.0 - edgeFadeoutFactor);
	}
	
	// same as a bove, but a lot more conservative version
    // fadeOut *= saturate( dot( edgesLRTB, float4( 0.9, 0.9, 0.9, 0.9 ) ) - 2.6 );
	
    // strength
	obscurance = U_PSCustom3.z * obscurance;
    // clamp
	obscurance = min(obscurance, U_PSCustom4.w);
    // fadeout
	obscurance *= fadeOut;

    // conceptually switch to occlusion with the meaning being visibility
    // (grows with visibility, occlusion == 1 implies full visibility), 
    // to be in line with what is more commonly used.
	float occlusion = 1.0 - obscurance;

    // modify the gradient
    // note: this cannot be moved to a later pass because of loss of precision after storing in the render target
	occlusion = pow(saturate(occlusion), U_PSCustom4.z);

	pOut.OcclusionEdge.x = occlusion; // Our final 'occlusion' term (0 means fully occluded, 1 means fully lit)
	pOut.OcclusionEdge.y = PackEdges(edgesLRTB);
	
	return pOut;
}
