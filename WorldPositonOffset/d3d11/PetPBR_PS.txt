		float4 ComputeRipple(float2 uv, float t)
		{
			//波纹贴图采样，并把采样的高度值扩展到-1到1。
			float4 ripple = S_Ripple.Sample(S_RippleSamplerState, uv);
			float gloss = ripple.y;
			ripple.yz = ripple.yz * 2.0 - 1.0;
			//获取波纹的时间,从A通道获取不同的波纹时间,
			float dropFrac = frac(ripple.a + t);
			//把时间限制在R通道内	
			float timeFrac = dropFrac - 1.0 + ripple.x;
			//做淡出处理
			float dropFactor = 1.0f-saturate( dropFrac);
			//计算最终的高度，用一个sin计算出随时间的振幅	 
			float final = dropFactor* sin(clamp(timeFrac * 9.0, 0.0, 4.0) * 3.1415926);
			return float4(ripple.yz * final, 1.0, (final ) * 0.2);
		}

		{

			screenTC *= U_ViewParam.xy;
			float depth = getSceneDepth(clipPos / clipPos.w, screenTC);
			
			return depth;
		}

float3 fresnelSchlick(float cosTheta, float3 F0)
{
	return F0 + (1.0-F0) * pow(1.0-cosTheta, 5.0);
}

float distributionGGX(float3 N, float3 H, float roughness)
{
	float a = roughness * roughness;
	float a2 = a * a;
	float NdotH = max(dot(N, H), 0.0);
	float NdotH2 = NdotH * NdotH;
	float nom = a2;
	float denom = (NdotH2 * (a2 - 1.0) + 1.0);
	denom = 3.141592654 * denom * denom;
	denom = max(denom, 0.00001);
	return nom / denom;
}

float geometrySchlickGGX(float NdotV, float roughness)
{
	float r = roughness + 1.0;
	float k = (r * r) / 8.0;
	
	float nom = NdotV;
	float denom = NdotV * (1.0 - k) + k;
	
	return nom / denom;
}

float geometrySmith(float3 N, float3 V, float3 L, float roughness)
{
	float NdotV = max(dot(N, V), 0.0);
	float NdotL = max(dot(N, L), 0.0);
	float ggx2 = geometrySchlickGGX(NdotV, roughness);
	float ggx1 = geometrySchlickGGX(NdotL, roughness);
	
	return ggx2 * ggx1;
}

float3 envBRDFApprox(float3 specularColor, float roughness, float NoV)
{
	float4 c0 = float4(-1.0,	-0.0275,	-0.572,		0.022);
	float4 c1 = float4(1.0,		0.0425,		1.04,		-0.04);
	
	float4 r = roughness * c0 + c1;
	float a004 = min(r.x * r.x, exp2(-9.28 * NoV)) * r.x + r.y;
	float2 AB = float2(-1.04, 1.04) * a004 + r.zw;
	return specularColor * AB.x + AB.y;
}

float3 thinTranslucency(float3 N, float3 L, float3 transmittanceColor)
{
	float transmittance = saturate(dot(-N, L) + 0.2);
	float fresnel = (1.0 - transmittance) * (1.0 - transmittance);
	fresnel *= fresnel;
	
	return transmittance * (1.0 - fresnel) * transmittanceColor;
}

PS_OUTPUT main(PS_INPUT IN,bool isFrontFacing : SV_IsFrontFace)
{
	PS_OUTPUT psOut = (PS_OUTPUT)0;

#ifdef PARABOLOID_MAPPING
	clip(IN.i_ClipDepth);
#endif

#ifdef SHADOWPASS
	#ifdef ALPHA_TEST

		#ifdef USEINSTANCE

			float4 color = S_Bump.Sample(S_BumpSamplerState, float3(IN.i_UV.xy,IN.i_Index.z));

			float4 color = S_Bump.Sample(S_BumpSamplerState, IN.i_UV.xy);
		#endif
		clip(color.b - matDiffuse.a);
	#endif
#else
	#ifdef MOTIONBLUR
		float2 vQX = IN.i_VelUV.xy;
	
		float fLenQX = length(vQX);
		vQX /= (fLenQX+0.01f);
		float fWeight = max(0.5f, min(fLenQX, 20.0f));
		vQX *= fWeight;
		psOut.color.xy = vQX;
		return psOut;
	#endif
	psOut.color.a=1.0f;
	#ifdef NoColor
		psOut.color = float4(0.0, 0.0, 0.0, 0.0);
		return psOut;
	#endif
	
		
	half4 bumpColor = 1.0;		
	float2 uv = IN.i_UV.xy;
	
	#ifdef DISSOLVE
	
		if(U_MaterialCustomParam1.y > 0)
		{
			float4x4 thresholdMatrix =
			{  1.0 / 17.0,  9.0 / 17.0,  3.0 / 17.0, 11.0 / 17.0,
			  13.0 / 17.0,  5.0 / 17.0, 15.0 / 17.0,  7.0 / 17.0,
			   4.0 / 17.0, 12.0 / 17.0,  2.0 / 17.0, 10.0 / 17.0,
			  16.0 / 17.0,  8.0 / 17.0, 14.0 / 17.0,  6.0 / 17.0
			};
			float2 pos = IN.i_PosInClip.xy;
			
			
			pos.xy = pos.xy/U_MaterialCustomParam1.z;
			
			int xindex = fmod(pos.x, 4);
			int yindex =fmod(pos.y, 4);
			
			float alpha = thresholdMatrix[xindex][yindex];//* _RowAccess[yindex];
			
			if(alpha < U_MaterialCustomParam1.y)
				discard;
		}
		else
		{
			if(S_Custom7.Sample(S_Custom7SamplerState, uv).a < U_MaterialFresnel.y)
			discard;	
		}
	#endif

	#ifdef USEINSTANCE
	
		int index = IN.i_Index.x*9;
		
		half4 materialDiffuse = U_PSCustom0[index];
		half3 emissiveColor = U_PSCustom0[index+1].xyz;
		half emissiveMinAmount = U_PSCustom0[index+1].w;
		half emissiveMaxAmount = U_PSCustom0[index+2].x;
		half3 coverColor = U_PSCustom0[index+2].yzw;		
		
		half3 maskRColor = U_PSCustom0[index+3].xyz;
		
		#ifdef HAIR
			half roughnessValue = U_PSCustom0[index+3].w;
			half metallicValue = U_PSCustom0[index+5].w;
		#endif
		#if defined(HAIR) || defined(TextureMask)
			half3 maskGColor = U_PSCustom0[index+4].xyz;
			half3 maskBColor = U_PSCustom0[index+5].xyz;
		#endif		
		
		half  coverScale = U_PSCustom0[index+4].w;	
		half  coverRange = U_PSCustom0[index+6].x;
		half todEmissive = U_PSCustom0[index+6].y;
		half dynamicEmissive = U_PSCustom0[index+6].z;
		half strokeStrength = U_PSCustom0[index+6].w;	
		
		half3 strokeColor = U_PSCustom0[index+7].xyz;
		half strokeScale = U_PSCustom0[index+7].w;

		half colorStrength = U_PSCustom0[index+8].x;
		half hdrParam = U_PSCustom0[index+8].y;
		half frequance = U_PSCustom0[index+8].z;	
		half coverCustom = U_PSCustom0[index+8].w;
	
		#ifdef BUMP_MAP
			bumpColor = S_Bump.Sample(S_BumpSamplerState, float3(uv,IN.i_Index.z)); 
		#else
			#ifdef ALPHA_TEST
				bumpColor = S_Bump.Sample(S_BumpSamplerState, float3(uv,IN.i_Index.z)); 	
			#endif
		#endif
	
		half4 diffuseColor = S_Diffuse.Sample(S_DiffuseSamplerState, float3(uv,IN.i_Index.y));		
		half4 mask = S_Custom2.Sample(S_Custom2SamplerState, float3(uv,IN.i_Index.w));	
		
		
	#else
		half4 materialDiffuse = U_MaterialDiffuse;
		half3 emissiveColor = U_PSGeneralRegister0.xyz;
		half emissiveMinAmount = U_PSGeneralRegister0.w;
		half emissiveMaxAmount = U_PSGeneralRegister1.x;
		half3 coverColor = U_PSGeneralRegister1.yzw;
		half3 maskRColor = U_PSGeneralRegister4.xyz;
	
		#ifdef HAIR
			half roughnessValue = U_PSGeneralRegister4.w;
			half metallicValue = U_PSGeneralRegister6.w;
		#endif	
		#if defined(HAIR) || defined(TextureMask)
			half3 maskGColor = U_PSGeneralRegister5.rgb;
			half3 maskBColor = U_PSGeneralRegister6.rgb;
		#endif
		half  coverScale = U_PSGeneralRegister5.w;		
		half  coverRange = U_PSGeneralRegister11.z;
		half todEmissive = U_PSGeneralRegister12.z;
		half dynamicEmissive = U_PSGeneralRegister12.w;
	
		half3 strokeColor = U_PSGeneralRegister9.xyz;
		half strokeScale = U_PSGeneralRegister9.w;
		half strokeStrength = U_PSGeneralRegister10.x;	
		half colorStrength = U_PSGeneralRegister10.y;
		half hdrParam = U_MaterialCustomParam1.x;
		half frequance = U_MaterialSpecular.w;	
		half coverCustom = U_PSGeneralRegister2.x;
		#ifdef BUMP_MAP
			bumpColor = S_Bump.Sample(S_BumpSamplerState, uv); 
		#else
			#ifdef ALPHA_TEST
				bumpColor = S_Bump.Sample(S_BumpSamplerState, uv); 	
			#endif
		#endif
	
		half4 diffuseColor = S_Diffuse.Sample(S_DiffuseSamplerState, uv);	
		half4 mask = S_Custom2.Sample(S_Custom2SamplerState, uv);
	#endif
	


	#ifdef ALPHA_TEST
		clip(bumpColor.b - materialDiffuse.a);
		bumpColor.b = 1.0f;
	#endif

	half3 baseMap = pow(diffuseColor.rgb, 2.2);
	
	#ifdef TextureMask
		half maskr = baseMap.r;
		half maskg = baseMap.b;
		half maskb = 1.0h - maskr - maskg;
		maskb = max(0.0h, maskb);
		half3 colorR = pow(maskRColor.rgb, 2.2) * maskr * baseMap.g;
		half3 colorG = pow(maskGColor.rgb, 2.2) * maskg * baseMap.g;
		half3 colorB = pow(maskBColor.rgb, 2.2) * maskb * baseMap.g;
		baseMap = colorR + colorG + colorB;
	#else
		#ifdef HAIR
			half3 colorR = pow(maskRColor.rgb, 2.2) * mask.r;
			half3 colorG = pow(maskGColor.rgb, 2.2) * mask.g;
			half3 colorB = pow(maskBColor.rgb, 2.2) * mask.b;
			float f = min(mask.r + mask.g + mask.b, 1.0);
			half3 colorF = colorR + colorG + colorB;
			baseMap = lerp(baseMap, baseMap * colorF, f);
		#else
			half3 colorMask = pow(maskRColor.rgb, 2.2);
			baseMap = lerp(baseMap, baseMap * colorMask, mask.r);
		#endif
	#endif
	
	
	half3 N = normalize(IN.i_Normal.xyz);
	
	#ifdef TWO_SIDES
		float frontValue = isFrontFacing ? 1.0f : -1.0f;
		N *= frontValue;
	#endif
	
	#ifdef USESHADOWMAP
		float4 shadowSpacePos[SHADOWSPLITE];
		ProjectToLightSpace(shadowSpacePos, IN.i_WPos.xyz, N, IN.i_UV.w);
	#endif

	half3 V = normalize(U_PS_CameraPosition.xyz - IN.i_WPos.xyz);
	half3 L = normalize(-U_MainLightDirection.xyz);
	
	half3 H = normalize(L+V);
	half NdotH = max(dot(N, H), 0.0h);
	half wetFinal = U_PSGeneralRegister8.w * NdotH;
	
	#ifdef BUMP_MAP
		half3 nn = N;
		half3 tn = normalize(IN.i_Tangent.xyz);
		half3 bn = normalize(IN.i_Binormal.xyz);

		tn = normalize(tn - nn * dot(tn, nn));
		bn = normalize(bn - nn * dot(bn, nn));

		half3x3 tbn = 
		{
			tn.x, bn.x, nn.x,
			tn.y, bn.y, nn.y,
			tn.z, bn.z, nn.z 
		};

		half3 bump = bumpColor.rgb * 2.0 - 1.0;
	
		#ifdef DETAIL_BUMP
			float2 detailUV = U_PSGeneralRegister13.x;	
			half4 detailBumpColor = S_Custom.Sample(S_CustomSamplerState, uv * detailUV);
			bump.xy += (detailBumpColor.xy - half2(0.5, 0.5)) * U_PSGeneralRegister13.y;
		#endif
		
		#ifdef RIPPLE
			float s = clamp(dot(N, float3(0.0,1.0,0.0)), 0.0, 1.0) ;
			s = pow(s, 5.0);
			s *= (1.f - U_PSGeneralRegister30.x);
			bump.xy  *= (1.0 - s * 0.9f);
			double time = U_PS_Time.x;
			float time01 = U_PS_Time.y * 0.1591549; //U_PS_Time.y / PI / 2
			//涟漪整体的透明度使用sin函数进行0-1的变化，当透明度变为0的时候，坐标进行偏移，以减轻雨滴的重复感。单张贴图透明度为0的时候会没有涟漪，所以采样两次，交替显示
			float2 rUV = IN.i_WPos.xz * 0.25f;
			
			float rOffset = floor(U_PS_Time.x * 0.5f) * 0.1f;
			float4 ripple = ComputeRipple(rUV + rOffset, time01 * 3.0f);
			
			rUV = IN.i_WPos.xz * 0.25f + 0.25f;
			rOffset = floor(U_PS_Time.x * 0.5f + 0.25f) * 0.1f;
			float4 ripple2 = ComputeRipple(rUV + float2(rOffset, -rOffset), time01 * 3.0f);
			
			float rLerp1 =  sin(time01 * 3.1415);
			float rLerp2 =  1.0 - rLerp1;
			bump.xy += rLerp1 * ripple.xy * s;
			bump.xy += rLerp2 * ripple2.xy * s;
		#endif

		bump.z = sqrt(1.0 - bump.x * bump.x - bump.y * bump.y);
		N = normalize(mul(tbn, bump));	
	
	#endif

	float coverLerp = dot(N, U_PS_GravityDirection.xyz);
	coverLerp += coverRange;
	coverLerp = coverLerp * coverScale;
	coverLerp = clamp(coverLerp, 0.0, 1.0);
	
	
	
	half3 coverMap = U_WeatherParam0.xyz;			
	coverMap = lerp(coverColor, coverMap, U_WeatherParam0.www * (1.0 - U_PSGeneralRegister30.x));
	coverMap = lerp(U_WeatherParam0.xyz, coverMap, coverCustom);
	coverLerp = lerp(coverLerp, coverLerp * U_WeatherParam0.w * (1.0 - U_PSGeneralRegister30.x), 1.0 - coverCustom);
			
	
	half metallic = 1.0;
	half roughness = 0.0001;
	half ao = 1.0;

	#ifdef HAIR
		metallic = metallicValue;
		roughness = roughnessValue;	
	#else
		metallic = mask.b;
		roughness = mask.g;
	#endif
	
	float r = U_WeatherParam1.x;
	roughness = lerp(roughness, r, coverLerp);
	
	//有涟漪的时候，给材质的高光整体加0.4		
	#ifdef RIPPLE
		roughness = clamp(roughness - 0.4f * (1.f - U_PSGeneralRegister30.x), 0.0, 1.0);
	#endif
	
	roughness = max(roughness , 1e-6);
	
	float m = U_WeatherParam1.y;
	metallic = lerp(metallic, m, coverLerp);
	
	half3 color = 0.0;
	
	half4 mainLightColor   = lerp(U_PBR_MainLightColor, U_RoleMainLightColor, U_PSGeneralRegister7.w);
	half4 skyLightColor    = lerp(U_PBR_SkyLightColor, U_RoleSkyLightColor, U_PSGeneralRegister7.w);
	half4 groundLightColor = lerp(U_PBR_GroundLightColor, U_RoleGroundLightColor, U_PSGeneralRegister7.w);
	half  blendThreshold   = lerp(U_PS_AmbientBlendThreshold.y, U_PS_AmbientBlendThreshold.z, U_PSGeneralRegister7.w);

	baseMap = lerp(baseMap, coverMap, coverLerp);
	half3 grayColor = baseMap.rgb * half3(0.333, 0.333, 0.333);
	float gray = grayColor.r + grayColor.g + grayColor.b;
	grayColor.rgb =  half3(gray,gray,gray);
	baseMap.rgb = lerp(baseMap.rgb,grayColor,U_MaterialCustomParam.x);		
	
	half3 albedo = baseMap * (1.0 - metallic);
	half3 specularColor = lerp(half3(0.04h, 0.04h, 0.04h), baseMap, metallic);

	specularColor = lerp(specularColor, half3(0.5h, 0.5h, 0.5h), wetFinal);
	color += calculateLightPower(N, V, L, mainLightColor.xyz * (1.0 - U_PSGeneralRegister8.x), baseMap, roughness, metallic);
	color = color*materialDiffuse.rgb;
	
	#ifdef USESHADOWMAP
		float shadowFactor = calcShadowFactor(
			shadowSpacePos, 
			IN.i_UV.w);
		shadowFactor = min(shadowFactor, 1.0);
		color *= shadowFactor;
	#endif

	color += calculateLightPower(N, V, -U_PS_ViewDirection.xyz, mainLightColor.xyz * U_PSGeneralRegister8.x, baseMap, roughness, metallic);

#ifdef CLUSTER_POINT_LIGHT
			color += calculateShadowPBRPointLight(IN.i_WPos, N, V, baseMap, roughness, metallic);

            float3 clusterLightColor = 
                calculateClusterPBRPointLight(IN.i_PosInClip.xy,
                                              IN.i_WPos.xyz,
                                              N,
                                              V,
                                              baseMap,
                                              roughness,
                                              metallic);
            color += clusterLightColor;
#endif

	
	half3 envBRDF = envBRDFApprox(specularColor, roughness, max(dot(N, V), 0.0));
	envBRDF = max(half3(0.0h, 0.0h, 0.0h), envBRDF);

	half3 ambientColor = ComputeAmbientLight(skyLightColor.xyz, groundLightColor.xyz, N, blendThreshold, U_PS_GravityDirection.xyz);

	envBRDF *= normalize(ambientColor);
	
	color += (albedo + envBRDF) * ambientColor;
	color *= ao;
	
	#ifdef STROKE
	
		half stroke = max(dot(V, N)*strokeScale + strokeStrength,0.0f);
		stroke = min(stroke,1.0f);
		stroke = (1.0 - stroke);
		color.rgb = lerp(color.rgb, strokeColor, stroke);
		color.rgb *= (colorStrength * stroke * 10.0 + 1.0);
	#endif
	
	#ifdef TextureMask
		diffuseColor.rgb = baseMap.rgb;
	#endif
	
	#ifdef TRANSPARENT_STROKE
		diffuseColor = float4(1,1,1,1);
		baseMap.rgb = float3(1,1,1);
		float stroke =  max(dot(V, N)*strokeScale + strokeStrength,0.0);
			
		stroke = min(stroke,1.0f);
		stroke = (1.0-stroke);
			
		color = half3(strokeColor*colorStrength);
		psOut.color.a = stroke;
	#endif

	half3 misColor = emissiveColor.xyz;
	half3 missiveColor = diffuseColor.rgb * misColor * diffuseColor.a;
	half3 minMissiveColor = missiveColor * emissiveMinAmount;
	half3 maxMissiveColor = missiveColor * emissiveMaxAmount;
	half todFactor = lerp(1.0, U_LightPower.x, todEmissive);
	half factor = frequance;	
	color += lerp(minMissiveColor, lerp(minMissiveColor, maxMissiveColor, half3(factor, factor, factor)), dynamicEmissive) * todFactor;
	
	float alphaaa = 1.0f;
	#ifdef SOFTBOARD
		float pixelDepth = 1.0f;
		float sceneDepth = 1.0f;
		float4 tempPos = 1.0f;
		tempPos = float4(IN.i_ClipPos.xy/IN.i_ClipPos.w, 0.0, 1.0);
		tempPos.x = tempPos.x * 0.5 + 0.5;
		tempPos.y = (1.0 - tempPos.y) * 0.5;
		sceneDepth = calcSceneDepth(IN.i_ClipPos);
		float4 vp;
		vp.x = dot(U_InvProjectMatrix[0], IN.i_ClipPos);
		vp.y = dot(U_InvProjectMatrix[1], IN.i_ClipPos);
		vp.z = dot(U_InvProjectMatrix[2], IN.i_ClipPos);
		vp.w = dot(U_InvProjectMatrix[3], IN.i_ClipPos);
		
		pixelDepth = vp.z / vp.w;
			
		alphaaa = saturate(abs(pixelDepth - sceneDepth) / IN.i_UV.z);
	#endif
	

	color *= hdrParam;
	color = color * IN.i_Fog.a + IN.i_Fog.rgb;
	psOut.color.rgb = pow(color, 1.0 / 2.2);
	psOut.color.a *= bumpColor.b * alphaaa;
	
	#ifdef EDITOR_MODE
		if(U_MaterialCustomParam.w > 0)
			psOut.color.rgb = U_MaterialCustomParam.xyz;			
	#endif
	
	#ifdef EditorHelp
		psOut.color.rgb = psOut.color.rgb * 0.00001 + float3(1.0,0.0,0.0);
		psOut.color.a = 1.0;
	#endif
	
	float overdrawlerp = ceil(U_PS_Time.w);
	psOut.color = psOut.color * (1.0 - overdrawlerp) + float4(U_PS_Time.w * overdrawlerp, U_PS_Time.w * overdrawlerp, U_PS_Time.w * overdrawlerp, overdrawlerp);
	
#endif

	return psOut;
}
