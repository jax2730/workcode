#include "SSSRCommon.h"

// Inputs
Texture2D<float> S_2DExt0;
#define g_roughness S_2DExt0
Texture2D<float3> S_2DExt1;
#define g_average_radiance S_2DExt1
Texture2D<float4> S_2DExt2;
#define g_in_radiance S_2DExt2
Texture2D<float4> S_2DExt3;
#define g_in_reprojected_radiance S_2DExt3
Texture2D<float> S_2DExt4;
#define g_in_variance S_2DExt4
Texture2D<float> S_2DExt5;
#define g_in_sample_count S_2DExt5

// Samplers
SamplerState S_2DExt0SamplerState;
#define g_linear_sampler S_2DExt0SamplerState

// Outputs
RWTexture2D<float4> S_2DExt6;
#define g_out_radiance S_2DExt6
RWTexture2D<float> S_2DExt7;
#define g_out_variance S_2DExt7
RWTexture2D<float> S_2DExt8;
#define g_out_sample_count S_2DExt8

StructuredBuffer<uint> B_CSCustom0;
#define g_denoiser_tile_list B_CSCustom0

float squre(float f) { return f * f; }

float3 FFX_DNSR_Reflections_SampleAverageRadiance(float2 uv) { return g_average_radiance.SampleLevel(g_linear_sampler, uv, 0.0f).xyz; }
float3 FFX_DNSR_Reflections_LoadRadiance(int2 pixel_coordinate) { return g_in_radiance.Load(int3(pixel_coordinate, 0)).xyz; }
float3 FFX_DNSR_Reflections_LoadRadianceReprojected(int2 pixel_coordinate) { return g_in_reprojected_radiance.Load(int3(pixel_coordinate, 0)).xyz; }
float FFX_DNSR_Reflections_LoadRoughness(int2 pixel_coordinate) { return squre(g_roughness.Load(int3(pixel_coordinate, 0))); }
float FFX_DNSR_Reflections_LoadVariance(int2 pixel_coordinate) { return g_in_variance.Load(int3(pixel_coordinate, 0)).x; }
float FFX_DNSR_Reflections_LoadNumSamples(int2 pixel_coordinate) { return g_in_sample_count.Load(int3(pixel_coordinate, 0)).x; }
void FFX_DNSR_Reflections_StoreTemporalAccumulation(int2 pixel_coordinate, float3 radiance, float variance) {
    g_out_radiance[pixel_coordinate] = radiance.xyzz;
    g_out_variance[pixel_coordinate] = variance.x;
}

#include "ffx_denoiser_reflections_config.h"
#define FFX_DNSR_REFLECTIONS_ESTIMATES_LOCAL_NEIGHBORHOOD
#include "ffx_denoiser_reflections_common.h"
#include "ffx_denoiser_reflections_resolve_temporal.h"

[numthreads(8, 8, 1)]
void main(int2 group_thread_id      : SV_GroupThreadID,
                uint group_index    : SV_GroupIndex,
                uint    group_id    : SV_GroupID) {
    uint  packed_coords               = g_denoiser_tile_list[group_id];
    int2  dispatch_thread_id          = int2(packed_coords & 0xffffu, (packed_coords >> 16) & 0xffffu) + group_thread_id;
    int2  dispatch_group_id           = dispatch_thread_id / 8;
    uint2 remapped_group_thread_id    = FFX_DNSR_Reflections_RemapLane8x8(group_index);
    uint2 remapped_dispatch_thread_id = dispatch_group_id * 8 + remapped_group_thread_id;

	FFX_DNSR_Reflections_ResolveTemporal(
        remapped_dispatch_thread_id, remapped_group_thread_id, 
        g_sssr_constants.g_buffer_dimensions, 
        g_sssr_constants.g_inv_buffer_dimensions, 
        g_sssr_constants.g_temporal_stability_factor);
}
