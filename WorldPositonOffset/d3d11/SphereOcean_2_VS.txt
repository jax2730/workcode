#include <Common_Uniform_VS.inl>
#include <Common_Function.inl>

float4  U_InvTransposeWorldMatrix[3];
float4	U_WorldViewProjectMatrix[4];	
float4  U_WorldMatrix[3];
float4  U_InvWorldMatrix[3];

float4  U_VSCustom0[3];
float4  U_VSCustom1[8];
float4  U_VSCustom2[3];

//#define FAR_HORIZONS

struct VS_INPUT
{
	float3 i_Pos		: POSITION0;
    float3 i_normal			: NORMAL0;
    float2 i_texcoord			: TEXCOORD0;
};

struct VS_OUTPUT
{
	float4	o_PosInClip			: SV_POSITION;
#ifndef FAR_HORIZONS
	float3	o_PosBitangentInWorld		: TEXCOORD0;
	float3 	o_PosTangeInWorld	: TEXCOORD1;
#endif
	float3  o_PosNorInWorld	: TEXCOORD2;

	float4 bumpCoord0 : TEXCOORD3;
	float4 bumpCoord1 : TEXCOORD4;
	
	float3 eyeVector  : TEXCOORD5;
	float4 clipPos	  : TEXCOORD7;

#ifndef FAR_HORIZONS
    float  wavePeak     : TEXCOORD6;
#endif

#if defined SOFT_BORDER	&& !defined FAR_HORIZONS
	float4 customParam: TEXCOORD8;
#endif
	float3 positionWS : TEXCOORD9;
	float3 centerPos  : TEXCOORD10;
	float4 fogColor   : COLOR0;

};

// wave functions
struct Wave {
	float amp;   // amplitude
	float sharp; // sharpness
	float freq;  // 2*PI / wavelength
	float phase; // speed * 2*PI / wavelength
	float3 dir;
};


VS_OUTPUT main(VS_INPUT IN)
{
	VS_OUTPUT vsOut = (VS_OUTPUT)0;

	float3 centerPos = float3(U_VSCustom0[2].x,U_VSCustom0[2].y,U_VSCustom0[2].z);
	vsOut.centerPos = centerPos;

	float4  pos =  float4(IN.i_Pos.xyz, 1.0f);
	float time = U_VS_Time.x ;
	float radius = U_VSCustom0[0].w;

	float3 positionWS;
	positionWS.x = dot(U_WorldMatrix[0], pos);
	positionWS.y = dot(U_WorldMatrix[1], pos);
	positionWS.z = dot(U_WorldMatrix[2], pos);

	float3 normalOnSph = positionWS - centerPos;
	float3 normalToOut = normalize(IN.i_normal.xyz);
	float3 outNormal;
	outNormal.x = dot(float4(normalToOut,1.0f),U_InvTransposeWorldMatrix[0]);
	outNormal.y = dot(float4(normalToOut,1.0f),U_InvTransposeWorldMatrix[1]);
	outNormal.z = dot(float4(normalToOut,1.0f),U_InvTransposeWorldMatrix[2]);
	outNormal = normalize(outNormal);
	float3 positionVecNormalized = normalize(normalOnSph);
	float3 vPosInWorld;
#ifndef FAR_HORIZONS
	positionWS -= centerPos;
	float sinPart = 0;
	float3 cosPart = float3(0,0,0);

	float sinPartNorm = 0;
	float3 cosPartNorm = float3(0,0,0);
	
	float3 tangent = float3(0,0,0);

	Wave wave[4] = 
	{
		{ U_VSCustom1[0].x, U_VSCustom1[0].y, U_VSCustom1[0].z, U_VSCustom1[0].w, float3(U_VSCustom1[4].xyz) },
		{ U_VSCustom1[1].x, U_VSCustom1[1].y, U_VSCustom1[1].z, U_VSCustom1[1].w, float3(U_VSCustom1[5].xyz)},
		{ U_VSCustom1[2].x, U_VSCustom1[2].y, U_VSCustom1[2].z, U_VSCustom1[2].w, float3(U_VSCustom1[6].xyz)},
		{ U_VSCustom1[3].x, U_VSCustom1[3].y, U_VSCustom1[3].z, U_VSCustom1[3].w, float3(U_VSCustom1[7].xyz)}
	};

	// float4 waveParams = _WaveParameters[waveIdx];
	for(int i=0; i<4; i++){
		float3 waveDir = wave[i].dir;
		float3 waveDirNormalized = normalize(waveDir);

		float A = wave[i].amp;
		float Q = wave[i].sharp;
		float w = wave[i].freq;
		float fi = wave[i].phase;
		
		float3 di = cross(positionVecNormalized, cross(positionVecNormalized -  waveDirNormalized, positionVecNormalized));
		float li = acos(dot(positionVecNormalized, waveDirNormalized)) * radius;
		
		sinPart += A * sin(w * li + fi * time);
		cosPart += Q * A * cos(w * li + fi * time) * di;

		sinPartNorm += Q * A * w * sin(w * li + fi * time);
		cosPartNorm += di * A * w * cos(w * li + fi * time);

		float3 diCross = cross(di, positionVecNormalized);
		tangent += diCross / length(diCross);
	}
	tangent = normalize(tangent);

	float3 wavePose;
	wavePose = positionWS + positionVecNormalized * sinPart + cosPart;
	//wavePose = float3((U_WorldMatrix[0][0]/radius) * wavePose.x,(U_WorldMatrix[1][1]/radius) * wavePose.y,(U_WorldMatrix[2][2]/radius) * wavePose.z);

	float3 waveNormal = positionVecNormalized - positionVecNormalized * sinPartNorm - cosPartNorm;
	waveNormal = normalize(waveNormal);
	wavePose = wavePose + centerPos;
	
    vsOut.wavePeak = sinPartNorm;
	
	// 在非球形上不连续，视觉上修复一下
	float3 deltaNor = waveNormal - positionVecNormalized;
	vsOut.o_PosNorInWorld = normalize(outNormal + deltaNor);

	pos.x = dot(U_InvWorldMatrix[0], float4(wavePose,1.0f));
	pos.y = dot(U_InvWorldMatrix[1], float4(wavePose,1.0f));
	pos.z = dot(U_InvWorldMatrix[2], float4(wavePose,1.0f));
	vPosInWorld = wavePose;

	vsOut.o_PosTangeInWorld.x = dot(U_WorldMatrix[0], float4(tangent, 0.0f));
    vsOut.o_PosTangeInWorld.y = dot(U_WorldMatrix[1], float4(tangent, 0.0f));
    vsOut.o_PosTangeInWorld.z = dot(U_WorldMatrix[2], float4(tangent, 0.0f));

	vsOut.o_PosBitangentInWorld = normalize(cross(vsOut.o_PosNorInWorld, vsOut.o_PosTangeInWorld));
#else  //FAR_HORIZONS
	vPosInWorld = positionWS;
	vsOut.o_PosNorInWorld = positionVecNormalized;
#endif //FAR_HORIZONS

	vsOut.positionWS = vPosInWorld;

	vsOut.o_PosInClip.x = dot(U_WorldViewProjectMatrix[0], pos);
	vsOut.o_PosInClip.y = dot(U_WorldViewProjectMatrix[1], pos);
	vsOut.o_PosInClip.z = dot(U_WorldViewProjectMatrix[2], pos);
	vsOut.o_PosInClip.w = dot(U_WorldViewProjectMatrix[3], pos);



	// FORM　OCEAN2_VS
	// calculate texture coordinates for normal map lookup
	
	float BumpScale = U_VSCustom0[1].x;
	float2 textureScale = U_VSCustom0[1].yz;
	float windSpeed = U_VSCustom0[1].w;
	

	// calculate texture coordinates for normal map lookup
	float2 tex_coord 		= float2(dot((vPosInWorld-centerPos),float3(0,1.0f,0)),dot((vPosInWorld-centerPos),float3(1.0f,0.0f,0.0f)));
	float realWindSpeed = windSpeed * 0.5 * fmod(U_VS_Time.x, 100.0f);
	/*
	
	vsOut.bumpCoord0.xy = tex_coord * textureScale 		  - vDirection * realWindSpeed * 0.01;
	vsOut.bumpCoord0.zw = tex_coord * textureScale * 2.0f - vDirection * realWindSpeed * 0.04;
	vsOut.bumpCoord0.z *= -1.0f;
	vsOut.bumpCoord1.xy = tex_coord * textureScale * 4.0f + vDirection * realWindSpeed * 0.08;
	vsOut.bumpCoord1.x *= -1.0f;
	*/

	
	vsOut.bumpCoord0.xy = textureScale;
	vsOut.bumpCoord0.z = windSpeed;
	vsOut.bumpCoord1.xyz = U_WindParams.xyz;
	
	vsOut.eyeVector = vPosInWorld.xyz - U_VS_CameraPosition.xyz; // eye position in world space

	float3 camToWorld =  U_VS_CameraPosition.xyz - vPosInWorld.xyz;	
	float4x4 gravityRotatePosMatrix;	
	gravityRotatePosMatrix[0] = U_VS_GravityRotationMatrix[0];
	gravityRotatePosMatrix[1] = U_VS_GravityRotationMatrix[1];
	gravityRotatePosMatrix[2] = U_VS_GravityRotationMatrix[2];
	gravityRotatePosMatrix[3] = U_VS_GravityRotationMatrix[3];
	camToWorld = mul(gravityRotatePosMatrix,float4(camToWorld.xyz,0.0f)).xyz;
	
	vsOut.fogColor = ComputeFogColor(camToWorld, U_FogParam.xy, U_FogParam.zw, U_FogRampParam, U_FogColorNear, U_FogColorFar);

#if defined SOFT_BORDER	&& !defined FAR_HORIZONS
	vsOut.customParam.xy = float2(dot((vPosInWorld-centerPos),float3(0,1.0f,0)),dot((vPosInWorld-centerPos),float3(1.0f,0.0f,0.0f)));	
#endif

	vsOut.clipPos = vsOut.o_PosInClip;
 	return vsOut;
}
