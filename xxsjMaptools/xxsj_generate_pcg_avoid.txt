namespace XXSJ_PCG_ROAD {

	const String gLevelPath = "levels/karstworld/";
	const String gExportPath = gLevelPath + "pcg/line_data";
	const float gBlockSize = 2048.0f;
	const float gRoadHalfWidth = 32.0f;

	struct PolylineSegment {
		enum TypeEnum : int {
			eUnknow = 0,
			eRoad = (1 << 0),
			eRiver = (1 << 1),
			eRailway = (1 << 2),
		};
		PolylineSegment() :startD(0.0f), endD(0.0f), type(eUnknow) {}
		union {
			struct {
				float startD;
				float endD;
				TypeEnum type;
			};
			float DATA[3];
		};
		std::vector<Vector2> points;
	};

	inline bool SegmentAABBIntersectsBlock(const Vector2& p0, const Vector2& p1, float width,float blockMinX, float blockMinY, float blockMaxX, float blockMaxY) {
		float minX = std::min(p0.x, p1.x) - width * 0.5f;
		float maxX = std::max(p0.x, p1.x) + width * 0.5f;
		float minY = std::min(p0.y, p1.y) - width * 0.5f;
		float maxY = std::max(p0.y, p1.y) + width * 0.5f;
		return !(maxX < blockMinX || minX > blockMaxX || maxY < blockMinY || minY > blockMaxY);
	}

	void RemoveNearDuplicatePoints(std::vector<Vector2>& points, float epsilon = 1e-3f) {
		std::vector<Vector2> result;
		if (points.empty()) return;
		result.push_back(points[0]);
		for (size_t i = 1; i < points.size(); ++i) {
			if (points[i].distance(result.back()) > epsilon) {
				result.push_back(points[i]);
			}
		}
		points = result;
	}

	void SplitPolylineByBlock(std::vector<Vector2>& points, float blockSize, PolylineSegment::TypeEnum type, float polylineWidth, std::map<EchoIndex2, std::vector<PolylineSegment>>& blockPolylines) {

		auto getBlocks = [&](const Vector2& p0, const Vector2& p1) {
			std::set<EchoIndex2> blocks;
			float minX = std::min(p0.x, p1.x) - polylineWidth * 0.5f;
			float maxX = std::max(p0.x, p1.x) + polylineWidth * 0.5f;
			float minY = std::min(p0.y, p1.y) - polylineWidth * 0.5f;
			float maxY = std::max(p0.y, p1.y) + polylineWidth * 0.5f;
			int blockMinI = static_cast<int>(std::floor(minX / blockSize));
			int blockMaxI = static_cast<int>(std::floor(maxX / blockSize));
			int blockMinJ = static_cast<int>(std::floor(minY / blockSize));
			int blockMaxJ = static_cast<int>(std::floor(maxY / blockSize));
			for (int i = blockMinI; i <= blockMaxI; ++i) {
				for (int j = blockMinJ; j <= blockMaxJ; ++j) {
					float blockMinX = i * blockSize;
					float blockMaxX = (i + 1) * blockSize;
					float blockMinY = j * blockSize;
					float blockMaxY = (j + 1) * blockSize;
					if (!(maxX < blockMinX || minX > blockMaxX || maxY < blockMinY || minY > blockMaxY)) {
						blocks.emplace(i, j);
					}
				}
			}
			return blocks;
			};

		RemoveNearDuplicatePoints(points, 1.0f);
		if (points.empty()) return;

		if (points.size() == 1) {
			PolylineSegment seg;
			seg.points = { points[0] };
			seg.startD = 0.0f;
			seg.endD = 0.0f;
			seg.type = type;
			std::set<EchoIndex2> blocks = getBlocks(points[0], points[0]);
			for (const auto& bi : blocks) {
				blockPolylines[bi].push_back(seg);
			}
			return;
		}

		std::vector<float> accDist(points.size(), 0.0f);
		for (size_t i = 1; i < points.size(); ++i) {
			accDist[i] = accDist[i - 1] + (points[i] - points[i - 1]).length();
		}

		PolylineSegment curSeg;
		std::set<EchoIndex2> curBlocks;

		for (size_t i = 0; i + 1 < points.size(); ++i) {
			Vector2 p0 = points[i];
			Vector2 p1 = points[i + 1];
			float d0 = accDist[i];
			float d1 = accDist[i + 1];

			std::set<EchoIndex2> segBlocks = getBlocks(p0, p1);

			if (curSeg.points.empty()) {
				curSeg.points.push_back(p0);
				curSeg.startD = d0;
				curSeg.type = type;
				curBlocks = segBlocks;
			}

			if (segBlocks == curBlocks) {
				curSeg.points.push_back(p1);
				curSeg.endD = d1;
			}
			else {
				if (curSeg.points.size() >= 2) {
					for (const auto& bi : curBlocks) {
						blockPolylines[bi].push_back(curSeg);
					}
				}

				curSeg = PolylineSegment();
				curSeg.points.push_back(p0);
				curSeg.points.push_back(p1);
				curSeg.startD = d0;
				curSeg.endD = d1;
				curSeg.type = type;
				curBlocks = segBlocks;
			}
		}

		if (curSeg.points.size() >= 2) {
			for (const auto& bi : curBlocks) {
				blockPolylines[bi].push_back(curSeg);
			}
		}
	}

	void ReadFileBin(const String& trainPath, std::vector<std::vector<Vector2>>& trains_data) {
		DataStreamPtr pDataStream(Root::instance()->GetFileDataStream(trainPath.c_str(), true));
		if (pDataStream.isNull()) return;
		int withHeight = 0;
		pDataStream->readData(&withHeight, 1);
		int lineNum = 0;
		pDataStream->readData(&lineNum, 1);
		trains_data.resize(lineNum);
		if (withHeight == 0) {
			for (int i = 0; i < lineNum; ++i) {

				std::vector<Vector2>& points = trains_data[i];
				int pointCount = 0;
				pDataStream->readData(&pointCount, 1);

				points.resize(pointCount);
				pDataStream->readData(points.data(), points.size());
			}
		}
		else {
			for (int i = 0; i < lineNum; ++i) {

				std::vector<Vector2>& points = trains_data[i];
				std::vector<Vector3> point3s;
				int pointCount = 0;
				pDataStream->readData(&pointCount, 1);

				points.resize(pointCount);
				point3s.resize(pointCount);
				pDataStream->readData(point3s.data(), point3s.size());
				for (int j = 0; j < pointCount; ++j) {
					points[j].x = point3s[j].x;
					points[j].y = point3s[j].z;
				}
			}
		}
	}

	void DealFileData(const String& filePath, PolylineSegment::TypeEnum data_type, std::map<EchoIndex2, std::vector<PolylineSegment>>& splinePageMaps) {

		std::vector<std::vector<Vector2>> trains_data;
		ReadFileBin(filePath, trains_data);

		int roadSize = (int)trains_data.size();
		for (int roadIndex = 0; roadIndex != roadSize; ++roadIndex) {
			SplitPolylineByBlock(trains_data[roadIndex], gBlockSize, data_type, gRoadHalfWidth, splinePageMaps);
		}
	}

	void DealRoadData_World_China() {

		std::filesystem::path exportPath = Root::instance()->getRootResourcePath() + gExportPath;
		std::filesystem::remove_all(exportPath);
		std::filesystem::create_directory(exportPath);

		std::map<EchoIndex2, std::vector<PolylineSegment>> splinePageMaps;

		if (true) {
			String trainPath = gLevelPath + "china_road.bin";
			DealFileData(trainPath, PolylineSegment::eRoad, splinePageMaps);
		}

		if (true) {
			String trainPath = gLevelPath + "china_river.bin";
			DealFileData(trainPath, PolylineSegment::eRiver, splinePageMaps);
		}

		if (true) {
			String trainPath = gLevelPath + "railway.bin";
			DealFileData(trainPath, PolylineSegment::eRailway, splinePageMaps);
		}

		if (!splinePageMaps.empty()) {
			{
				DataStreamPtr pDataStream(Root::instance()->CreateDataStream((std::string(gExportPath + "/pcg.page")).c_str()));
				uint32 splinePageMapsSize = (uint32)splinePageMaps.size();
				pDataStream->writeData(&splinePageMapsSize);
				for (auto& iter : splinePageMaps) {
					pDataStream->writeData(&iter.first.x);
					pDataStream->writeData(&iter.first.z);
				}
			}
			for (auto& iter : splinePageMaps) {
				DataStreamPtr pDataStream(Root::instance()->CreateDataStream((std::string(gExportPath + "/section_") + std::to_string(iter.first.x) + "_" + std::to_string(iter.first.z) + std::string(".page")).c_str()));
				int splineSize = (int)iter.second.size();
				pDataStream->writeData(&splineSize);
				for (int i = 0; i != splineSize; ++i) {
					PolylineSegment& spline = iter.second[i];
					pDataStream->writeData(spline.DATA, 3);
					uint32 pointSize = (uint32)spline.points.size();
					pDataStream->writeData(&pointSize);
					if (pointSize != 0) {
						pDataStream->writeData(spline.points.data(), spline.points.size());
					}
				}
			}
		}

	}
